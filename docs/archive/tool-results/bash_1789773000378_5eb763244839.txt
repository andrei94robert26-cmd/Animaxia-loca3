     1→/**
     2→ * modules/search/search.js — FAZA 16-a: Neon PostgreSQL full-text search
     3→ * engine over the `searchdocs` denormalized index table (synced from
     4→ * `content`). Engine label: 'animaxia-neon-fts'.
     5→ *
     6→ * FAZA 17-a hardening (all additive — FAZA 16 contracts intact):
     7→ *   • OFFSET pagination capped at 10000 rows: page*limit > 10000 without a
     8→ *     cursor → 400 { error, useCursor: true } (offsetCapped counter);
     9→ *   • keyset (cursor) pagination: ?cursor=base64url({sort-specific payload +
    10→ *     p: pageNumber}) — per-sort WHERE continuations (score/year/rating/title,
    11→ *     tie-break on contentid; filter-only + no q → contentid keyset);
    12→ *     responses gain nextCursor (string|null) + paginationMode 'offset'|'cursor';
    13→ *   • single-flight coalescing: identical concurrent /advanced requests share
    14→ *     one promise (coalesced counter, map cleared on settle);
    15→ *   • query timeout: runAdvancedQuery races an 8s budget → 504
    16→ *     { error: 'Search query timeout' } (errors counter, loser logged);
    17→ *   • searchOpsSnapshot() exports { coalesced, errors, offsetCapped } for the
    18→ *     stability endpoint; runSearchLoad(q) is the stress-lab entry point.
    19→ *
    20→ * Index: syncIndex() upserts every content row into searchdocs with a
    21→ * to_tsvector('simple', …) document (title + genres + first 500 chars of the
    22→ * overview) plus year / rating / genres[] / actors[] / studios[] / category
    23→ * extracted from the JSONB metadata. Boot is lazy — the first query triggers
    24→ * ensureSynced(); 'content:changed' debounces a background resync (3s, async
    25→ * lock coalescing).
    26→ *
    27→ * Router (mount: app.use('/api/search', searchRouter)):
    28→ *   GET  /advanced?q=&type=&genre=&year=&minRating=&actor=&studio=&category=
    29→ *        &sort=score|year|rating|title&fuzzy=true&limit=20&page=1&cursor=
    30→ *        → { took, total, max_score, page, limit, engine, data: [...],
    31→ *             nextCursor, paginationMode }
    32→ *   GET  /autocomplete?q= → { suggestions: [{ id, title, type, thumbnail, rating }] }
    33→ *   GET  /filters         → { genres, actors, years, categories, studios }
    34→ *   POST /index           → { ok, indexed }                       (requireAuth+requireAdmin)
    35→ *   GET  /status          → { docs, engine, backend, extensions, lastSync, dirty,
    36→ *                             latency: { p50Ms, p95Ms, samples }, capacity }
    37→ *
    38→ * Text match: tsv @@ websearch_to_tsquery('simple', q). When that yields 0
    39→ * hits and fuzzy !== 'false', a trigram fallback runs (title % q OR per-word
    40→ * ILIKE). Combined score: ts_rank(tsv, q) * 2 + similarity(title, q) +
    41→ * (title ILIKE 'q%' → +0.5). Empty q + filters → filter-only query (no text
    42→ * score). Rolling p50/p95 latency over the last 50 queries feeds /status.
    43→ */

    72→/** base64url encode/decode (no padding issues, URL-safe). */
    73→function b64urlEncode(obj) {
    74→  return Buffer.from(JSON.stringify(obj), 'utf8')
    75→    .toString('base64')
    76→    .replace(/\+/g, '-')
    77→    .replace(/\//g, '_')
    78→    .replace(/=+$/, '');
    79→}

    81→function b64urlDecode(text) {
    82→  try {
    83→    const padded = String(text).replace(/-/g, '+').replace(/_/g, '/');
    84→    return JSON.parse(Buffer.from(padded, 'base64').toString('utf8'));
    85→  } catch {
    86→    return null;
    87→  }
    88→}

    90→/**
    91→ * Build the next cursor from the LAST row of the current page, per sort.
    92→ * Filter-only + no q → effective sort is contentid (see effectiveSort).
    93→ */
    94→function buildNextCursor({ sort, lastRow, pageNumber }) {
    95→  if (!lastRow) return null;
    96→  const id = Number(lastRow.id);
    97→  if (!Number.isFinite(id)) return null;
    98→  let payload = null;
    99→  if (sort === 'year') payload = { y: lastRow._year ?? null, id };
   100→  else if (sort === 'rating') payload = { r: lastRow._rating ?? null, id };
   101→  else if (sort === 'title') payload = { t: lastRow._title ?? null, id };
   102→  else if (sort === 'score') payload = { s: Number(lastRow._score ?? 0), id };
   103→  else payload = { id }; // contentid keyset (filter-only)
   104→  payload.p = pageNumber; // page estimate — incremented on each cursor hop
   105→  return b64urlEncode(payload);
   106→}

   108→/**
   109→ * Decode + validate the cursor payload for a given effective sort.
   110→ * @returns {{ payload: object, where: string, params: any[] } | { error: string }}
   111→ */
   112→function cursorContinuation(effectiveSort, cursorText) {
   113→  const payload = b64urlDecode(cursorText);
   114→  if (!payload || typeof payload !== 'object' || !Number.isFinite(Number(payload.id))) {
   115→    return { error: 'Invalid cursor' };
   116→  }
   117→  const id = Number(payload.id);
   118→  const params = [];
   119→  const P = (v) => {
   120→    params.push(v);
   121→    return `$${params.length}`;
   122→  };
   123→
   124→  if (effectiveSort === 'score') {
   125→    const s = Number(payload.s);
   126→    if (!Number.isFinite(s)) return { error: 'Invalid cursor (missing s)' };
   127→    const sp = P(s);
   128→    const ip = P(id);
   129→    return { payload, where: `(t.score < ${sp} OR (t.score = ${sp} AND t.contentid > ${ip}))`, params };
   130→  }
   131→  if (effectiveSort === 'year') {
   132→    const y = Number(payload.y);
   133→    if (!Number.isFinite(y)) return { error: 'Invalid cursor (missing y)' };
   134→    const yp = P(y);
   135→    const ip = P(id);
   136→    return { payload, where: `(t.year < ${yp} OR (t.year = ${yp} AND t.contentid > ${ip}))`, params };
   137→  }
   138→  if (effectiveSort === 'rating') {
   139→    const r = Number(payload.r);
   140→    if (!Number.isFinite(r)) return { error: 'Invalid cursor (missing r)' };
   141→    const rp = P(r);
   142→    const ip = P(id);
   143→    return { payload, where: `(t.rating < ${rp} OR (t.rating = ${rp} AND t.contentid > ${ip}))`, params };
   144→  }
   145→  if (effectiveSort === 'title') {
   146→    const t = payload.t;
   147→    if (typeof t !== 'string') return { error: 'Invalid cursor (missing t)' };
   148→    const tp = P(t);
   149→    const ip = P(id);
   150→    return { payload, where: `(t.title > ${tp} OR (t.title = ${tp} AND t.contentid > ${ip}))`, params };
   151→  }
   152→  // contentid keyset (filter-only / no q)
   153→  const ip = P(id);
   154→  return { payload, where: `t.contentid > ${ip}`, params };
   155→}

   159→/** Coerce a metadata value to a clean string array (strings or {name} objects). */
   160→function toNameArray(value) {
   161→  if (value == null) return [];
   162→  const list = Array.isArray(value) ? value : [value];
   163→  const out = [];
   164→  for (const item of list) {
   165→    if (typeof item === 'string') {
   166→      const name = item.trim();
   167→      if (name) out.push(name);
   168→    } else if (item && typeof item === 'object') {
   169→      const name = String(item.name ?? '').trim();
   170→      if (name) out.push(name);
   171→    }
   172→  }
   173→  return out;
   174→}

   181→function toFloatOrNull(value) {

   186→/** Build the FTS document text: title + genres + first 500 chars of overview. */
   187→function buildTsvText(title, genres, meta) {
   188→  const overview = String(meta.overview || meta.synopsis || meta.description || '');
   189→  return [String(title || ''), genres.join(' '), overview.slice(0, 500)].join(' ').trim();
   190→}

   193→
   194→let syncPromise = null; // async lock — concurrent ensureSynced() calls coalesce
   195→let lastSync = null; // ISO timestamp of the last successful sync
   196→let dirty = true; // set at boot and on every content:changed
   197→
   198→async function syncIndex() {
   199→  const raw = await rows(
   200→    'SELECT id, externalid, type, title, metadata, thumbnail, createdat FROM content',
   201→  );
   202→
   203→  let synced = 0;
   204→  for (let i = 0; i < raw.length; i += SYNC_CHUNK) {
   205→    const slice = raw.slice(i, i + SYNC_CHUNK);
   206→    const values = [];
   207→    const tuples = slice.map((row, j) => {
   208→      const meta = parseJsonb(row.metadata, {});
   209→      const genres = toNameArray(meta.genres);
   210→      const actors = toNameArray(meta.actors ?? meta.cast);
   211→      const studios = toNameArray(meta.studios ?? meta.studio);
   212→      const tsvText = buildTsvText(row.title, genres, meta);
   213→      // FAZA 13 (sharding): searchdocs carries its shard id = the catalog
   214→      // RANGE shard of the contentid it indexes (ids < 1B → 0, legacy-safe).
   215→      const shardid = computeShardIdFromContentId(row.id);
   216→      const base = j * 12;
   217→      values.push(
   218→        Number(row.id),
   219→        row.externalid ?? null,
   220→        row.type ?? null,
   221→        row.title ?? 'untitled',
   222→        tsvText,
   223→        toIntOrNull(meta.year),
   224→        toFloatOrNull(meta.rating ?? meta.score),
   225→        genres,
   226→        actors,
   227→        studios,
   228→        row.type ?? null,
   229→        shardid,
   230→      );
   231→      const b = (k) => `$${base + k}`;
   232→      return `(${b(1)},${b(2)},${b(3)},${b(4)},to_tsvector('simple', ${b(5)}),${b(6)},${b(7)},${b(8)}::text[],${b(9)}::text[],${b(10)}::text[],${b(11)},${b(12)},NOW())`;
   233→    });
   234→    await rows(
   235→      `INSERT INTO searchdocs
   236→         (contentid, externalid, type, title, tsv, year, rating, genres, actors, studios, category, shardid, syncedat)
   237→       VALUES ${tuples.join(', ')}
   238→       ON CONFLICT (contentid) DO UPDATE SET
   239→         externalid = EXCLUDED.externalid,
   240→         type = EXCLUDED.type,
   241→         title = EXCLUDED.title,
   242→         tsv = EXCLUDED.tsv,
   243→         year = EXCLUDED.year,
   244→         rating = EXCLUDED.rating,
   245→         genres = EXCLUDED.genres,
   246→         actors = EXCLUDED.actors,
   247→         studios = EXCLUDED.studios,
   248→         category = EXCLUDED.category,
   249→         shardid = EXCLUDED.shardid,
   250→         syncedat = NOW()`,
   251→      values,
   252→    );
   253→    synced += slice.length;
   254→  }
   255→
   256→  // Drop stale docs (content rows deleted since the last sync).
   257→  try {
   258→    await rows('DELETE FROM searchdocs WHERE contentid NOT IN (SELECT id FROM content)');
   259→  } catch (err) {
   260→    logger.warn(`[search] stale-doc cleanup skipped: ${err.message}`);
   261→  }
   262→
   263→  lastSync = new Date().toISOString();
   264→  dirty = false;
   265→  return synced;
   266→}

   268→/** Rebuild once under an async lock; concurrent callers await the same run. */
   269→async function ensureSynced() {
   270→  if (!dirty && lastSync) return;
   271→  if (!syncPromise) {
   272→    syncPromise = syncIndex().finally(() => {
   273→      syncPromise = null;
   274→    });
   275→  }
   276→  await syncPromise;
   277→}
   278→
   279→/** Forced full resync (POST /index): waits for any in-flight sync, then runs one. */
   280→async function forceResync() {
   281→  while (syncPromise) {
   282→    await syncPromise.catch(() => {});
   283→  }
   284→  syncPromise = syncIndex().finally(() => {
   285→    syncPromise = null;
   286→  });
   287→  return syncPromise;
   288→}

   290→// content:changed → dirty immediately; debounced background resync (3s).
   291→let resyncTimer = null;
   292→events.on('content:changed', () => {
   293→  dirty = true;
   294→  if (resyncTimer) clearTimeout(resyncTimer);
   295→  resyncTimer = setTimeout(() => {
   296→    resyncTimer = null;
   297→    ensureSynced().catch((err) => logger.warn(`[search] background resync failed: ${err.message}`));
   298→  }, SYNC_DEBOUNCE_MS);
   299→  if (typeof resyncTimer.unref === 'function') resyncTimer.unref();
   300→});

   311→function percentile(sorted, p) {
   312→  if (sorted.length === 0) return 0;
   313→  const idx = Math.min(sorted.length - 1, Math.max(0, Math.ceil(p * sorted.length) - 1));
   314→  return Number(sorted[idx].toFixed(2));
   315→}

   347→/**
   348→ * Run one advanced query against searchdocs (joined back to content).
   349→ * Filter fragments are built once and shared by the count + data statements.
   350→ * FAZA 17-a: `cursor` (base64url payload) switches to keyset pagination —
   351→ * the ORDER BY becomes the cursor continuation WHERE and OFFSET is dropped;
   352→ * `effectiveSort` is what actually orders the result (contentid keyset when
   353→ * there is no text query and the caller asked for the default 'score' sort,
   354→ * because score would be constant 0 for every row).
   355→ */
   356→async function runAdvancedQuery({
   357→  q,
   358→  type,
   359→  genre,
   360→  year,
   361→  minRating,
   362→  actor,
   363→  studio,
   364→  category,
   365→  sort,
   366→  fuzzyEnabled,
   367→  limit,
   368→  page,
   369→  cursor = null,
   370→}) {
   371→  const params = [];
   372→  const P = (v) => {
   373→    params.push(v);
   374→    return `$${params.length}`;
   375→  };
   376→
   377→  // Effective sort: filter-only (no q) with default 'score' → contentid keyset.
   378→  const effectiveSort = !q && sort === 'score' ? 'contentid' : sort;
   379→
   380→  // Keyset continuation (cursor present) — validated per effective sort.
   381→  let cursorPayload = null;
   382→  let cursorWhere = null;
   383→  let cursorParams = [];
   384→  if (cursor) {
   385→    const decoded = cursorContinuation(effectiveSort, cursor);
   386→    if (decoded.error) {
   387→      const err = new Error(decoded.error);
   388→      err.status = 400;
   389→      throw err;
   390→    }
   391→    cursorPayload = decoded.payload;
   392→    cursorWhere = decoded.where;
   393→    cursorParams = decoded.params;
   394→  }
   395→
   396→  // Field filters (SQL-side, case-insensitive where array containment).
   397→  const filters = [];
   398→  if (type) filters.push(`s.type = ${P(type)}`);
   399→  if (genre) filters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${P(genre)})`);
   400→  if (year != null) filters.push(`s.year = ${P(year)}`);
   401→  if (actor) {
   402→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${P(actor)} || '%')`);
   403→  }
   404→  if (studio) {
   405→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${P(studio)} || '%')`);
   406→  }
   407→  if (category) filters.push(`s.category = ${P(category)}`);
   408→  if (minRating != null) filters.push(`s.rating >= ${P(minRating)}`);
   409→  const filterSql = filters.length ? ` AND ${filters.join(' AND ')}` : '';
   410→
   411→  // Text score expression (combined: ts_rank * 2 + similarity + prefix bonus).
   412→  let scoreExpr = '0';
   413→  let textMatch = '';
   414→  if (q) {
   415→    const qp = P(q);
   416→    scoreExpr =
   417→      `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qp})) * 2` +
   418→      ` + similarity(s.title, ${qp})` +
   419→      ` + CASE WHEN s.title ILIKE ${qp} || '%' THEN 0.5 ELSE 0 END`;
   420→    textMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qp})`;
   421→  }
   422→
   423→  const whereFor = (matchSql) => `WHERE (${matchSql || 'TRUE'})${filterSql}`;
   424→
   425→  // 1. FTS pass (or filter-only when q is empty).
   426→  let matchSql = textMatch;
   427→  let countRow = await one(
   428→    `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   429→    params,
   430→  );
   431→
   432→  // 2. Trigram fuzzy fallback when the FTS pass matched nothing.
   433→  //    `title % q` (pg_trgm, default 0.3 threshold) plus a looser
   434→  //    similarity() > 0.1 guard so short typo'd queries still reach long
   435→  //    titles (e.g. 'dunee' → 'Dune: Part Two - Wikipedia', sim ≈ 0.15).
   436→  if (q && fuzzyEnabled && Number(countRow?.total ?? 0) === 0) {
   437→    const words = q.split(/\s+/).filter(Boolean);
   438→    const trigramClauses = [];
   439→    for (const candidate of [q, ...words]) {
   440→      const ph = P(`%${candidate}%`);
   441→      trigramClauses.push(`s.title ILIKE ${ph}`);
   442→    }
   443→    const qSim = P(q);
   444→    trigramClauses.push(`(s.title % ${qSim} OR similarity(s.title, ${qSim}) > 0.1)`);
   445→    matchSql = `(${trigramClauses.join(' OR ')})`;
   446→    countRow = await one(
   447→      `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   448→      params,
   449→    );
   450→  }
   451→
   452→  const total = Number(countRow?.total ?? 0);
   453→  const maxScore = Number(Number(countRow?.maxscore ?? 0).toFixed(6));
   454→
   455→  // 3. Data page — subquery keeps the score expression, then joins content.
   456→  //    FAZA 17-a: expose year/rating/title to JS for nextCursor construction;
   457→  //    cursor mode replaces OFFSET with the keyset continuation predicate.
   458→  const orderExpr =
   459→    effectiveSort === 'year'
   460→      ? 't.year DESC NULLS LAST, t.contentid ASC'
   461→      : effectiveSort === 'rating'
   462→        ? 't.rating DESC NULLS LAST, t.score DESC, t.contentid ASC'
   463→        : effectiveSort === 'title'
   464→          ? 't.title ASC, t.contentid ASC'
   465→          : effectiveSort === 'contentid'
   466→            ? 't.contentid ASC'
   467→            : 't.score DESC, t.contentid ASC';
   468→
   469→  const limitPh = P(limit);
   470→  const offsetPh = P((page - 1) * limit);
   471→  // Cursor mode: continuation params come FIRST ($1..$n), then LIMIT; the
   472→  // filter/score params are re-bound only inside the subquery — so build a
   473→  // dedicated statement for each mode instead of renumbering dynamically.
   474→  let dataSql;
   475→  let dataParams;
   476→  if (cursorWhere) {
   477→    // Rebuild the subquery with a fresh parameter namespace for the cursor run
   478→    // (filters + score expression are re-parameterized after the cursor slots).
   479→    const subParams = [...cursorParams, limit];
   480→    const sub = (v) => {
   481→      subParams.push(v);
   482→      return `$${subParams.length}`;
   483→    };
   484→    const cFilters = [];
   485→    if (type) cFilters.push(`s.type = ${sub(type)}`);
   486→    if (genre) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${sub(genre)})`);
   487→    if (year != null) cFilters.push(`s.year = ${sub(year)}`);
   488→    if (actor) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${sub(actor)} || '%')`);
   489→    if (studio) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${sub(studio)} || '%')`);
   490→    if (category) cFilters.push(`s.category = ${sub(category)}`);
   491→    if (minRating != null) cFilters.push(`s.rating >= ${sub(minRating)}`);
   492→    const cFilterSql = cFilters.length ? ` AND ${cFilters.join(' AND ')}` : '';
   493→
   494→    let cScoreExpr = '0';
   495→    let cMatch = 'TRUE';
   496→    // Single q slot, bound LAST (after cursor params + LIMIT) so every
   497→    // placeholder references a parameter that exists in the bind array.
   498→    let qParam = null;
   499→    if (q) {
   500→      qParam = sub(String(q));
   501→      cScoreExpr =
   502→        `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qParam})) * 2` +
   503→        ` + similarity(s.title, ${qParam})` +
   504→        ` + CASE WHEN s.title ILIKE ${qParam} || '%' THEN 0.5 ELSE 0 END`;
   505→      cMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qParam})`;
   506→    }
   507→    // Fuzzy fallback consistency: when the FTS pass matched nothing the offset
   508→    // run switched matchSql — mirror that for the cursor run using `total`.
   509→    if (q && fuzzyEnabled && total === 0) {
   510→      const words = q.split(/\s+/).filter(Boolean);
   511→      const clauses = [q, ...words].map((candidate) => `s.title ILIKE ${sub(`%${candidate}%`)}`);
   512→      clauses.push(`(s.title % ${qParam} OR similarity(s.title, ${qParam}) > 0.1)`);
   513→      cMatch = `(${clauses.join(' OR ')})`;
   514→    }
   515→
   516→    dataSql =
   517→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   518→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   519→       FROM (
   520→         SELECT s.contentid, s.year, s.rating, s.title, ${cScoreExpr} AS score
   521→         FROM searchdocs s
   522→         WHERE (${cMatch})${cFilterSql}
   523→       ) t
   524→       JOIN content c ON c.id = t.contentid
   525→       WHERE ${cursorWhere}
   526→       ORDER BY ${orderExpr}
   527→       LIMIT $${cursorParams.length + 1}::int`;
   528→    // LIMIT is the placeholder right after the cursor params (::int — a bare
   529→    // parameter in LIMIT cannot be type-inferred by Postgres); the q/filter
   530→    // slots follow it so every bound parameter is referenced by the statement.
   531→    dataParams = subParams;
   532→  } else {
   533→    dataSql =
   534→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   535→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   536→       FROM (
   537→         SELECT s.contentid, s.year, s.rating, s.title, ${scoreExpr} AS score
   538→         FROM searchdocs s
   539→         ${whereFor(matchSql)}
   540→       ) t
   541→       JOIN content c ON c.id = t.contentid
   542→       ORDER BY ${orderExpr}
   543→       LIMIT ${limitPh} OFFSET ${offsetPh}`;
   544→    dataParams = params;
   545→  }
   546→  const dataRows = await rows(dataSql, dataParams);
   547→
   548→  // Keyset paging estimates the page number by incrementing the cursor's `p`.
   549→  const resultPage = cursor ? (Number(cursorPayload?.p) || 1) + 1 : page;
   550→  const normalized = dataRows.map(normalizeHit).filter(Boolean);
   551→  const lastRaw = dataRows.length ? dataRows[dataRows.length - 1] : null;
   552→
   553→  return {
   554→    total,
   555→    max_score: maxScore,
   556→    data: normalized,
   557→    page: resultPage,
   558→    nextCursor: buildNextCursor({ sort: effectiveSort, lastRow: lastRaw, pageNumber: resultPage }),
   559→    paginationMode: cursor ? 'cursor' : 'offset',
   560→  };
   561→}

   565→/**
   566→ * FAZA 17-a: identical concurrent /advanced requests share one promise.
   567→ * Key = normalized query params; the map entry is removed on settle.
   568→ */
   569→const inflightSearch = new Map();
   570→
   571→function coalesceAdvanced(normalizedKey, run) {
   572→  const existing = inflightSearch.get(normalizedKey);
   573→  if (existing) {
   574→    searchOps.coalesced += 1;
   575→    return existing;
   576→  }
   577→  const promise = run().finally(() => {
   578→    inflightSearch.delete(normalizedKey);
   579→  });
   580→  inflightSearch.set(normalizedKey, promise);
   581→  return promise;
   582→}

   584→/** Race a query promise against the 8s budget (loser is logged, not awaited). */
   585→function withQueryTimeout(promise, describe) {
   586→  let timer = null;
   587→  const budget = new Promise((_, reject) => {
   588→    timer = setTimeout(() => {
   589→      const err = new Error('Search query timeout');
   590→      err.status = 504;
   591→      reject(err);
   592→    }, QUERY_TIMEOUT_MS);
   593→    if (typeof timer.unref === 'function') timer.unref();
   594→  });
   595→  return promise.then(
   596→    (value) => {
   597→      clearTimeout(timer);
   598→      return value;
   599→    },
   600→    (err) => {
   601→      clearTimeout(timer);
   602→      throw err;
   603→    },
   604→  ).finally(() => {
   605→    clearTimeout(timer);
   606→    promise.catch((loser) => {
   607→      // The lost race (if it also failed) must not become an unhandledRejection.
   608→      logger.warn(`[search] query lost the timeout race (${describe}): ${loser?.message ?? loser}`);
   609→    });
   610→  });
   611→}

   663→  const fuzzyEnabled = fuzzy !== 'false'; // default on
   664→  const limit = Math.min(50, Math.max(1, parseInt(req.query.limit, 10) || 20));
   665→  const page = Math.max(1, parseInt(req.query.page, 10) || 1);
