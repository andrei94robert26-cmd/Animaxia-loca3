    66→function formatUptime(sec: number): string {
    67→  const h = Math.floor(sec / 3600);
    68→  const m = Math.floor((sec % 3600) / 60);
    69→  const s = Math.floor(sec % 60);
    70→  if (h > 0) return `${h}h ${m}m`;
    71→  if (m > 0) return `${m}m ${s}s`;
    72→  return `${s}s`;
    73→}
    74→
    75→function formatBytes(b?: number): string {
    76→  if (!b || b <= 0) return '0 B';
    77→  const units = ['B', 'KB', 'MB', 'GB'];
    78→  let v = b;
    79→  let i = 0;
    80→  while (v >= 1024 && i < units.length - 1) {
    81→    v /= 1024;
    82→    i += 1;
    83→  }
    84→  return `${v >= 100 ? Math.round(v) : v.toFixed(1)} ${units[i]}`;
    85→}
    86→
    87→function formatMs(ms?: number): string {
    88→  if (ms == null) return '—';
    89→  if (ms < 1000) return `${Math.round(ms)}ms`;
    90→  return `${(ms / 1000).toFixed(1)}s`;
    91→}
    92→
    93→function formatDate(iso?: string | null): string {
    94→  if (!iso) return '—';
    95→  const d = new Date(iso);
    96→  if (Number.isNaN(d.getTime())) return '—';
    97→  return d.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
    98→}

   100→function getErrorMessage(e: unknown): string {
   101→  return e instanceof Error ? e.message : 'Request failed';
   102→}
