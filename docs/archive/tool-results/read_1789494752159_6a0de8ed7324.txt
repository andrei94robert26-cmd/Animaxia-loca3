    56→/** Renders AI output: bullet lines ("- " / "• ") become styled list items. */
    57→function AiText({ text }: { text: string }) {
    58→  const lines = text
    59→    .split('\n')
    60→    .map((l) => l.trim())
    61→    .filter(Boolean);
    62→  return (
    63→    <div className="space-y-2.5">
    64→      {lines.map((line, i) => {
    65→        if (line.startsWith('- ') || line.startsWith('• ')) {
    66→          return (
    67→            <div key={i} className="flex gap-2.5">
    68→              <span className="mt-[9px] h-1.5 w-1.5 shrink-0 rounded-full bg-rose-400" aria-hidden />
    69→              <p className="text-sm leading-relaxed text-zinc-300">{line.slice(2)}</p>
    70→            </div>
    71→          );
    72→        }
    73→        return (
    74→          <p key={i} className="text-sm leading-relaxed text-zinc-300">
    75→            {line}
    76→          </p>
    77→        );
    78→      })}
    79→    </div>
    80→  );

   133→/**
   134→ * Library content picker: debounced (300 ms) autocomplete dropdown with
   135→ * thumbnails, keyboard (Escape) and click-outside dismissal, and a selected
   136→ * chip. Reused by the Summarize and Chat tabs.
   137→ */
   138→function ContentPicker({
   139→  selected,
   140→  onSelect,
   141→  label,
   142→  inputId,
   143→}: {
   144→  selected: Suggestion | null;
   145→  onSelect: (s: Suggestion | null) => void;
   146→  label: string;
   147→  inputId: string;
   148→}) {
   149→  const [query, setQuery] = useState('');
   150→  const [open, setOpen] = useState(false);
   151→  const [suggestions, setSuggestions] = useState<Suggestion[]>([]);
   152→  const [loading, setLoading] = useState(false);
   153→  const rootRef = useRef<HTMLDivElement>(null);
   154→
   155→  /* Debounced autocomplete (300 ms while typing; instant clear otherwise —
   156→     state updates happen inside the async timer to avoid sync setState). */
   157→  useEffect(() => {
   158→    const q = query.trim();
   159→    const timer = setTimeout(
   160→      () => {
   161→        if (!open || q.length === 0) {
   162→          setSuggestions([]);
   163→          return;
   164→        }
   165→        setLoading(true);
   166→        platform
   167→          .autocomplete(q)
   168→          .then((res) => setSuggestions(res.suggestions))
   169→          .catch(() => setSuggestions([]))
   170→          .finally(() => setLoading(false));
   171→      },
   172→      open && q.length > 0 ? 300 : 0,
   173→    );
   174→    return () => clearTimeout(timer);
   175→  }, [query, open]);
   176→
   177→  /* Click-outside closes the dropdown. */
   178→  useEffect(() => {
   179→    if (!open) return;
   180→    const onDown = (e: MouseEvent) => {
   181→      if (rootRef.current && !rootRef.current.contains(e.target as Node)) setOpen(false);
   182→    };
   183→    document.addEventListener('mousedown', onDown);
   184→    return () => document.removeEventListener('mousedown', onDown);
   185→  }, [open]);
   186→
   187→  return (
   188→    <div className="space-y-2" ref={rootRef}>
   189→      <label htmlFor={inputId} className="text-sm font-semibold text-zinc-300">
   190→        {label}
   191→      </label>
   192→      <div className="relative">
   193→        <Search
   194→          className="pointer-events-none absolute left-3.5 top-1/2 h-4 w-4 -translate-y-1/2 text-zinc-500"
   195→          aria-hidden
   196→        />
   197→        <Input
   198→          id={inputId}
   199→          type="text"
   200→          role="combobox"
   201→          aria-expanded={open}
   202→          aria-controls={`${inputId}-listbox`}
   203→          aria-autocomplete="list"
   204→          autoComplete="off"
   205→          value={query}
   206→          placeholder="Search the library…"
   207→          onChange={(e) => {
   208→            setQuery(e.target.value);
   209→            setOpen(true);
   210→          }}
   211→          onFocus={() => setOpen(true)}
   212→          onKeyDown={(e) => {
   213→            if (e.key === 'Escape') setOpen(false);
   214→          }}
   215→          className="h-11 rounded-xl border-white/10 bg-white/[0.04] pl-10 text-sm text-zinc-100 placeholder:text-zinc-500 focus-visible:border-rose-500/50 focus-visible:ring-rose-500/30"
   216→        />
   217→        {open && query.trim().length > 0 && (
   218→          <div
   219→            id={`${inputId}-listbox`}
   220→            role="listbox"
   221→            aria-label="Library suggestions"
   222→            className="absolute z-30 mt-2 max-h-72 w-full overflow-y-auto rounded-xl border border-white/10 bg-[#12121c] p-2 shadow-2xl shadow-black/70 custom-scrollbar"
   223→          >
   224→            {loading ? (
   225→              <p className="px-2 py-3 text-xs text-zinc-500">Searching the library…</p>
   226→            ) : suggestions.length === 0 ? (
   227→              <p className="px-2 py-3 text-xs text-zinc-500">No titles match that search.</p>
   228→            ) : (
   229→              suggestions.map((s) => (
   230→                <button
   231→                  key={s.id}
   232→                  type="button"
   233→                  role="option"
   234→                  aria-selected={selected?.id === s.id}
   235→                  onClick={() => {
   236→                    onSelect(s);
   237→                    setQuery('');
   238→                    setOpen(false);
   239→                  }}
   240→                  className="flex w-full items-center gap-3 rounded-lg p-2 text-left transition-colors hover:bg-white/5 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-rose-500/60"
   241→                >
   242→                  <span className="h-12 w-9 shrink-0 overflow-hidden rounded-md border border-white/10 bg-white/[0.03]">
   243→                    {s.thumbnail ? (
   244→                      <img src={s.thumbnail} alt="" loading="lazy" className="h-full w-full object-cover" />
   245→                    ) : (
   246→                      <BrandPlaceholder title={s.title} />
   247→                    )}
   248→                  </span>
   249→                  <span className="min-w-0 flex-1">
   250→                    <span className="block truncate text-sm font-semibold text-zinc-100">{s.title}</span>
   251→                  </span>
   252→                  <TypeBadge type={s.type} />
   253→                </button>
   254→              ))
   255→            )}
   256→          </div>
   257→        )}
   258→      </div>
   259→      {selected && (
   260→        <div className="flex items-center gap-2">
   261→          <span className="inline-flex max-w-full items-center gap-2 rounded-full border border-rose-500/30 bg-rose-500/10 py-1 pl-1 pr-2 text-sm text-rose-200">
   262→            <span className="h-7 w-7 shrink-0 overflow-hidden rounded-full border border-white/10 bg-white/[0.03]">
   263→              {selected.thumbnail ? (
   264→                <img src={selected.thumbnail} alt="" className="h-full w-full object-cover" />
   265→              ) : (
   266→                <BrandPlaceholder title={selected.title} />
   267→              )}
   268→            </span>
   269→            <span className="truncate font-medium">{selected.title}</span>
   270→            <button
   271→              type="button"
   272→              onClick={() => onSelect(null)}
   273→              aria-label={`Clear ${selected.title}`}
   274→              className="rounded-full p-1 text-rose-300/70 transition hover:bg-rose-500/20 hover:text-rose-100"
   275→            >
   276→              <X className="h-3.5 w-3.5" aria-hidden />
   277→            </button>
   278→          </span>
   279→        </div>
   280→      )}
   281→    </div>
   282→  );
   283→}

   366→function RecommendationsTab({ user }: { user: PlatformUser | null }) {
   367→  const [picks, setPicks] = useState<AiRecommendation[] | null>(null);
   368→  const [provider, setProvider] = useState('');
   369→  const [loading, setLoading] = useState(false);
   370→  const [error, setError] = useState<string | null>(null);
   371→  const [upgrade, setUpgrade] = useState<string | null>(null);
   372→
   373→  async function fetchPicks() {
   374→    if (loading) return;
   375→    setLoading(true);
   376→    setError(null);
   377→    setUpgrade(null);
   378→    try {
   379→      const res = await platform.aiRecommendations(8);
   380→      setPicks(res.data);
   381→      setProvider(res.provider);
   382→    } catch (e) {
   383→      if (e instanceof PlatformError && e.upgrade) setUpgrade(e.message);
   384→      else setError(e instanceof Error ? e.message : 'Could not get AI picks');
   385→    } finally {
   386→      setLoading(false);
   387→    }
   388→  }
   389→
   390→  return (
   391→    <div className="space-y-6">
   392→      {/* Hero */}
   393→      <div className="rounded-2xl bg-gradient-to-br from-rose-500/50 via-fuchsia-500/30 to-amber-500/40 p-[1px]">
   394→        <div className="rounded-2xl bg-[#0d0d14] p-5 sm:p-7">
   395→          <div className="flex flex-wrap items-end justify-between gap-4">
   396→            <div>
   397→              <h2 className="flex items-center gap-2.5 text-xl font-black tracking-tight text-white sm:text-2xl">
   398→                <Dices className="h-6 w-6 text-rose-400" aria-hidden />
   399→                AI picks for you
   400→              </h2>
   401→              <p className="mt-1 max-w-xl text-sm text-zinc-400">
   402→                {user
   403→                  ? 'Eight picks ranked by the Animaxia LLM against your taste profile and watch history.'
   404→                  : 'Eight LLM-ranked picks from the library. Sign in for picks personalized to your taste profile.'}
   405→              </p>
   406→            </div>
   407→            <Button
   408→              type="button"
   409→              size="lg"
   410→              onClick={fetchPicks}
   411→              disabled={loading}
   412→              className="bg-rose-600 font-bold text-white hover:bg-rose-500"
   413→            >
   414→              {loading ? <Loader2 className="animate-spin" aria-hidden /> : <Dices aria-hidden />}
   415→              {loading ? 'Thinking…' : picks ? 'Regenerate' : 'Get AI picks'}
   416→            </Button>
   417→          </div>
   418→        </div>
   419→      </div>
   420→
   421→      {upgrade && <UpgradeCard message={upgrade} />}
   422→      {error && !upgrade && <ErrorState message={error} onRetry={fetchPicks} />}
   423→      {loading && <LoadingCards count={8} />}
   424→
   425→      {!loading && picks && picks.length === 0 && (
   426→        <EmptyState
   427→          title="No picks right now"
   428→          message="The library is too sparse for the LLM to work with — add a few more titles and try again."
   429→        />
   430→      )}
   431→
   432→      {!loading && picks && picks.length > 0 && (
   433→        <>
   434→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   435→            {picks.map((rec) => (
   436→              <RecommendationCard key={rec.content.id} rec={rec} />
   437→            ))}
   438→          </div>
   439→          <p className="flex flex-wrap items-center gap-2 text-xs text-zinc-500">
   440→            Picks generated by Animaxia Intelligence <ProviderBadge provider={provider} />
   441→          </p>
   442→        </>
   443→      )}
   444→    </div>
   445→  );

   496→function SummarizeTab({ user }: { user: PlatformUser | null }) {
   497→  const [selected, setSelected] = useState<Suggestion | null>(null);
   498→  const [result, setResult] = useState<{ summary: string; provider: string } | null>(null);
   499→  const [loading, setLoading] = useState(false);
   500→  const [error, setError] = useState<string | null>(null);
   501→  const [upgrade, setUpgrade] = useState<string | null>(null);
   502→
   503→  async function summarize() {
   504→    if (!selected || loading) return;
   505→    setLoading(true);
   506→    setError(null);
   507→    setUpgrade(null);
   508→    try {
   509→      const res = await platform.aiSummarize(selected.id);
   510→      setResult({ summary: res.summary, provider: res.provider });
   511→    } catch (e) {
   512→      if (e instanceof PlatformError && e.upgrade) setUpgrade(e.message);
   513→      else setError(e instanceof Error ? e.message : 'Could not generate a summary');
   514→    } finally {
   515→      setLoading(false);
   516→    }
   517→  }
   518→
   519→  return (
   520→    <div className="mx-auto max-w-3xl space-y-6">
   521→      <Card className="border-white/10 bg-white/[0.03]">
   522→        <CardContent className="space-y-4 p-5 sm:p-6">
   523→          <ContentPicker
   524→            selected={selected}
   525→            onSelect={(s) => {
   526→              setSelected(s);
   527→              setResult(null);
   528→            }}

   755→function ChatTab({ user }: { user: PlatformUser | null }) {
   756→  const [selected, setSelected] = useState<Suggestion | null>(null);
   757→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   758→  const [input, setInput] = useState('');
   759→  const [busy, setBusy] = useState(false);
   760→  const [quotaError, setQuotaError] = useState<string | null>(null);
   761→  const scrollRef = useRef<HTMLDivElement>(null);
   762→
   763→  useEffect(() => {
   764→    const el = scrollRef.current;
   765→    if (el) el.scrollTop = el.scrollHeight;
   766→  }, [messages, busy]);
   767→
   768→  if (!user) {
   769→    return (
   770→      <div className="mx-auto max-w-3xl">
   771→        <div className="flex flex-col items-center justify-center gap-3 rounded-2xl border border-white/10 bg-white/[0.02] p-10 text-center">
   772→          <div className="rounded-full bg-rose-500/10 p-3 text-rose-300">
   773→            <Sparkles className="h-6 w-6" aria-hidden />
   774→          </div>
   775→          <div>
   776→            <p className="font-semibold text-zinc-100">Sign in to chat with Animaxia AI</p>
   777→            <p className="mt-1 max-w-md text-sm text-zinc-400">
   778→              Ask anything about any title in the library — characters, plots, watch order, hidden gems.
   779→            </p>
   780→          </div>
   781→          <Button
   782→            type="button"
   783→            className="bg-rose-600 font-bold text-white hover:bg-rose-500"
   784→            onClick={() => navigate('/auth')}
   785→          >
   786→            Sign in to chat
   787→          </Button>
   788→        </div>
   789→      </div>
   790→    );
   791→  }
   792→
   793→  async function send() {
   794→    const text = input.trim();
   795→    if (!text || !selected || busy) return;
   796→    setBusy(true);
   797→    setQuotaError(null);
   798→    setMessages((prev) => [...prev, { role: 'user', text }]);
   799→    setInput('');
   800→    try {
   801→      const res = await platform.aiChat(selected.id, text);
   802→      setMessages((prev) => [...prev, { role: 'ai', text: res.reply }]);
   803→    } catch (e) {
   804→      if (e instanceof PlatformError && e.upgrade) {
   805→        setQuotaError(e.message);
   806→      } else {
   807→        setMessages((prev) => [...prev, { role: 'ai', text: e instanceof Error ? e.message : 'Something went wrong.' }]);
   808→      }
   809→    } finally {
   810→      setBusy(false);
   811→    }
   812→  }
   813→
   814→  const onKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   815→    if (e.key === 'Enter' && !e.shiftKey) {
   816→      e.preventDefault();
   817→      void send();
   818→    }
   819→  };
   820→
   821→  return (
   822→    <div className="mx-auto max-w-3xl space-y-4">
   823→      <Card className="border-white/10 bg-white/[0.03]">
   824→        <CardContent className="p-5 sm:p-6">
   825→          <ContentPicker
   826→            selected={selected}
   827→            onSelect={(s) => {
   828→              setSelected(s);
   829→              setMessages([]);
   830→              setQuotaError(null);
   831→            }}
   832→            label="Pick a title to talk about"
   833→            inputId="chat-picker"
   834→          />
   835→        </CardContent>
   836→      </Card>
   837→
   838→      <div
   839→        ref={scrollRef}
   840→        className="custom-scrollbar max-h-96 space-y-3 overflow-y-auto rounded-2xl border border-white/10 bg-black/20 p-4"
   841→        aria-live="polite"
   842→        aria-label="Chat conversation"
   843→      >
   844→        {messages.length === 0 && !busy && !quotaError && (
   845→          <p className="py-8 text-center text-xs leading-relaxed text-zinc-500">
   846→            {selected
   847→              ? `Ask anything about ${selected.title} — the AI knows the title's details and your conversation history.`
   848→              : 'Pick a title above, then ask anything — characters, plot, where to start…'}
   849→          </p>
   850→        )}
   851→        {messages.map((m, i) => (
   852→          <div key={i} className={cn('flex gap-2.5', m.role === 'user' ? 'justify-end' : 'justify-start')}>
   853→            {m.role === 'ai' && (
   854→              <span className="mt-1 grid h-7 w-7 shrink-0 place-items-center rounded-full bg-gradient-to-br from-rose-500/30 to-fuchsia-500/30">
   855→                <Sparkles className="h-3.5 w-3.5 text-rose-300" aria-hidden />
   856→              </span>
   857→            )}
   858→            <div
   859→              className={cn(
   860→                'max-w-[85%] whitespace-pre-wrap break-words px-3.5 py-2 text-sm leading-relaxed',
   861→                m.role === 'user'
   862→                  ? 'rounded-2xl rounded-br-sm bg-rose-600/20 text-rose-50'
   863→                  : 'rounded-2xl rounded-bl-sm bg-white/5 text-zinc-200',
   864→              )}
   865→            >
   866→              {m.text}
   867→            </div>
   868→          </div>
   869→        ))}
   870→        {busy && (
   871→          <div className="flex justify-start">
   872→            <span className="mt-1 grid h-7 w-7 shrink-0 place-items-center rounded-full bg-gradient-to-br from-rose-500/30 to-fuchsia-500/30">
   873→              <Sparkles className="h-3.5 w-3.5 text-rose-300" aria-hidden />
   874→            </span>
   875→            <div className="rounded-2xl rounded-bl-sm bg-white/5 px-3.5 py-2">
   876→              <TypingDots />
   877→            </div>
   878→          </div>
   879→        )}
   880→      </div>
   881→
   882→      {quotaError && <UpgradeCard message={quotaError} />}
   883→
   884→      <div className="flex items-end gap-2">
   885→        <Textarea
   886→          value={input}
   887→          onChange={(e) => setInput(e.target.value)}
   888→          onKeyDown={onKeyDown}
   889→          placeholder={selected ? `Ask about ${selected.title}…` : 'Pick a title first…'}
   890→          aria-label="Message Animaxia AI"
   891→          rows={2}
   892→          maxLength={1000}
   893→          disabled={!selected}
   894→          className="min-h-11 flex-1 rounded-xl border-white/10 bg-white/[0.04] text-sm text-zinc-100 placeholder:text-zinc-500 focus-visible:border-rose-500/50 focus-visible:ring-rose-500/30 disabled:opacity-50"
   895→        />
   896→        <Button
   897→          type="button"
   898→          onClick={() => void send()}
   899→          disabled={busy || !selected || !input.trim()}
   900→          aria-label="Send message"
   901→          className="h-11 bg-rose-600 font-bold text-white hover:bg-rose-500"
   902→        >
   903→          {busy ? <Loader2 className="animate-spin" aria-hidden /> : <Send aria-hidden />}
   904→        </Button>
   905→      </div>
   906→      <div className="flex items-center justify-between gap-3">
   907→        <p className="text-[11px] text-zinc-600">Enter to send · Shift+Enter for a new line</p>
   908→        {messages.length > 0 && (
   909→          <button
   910→            type="button"
   911→            onClick={() => {
   912→              setMessages([]);
   913→              setQuotaError(null);
   914→            }}
   915→            className="text-[11px] font-semibold text-zinc-500 transition hover:text-zinc-300"
   916→          >
   917→            Clear conversation
   918→          </button>
   919→        )}
   920→      </div>
   921→    </div>
   922→  );
   923→}
