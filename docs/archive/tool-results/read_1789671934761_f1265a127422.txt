     1→/**
     2→ * ai/RagLabTab.tsx — "RAG Lab" segment of the mobile AI hub (NEW in Phase 6).
     3→ *
     4→ * A public playground over the Animaxia RAG surface (pgvector embeddings):
     5→ *
     6→ *   Recommend  POST /api/ai/rag/recommend { query, limit }  → vector-ranked
     7→ *              picks with reasons + score badges (works signed-out — anonymous

    16→ * Every mode handles the full state matrix: inline validation, loading
    17→ * (RowSkeleton / ActivityIndicator), ErrorState + Retry, 402 UpgradeCard,
    18→ * provider/degraded badges and the per-plan quota meter. Responses are
    19→ * treated defensively (?? [] / ?? '' guards) so a malformed payload can

    67→/* ------------------------------ shared helpers ----------------------------- */
    68→
    69→function isQuotaError(err: unknown): boolean {
    70→  return isApiError(err) && (err.upgrade || err.status === 402);
    71→}
    72→
    73→/** "42" → 42; "" / "abc" / "-1" → null. */
    74→function parseContentId(raw: string): number | null {
    75→  const n = Number.parseInt(raw.trim(), 10);
    76→  return Number.isFinite(n) && n > 0 ? n : null;
    77→}
    78→
    79→function quotaMessage(err: unknown): string {
    80→  return isApiError(err) ? err.message : 'Daily AI quota reached';
    81→}
    82→
    83→function ragError(err: unknown): string {
    84→  return isApiError(err) ? err.message : 'The RAG engine is unreachable';
    85→}

   196→  const { width } = useWindowDimensions();
   197→  const [query, setQuery] = useState('');
   198→  const [limit, setLimit] = useState<number>(6);
   199→  const [loading, setLoading] = useState(false);
   200→  const [error, setError] = useState<string | null>(null);
   201→  const [quotaBlocked, setQuotaBlocked] = useState<string | null>(null);
   202→  const [result, setResult] = useState<AiRagRecommendResponse | null>(null);
   203→
   204→  const run = useCallback(async () => {
   205→    const q = query.trim();
   206→    if (!q || loading) return;
   207→    setLoading(true);
   208→    setError(null);
   209→    setQuotaBlocked(null);
   210→    try {
   211→      setResult(await aiApi.aiRagRecommend(q, limit));
   212→    } catch (err) {
   213→      setResult(null);
   214→      if (isQuotaError(err)) setQuotaBlocked(quotaMessage(err));
   215→      else setError(ragError(err));

   217→      setLoading(false);
   218→    }
   219→  }, [query, limit, loading]);
   220→
   221→  // Two responsive poster columns inside the hub's horizontal padding.
   222→  const cardWidth = Math.min(190, Math.max(120, Math.floor((width - spacing.lg * 2 - spacing.md) / 2)));

   225→  return (
   226→    <View style={{ gap: spacing.md }}>
   227→      <Text style={styles.paneHint}>
   228→        Describe what you feel like — the vector engine ranks the whole library and explains every pick.
   229→      </Text>
   230→      <TextInput
   231→        style={styles.input}
   232→        value={query}
   233→        onChangeText={setQuery}
   234→        placeholder="e.g. dark sci-fi anime with mechs"
   235→        placeholderTextColor={colors.textDim}
   236→        returnKeyType="search"
   237→        autoCapitalize="none"
   238→        autoCorrect={false}
   239→        maxLength={200}
   240→        accessibilityLabel="Recommendation query"
   241→        onSubmitEditing={run}

   248→            <Pressable
   249→              key={n}
   250→              accessibilityRole="button"
   251→              accessibilityLabel={`${n} results`}
   252→              accessibilityState={{ selected: active }}
   253→              style={[styles.limitChip, active && styles.limitChipActive]}
   254→              onPress={() => setLimit(n)}
   255→            >
   256→              <Text style={[styles.limitText, active && styles.limitTextActive]}>{n}</Text>
   257→            </Pressable>
   258→          );
   259→        })}
   260→      </View>
   261→      <RunButton label="Find titles" onPress={run} disabled={!query.trim()} busy={loading} />
   262→
   263→      {loading ? <RowSkeleton /> : null}
   264→      {!loading && quotaBlocked ? <UpgradeCard message={quotaBlocked} /> : null}
   265→      {!loading && error ? <ErrorState message={error} onRetry={run} /> : null}
   266→
   267→      {!loading && result ? (
   268→        <View style={{ gap: spacing.md }}>
   269→          <View style={styles.metaRow}>
   270→            <ProviderBadge provider={result.provider} degraded={result.degraded} />
   271→            {result.personalized ? (
   272→              <View style={styles.personalizedChip}>
   273→                <Ionicons name="person" size={9} color={colors.roseSoft} />
   274→                <Text style={styles.personalizedText}>personalized ranking</Text>
   275→              </View>
   276→            ) : null}
   277→          </View>
   278→          {result.intro ? <Text style={styles.lead}>{result.intro}</Text> : null}
   279→          {picks.length === 0 ? (
   280→            <Text style={styles.emptyText}>Nothing matched — try a broader description.</Text>
   281→          ) : (
   282→            <View style={styles.picksWrap}>
   283→              {picks.map((pick, index) => (
   284→                <View key={`${pick.id}-${index}`} style={[styles.pickCell, { width: cardWidth }]}>
   285→                  <ContentCard item={pick} width={cardWidth} reason={pick.reason} />
   286→                  <View style={styles.scoreBadge}>
   287→                    <Ionicons name="sparkles" size={9} color={colors.roseSoft} />
   288→                    <Text style={styles.scoreText}>{Number(pick.score ?? 0).toFixed(2)}</Text>
   289→                  </View>
   290→                </View>
   291→              ))}
   292→            </View>
   293→          )}
   294→          {result.quota ? <QuotaMeter quota={result.quota} /> : null}
   295→        </View>
   296→      ) : null}
   297→    </View>
   298→  );
   299→}

   474→/* -------------------------------- Chat mode -------------------------------- */
   475→
   476→interface RagTurn {
   477→  role: 'user' | 'assistant';
   478→  content: string;
   479→  sources?: { id: number; title: string; score: number }[];
   480→  provider?: string;
   481→  degraded?: boolean;
   482→}
   483→
   484→function ChatPane(): React.JSX.Element {
   485→  const { user } = useAuth();
   486→  const [idDraft, setIdDraft] = useState('');
   487→  const [draft, setDraft] = useState('');
   488→  const [turns, setTurns] = useState<RagTurn[]>([]);
   489→  /** The content id the current conversation belongs to (switch → fresh chat). */
   490→  const [conversationId, setConversationId] = useState<number | null>(null);
   491→  const [sending, setSending] = useState(false);
   492→  const [error, setError] = useState<string | null>(null);
   493→  const [quotaBlocked, setQuotaBlocked] = useState<string | null>(null);
   494→  const [quota, setQuota] = useState<AiCloudQuota | null>(null);
   495→
   496→  const id = parseContentId(idDraft);
   497→
   498→  const history = useMemo<AiChatTurn[]>(
   499→    () => turns.slice(-8).map((t) => ({ role: t.role, content: t.content })),
   500→    [turns],
   501→  );
   502→
   503→  const send = useCallback(async () => {
   504→    const message = draft.trim();
   505→    if (!message || !id || sending) return;
   506→    setError(null);
   507→    setQuotaBlocked(null);
   508→    setSending(true);
   509→    setDraft('');
   510→
   511→    // New content id → the old conversation is irrelevant: start clean.
   512→    const switching = conversationId !== id;
   513→    const hist = switching ? [] : history;
   514→    setConversationId(id);
   515→    setTurns((prev) =>
   516→      switching ? [{ role: 'user', content: message }] : [...prev, { role: 'user', content: message }],
   517→    );
   518→
   519→    try {
   520→      const res = await aiApi.aiRagChat(id, message, hist);
   521→      setTurns((prev) => [
   522→        ...prev,
   523→        {
   524→          role: 'assistant',
   525→          content: res.reply,
   526→          sources: res.sources ?? [],
   527→          provider: res.provider,
   528→          degraded: res.degraded,
   529→        },
   530→      ]);
   531→      if (res.quota) setQuota(res.quota);
   532→    } catch (err) {
   533→      // Roll back the optimistic user bubble and give the draft back.
   534→      setTurns((prev) => prev.slice(0, -1));
   535→      setDraft(message);
   536→      if (isQuotaError(err)) setQuotaBlocked(quotaMessage(err));
   537→      else setError(ragError(err));
   538→    } finally {
   539→      setSending(false);
   540→    }
   541→  }, [draft, id, sending, conversationId, history]);
   542→
   543→  if (!user) {
   544→    return (
   545→      <SignInCard message="Sign in to chat with a title — every answer is grounded in the library metadata (RAG)." />
   546→    );
   547→  }
   548→
   549→  return (
   550→    <KeyboardAvoidingView behavior={Platform.OS === 'ios' ? 'padding' : undefined} style={{ gap: spacing.md }}>
   551→      <Text style={styles.paneHint}>
   552→        Pick a title and ask anything about it — every answer lists the sources it was grounded on.
   553→      </Text>
   554→      <IdInput value={idDraft} onChange={setIdDraft} />
   555→
   556→      {turns.length === 0 && !sending ? (
   557→        <View style={styles.emptyChat}>
   558→          <Ionicons name="chatbubbles-outline" size={20} color={colors.textDim} />
   559→          <Text style={styles.emptyChatText}>
   560→            No messages yet — your last 8 turns are sent back as context.
   561→          </Text>
   562→        </View>
   563→      ) : null}
   564→
   565→      <View style={styles.chatLog}>
   566→        {turns.map((turn, index) => (
   567→          <View
   568→            key={`${turn.role}-${index}`}
   569→            style={[styles.bubble, turn.role === 'user' ? styles.bubbleUser : styles.bubbleAi]}
   570→          >
   571→            <Text style={turn.role === 'user' ? styles.bubbleUserText : styles.bubbleAiText}>
   572→              {turn.content}
   573→            </Text>
   574→            {turn.role === 'assistant' && turn.sources && turn.sources.length > 0 ? (
   575→              <View style={styles.sourceRow}>
   576→                <Text style={styles.sourceLabel}>based on:</Text>
   577→                {turn.sources.slice(0, 4).map((s) => (
   578→                  <Pressable
   579→                    key={`${s.id}-${s.title}`}
   580→                    accessibilityRole="button"
   581→                    accessibilityLabel={`Open ${s.title}`}
   582→                    style={styles.sourceChip}
   583→                    onPress={() => navigateTo('Content', { id: s.id })}
   584→                  >
   585→                    <Text style={styles.sourceChipText} numberOfLines={1}>
   586→                      {s.title} · {Number(s.score ?? 0).toFixed(2)}
   587→                    </Text>
   588→                  </Pressable>
   589→                ))}
   590→              </View>
   591→            ) : null}
   592→            {turn.role === 'assistant' ? (
   593→              <ProviderBadge provider={turn.provider} degraded={turn.degraded} />
   594→            ) : null}
   595→          </View>
   596→        ))}
   597→        {sending ? (
   598→          <View style={[styles.bubble, styles.bubbleAi]}>
   599→            <Text style={styles.bubbleAiText}>reading the library…</Text>
   600→          </View>
   601→        ) : null}
   602→      </View>
   603→
   604→      {quotaBlocked ? <UpgradeCard message={quotaBlocked} /> : null}
   605→      {error ? <ErrorState message={error} compact onRetry={() => setError(null)} /> : null}
   606→      {quota ? <QuotaMeter quota={quota} /> : null}
   607→
   608→      <View style={styles.inputRow}>
   609→        <TextInput
   610→          style={styles.input}
   611→          value={draft}
   612→          onChangeText={setDraft}
   613→          placeholder={id ? 'Ask about the title…' : 'Enter a content id first…'}
   614→          placeholderTextColor={colors.textDim}
   615→          multiline
   616→          maxLength={2000}
   617→          accessibilityLabel="Message the RAG chat"
   618→        />
   619→        <Pressable
   620→          accessibilityRole="button"
   621→          accessibilityLabel="Send message"
   622→          style={[styles.sendButton, (!draft.trim() || !id || sending) && styles.sendButtonDisabled]}
   623→          disabled={!draft.trim() || !id || sending}
   624→          onPress={send}
   625→        >
   626→          <Ionicons name="arrow-up" size={17} color="#fff" />
   627→        </Pressable>
   628→      </View>
   629→    </KeyboardAvoidingView>
   630→  );
   631→}
