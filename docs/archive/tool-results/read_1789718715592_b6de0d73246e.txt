   117→function SocialHub({ route }: { route: AppRoute }) {
   118→  const { user } = useAuth();
   119→  const initialTab = route.query.get('tab');
   120→  const [tab, setTab] = useState<SocialTab>(
   121→    TABS.some((t) => t.value === initialTab) ? (initialTab as SocialTab) : 'feed',
   122→  );
   123→
   124→  /* ONE socket while the hub is mounted (anon ok): presence + live ticker. */
   125→  const [online, setOnline] = useState<number | null>(null);
   126→  const [liveEvents, setLiveEvents] = useState<ActivityEvent[]>([]);
   127→
   128→  useEffect(() => {
   129→    const socket = createSocialSocket();
   130→    socket.on('hello', (d: { online?: number }) => {
   131→      if (typeof d.online === 'number') setOnline(d.online);
   132→    });
   133→    socket.on('presence:online', (d: { online: number }) => setOnline(d.online));
   134→    socket.on('social:activity', (event: ActivityEvent) => {
   135→      if (!event || typeof event !== 'object' || event.id == null) return;
   136→      setLiveEvents((prev) => (prev.some((e) => e.id === event.id) ? prev : [event, ...prev].slice(0, 30)));
   137→    });

   139→      socket.removeAllListeners();
   140→      socket.disconnect();
   141→    };
   142→  }, []);

   150→              <span className="relative flex h-1.5 w-1.5">
   151→                <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-400 opacity-75" />
   152→                <span className="relative inline-flex h-1.5 w-1.5 rounded-full bg-emerald-400" />
   153→              </span>
   154→              {online} online

   158→            <Button
   159→              type="button"
   160→              size="sm"
   161→              className="bg-rose-600 font-bold text-white hover:bg-rose-500"

   205→function FeedTab({ liveEvents }: { liveEvents: ActivityEvent[] }) {
   206→  const { user } = useAuth();
   207→  const feed = useApi(`social-feed-v2-${user ? `u${user.id}` : 'anon'}`, () =>
   208→    user ? social.feed() : social.publicFeed(),
   209→  );
   210→
   211→  /* Cursor pagination — same accumulate-locally pattern as feed-view. */
   212→  const [moreItems, setMoreItems] = useState<ActivityEvent[]>([]);
   213→  const [moreCursor, setMoreCursor] = useState<string | null | undefined>(undefined);
   214→  const [loadingMore, setLoadingMore] = useState(false);
   215→  const [moreError, setMoreError] = useState<string | null>(null);
   216→
   217→  useEffect(() => {
   218→    setMoreItems([]);
   219→    setMoreCursor(undefined);
   220→    setMoreError(null);
   221→  }, [feed.data]);
   222→
   223→  const items = useMemo(() => [...(feed.data?.data ?? []), ...moreItems], [feed.data, moreItems]);
   224→  const activeCursor = moreCursor !== undefined ? moreCursor : (feed.data?.nextCursor ?? null);
   225→
   226→  async function loadMore() {
   227→    if (!activeCursor || loadingMore) return;
   228→    setLoadingMore(true);
   229→    setMoreError(null);
   230→    try {
   231→      const res = user ? await social.feed(activeCursor) : await social.publicFeed(activeCursor);
   232→      setMoreItems((prev) => [...prev, ...res.data]);
   233→      setMoreCursor(res.nextCursor);
   234→    } catch (e) {
   235→      setMoreError(e instanceof Error ? e.message : 'Could not load more activity');
   236→    } finally {
   237→      setLoadingMore(false);
   238→    }
   239→  }
   240→
   241→  if (feed.loading && !feed.data) return <FeedSkeleton />;
   242→  if (feed.error && !feed.data) return <ErrorState message={feed.error} onRetry={feed.refetch} />;
   243→
   244→  /* Live events on top, deduped against everything already listed. */
   245→  const listedIds = new Set(items.map((e) => e.id));
   246→  const fresh = liveEvents.filter((e) => !listedIds.has(e.id));
   247→
   248→  return (
   249→    <div className="mx-auto max-w-3xl space-y-4">
   250→      {fresh.length > 0 && (
   251→        <p className="flex items-center gap-1.5 text-[11px] font-bold uppercase tracking-widest text-rose-300">
   252→          <Radio className="h-3.5 w-3.5 animate-pulse" aria-hidden /> Live
   253→        </p>
   254→      )}
   255→      {fresh.map((event) => (
   256→        <ActivityCard key={`live-${event.id}`} event={event} live />
   257→      ))}
   258→      {fresh.length > 0 && items.length > 0 && <div className="h-px bg-white/10" role="separator" />}
   259→
   260→      {items.length === 0 && fresh.length === 0 ? (
   261→        <EmptyState
   262→          icon={<Clapperboard className="h-6 w-6" aria-hidden />}
   263→          title="Nothing here yet"
   264→          message="Reviews, follows and watch parties all show up here as they happen. Be the first — follow someone or start a party."
   265→          action={
   266→            <Button variant="outline" className="border-white/15 bg-white/5 hover:bg-white/10" onClick={() => navigate('/home')}>
   267→              Browse the library
   268→            </Button>
   269→          }
   270→        />
   271→      ) : (
   272→        <ol className="space-y-3" aria-label="Social activity">
   273→          {items.map((event, i) => (
   274→            <li key={`${event.type}-${event.id}-${i}`}>
   275→              <ActivityCard event={event} />
   276→            </li>
   277→          ))}
   278→        </ol>
   279→      )}
   280→
   281→      {activeCursor && (
   282→        <div className="space-y-2 pt-1">
   283→          {moreError && (
   284→            <p className="text-center text-xs text-rose-300" role="alert">
   285→              {moreError}
   286→            </p>
   287→          )}
   288→          <Button
   289→            type="button"
   290→            variant="outline"
   291→            onClick={() => void loadMore()}
   292→            disabled={loadingMore}
   293→            className="w-full border-white/15 bg-white/5 font-semibold hover:bg-white/10"
   294→          >
   295→            {loadingMore ? <Loader2 className="animate-spin" aria-hidden /> : <ChevronDown aria-hidden />}
   296→            {loadingMore ? 'Loading…' : 'Load more'}
   297→          </Button>
   298→        </div>
   299→      )}
   300→    </div>
   301→  );
   302→}

   360→function PeopleTab() {
   361→  const { user } = useAuth();
   362→  const [query, setQuery] = useState('');
   363→  const [debouncedQ, setDebouncedQ] = useState('');
   364→  const [profileId, setProfileId] = useState<number | null>(null);
   365→
   366→  /* Debounced search (400 ms). */
   367→  useEffect(() => {
   368→    const timer = setTimeout(() => setDebouncedQ(query.trim()), 400);
   369→    return () => clearTimeout(timer);
   370→  }, [query]);
   371→
   372→  const people = useApi(
   373→    `social-people-${debouncedQ.toLowerCase()}`,
   374→    () => social.people(debouncedQ || undefined, 12),
   375→    { enabled: !!user },
   376→  );
   377→  const following = useApi(`social-following-${user?.id ?? 'anon'}`, () => social.following(), { enabled: !!user });
   378→
   379→  /* Optimistic follow overrides shared by every list below. */
   380→  const [overrides, setOverrides] = useState<Record<number, boolean>>({});
   381→  const [busy, setBusy] = useState<Record<number, boolean>>({});
   382→
   383→  function isFollowing(u: SocialUserCard): boolean {
   384→    return overrides[u.id] ?? u.isfollowing ?? false;
   385→  }
   386→
   387→  async function toggleFollow(u: SocialUserCard) {
   388→    if (!user) {
   389→      toast('Sign in to follow people');
   390→      navigate('/auth');
   391→      return;
   392→    }
   393→    if (busy[u.id]) return;
   394→    const next = !isFollowing(u);
   395→    setOverrides((prev) => ({ ...prev, [u.id]: next }));
   396→    setBusy((prev) => ({ ...prev, [u.id]: true }));
   397→    try {
   398→      await (next ? social.follow(u.id) : social.unfollow(u.id));
   399→    } catch (err) {
   400→      setOverrides((prev) => ({ ...prev, [u.id]: !next }));
   401→      toast.error(err instanceof Error ? err.message : 'Could not update follows');
   402→    } finally {
   403→      setBusy((prev) => ({ ...prev, [u.id]: false }));
   404→    }
   405→  }
   406→
   407→  if (!user) {
   408→    return (
   409→      <SignInPrompt
   410→        title="Sign in to find people"
   411→        message="Follow reviewers with your taste — the People tab ranks the community so you can build your feed."
   412→      />
   413→    );
   414→  }
   415→
   416→  return (
   417→    <div className="space-y-8">
   418→      <div className="mx-auto max-w-2xl">
   419→        <div className="relative">
   420→          <Search className="pointer-events-none absolute left-3.5 top-1/2 h-4 w-4 -translate-y-1/2 text-zinc-500" aria-hidden />
   421→          <label htmlFor="people-search" className="sr-only">
   422→            Search people
   423→          </label>
   424→          <Input
   425→            id="people-search"
   426→            value={query}
   427→            onChange={(e) => setQuery(e.target.value)}
   428→            placeholder="Search people by username…"
   429→            autoComplete="off"
   430→            className="h-11 rounded-xl border-white/10 bg-white/[0.04] pl-10 text-sm text-zinc-100 placeholder:text-zinc-500 focus-visible:border-rose-500/50 focus-visible:ring-rose-500/30"
   431→          />
   432→        </div>
   433→      </div>
   434→
   435→      <Section
   436→        title={debouncedQ ? `Results for “${debouncedQ}”` : 'Suggested people'}
   437→        subtitle="Most active members of the community."
   438→      >
   439→        {people.loading && !people.data ? (
   440→          <PeopleGridSkeleton />
   441→        ) : people.error ? (
   442→          <ErrorState message={people.error} onRetry={people.refetch} />
   443→        ) : (people.data?.data ?? []).length === 0 ? (
   444→          <EmptyState
   445→            icon={<Users className="h-6 w-6" aria-hidden />}
   446→            title={debouncedQ ? 'No one matches that search' : 'No people yet'}
   447→            message={debouncedQ ? 'Try a different username.' : 'The community will show up here as people join.'}
   448→          />
   449→        ) : (
   450→          <ul className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3" aria-label="Suggested people">
   451→            {(people.data?.data ?? []).map((u) => (
   452→              <li key={u.id}>
   453→                <PersonCard person={u} following={isFollowing(u)} busy={!!busy[u.id]} onToggle={() => void toggleFollow(u)} onOpen={() => setProfileId(u.id)} />
   454→              </li>
   455→            ))}
   456→          </ul>
   457→        )}
   458→      </Section>
   459→
   460→      <Section title="You follow" subtitle="People whose activity fills your Feed tab.">
   461→        {following.loading && !following.data ? (
   462→          <div className="space-y-2" aria-busy="true">
   463→            <Skeleton className="h-16 w-full rounded-xl" />
   464→            <Skeleton className="h-16 w-full rounded-xl" />
   465→          </div>
   466→        ) : following.error ? (
   467→          <ErrorState message={following.error} onRetry={following.refetch} compact />
   468→        ) : (following.data?.data ?? []).length === 0 ? (
   469→          <EmptyState
   470→            icon={<UserPlus className="h-6 w-6" aria-hidden />}
   471→            title="You don't follow anyone yet"
   472→            message="Follow a few people above and their reviews, comments and parties will light up your feed."
   473→          />
   474→        ) : (
   475→          <ul className="grid gap-3 sm:grid-cols-2" aria-label="People you follow">
   476→            {(following.data?.data ?? []).map((u) => (
   477→              <li key={u.id}>
   478→                <PersonCard
   479→                  person={u}
   480→                  following={isFollowing(u)}
   481→                  busy={!!busy[u.id]}
   482→                  onToggle={() => void toggleFollow(u)}
   483→                  onOpen={() => setProfileId(u.id)}
   484→                />
   485→              </li>
   486→            ))}
   487→          </ul>
   488→        )}
   489→      </Section>
   490→
   491→      <ProfileDialog
   492→        userId={profileId}
   493→        onClose={() => setProfileId(null)}
   494→        isFollowing={(u) => isFollowing(u)}
   495→        busy={(id) => !!busy[id]}
   496→        onToggle={(u) => void toggleFollow(u)}
   497→      />
   498→    </div>
   499→  );
   500→}

   507→  onOpen,
   508→}: {
   509→  person: SocialUserCard;
   510→  following: boolean;
   511→  busy: boolean;
   512→  onToggle: () => void;
   513→  onOpen: () => void;
   514→}) {
   515→  return (
   516→    <Card className="group border-white/10 bg-white/[0.03] transition-colors hover:border-white/20">
   517→      <CardContent className="flex items-center gap-3 p-4">
   518→        <button
   519→          type="button"
   520→          onClick={onOpen}
   521→          aria-label={`Open ${person.username}'s profile`}
   522→          className="flex min-w-0 flex-1 items-center gap-3 rounded-lg text-left focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-rose-500/60"
   523→        >
   524→          <UserAvatar username={person.username} avatar={person.avatar} className="h-11 w-11 shrink-0" />
   525→          <span className="min-w-0 flex-1">

   776→function ReviewsTab() {
   777→  const { user } = useAuth();
   778→  const reviews = useApi('social-recent-reviews', () => social.recentReviews(20));
   779→
   780→  /* Optimistic like toggles keyed by review id. */
   781→  const [likes, setLikes] = useState<Record<number, { likes: number; liked: boolean }>>({});
   782→  const [busy, setBusy] = useState<Record<number, boolean>>({});
   783→
   784→  async function toggleLike(id: number) {
   785→    if (!user) {
   786→      toast('Sign in to like reviews');
   787→      navigate('/auth');
   788→      return;
   789→    }
   790→    if (busy[id]) return;
   791→    const current =
   792→      likes[id] ??
   793→      (() => {
   794→        const r = reviews.data?.data.find((x) => x.id === id);
   795→        return r ? { likes: r.likes, liked: r.liked } : null;
   796→      })();
   797→    if (!current) return;
   798→    const next = { likes: current.liked ? current.likes - 1 : current.likes + 1, liked: !current.liked };
   799→    setLikes((prev) => ({ ...prev, [id]: next }));
   800→    setBusy((prev) => ({ ...prev, [id]: true }));
   801→    try {
   802→      const res = await social.likeReview(id);
   803→      setLikes((prev) => ({ ...prev, [id]: { likes: res.likes, liked: res.liked } }));
   804→    } catch (err) {
   805→      setLikes((prev) => ({ ...prev, [id]: current }));
   806→      toast.error(err instanceof Error ? err.message : 'Could not like the review');
   807→    } finally {
   808→      setBusy((prev) => ({ ...prev, [id]: false }));
   809→    }
   810→  }
   811→
   812→  if (reviews.loading && !reviews.data) {
   813→    return (
   814→      <div className="mx-auto max-w-3xl space-y-3" aria-busy="true" aria-label="Loading reviews">
   815→        {Array.from({ length: 4 }).map((_, i) => (
   816→          <Skeleton key={i} className="h-36 w-full rounded-xl" />
   817→        ))}
   818→      </div>
   819→    );
   820→  }
   821→  if (reviews.error && !reviews.data) return <ErrorState message={reviews.error} onRetry={reviews.refetch} />;
   822→
   823→  const list = reviews.data?.data ?? [];
   824→
   825→  return (
   826→    <div className="mx-auto max-w-3xl space-y-4">
   827→      {list.length === 0 ? (
   828→        <EmptyState
   829→          icon={<Star className="h-6 w-6" aria-hidden />}
   830→          title="No reviews yet"
   831→          message="Open any title and share your verdict — the community's freshest reviews show up here."
   832→          action={
   833→            <Button variant="outline" className="border-white/15 bg-white/5 hover:bg-white/10" onClick={() => navigate('/home')}>
   834→              Find something to review
   835→            </Button>
   836→          }
   837→        />
   838→      ) : (
   839→        <ol className="space-y-3" aria-label="Recent community reviews">
   840→          {list.map((r) => {
   841→            const likeState = likes[r.id] ?? { likes: r.likes, liked: r.liked };
   842→            return (
   843→              <li key={r.id}>
   844→                <article className="rounded-xl border border-white/10 bg-white/[0.03] p-4 transition-colors hover:border-white/20">
   845→                  <div className="flex items-center gap-2.5">
   846→                    <UserAvatar username={r.user.username} avatar={r.user.avatar} className="h-8 w-8" />
   847→                    <div className="min-w-0 flex-1">
   848→                      <p className="truncate text-sm font-bold text-zinc-100">{r.user.username}</p>
   849→                      <p className="text-[11px] text-zinc-500">{timeAgo(r.createdat)}</p>
   850→                    </div>
   851→                    <RatingPill rating={r.rating} />
   852→                  </div>
   853→                  {r.title ? <p className="mt-2.5 text-sm font-semibold text-zinc-100">{r.title}</p> : null}
   854→                  <p className="mt-1 line-clamp-3 break-words text-sm text-zinc-400">{r.body}</p>
   855→                  <div className="mt-2.5">
   856→                    <ContentMiniCard content={r.content} compact />
   857→                  </div>
   858→                  <div className="mt-2.5 flex items-center justify-between gap-3">
   859→                    <Button
   860→                      type="button"
   861→                      variant="ghost"
   862→                      size="sm"
   863→                      onClick={() => void toggleLike(r.id)}
   864→                      disabled={!!busy[r.id]}
   865→                      aria-pressed={likeState.liked}
   866→                      className={cn(
   867→                        'gap-1.5 text-zinc-400 hover:bg-white/5 hover:text-white',
   868→                        likeState.liked && 'text-rose-300 hover:text-rose-200',
   869→                      )}
   870→                    >
   871→                      {busy[r.id] ? (
   872→                        <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   873→                      ) : (
   874→                        <Star className={cn('h-4 w-4', likeState.liked && 'fill-rose-400')} aria-hidden />
   875→                      )}
   876→                      {likeState.likes} {likeState.likes === 1 ? 'like' : 'likes'}
   877→                    </Button>
   878→                  </div>
   879→                </article>
   880→              </li>
   881→            );
   882→          })}
   883→        </ol>

  1083→function StartPartyDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (open: boolean) => void }) {
  1084→  const { user } = useAuth();
  1085→  const [query, setQuery] = useState('');
  1086→  const [debouncedQ, setDebouncedQ] = useState('');
  1087→  const [results, setResults] = useState<DbContent[]>([]);
  1088→  const [searching, setSearching] = useState(false);
  1089→  const [creatingId, setCreatingId] = useState<number | null>(null);
  1090→
  1091→  useEffect(() => {
  1092→    const timer = setTimeout(() => setDebouncedQ(query.trim()), 400);
  1093→    return () => clearTimeout(timer);
  1094→  }, [query]);
  1095→
  1096→  useEffect(() => {
  1097→    if (!open) return;
  1098→    let cancelled = false;
  1099→    if (debouncedQ.length === 0) {
  1100→      setResults([]);
  1101→      return;
  1102→    }
  1103→    setSearching(true);
  1104→    api
  1105→      .all({ search: debouncedQ, limit: 12 })
  1106→      .then((res) => {
  1107→        if (!cancelled) setResults(res.data);
  1108→      })
  1109→      .catch(() => {
  1110→        if (!cancelled) setResults([]);
  1111→      })
  1112→      .finally(() => {
  1113→        if (!cancelled) setSearching(false);
  1114→      });
  1115→    return () => {
  1116→      cancelled = true;
  1117→    };
  1118→  }, [debouncedQ, open]);
  1119→
  1120→  async function createParty(content: DbContent) {
  1121→    if (!user || creatingId != null) return;
  1122→    setCreatingId(content.id);
  1123→    try {
  1124→      const res = await social.createParty({ contentId: content.id });
  1125→      toast.success('Watch party started — share your invite code');
  1126→      onOpenChange(false);
  1127→      navigate(`/social/party/${res.party.id}`);
  1128→    } catch (err) {
  1129→      toast.error(err instanceof Error ? err.message : 'Could not start the party');
  1130→    } finally {
  1131→      setCreatingId(null);
  1132→    }
  1133→  }
  1134→
  1135→  return (
  1136→    <Dialog open={open} onOpenChange={onOpenChange}>
  1137→      <DialogContent className="max-h-[85vh] max-w-xl overflow-y-auto border-white/10 bg-[#12121c] text-zinc-100 custom-scrollbar">
  1138→        <DialogHeader>
  1139→          <DialogTitle className="text-lg font-black text-white">Start a watch party</DialogTitle>
  1140→          <DialogDescription>
  1141→            Pick a library title — you become the host and your playback syncs everyone in the room.
  1142→          </DialogDescription>
  1143→        </DialogHeader>
  1144→
  1145→        <div className="relative">
  1146→          <Search className="pointer-events-none absolute left-3.5 top-1/2 h-4 w-4 -translate-y-1/2 text-zinc-500" aria-hidden />
  1147→          <label htmlFor="party-content-search" className="sr-only">
  1148→            Search titles
  1149→          </label>
  1150→          <Input
  1151→            id="party-content-search"
  1152→            value={query}
  1153→            onChange={(e) => setQuery(e.target.value)}
  1154→            placeholder="Search the library…"
  1155→            autoComplete="off"
  1156→            className="h-11 rounded-xl border-white/10 bg-white/[0.04] pl-10 text-sm text-zinc-100 placeholder:text-zinc-500 focus-visible:border-rose-500/50 focus-visible:ring-rose-500/30"
  1157→          />
  1158→        </div>
  1159→
  1160→        <div className="min-h-[120px]">
  1161→          {searching ? (
  1162→            <div className="grid grid-cols-2 gap-2 sm:grid-cols-3" aria-busy="true" aria-label="Searching titles">
  1163→              {Array.from({ length: 6 }).map((_, i) => (
  1164→                <Skeleton key={i} className="h-[104px] rounded-xl" />
  1165→              ))}
  1166→            </div>
  1167→          ) : debouncedQ.length === 0 ? (
  1168→            <p className="py-8 text-center text-sm text-zinc-500">Type at least one character to search the library.</p>
  1169→          ) : results.length === 0 ? (
  1170→            <p className="py-8 text-center text-sm text-zinc-500">No titles match that search.</p>
  1171→          ) : (
  1172→            <ul className="grid grid-cols-2 gap-2 sm:grid-cols-3" aria-label="Search results">
  1173→              {results.map((c) => (
  1174→                <li key={c.id}>
  1175→                  <button
  1176→                    type="button"
  1177→                    onClick={() => void createParty(c)}
  1178→                    disabled={creatingId != null}
  1179→                    className="group flex w-full flex-col gap-2 rounded-xl border border-white/10 bg-white/[0.03] p-2 text-left transition hover:border-rose-500/40 hover:bg-white/[0.06] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-rose-500/60 disabled:opacity-60"
  1180→                  >
  1181→                    <span className="aspect-[2/3] w-full overflow-hidden rounded-lg border border-white/10 bg-white/[0.03]">
  1182→                      {c.thumbnail ? (
  1183→                        <img src={c.thumbnail} alt="" loading="lazy" className="h-full w-full object-cover" />
  1184→                      ) : (
  1185→                        <BrandPlaceholder title={c.title} />

  1187→                    </span>

  1194→                </li>
