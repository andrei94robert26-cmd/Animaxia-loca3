   205→   205→function FeedTab({ liveEvents }: { liveEvents: ActivityEvent[] }) {
   206→   206→  const { user } = useAuth();
   207→   207→  const feed = useApi(`social-feed-v2-${user ? `u${user.id}` : 'anon'}`, () =>
   208→   208→    user ? social.feed() : social.publicFeed(),
   209→   209→  );
   210→   210→
   211→   211→  /* Cursor pagination — same accumulate-locally pattern as feed-view. */
   212→   212→  const [moreItems, setMoreItems] = useState<ActivityEvent[]>([]);
   213→   213→  const [moreCursor, setMoreCursor] = useState<string | null | undefined>(undefined);
   214→   214→  const [loadingMore, setLoadingMore] = useState(false);
   215→   215→  const [moreError, setMoreError] = useState<string | null>(null);
   216→   216→
   217→   217→  useEffect(() => {
   218→   218→    setMoreItems([]);
   219→   219→    setMoreCursor(undefined);
   220→   220→    setMoreError(null);
   221→   221→  }, [feed.data]);
   222→   222→
   223→   223→  const items = useMemo(() => [...(feed.data?.data ?? []), ...moreItems], [feed.data, moreItems]);
   224→   224→  const activeCursor = moreCursor !== undefined ? moreCursor : (feed.data?.nextCursor ?? null);
   225→   225→
   226→   226→  async function loadMore() {
   227→   227→    if (!activeCursor || loadingMore) return;
   228→   228→    setLoadingMore(true);
   229→   229→    setMoreError(null);
   230→   230→    try {
   231→   231→      const res = user ? await social.feed(activeCursor) : await social.publicFeed(activeCursor);
   232→   232→      setMoreItems((prev) => [...prev, ...res.data]);
   233→   233→      setMoreCursor(res.nextCursor);
   234→   234→    } catch (e) {
   235→   235→      setMoreError(e instanceof Error ? e.message : 'Could not load more activity');
   236→   236→    } finally {
   237→   237→      setLoadingMore(false);
   238→   238→    }
   239→   239→  }
   240→   240→
   241→   241→  if (feed.loading && !feed.data) return <FeedSkeleton />;
   242→   242→  if (feed.error && !feed.data) return <ErrorState message={feed.error} onRetry={feed.refetch} />;
   243→   243→
   244→   244→  /* Live events on top, deduped against everything already listed. */
   245→   245→  const listedIds = new Set(items.map((e) => e.id));
   246→   246→  const fresh = liveEvents.filter((e) => !listedIds.has(e.id));
   247→   247→
   248→   248→  return (
   249→   249→    <div className="mx-auto max-w-3xl space-y-4">
   250→   250→      {fresh.length > 0 && (
   251→   251→        <p className="flex items-center gap-1.5 text-[11px] font-bold uppercase tracking-widest text-rose-300">
   252→   252→          <Radio className="h-3.5 w-3.5 animate-pulse" aria-hidden /> Live
   253→   253→        </p>
   254→   254→      )}
   255→   255→      {fresh.map((event) => (
   256→   256→        <ActivityCard key={`live-${event.id}`} event={event} live />
   257→   257→      ))}
   258→   258→      {fresh.length > 0 && items.length > 0 && <div className="h-px bg-white/10" role="separator" />}
   259→   259→
   260→   260→      {items.length === 0 && fresh.length === 0 ? (
   261→   261→        <EmptyState
   262→   262→          icon={<Clapperboard className="h-6 w-6" aria-hidden />}
   263→   263→          title="Nothing here yet"
   264→   264→          message="Reviews, follows and watch parties all show up here as they happen. Be the first — follow someone or start a party."
   265→   265→          action={
   266→   266→            <Button variant="outline" className="border-white/15 bg-white/5 hover:bg-white/10" onClick={() => navigate('/home')}>
   267→   267→              Browse the library
   268→   268→            </Button>
   269→   269→          }
   270→   270→        />
   271→   271→      ) : (
   272→   272→        <ol className="space-y-3" aria-label="Social activity">
   273→   273→          {items.map((event, i) => (
   274→   274→            <li key={`${event.type}-${event.id}-${i}`}>
   275→   275→              <ActivityCard event={event} />
   276→   276→            </li>
   277→   277→          ))}
   278→   278→        </ol>
   279→   279→      )}
   280→   280→
   281→   281→      {activeCursor && (
   282→   282→        <div className="space-y-2 pt-1">
   283→   283→          {moreError && (
   284→   284→            <p className="text-center text-xs text-rose-300" role="alert">
   285→   285→              {moreError}
   286→   286→            </p>
   287→   287→          )}
   288→   288→          <Button
   289→   289→            type="button"
   290→   290→            variant="outline"
   291→   291→            onClick={() => void loadMore()}
   292→   292→            disabled={loadingMore}
   293→   293→            className="w-full border-white/15 bg-white/5 font-semibold hover:bg-white/10"
   294→   294→          >
   295→   295→            {loadingMore ? <Loader2 className="animate-spin" aria-hidden /> : <ChevronDown aria-hidden />}
   296→   296→            {loadingMore ? 'Loading…' : 'Load more'}
   297→   297→          </Button>
   298→   298→        </div>
   299→   299→      )}
   300→   300→    </div>
   301→   301→  );
   302→   302→}

   360→   360→function PeopleTab() {

   362→   362→  const [query, setQuery] = useState('');
   363→   363→  const [debouncedQ, setDebouncedQ] = useState('');
   364→   364→  const [profileId, setProfileId] = useState<number | null>(null);
   365→   365→
   366→   366→  /* Debounced search (400 ms). */
   367→   367→  useEffect(() => {
   368→   368→    const timer = setTimeout(() => setDebouncedQ(query.trim()), 400);
   369→   369→    return () => clearTimeout(timer);
   370→   370→  }, [query]);
   371→   371→
   372→   372→  const people = useApi(
   373→   373→    `social-people-${debouncedQ.toLowerCase()}`,
   374→   374→    () => social.people(debouncedQ || undefined, 12),
   375→   375→    { enabled: !!user },
   376→   376→  );
   377→   377→  const following = useApi(`social-following-${user?.id ?? 'anon'}`, () => social.following(), { enabled: !!user });
   378→   378→
   379→   379→  /* Optimistic follow overrides shared by every list below. */
   380→   380→  const [overrides, setOverrides] = useState<Record<number, boolean>>({});
   381→   381→  const [busy, setBusy] = useState<Record<number, boolean>>({});
   382→   382→
   383→   383→  function isFollowing(u: SocialUserCard): boolean {
   384→   384→    return overrides[u.id] ?? u.isfollowing ?? false;
   385→   385→  }
   386→   386→
   387→   387→  async function toggleFollow(u: SocialUserCard) {
   388→   388→    if (!user) {
   389→   389→      toast('Sign in to follow people');
   390→   390→      navigate('/auth');
   391→   391→      return;
   392→   392→    }
   393→   393→    if (busy[u.id]) return;
   394→   394→    const next = !isFollowing(u);
   395→   395→    setOverrides((prev) => ({ ...prev, [u.id]: next }));
   396→   396→    setBusy((prev) => ({ ...prev, [u.id]: true }));
   397→   397→    try {
   398→   398→      await (next ? social.follow(u.id) : social.unfollow(u.id));
   399→   399→    } catch (err) {
   400→   400→      setOverrides((prev) => ({ ...prev, [u.id]: !next }));
   401→   401→      toast.error(err instanceof Error ? err.message : 'Could not update follows');
   402→   402→    } finally {
   403→   403→      setBusy((prev) => ({ ...prev, [u.id]: false }));
   404→   404→    }
   405→   405→  }
   406→   406→
   407→   407→  if (!user) {
   408→   408→    return (
   409→   409→      <SignInPrompt
   410→   410→        title="Sign in to find people"
   411→   411→        message="Follow reviewers with your taste — the People tab ranks the community so you can build your feed."
   412→   412→      />
   413→   413→    );
   414→   414→  }
   415→   415→
   416→   416→  return (
   417→   417→    <div className="space-y-8">
   418→   418→      <div className="mx-auto max-w-2xl">
   419→   419→        <div className="relative">
   420→   420→          <Search className="pointer-events-none absolute left-3.5 top-1/2 h-4 w-4 -translate-y-1/2 text-zinc-500" aria-hidden />
   421→   421→          <label htmlFor="people-search" className="sr-only">
   422→   422→            Search people
   423→   423→          </label>
   424→   424→          <Input
   425→   425→            id="people-search"
   426→   426→            value={query}
   427→   427→            onChange={(e) => setQuery(e.target.value)}
   428→   428→            placeholder="Search people by username…"
   429→   429→            autoComplete="off"
   430→   430→            className="h-11 rounded-xl border-white/10 bg-white/[0.04] pl-10 text-sm text-zinc-100 placeholder:text-zinc-500 focus-visible:border-rose-500/50 focus-visible:ring-rose-500/30"
   431→   431→          />
   432→   432→        </div>
   433→   433→      </div>
   434→   434→
   435→   435→      <Section
   436→   436→        title={debouncedQ ? `Results for “${debouncedQ}”` : 'Suggested people'}
   437→   437→        subtitle="Most active members of the community."
   438→   438→      >
   439→   439→        {people.loading && !people.data ? (
   440→   440→          <PeopleGridSkeleton />
   441→   441→        ) : people.error ? (
   442→   442→          <ErrorState message={people.error} onRetry={people.refetch} />
   443→   443→        ) : (people.data?.data ?? []).length === 0 ? (
   444→   444→          <EmptyState
   445→   445→            icon={<Users className="h-6 w-6" aria-hidden />}
   446→   446→            title={debouncedQ ? 'No one matches that search' : 'No people yet'}
   447→   447→            message={debouncedQ ? 'Try a different username.' : 'The community will show up here as people join.'}
   448→   448→          />
   449→   449→        ) : (
   450→   450→          <ul className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3" aria-label="Suggested people">
   451→   451→            {(people.data?.data ?? []).map((u) => (
   452→   452→              <li key={u.id}>
   453→   453→                <PersonCard person={u} following={isFollowing(u)} busy={!!busy[u.id]} onToggle={() => void toggleFollow(u)} onOpen={() => setProfileId(u.id)} />
   454→   454→              </li>
   455→   455→            ))}
   456→   456→          </ul>
   457→   457→        )}
   458→   458→      </Section>
   459→   459→
   460→   460→      <Section title="You follow" subtitle="People whose activity fills your Feed tab.">
   461→   461→        {following.loading && !following.data ? (
   462→   462→          <div className="space-y-2" aria-busy="true">
   463→   463→            <Skeleton className="h-16 w-full rounded-xl" />
   464→   464→            <Skeleton className="h-16 w-full rounded-xl" />
   465→   465→          </div>
   466→   466→        ) : following.error ? (
   467→   467→          <ErrorState message={following.error} onRetry={following.refetch} compact />
   468→   468→        ) : (following.data?.data ?? []).length === 0 ? (
   469→   469→          <EmptyState
   470→   470→            icon={<UserPlus className="h-6 w-6" aria-hidden />}
   471→   471→            title="You don't follow anyone yet"
   472→   472→            message="Follow a few people above and their reviews, comments and parties will light up your feed."
   473→   473→          />
   474→   474→        ) : (
   475→   475→          <ul className="grid gap-3 sm:grid-cols-2" aria-label="People you follow">
   476→   476→            {(following.data?.data ?? []).map((u) => (
   477→   477→              <li key={u.id}>
   478→   478→                <PersonCard
   479→   479→                  person={u}
   480→   480→                  following={isFollowing(u)}
   481→   481→                  busy={!!busy[u.id]}
