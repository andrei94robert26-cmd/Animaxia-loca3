    19→function platformUrl(path: string, params?: Record<string, string | number | boolean | undefined | null>): string {
    20→  const search = new URLSearchParams();
    21→  search.set('XTransformPort', String(BACKEND_PORT));
    22→  if (params) {
    23→    for (const [key, value] of Object.entries(params)) {
    24→      if (value !== undefined && value !== null && value !== '') search.set(key, String(value));
    25→    }
    26→  }
    27→  return `${path}?${search.toString()}`;
    28→}

    45→const ACCESS_KEY = 'animaxia.accessToken';
    46→const REFRESH_KEY = 'animaxia.refreshToken';

    48→export const tokenStore = {
    49→  get access(): string | null {
    50→    try {
    51→      return window.localStorage.getItem(ACCESS_KEY);
    52→    } catch {
    53→      return null;
    54→    }
    55→  },
    56→  get refresh(): string | null {
    57→    try {
    58→      return window.localStorage.getItem(REFRESH_KEY);
    59→    } catch {
    60→      return null;
    61→    }
    62→  },
    63→  set(access: string, refresh: string) {
    64→    try {
    65→      window.localStorage.setItem(ACCESS_KEY, access);
    66→      window.localStorage.setItem(REFRESH_KEY, refresh);
    67→    } catch {
    68→      /* private mode — session will not persist */
    69→    }
    70→  },
    71→  clear() {
    72→    try {
    73→      window.localStorage.removeItem(ACCESS_KEY);
    74→      window.localStorage.removeItem(REFRESH_KEY);
    75→    } catch {
    76→      /* noop */
    77→    }
    78→  },
    79→};

    83→async function rawRequest<T>(path: string, init?: RequestInit): Promise<T> {
    84→  let res: Response;
    85→  try {
    86→    res = await fetch(path, {
    87→      ...init,
    88→      headers: {
    89→        'Content-Type': 'application/json',
    90→        ...(init?.headers as Record<string, string> | undefined),
    91→      },
    92→    });
    93→  } catch {
    94→    throw new PlatformError('Cannot reach the Animaxia backend — it may be restarting', 0);
    95→  }
    96→  let body: unknown = null;
    97→  try {
    98→    body = await res.json();
    99→  } catch {
   100→    body = null;
   101→  }
   102→  if (!res.ok) {
   103→    const b = (body as { error?: string; upgrade?: boolean; code?: string } | null) ?? {};
   104→    throw new PlatformError(b.error || `Request failed with status ${res.status}`, res.status, b.upgrade, b.code);
   105→  }
   106→  return body as T;
   107→}

   109→/** Authenticated request with one automatic token-refresh retry on 401. */
   110→export async function authRequest<T>(path: string, init?: RequestInit): Promise<T> {
   111→  const access = tokenStore.access;
   112→  const doFetch = () =>
   113→    rawRequest<T>(path, {
   114→      ...init,
   115→      headers: { ...(init?.headers as Record<string, string> | undefined), ...(access ? { Authorization: `Bearer ${access}` } : {}) },
   116→    });
   117→  try {
   118→    return await doFetch();
   119→  } catch (err) {
   120→    if (err instanceof PlatformError && err.status === 401 && tokenStore.refresh) {
   121→      const refreshed = await tryRefresh();
   122→      if (refreshed) {
   123→        return authRequest<T>(path, init); // retry with the new token
   124→      }
   125→    }
   126→    throw err;
   127→  }
   128→}
   129→
   130→/** Attempt a refresh-token rotation. Returns true on success. */
   131→export async function tryRefresh(): Promise<boolean> {
   132→  const refresh = tokenStore.refresh;
   133→  if (!refresh) return false;
   134→  try {
   135→    const res = await rawRequest<{ ok: true; accessToken: string; refreshToken: string }>('/api/auth/refresh?XTransformPort=4000', {
   136→      method: 'POST',
   137→      body: JSON.stringify({ refreshToken: refresh }),
   138→    });
   139→    tokenStore.set(res.accessToken, res.refreshToken);
   140→    return true;
   141→  } catch {
   142→    tokenStore.clear();
   143→    return false;
   144→  }
   145→}

   147→/** Anonymous request (no Bearer header). */
   148→export async function anonRequest<T>(path: string, init?: RequestInit): Promise<T> {
   149→  return rawRequest<T>(path, init);
