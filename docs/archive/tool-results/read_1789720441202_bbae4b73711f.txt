   211→/**
   212→ * Library content picker: debounced (300 ms) autocomplete dropdown with
   213→ * thumbnails, keyboard (Escape) and click-outside dismissal, and a selected
   214→ * chip. Reused by the Summarize and Chat tabs.
   215→ */
   216→function ContentPicker({
   217→  selected,
   218→  onSelect,
   219→  label,
   220→  inputId,
   221→}: {
   222→  selected: Suggestion | null;
   223→  onSelect: (s: Suggestion | null) => void;
   224→  label: string;
   225→  inputId: string;
   226→}) {
   227→  const [query, setQuery] = useState('');
   228→  const [open, setOpen] = useState(false);
   229→  const [suggestions, setSuggestions] = useState<Suggestion[]>([]);
   230→  const [loading, setLoading] = useState(false);
   231→  const rootRef = useRef<HTMLDivElement>(null);
   232→
   233→  /* Debounced autocomplete (300 ms while typing; instant clear otherwise —
   234→     state updates happen inside the async timer to avoid sync setState). */
   235→  useEffect(() => {
   236→    const q = query.trim();
   237→    const timer = setTimeout(
   238→      () => {
   239→        if (!open || q.length === 0) {
   240→          setSuggestions([]);
   241→          return;
   242→        }
   243→        setLoading(true);
   244→        platform
   245→          .autocomplete(q)
   246→          .then((res) => setSuggestions(res.suggestions))
   247→          .catch(() => setSuggestions([]))
   248→          .finally(() => setLoading(false));
   249→      },
   250→      open && q.length > 0 ? 300 : 0,
   251→    );
   252→    return () => clearTimeout(timer);
   253→  }, [query, open]);
   254→
   255→  /* Click-outside closes the dropdown. */
   256→  useEffect(() => {
   257→    if (!open) return;
   258→    const onDown = (e: MouseEvent) => {
   259→      if (rootRef.current && !rootRef.current.contains(e.target as Node)) setOpen(false);
   260→    };
   261→    document.addEventListener('mousedown', onDown);
   262→    return () => document.removeEventListener('mousedown', onDown);
   263→  }, [open]);
   264→
   265→  return (
   266→    <div className="space-y-2" ref={rootRef}>
   267→      <label htmlFor={inputId} className="text-sm font-semibold text-zinc-300">
   268→        {label}
   269→      </label>
   270→      <div className="relative">
   271→        <Search
   272→          className="pointer-events-none absolute left-3.5 top-1/2 h-4 w-4 -translate-y-1/2 text-zinc-500"
   273→          aria-hidden
   274→        />
   275→        <Input
   276→          id={inputId}
   277→          type="text"
   278→          role="combobox"
   279→          aria-expanded={open}
   280→          aria-controls={`${inputId}-listbox`}
   281→          aria-autocomplete="list"
   282→          autoComplete="off"
   283→          value={query}
   284→          placeholder="Search the library…"
   285→          onChange={(e) => {
   286→            setQuery(e.target.value);
   287→            setOpen(true);
   288→          }}
   289→          onFocus={() => setOpen(true)}
   290→          onKeyDown={(e) => {
   291→            if (e.key === 'Escape') setOpen(false);
   292→          }}
   293→          className="h-11 rounded-xl border-white/10 bg-white/[0.04] pl-10 text-sm text-zinc-100 placeholder:text-zinc-500 focus-visible:border-rose-500/50 focus-visible:ring-rose-500/30"
   294→        />
   295→        {open && query.trim().length > 0 && (
   296→          <div
   297→            id={`${inputId}-listbox`}
   298→            role="listbox"
   299→            aria-label="Library suggestions"
   300→            className="absolute z-30 mt-2 max-h-72 w-full overflow-y-auto rounded-xl border border-white/10 bg-[#12121c] p-2 shadow-2xl shadow-black/70 custom-scrollbar"
   301→          >
   302→            {loading ? (
   303→              <p className="px-2 py-3 text-xs text-zinc-500">Searching the library…</p>
   304→            ) : suggestions.length === 0 ? (
   305→              <p className="px-2 py-3 text-xs text-zinc-500">No titles match that search.</p>
   306→            ) : (
   307→              suggestions.map((s) => (
   308→                <button
   309→                  key={s.id}
   310→                  type="button"
   311→                  role="option"
   312→                  aria-selected={selected?.id === s.id}
   313→                  onClick={() => {
   314→                    onSelect(s);
   315→                    setQuery('');
   316→                    setOpen(false);
   317→                  }}
   318→                  className="flex w-full items-center gap-3 rounded-lg p-2 text-left transition-colors hover:bg-white/5 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-rose-500/60"
   319→                >
   320→                  <span className="h-12 w-9 shrink-0 overflow-hidden rounded-md border border-white/10 bg-white/[0.03]">
   321→                    {s.thumbnail ? (
   322→                      <img src={s.thumbnail} alt="" loading="lazy" className="h-full w-full object-cover" />
   323→                    ) : (
   324→                      <BrandPlaceholder title={s.title} />
   325→                    )}
   326→                  </span>
   327→                  <span className="min-w-0 flex-1">
   328→                    <span className="block truncate text-sm font-semibold text-zinc-100">{s.title}</span>
   329→                  </span>
   330→                  <TypeBadge type={s.type} />
   331→                </button>
   332→              ))
   333→            )}
   334→          </div>
   335→        )}
   336→      </div>
   337→      {selected && (
   338→        <div className="flex items-center gap-2">
   339→          <span className="inline-flex max-w-full items-center gap-2 rounded-full border border-rose-500/30 bg-rose-500/10 py-1 pl-1 pr-2 text-sm text-rose-200">
   340→            <span className="h-7 w-7 shrink-0 overflow-hidden rounded-full border border-white/10 bg-white/[0.03]">
   341→              {selected.thumbnail ? (
   342→                <img src={selected.thumbnail} alt="" className="h-full w-full object-cover" />
   343→              ) : (
   344→                <BrandPlaceholder title={selected.title} />
   345→              )}
   346→            </span>
   347→            <span className="truncate font-medium">{selected.title}</span>
   348→            <button
   349→              type="button"
   350→              onClick={() => onSelect(null)}
   351→              aria-label={`Clear ${selected.title}`}
   352→              className="rounded-full p-1 text-rose-300/70 transition hover:bg-rose-500/20 hover:text-rose-100"
   353→            >
   354→              <X className="h-3.5 w-3.5" aria-hidden />
   355→            </button>
   356→          </span>
   357→        </div>
   358→      )}
   359→    </div>
   360→  );
   361→}

   514→                AI picks for you

  1352→function AssistantTab({ user }: { user: PlatformUser | null }) {

  1354→  const [input, setInput] = useState('');
  1355→  const [busy, setBusy] = useState(false);

  1357→  const [historyLoading, setHistoryLoading] = useState(true);
  1358→  const scrollRef = useRef<HTMLDivElement>(null);
  1359→

  1361→    const el = scrollRef.current;
  1362→    if (el) el.scrollTop = el.scrollHeight;

  1364→
  1365→  /* Restore the persisted conversation (oldest → newest) on mount. */

  1367→    if (!user) {

  1382→        }
  1383→        turns.sort((a, b) => a.at - b.at);
  1384→        setMessages(turns.map(({ role, text }) => ({ role, text })));
  1385→      })
  1386→      .catch(() => {
  1387→        /* history is best-effort — start a fresh conversation */
  1388→      })
  1389→      .finally(() => {

  1425→          meta: {
  1426→            provider: res.provider,
  1427→            degraded: res.degraded,
  1428→            toolsUsed: res.toolsUsed ?? [],
  1429→            memoriesUsed: res.memoriesUsed ?? 0,
  1430→            refused: Boolean(res.refused),
  1431→          },
  1432→        },
  1433→      ]);

  1454→    }
  1455→  };
  1456→
  1457→  return (
