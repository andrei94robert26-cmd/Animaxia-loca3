     4→ * party-room.tsx — Watch party room (#/social/party/:id). Synchronized player
     5→ * (host drives playback via realtime `party:control`, members follow with a
     6→ * 1.5s drift tolerance), live chat with typing indicators, member presence
     7→ * and a REST fallback for chat when the socket is down. Reuses the
     8→ * player-view rendering approach (mp4 / m3u8 via hls.js, embeds without sync).
     9→ */

    84→export function PartyRoom({ partyId }: { partyId: number }) {
    85→  const { user } = useAuth();
    86→  const validId = Number.isInteger(partyId) && partyId > 0;
    87→
    88→  const detail = useApi(`social-party-${partyId}`, () => social.partyDetail(partyId), { enabled: validId });
    89→  const current = detail.data && detail.data.party.id === partyId ? detail.data : null;
    90→
    91→  const [ended, setEnded] = useState(false);
    92→  const [members, setMembers] = useState<SocketMemberRow[]>([]);
    93→  const [messages, setMessages] = useState<ChatMessage[]>([]);
    94→  const [typingUser, setTypingUser] = useState<string | null>(null);
    95→  const [socketConnected, setSocketConnected] = useState(false);
    96→  const [lastPersisted, setLastPersisted] = useState<boolean | null>(null);
    97→  const [leaving, setLeaving] = useState(false);
    98→  const [copied, setCopied] = useState(false);
    99→  const [chatInput, setChatInput] = useState('');
   100→  const [lastState, setLastState] = useState<PartyState | null>(null);
   101→
   102→  const socketRef = useRef<Socket | null>(null);
   103→  const videoRef = useRef<HTMLVideoElement>(null);
   104→  const chatScrollRef = useRef<HTMLDivElement>(null);
   105→  const typingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   106→  const lastTypingEmitRef = useRef(0);
   107→  const lastStateRef = useRef<PartyState | null>(null);
   108→  const seededRef = useRef(false);
   109→  const isHostRef = useRef(false);
   110→
   111→  const party = current?.party ?? null;
   112→  const isHost = Boolean(user && party && party.host.id === user.id);
   113→  const partyActive = Boolean(party && party.status === 'active') && !ended;
   114→
   115→  /* Player sources — same client and rendering rules as player-view (with a
   116→     source switcher: the host's switch broadcasts a `source` control). */
   117→  const contentId = party?.content.id ?? 0;
   118→  const player = useApi(`party-player-${contentId}`, () => api.player(contentId), { enabled: contentId > 0 });
   119→  const playerData = player.data && player.data.content.id === contentId ? player.data : null;
   120→  const [sourceIndex, setSourceIndex] = useState(0);
   121→  const sources = playerData?.sources ?? [];
   122→  const source: PlayerSource | null =
   123→    sources.length > 0 ? sources[Math.min(sourceIndex, sources.length - 1)] : null;
   124→
   125→  /* Reset the picked source when the underlying player payload changes. */
   126→  useEffect(() => {
   127→    setSourceIndex(0);
   128→  }, [playerData]);
   129→
   130→  /* Sync isHost into a ref so socket handlers (registered once) stay fresh. */
   131→  useEffect(() => {
   132→    isHostRef.current = isHost;
   133→  }, [isHost]);
   134→
   135→  /* Seed local state from the REST detail (members / messages / state). */
   136→  useEffect(() => {
   137→    if (!current) return;
   138→    setMembers(current.members.map((m) => ({ userid: m.userid, username: m.username, avatar: m.avatar, online: false })));
   139→    setMessages(current.messages ?? []);
   140→    setLastState(current.state);
   141→    lastStateRef.current = current.state;
   142→    seededRef.current = false;
   143→  }, [current]);
   144→
   145→  /* Apply a party state to the <video> element (seek + pause/play + rate). */
   146→  function applyStateToVideo(state: PartyState): void {
   147→    const v = videoRef.current;
   148→    if (!v || !Number.isFinite(state.positionSec) || state.positionSec < 0) return;
   149→    try {
   150→      if (Math.abs(v.currentTime - state.positionSec) > 0.3 || v.readyState < 1) {
   151→        v.currentTime = state.positionSec;
   152→      }
   153→    } catch {
   154→      /* seeking before metadata is loaded — the loadedmetadata seed retries */
   155→      return;
   156→    }
   157→    if (Number.isFinite(state.playbackRate) && state.playbackRate > 0) v.playbackRate = state.playbackRate;
   158→    if (state.isPaused) v.pause();
   159→    else v.play().catch(() => {});
   160→  }
   161→
   162→  /* Seed the video position once the join ack (or REST state) arrives. */
   163→  function seedFromState(state: PartyState): void {
   164→    lastStateRef.current = state;
   165→    setLastState(state);
   166→    if (seededRef.current) return;
   167→    seededRef.current = true;
   168→    if (state.positionSec <= 0.5 && state.isPaused) return; // nothing to seed
   169→    applyStateToVideo(state);
   170→  }
   171→
   172→  /* Realtime socket — created for signed-in users while the party is active. */
   173→  useEffect(() => {
   174→    if (!user || !partyActive || !validId) return;
   175→
   176→    /* Idempotent REST join (also keeps membership when the socket degrades). */
   177→    social.partyJoin(partyId).catch(() => {
   178→      /* silent — the socket join below retries membership too */
   179→    });
   180→
   181→    const socket = createSocialSocket();
   182→    socketRef.current = socket;
   183→
   184→    /* The realtime service finishes its per-connection bootstrap (an async
   185→       profile fetch) before it registers the `party:join` handler — a join
   186→       emitted the instant `connect` fires can land too early and be dropped.
   187→       Join is idempotent server-side, so: emit immediately, re-emit on `hello`
   188→       (server sends it right after bootstrap) and retry until the join ack
   189→       (`party:state`) arrives. */
   190→    let joinTimer: ReturnType<typeof setTimeout> | null = null;
   191→    let joinAttempts = 0;
   192→    let stateAcked = false;
   193→    const emitJoin = (): void => {
   194→      socket.emit('party:join', { partyId });
   195→      joinAttempts += 1;
   196→      if (!stateAcked && joinAttempts < 6 && !joinTimer) {
   197→        joinTimer = setTimeout(() => {
   198→          joinTimer = null;
   199→          if (!stateAcked) emitJoin();
   200→        }, 1500);
   201→      }
   202→    };
   203→
   204→    socket.on('connect', () => {
   205→      setSocketConnected(true);
   206→      stateAcked = false;
   207→      if (joinTimer) {
   208→        clearTimeout(joinTimer);
   209→        joinTimer = null;
   210→      }
   211→      joinAttempts = 0;
   212→      emitJoin();
   213→    });
   214→    /* `hello` arrives after the server registered its handlers — a join sent
   215→       here always reaches the listener (covers both first connect and any
   216→       reconnect). */
   217→    socket.on('hello', () => {
   218→      if (!stateAcked) emitJoin();
   219→    });
   220→    socket.on('disconnect', () => setSocketConnected(false));
   221→    socket.on('connect_error', () => setSocketConnected(false));
   222→
   223→    socket.on('party:state', (d: PartyStateEvent) => {
   224→      if (d.partyId !== partyId) return;
   225→      stateAcked = true;
   226→      if (joinTimer) {
   227→        clearTimeout(joinTimer);
   228→        joinTimer = null;
   229→      }
   230→      setMembers(d.members ?? []);
   231→      seedFromState(d.state);
   232→      if (d.degraded) toast('Realtime service is in offline mode — messages may not be saved');
   233→    });
   234→
   235→    socket.on('party:member-joined', (d: { partyId: number; user: SocketMemberRow }) => {
   236→      if (d.partyId !== partyId) return;
   237→      setMembers((prev) =>
   238→        prev.some((m) => m.userid === d.user.userid)
   239→          ? prev.map((m) => (m.userid === d.user.userid ? { ...m, online: true } : m))
   240→          : [...prev, { ...d.user, online: true }],
   241→      );
   242→    });
   243→
   244→    socket.on('party:member-left', (d: { partyId: number; user: { userid: number } }) => {
   245→      if (d.partyId !== partyId) return;
   246→      setMembers((prev) => prev.filter((m) => m.userid !== d.user.userid));
   247→    });
   248→
   249→    socket.on('party:control', (d: PartyControlEvent) => {
   250→      if (d.partyId !== partyId) return;
   251→      lastStateRef.current = d.state;
   252→      setLastState(d.state);
   253→      if (isHostRef.current) return; // the host drives — never loop back
   254→      if (d.action === 'source' && Number.isInteger(d.state.sourceIndex) && d.state.sourceIndex >= 0) {
   255→        setSourceIndex(d.state.sourceIndex);
   256→      }
   257→      const v = videoRef.current;
   258→      if (!v) return;
   259→      const drift = Number.isFinite(d.state.positionSec) ? Math.abs(v.currentTime - d.state.positionSec) : 0;
   260→      if (d.action === 'seek' || d.action === 'source' || drift > 1.5) {
   261→        try {
   262→          v.currentTime = d.state.positionSec;
   263→        } catch {
   264→          /* not seekable yet */
   265→        }
   266→      }
   267→      if (Number.isFinite(d.state.playbackRate) && d.state.playbackRate > 0 && d.state.playbackRate !== v.playbackRate) {
   268→        v.playbackRate = d.state.playbackRate;
   269→      }
   270→      if (d.state.isPaused && !v.paused) v.pause();
   271→      else if (!d.state.isPaused && v.paused) v.play().catch(() => {});
   272→    });
   273→
   274→    socket.on('party:chat', (d: PartyChatEvent) => {
   275→      if (d.partyId !== partyId) return;
   276→      setLastPersisted(d.persisted !== false);
   277→      setTypingUser((current) => (current === d.message.user.username ? null : current));
   278→      setMessages((prev) => {
   279→        if (d.message.id != null && prev.some((m) => m.id === d.message.id)) return prev;
   280→        const next = [...prev, d.message];
   281→        return next.length > 200 ? next.slice(next.length - 200) : next;
   282→      });
   283→    });
   284→
   285→    socket.on('party:typing', (d: { partyId: number; user: { userid: number; username: string } }) => {
   286→      if (d.partyId !== partyId || !d.user) return;
   287→      if (user && d.user.userid === user.id) return;
   288→      setTypingUser(d.user.username);
   289→      if (typingTimerRef.current) clearTimeout(typingTimerRef.current);
   290→      typingTimerRef.current = setTimeout(() => setTypingUser(null), 3000);
   291→    });
   292→
   293→    socket.on('party:error', (d: { code: string; message: string }) => {
   294→      if (d.code === 'ended') {
   295→        setEnded(true);
   296→        toast('This watch party has ended');
   297→        return;
   298→      }
   299→      if (d.code === 'auth') return; // anon flow is handled by the sign-in prompt
   300→      toast.error(d.message || 'Watch party error');
   301→    });
   302→
   303→    return () => {
   304→      if (joinTimer) clearTimeout(joinTimer);
   305→      socket.removeAllListeners();
   306→      socket.disconnect();
   307→      socketRef.current = null;
   308→      setSocketConnected(false);
   309→    };
   310→  }, [user?.id, partyId, partyActive, validId, user]);
   311→  /* NOTE: `user` is listed for identity freshness (same id, refreshed profile);
   312→     the effect only ever creates one socket at a time. */
   313→
   314→  /* Host: periodic state snapshot to the monolith (fire-and-forget, 10s). */
   315→  useEffect(() => {
   316→    if (!isHost || !partyActive) return;
   317→    const timer = setInterval(() => {
   318→      const v = videoRef.current;
   319→      if (!v) return;
   320→      social
   321→        .partySync(partyId, {
   322→          positionSec: v.currentTime,
   323→          isPaused: v.paused,
   324→          playbackRate: Number.isFinite(v.playbackRate) && v.playbackRate > 0 ? v.playbackRate : 1,
   325→          sourceIndex,
   326→          updatedAt: new Date().toISOString(),
   327→        })
   328→        .then((res) => {
   329→          lastStateRef.current = res.state;
   330→        })
   331→        .catch(() => {});
   332→    }, 10_000);
   333→    return () => clearInterval(timer);
   334→  }, [isHost, partyActive, partyId, sourceIndex]);
   335→
   336→  /* HLS (m3u8) sources — identical to player-view's dynamic hls.js attach. */
   337→  useEffect(() => {
   338→    const url = source?.url;
   339→    const el = videoRef.current;
   340→    if (source?.type !== 'm3u8' || !url || !el) return;
   341→    let hls: { destroy: () => void } | null = null;
   342→    let cancelled = false;
   343→    (async () => {
   344→      try {
   345→        const mod = await import('hls.js');
   346→        const Hls = mod.default;
   347→        if (cancelled) return;
   348→        if (Hls.isSupported()) {
   349→          const instance = new Hls({ enableWorker: true });
   350→          instance.loadSource(url);
   351→          instance.attachMedia(el);
   352→          hls = instance;
   353→        } else if (el.canPlayType('application/vnd.apple.mpegurl')) {
   354→          el.src = url;
   355→        }
   356→      } catch {
   357→        /* hls failed to init */
   358→      }
   359→    })();
   360→    return () => {
   361→      cancelled = true;
   362→      hls?.destroy();
   363→    };
   364→  }, [source?.type, source?.url]);
   365→
   366→  /* Auto-scroll chat to the newest message. */
   367→  useEffect(() => {
   368→    const el = chatScrollRef.current;
   369→    if (el) el.scrollTop = el.scrollHeight;
   370→  }, [messages.length, typingUser]);
   371→
   372→  /* Clear the typing timer on unmount. */
   373→  useEffect(() => {
   374→    return () => {
   375→      if (typingTimerRef.current) clearTimeout(typingTimerRef.current);
   376→    };
   377→  }, []);
   378→
   379→  /* ------------------------------ host controls ----------------------------- */
   380→
   381→  function emitControl(action: 'play' | 'pause' | 'seek' | 'rate' | 'source', overrideIndex?: number): void {
   382→    const socket = socketRef.current;
   383→    const v = videoRef.current;
   384→    if (!socket || !isHost) return;
   385→    socket.emit('party:control', {
   386→      partyId,
   387→      action,
   388→      positionSec: v?.currentTime ?? 0,
   389→      playbackRate: v?.playbackRate ?? 1,
   390→      sourceIndex: overrideIndex ?? sourceIndex,
   391→    });
   392→  }
   393→
   394→  /** Host picks a different stream source — members follow via `source`. */
   395→  function switchSource(nextIndex: number): void {
   396→    setSourceIndex(nextIndex);
   397→    if (isHost) emitControl('source', nextIndex);
   398→  }
   399→
   400→  /* --------------------------------- actions -------------------------------- */
   401→
   402→  function sendChat(): void {
   403→    const body = chatInput.trim();
   404→    if (!body || !user) return;
   405→    setChatInput('');
   406→    const socket = socketRef.current;
   407→    if (socket && socket.connected) {
   408→      socket.emit('party:chat', { partyId, body });
   409→      return;
   410→    }
   411→    /* REST fallback (socket down): optimistic append, replaced by the reply. */
   412→    const temp: ChatMessage = {
   413→      id: `tmp-${Date.now()}`,
   414→      body,
   415→      createdat: new Date().toISOString(),
   416→      user: { id: user.id, username: user.username, avatar: user.avatar },
   417→    };
   418→    setMessages((prev) => [...prev, temp]);
   419→    social
   420→      .partySendMessage(partyId, body)
   421→      .then((res) => {
   422→        setLastPersisted(true);
   423→        setMessages((prev) => prev.map((m) => (m.id === temp.id ? res.message : m)));
   424→      })
   425→      .catch((err) => {
   426→        setMessages((prev) => prev.filter((m) => m.id !== temp.id));
   427→        toast.error(err instanceof Error ? err.message : 'Could not send the message');
   428→      });
   429→  }
   430→
   431→  function handleChatInputChange(value: string): void {
   432→    setChatInput(value);
   433→    const socket = socketRef.current;
   434→    const now = Date.now();
   435→    if (socket && socket.connected && now - lastTypingEmitRef.current > 2000) {
   436→      lastTypingEmitRef.current = now;
   437→      socket.emit('party:typing', { partyId });
   438→    }
   439→  }
   440→
   441→  async function copyInviteCode(): Promise<void> {
   442→    if (!party?.code) return;
   443→    try {
   444→      await navigator.clipboard.writeText(party.code);
   445→      setCopied(true);
   446→      toast.success('Invite code copied');
   447→      setTimeout(() => setCopied(false), 1500);
   448→    } catch {
   449→      toast.error('Clipboard is not available in this browser');
   450→    }
   451→  }
   452→
   453→  async function leaveParty(): Promise<void> {
   454→    if (leaving) return;
   455→    setLeaving(true);
   456→    try {
   457→      socketRef.current?.emit('party:leave', { partyId });
   458→      await social.partyLeave(partyId);
   459→      if (isHost) toast('Watch party ended');
   460→      else toast('Left the watch party');
   461→    } catch (err) {
   462→      toast.error(err instanceof Error ? err.message : 'Could not leave the party');
   463→    } finally {
   464→      navigate('/social');
   465→    }
   466→  }
   467→
   468→  /* -------------------------------- rendering ------------------------------- */
   469→
   470→  if (!validId) {
   471→    return (
   472→      <Container className="py-20">
   473→        <EmptyState
   474→          title="Watch party not found"
   475→          message="The URL is missing a valid party id."
   476→          action={
   477→            <Button className="bg-rose-600 font-bold hover:bg-rose-500" onClick={() => navigate('/social')}>
   478→              Back to Social
   479→            </Button>
   480→          }
   481→        />
   482→      </Container>
   483→    );
   484→  }
   485→
   486→  if (detail.loading && !current) {
   487→    return (
   488→      <Container className="py-8 sm:py-10">
   489→        <div className="mx-auto w-full max-w-6xl space-y-5">
   490→          <div className="flex items-center gap-3">
   491→            <Skeleton className="h-9 w-9 rounded-lg" />
   492→            <div className="flex-1 space-y-2">
   493→              <Skeleton className="h-6 w-72 max-w-full" />
   494→              <Skeleton className="h-4 w-44" />
   495→            </div>
   496→            <Skeleton className="h-9 w-24 rounded-lg" />
   497→          </div>
   498→          <Skeleton className="aspect-video w-full rounded-2xl" />
   499→          <div className="grid gap-5 lg:grid-cols-3">
   500→            <Skeleton className="h-40 rounded-2xl lg:col-span-2" />
   501→            <Skeleton className="h-64 rounded-2xl" />
   502→          </div>
   503→        </div>
   504→      </Container>
   505→    );
   506→  }
   507→
   508→  if (detail.error && !current) {
   509→    if (isNotFound(detail.error)) {
   510→      return (
   511→        <Container className="py-20">
   512→          <EmptyState
   513→            title="Party ended or not found"
   514→            message="This watch party is no longer active."
   515→            action={
   516→              <Button className="bg-rose-600 font-bold hover:bg-rose-500" onClick={() => navigate('/social')}>
   517→                <ArrowLeft aria-hidden /> Back to Social
   518→              </Button>
   519→            }
   520→          />
   521→        </Container>
   522→      );
   523→    }
   524→    return (
   525→      <Container className="py-20">
   526→        <ErrorState message={detail.error} onRetry={detail.refetch} />
   527→      </Container>
   528→    );
   529→  }
   530→
   531→  if (!party) return null;
   532→
   533→  if (ended || party.status !== 'active') {
   534→    return (
   535→      <Container className="py-20">
   536→        <EmptyState
   537→          icon={<Users className="h-6 w-6" aria-hidden />}
   538→          title="Party ended or not found"
   539→          message="This watch party has ended. Start a new one from the Social hub — or join another live party."
   540→          action={
   541→            <div className="flex flex-wrap justify-center gap-2">
   542→              <Button className="bg-rose-600 font-bold hover:bg-rose-500" onClick={() => navigate('/social')}>
   543→                <ArrowLeft aria-hidden /> Back to Social
   544→              </Button>
   545→              <Button variant="outline" className="border-white/15 bg-white/5 hover:bg-white/10" onClick={() => navigate(`/content/${party.content.id}`)}>
   546→                Open {party.content.title}
   547→              </Button>
   548→            </div>
   549→          }
   550→        />
   551→      </Container>
   552→    );
