    93→export function CategoryView({ route }: { route: AppRoute }) {
    94→  const type = route.params[0] || 'all';
    95→
    96→  /* Reset pagination whenever the category changes (adjust during render). */
    97→  const [page, setPage] = useState(1);
    98→  const [lastType, setLastType] = useState(type);
    99→  if (lastType !== type) {
   100→    setLastType(type);
   101→    setPage(1);
   102→  }
   103→
   104→  const [newsSource, setNewsSource] = useState('all');
   105→  const [sportTab, setSportTab] = useState<'football' | 'f1'>('football');
   106→  const [competition, setCompetition] = useState('PL');
   107→  const [addOpen, setAddOpen] = useState(false);
   108→
   109→  const isNews = type === 'news';
   110→  const isSports = type === 'sports';
   111→  const isSpecial = SPECIAL_CATEGORIES.has(type);
   112→  const virtual = VIRTUAL_CATEGORIES[type];
   113→  const isVirtual = Boolean(virtual) && !isSpecial;
   114→  const isGrid = !isNews && !isSports && !isSpecial && !isVirtual;
   115→  const hasStrip = type === 'movie' || type === 'tv' || type === 'anime';
   116→
   117→  const meta =
   118→    CATEGORY_META[type] ??
   119→    (virtual
   120→      ? { title: virtual.title, subtitle: virtual.subtitle }
   121→      : {
   122→          title: capitalize(type),
   123→          subtitle: 'Browse every title Animaxia has filed under this category.',
   124→        });
   125→
   126→  const listRes = useApi(
   127→    `cat-${type}-${page}`,
   128→    () => api.all({ type: type === 'all' ? undefined : type, page, limit: 24 }),
   129→    { enabled: isGrid },
   130→  );
   131→
   132→  /* Virtual sidebar categories: one full-catalog fetch + client-side filter. */
   133→  const virtualRes = useApi(`cat-virtual-${type}`, () => api.all({ limit: 100 }), { enabled: isVirtual });
   134→  const virtualItems = useMemo<NormalizedItem[]>(() => {
   135→    if (!virtual || !virtualRes.data) return [];
   136→    const rows = virtualRes.data.data.filter(virtual.filter);
   137→    const sorted = virtual.sort ? [...rows].sort(virtual.sort) : rows;
   138→    return sorted.map(dbToItem);
   139→  }, [virtual, virtualRes.data]);
   140→
   141→  const stripRes = useApi(
   142→    `cat-strip-${type}`,
   143→    async (): Promise<NormalizedItem[]> => {
   144→      if (type === 'movie') return (await api.moviesTrending('day')).data.map(mediaToItem);
   145→      if (type === 'tv') return (await api.tvPopular()).data.map(mediaToItem);
   146→      return (await api.animeSeason()).data.map(animeToItem);
   147→    },
   148→    { enabled: hasStrip },
   149→  );
   150→
   151→  const newsRes = useApi(`cat-news-${newsSource}`, () => api.news(newsSource, 24), { enabled: isNews });
   152→
   153→  const standingsRes = useApi(`fb-standings-${competition}`, () => api.footballStandings(competition), {
   154→    enabled: isSports,
   155→  });
   156→  const matchesRes = useApi(`fb-matches-${competition}`, () => api.footballMatches(competition, 10), {
   157→    enabled: isSports,
   158→  });
   159→  const f1StandingsRes = useApi('f1-standings', () => api.f1Standings(), {
   160→    enabled: isSports && sportTab === 'f1',
   161→  });
   162→  const f1ScheduleRes = useApi('f1-schedule', () => api.f1Schedule(), {
   163→    enabled: isSports && sportTab === 'f1',
   164→  });
   165→
   166→  const items = useMemo(() => (listRes.data?.data ?? []).map(dbToItem), [listRes.data]);
   167→  const gridRef = useRef<HTMLDivElement>(null);
   168→
   169→  const stripMeta =
   170→    type === 'movie'
   171→      ? { title: 'Trending on TMDB', subtitle: 'Today’s most-watched films worldwide' }
   172→      : type === 'tv'
   173→        ? { title: 'Popular on TMDB', subtitle: 'Series people cannot stop watching' }
   174→        : { title: 'Airing this season', subtitle: 'Currently broadcasting in Japan' };
   175→
   176→  const goToPage = (next: number) => {
   177→    setPage(next);
   178→    gridRef.current?.scrollIntoView({ behavior: 'smooth', block: 'start' });
   179→  };
   180→
   181→  return (
   182→    <Container className="space-y-8 py-8 sm:py-10">
   183→      <PageHeader title={meta.title} subtitle={meta.subtitle}>
   184→        {isGrid && !listRes.loading && listRes.data && (
   185→          <p className="text-sm font-medium text-zinc-500">
   186→            Showing {listRes.data.data.length} of {listRes.data.total}{' '}
   187→            {listRes.data.total === 1 ? 'title' : 'titles'}
   188→          </p>
   189→        )}
   190→      </PageHeader>
   191→
   192→      {isGrid && (
   193→        <>
   194→          {type === 'mined' && (
   195→            <Alert className="border-rose-500/20 bg-rose-500/5">
   196→              <Pickaxe className="h-4 w-4 text-rose-400" aria-hidden />
   197→              <AlertTitle className="text-zinc-100">Extracted by the Animaxia Miner</AlertTitle>
   198→              <AlertDescription className="text-zinc-400">
   199→                <span>
   200→                  Mined titles are extracted by the Animaxia Miner from any URL — open the Admin
   201→                  panel to mine a new stream.
   202→                </span>
   203→                <Button
   204→                  size="sm"
   205→                  onClick={() => navigate('/admin')}
   206→                  className="mt-2 bg-rose-600 font-semibold text-white hover:bg-rose-500"
   207→                >
   208→                  Open Admin panel
   209→                </Button>
   210→              </AlertDescription>
   211→            </Alert>
   212→          )}
   213→
   214→          {hasStrip &&
   215→            (stripRes.error && !stripRes.loading && !stripRes.data ? (
   216→              <ErrorState message={stripRes.error} onRetry={stripRes.refetch} compact />
   217→            ) : (
   218→              <ContentRow
   219→                title={stripMeta.title}
   220→                subtitle={stripMeta.subtitle}
   221→                items={stripRes.data ?? []}
   222→                loading={stripRes.loading}
   223→                emptyMessage="Live provider data is unavailable right now."
   224→              />
   225→            ))}
   226→
   227→          <div ref={gridRef} className="scroll-mt-24 space-y-6">
   228→            {listRes.error && !listRes.loading && !listRes.data ? (
   229→              <ErrorState message={listRes.error} onRetry={listRes.refetch} />
   230→            ) : (
   231→              <ContentGrid
   232→                items={items}
   233→                loading={listRes.loading}
   234→                skeletonCount={12}
   235→                emptyTitle={type === 'all' ? 'Your library is empty' : `No ${meta.title.toLowerCase()} yet`}
   236→                emptyMessage="Nothing has been added to this category so far — add titles from any detail page."
   237→              />
   238→            )}
   239→            <Pagination page={page} totalPages={listRes.data?.totalPages ?? 1} onPage={goToPage} />
   240→          </div>
   241→        </>
   242→      )}
   243→
   244→      {isSpecial && (
   245→        <>
   246→          {type === 'my-list' && <MyListPanel />}
   247→          {type === 'my-collections' && <CollectionsPanel />}
   248→          {type === 'manage' && <ManagePanel />}
   249→          {type === 'tvshows' && <TvShowsPanel />}
   250→        </>
   251→      )}
   252→
   253→      {isVirtual && (
   254→        <>
   255→          {virtualRes.error && !virtualRes.loading && !virtualRes.data ? (
   256→            <ErrorState message={virtualRes.error} onRetry={virtualRes.refetch} />
   257→          ) : (
   258→            <ContentGrid
   259→              items={virtualItems}
   260→              loading={virtualRes.loading}
   261→              skeletonCount={9}
   262→              emptyTitle={`Încă nimic în ${meta.title}`}
   263→              emptyMessage={
   264→                virtual?.emptyHint ??
   265→                'Această categorie se umple automat cu tot ce încarci tu — adaugă conținut din orice sursă.'
   266→              }
   267→              emptyAction={
   268→                <Button onClick={() => setAddOpen(true)} className="mt-4 bg-rose-600 font-bold hover:bg-rose-500">
   269→                  <Plus className="mr-1.5 h-4 w-4" aria-hidden />
   270→                  Adaugă conținut
   271→                </Button>
   272→              }
   273→            />
   274→          )}
   275→          {!virtualRes.loading && virtualItems.length > 0 && (
   276→            <p className="text-sm font-medium text-zinc-500">
   277→              {virtualItems.length} {virtualItems.length === 1 ? 'titlu' : 'titluri'} în {meta.title}
   278→            </p>
   279→          )}
   280→        </>
   281→      )}
   282→
   283→      <AddContentDialog open={addOpen} onOpenChange={setAddOpen} />
   284→
   285→      {isNews && <NewsPanel source={newsSource} onSource={setNewsSource} res={newsRes} />}
   286→
   287→      {isSports && (
   288→        <Tabs defaultValue="football" onValueChange={(value) => setSportTab(value as 'football' | 'f1')}>
   289→          <TabsList className="h-10 w-full justify-start gap-1 rounded-xl border border-white/10 bg-white/[0.03] p-1 sm:w-auto">
   290→            <TabsTrigger
   291→              value="football"
   292→              className="px-4 text-zinc-400 hover:text-zinc-100 data-[state=active]:bg-rose-600 data-[state=active]:text-white data-[state=active]:shadow-rose-900/40 sm:px-5"
   293→            >
   294→              Football
   295→            </TabsTrigger>
   296→            <TabsTrigger
   297→              value="f1"
   298→              className="px-4 text-zinc-400 hover:text-zinc-100 data-[state=active]:bg-rose-600 data-[state=active]:text-white data-[state=active]:shadow-rose-900/40 sm:px-5"
   299→            >
   300→              Formula 1
   301→            </TabsTrigger>
   302→          </TabsList>
   303→          <TabsContent value="football" className="pt-6">
   304→            <FootballPanel
   305→              competition={competition}
   306→              onCompetition={setCompetition}
   307→              standingsRes={standingsRes}
   308→              matchesRes={matchesRes}
   309→            />
   310→          </TabsContent>
   311→          <TabsContent value="f1" className="pt-6">
   312→            <F1Panel standingsRes={f1StandingsRes} scheduleRes={f1ScheduleRes} />
   313→          </TabsContent>
   314→        </Tabs>
   315→      )}
   316→    </Container>
   317→  );
   318→}

   320→/* ------------------------------ news panel ------------------------------- */
   321→
   322→function NewsPanel({
   323→  source,
   324→  onSource,
   325→  res,
   326→}: {
   327→  source: string;
   328→  onSource: (value: string) => void;
   329→  res: UseApiResult<{ data: NewsItem[] }>;
   330→}) {
   331→  const items = useMemo(() => (res.data?.data ?? []).map(newsToItem), [res.data]);
   332→  return (
   333→    <div className="space-y-6">
   334→      <div className="flex flex-wrap items-center gap-2" role="group" aria-label="News source filter">
   335→        {NEWS_SOURCES.map((s) => (
   336→          <FilterChip key={s.value} active={source === s.value} onClick={() => onSource(s.value)}>
   337→            {s.label}
   338→          </FilterChip>
   339→        ))}
   340→      </div>
   341→      {res.error && !res.loading && !res.data ? (
   342→        <ErrorState message={res.error} onRetry={res.refetch} />
   343→      ) : (
   344→        <ContentGrid
   345→          items={items}
   346→          loading={res.loading}
   347→          skeletonCount={12}
   348→          emptyTitle="No headlines right now"
   349→          emptyMessage="This feed returned no recent articles — try another source."
   350→        />
   351→      )}
   352→    </div>
   353→  );
   354→}

   356→/* ---------------------------- football panel ----------------------------- */
   357→
   358→function FootballPanel({
   359→  competition,
   360→  onCompetition,
   361→  standingsRes,
   362→  matchesRes,
   363→}: {
   364→  competition: string;
   365→  onCompetition: (code: string) => void;
   366→  standingsRes: UseApiResult<{ data: { competition: string; table: FootballTableEntry[] } }>;
   367→  matchesRes: UseApiResult<{ data: FootballMatch[] }>;
   368→}) {
   369→  const competitionName = COMPETITIONS.find((c) => c.code === competition)?.name ?? competition;
   370→  return (
   371→    <div className="space-y-8">
   372→      <div className="space-y-3">
   373→        <h2 className="text-lg font-bold text-zinc-100">Competition</h2>
   374→        <div className="flex flex-wrap gap-2" role="group" aria-label="Competition filter">
   375→          {COMPETITIONS.map((c) => (
   376→            <FilterChip key={c.code} active={competition === c.code} onClick={() => onCompetition(c.code)}>
   377→              {c.name}
   378→            </FilterChip>
   379→          ))}
   380→        </div>
   381→      </div>
   382→
   383→      <Section title="Standings" subtitle={`Current ${competitionName} table`}>
   384→        {standingsRes.loading ? (
   385→          <Skeleton className="h-80 w-full rounded-xl" />
   386→        ) : standingsRes.error ? (
   387→          <ErrorState message={standingsRes.error} onRetry={standingsRes.refetch} compact />
   388→        ) : !standingsRes.data || standingsRes.data.data.table.length === 0 ? (
   389→          <EmptyState title="No standings available" message="This competition has no table data right now." />
   390→        ) : (
   391→          <StandingsTable entries={standingsRes.data.data.table} />
   392→        )}
   393→      </Section>
   394→
   395→      <Section title="Fixtures & results" subtitle="Upcoming matches and recent scores">
   396→        {matchesRes.loading ? (
   397→          <Skeleton className="h-72 w-full rounded-xl" />
   398→        ) : matchesRes.error ? (
   399→          <ErrorState message={matchesRes.error} onRetry={matchesRes.refetch} compact />
   400→        ) : !matchesRes.data || matchesRes.data.data.length === 0 ? (
   401→          <EmptyState title="No matches scheduled" message="There are no upcoming fixtures for this competition." />
   402→        ) : (
   403→          <div className="overflow-hidden rounded-xl border border-white/10 bg-white/[0.02]">
   404→            <div className="divide-y divide-white/5">
   405→              {matchesRes.data.data.map((match) => (
   406→                <MatchRow key={match.id} match={match} />
   407→              ))}
   408→            </div>
   409→          </div>
   410→        )}
   411→      </Section>
   412→    </div>
   413→  );
   414→}

   473→function MatchRow({ match }: { match: FootballMatch }) {
   474→  const finished =
   475→    match.status === 'FINISHED' && match.score.home != null && match.score.away != null;
   476→  const date = new Date(match.utcDate);
   477→  const dateLabel = date.toLocaleDateString('en-US', {
   478→    weekday: 'short',
   479→    month: 'short',
   480→    day: 'numeric',
   481→    timeZone: 'UTC',
   482→  });
   483→  const timeLabel = date.toLocaleTimeString('en-US', {
   484→    hour: '2-digit',
   485→    minute: '2-digit',
   486→    hour12: false,
   487→    timeZone: 'UTC',
   488→  });
   489→  return (
   490→    <div className="flex flex-col gap-3 px-4 py-4 sm:flex-row sm:items-center sm:justify-between">
   491→      <div className="grid min-w-0 flex-1 grid-cols-[1fr_auto_1fr] items-center gap-3">
   492→        <div className="flex min-w-0 items-center justify-end gap-2">
   493→          <span className="truncate text-right text-sm font-medium text-zinc-200">{match.home}</span>
   494→          {match.homeCrest ? (
   495→            <img src={match.homeCrest} alt="" aria-hidden loading="lazy" className="h-6 w-6 shrink-0 object-contain" />
   496→          ) : null}
   497→        </div>
   498→        <div className="w-14 text-center text-sm font-bold">
   499→          {finished ? (
   500→            <span className="text-emerald-400">
   501→              {match.score.home} – {match.score.away}
   502→            </span>
   503→          ) : (
   504→            <span className="text-zinc-500">vs</span>
   505→          )}
   506→        </div>
   507→        <div className="flex min-w-0 items-center gap-2">
   508→          {match.awayCrest ? (
   509→            <img src={match.awayCrest} alt="" aria-hidden loading="lazy" className="h-6 w-6 shrink-0 object-contain" />
   510→          ) : null}
   511→          <span className="truncate text-sm font-medium text-zinc-200">{match.away}</span>
   512→        </div>
   513→      </div>
   514→      <div className="flex items-center gap-3 sm:flex-col sm:items-end sm:gap-1">
   515→        <span className="text-xs text-zinc-500">
   516→          {dateLabel} · {timeLabel} UTC
   517→        </span>
   518→        <StatusBadge status={match.status} />
   519→      </div>
   520→    </div>
   521→  );
   522→}

   617→        {race.round}
   618→      </span>
   619→      <div className="min-w-0 flex-1">
   620→        <p className="truncate font-semibold text-zinc-100">{race.race}</p>
   621→        <p className="mt-0.5 truncate text-xs text-zinc-500">

   628→      </div>

   631→}
