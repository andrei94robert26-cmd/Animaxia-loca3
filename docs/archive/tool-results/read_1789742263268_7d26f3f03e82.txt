    93→    93→export function CategoryView({ route }: { route: AppRoute }) {
    94→    94→  const type = route.params[0] || 'all';
    95→    95→
    96→    96→  /* Reset pagination whenever the category changes (adjust during render). */
    97→    97→  const [page, setPage] = useState(1);
    98→    98→  const [lastType, setLastType] = useState(type);
    99→    99→  if (lastType !== type) {
   100→   100→    setLastType(type);
   101→   101→    setPage(1);
   102→   102→  }
   103→   103→
   104→   104→  const [newsSource, setNewsSource] = useState('all');
   105→   105→  const [sportTab, setSportTab] = useState<'football' | 'f1'>('football');
   106→   106→  const [competition, setCompetition] = useState('PL');

   108→   108→
   109→   109→  const isNews = type === 'news';
   110→   110→  const isSports = type === 'sports';
   111→   111→  const isSpecial = SPECIAL_CATEGORIES.has(type);
   112→   112→  const virtual = VIRTUAL_CATEGORIES[type];
   113→   113→  const isVirtual = Boolean(virtual) && !isSpecial;
   114→   114→  const isGrid = !isNews && !isSports && !isSpecial && !isVirtual;
   115→   115→  const hasStrip = type === 'movie' || type === 'tv' || type === 'anime';
   116→   116→
   117→   117→  const meta =
   118→   118→    CATEGORY_META[type] ??

   122→   122→          title: capitalize(type),

   124→   124→        });
   125→   125→
   126→   126→  const listRes = useApi(
   127→   127→    `cat-${type}-${page}`,
   128→   128→    () => api.all({ type: type === 'all' ? undefined : type, page, limit: 24 }),
   129→   129→    { enabled: isGrid },
   130→   130→  );
   131→   131→
   132→   132→  /* Virtual sidebar categories: one full-catalog fetch + client-side filter. */
   133→   133→  const virtualRes = useApi(`cat-virtual-${type}`, () => api.all({ limit: 100 }), { enabled: isVirtual });
   134→   134→  const virtualItems = useMemo<NormalizedItem[]>(() => {
   135→   135→    if (!virtual || !virtualRes.data) return [];

   137→   137→    const sorted = virtual.sort ? [...rows].sort(virtual.sort) : rows;
   138→   138→    return sorted.map(dbToItem);
   139→   139→  }, [virtual, virtualRes.data]);
   140→   140→
   141→   141→  const stripRes = useApi(
   142→   142→    `cat-strip-${type}`,
   143→   143→    async (): Promise<NormalizedItem[]> => {
   144→   144→      if (type === 'movie') return (await api.moviesTrending('day')).data.map(mediaToItem);
   145→   145→      if (type === 'tv') return (await api.tvPopular()).data.map(mediaToItem);
   146→   146→      return (await api.animeSeason()).data.map(animeToItem);
   147→   147→    },
   148→   148→    { enabled: hasStrip },
   149→   149→  );
   150→   150→
   151→   151→  const newsRes = useApi(`cat-news-${newsSource}`, () => api.news(newsSource, 24), { enabled: isNews });
   152→   152→
   153→   153→  const standingsRes = useApi(`fb-standings-${competition}`, () => api.footballStandings(competition), {
   154→   154→    enabled: isSports,
   155→   155→  });
   156→   156→  const matchesRes = useApi(`fb-matches-${competition}`, () => api.footballMatches(competition, 10), {
   157→   157→    enabled: isSports,
   158→   158→  });
   159→   159→  const f1StandingsRes = useApi('f1-standings', () => api.f1Standings(), {
   160→   160→    enabled: isSports && sportTab === 'f1',
   161→   161→  });
   162→   162→  const f1ScheduleRes = useApi('f1-schedule', () => api.f1Schedule(), {
   163→   163→    enabled: isSports && sportTab === 'f1',
   164→   164→  });
   165→   165→
   166→   166→  const items = useMemo(() => (listRes.data?.data ?? []).map(dbToItem), [listRes.data]);
   167→   167→  const gridRef = useRef<HTMLDivElement>(null);
   168→   168→
   169→   169→  const stripMeta =
   170→   170→    type === 'movie'
   171→   171→      ? { title: 'Trending on TMDB', subtitle: 'Today’s most-watched films worldwide' }
   172→   172→      : type === 'tv'
   173→   173→        ? { title: 'Popular on TMDB', subtitle: 'Series people cannot stop watching' }
   174→   174→        : { title: 'Airing this season', subtitle: 'Currently broadcasting in Japan' };
   175→   175→
   176→   176→  const goToPage = (next: number) => {
   177→   177→    setPage(next);
   178→   178→    gridRef.current?.scrollIntoView({ behavior: 'smooth', block: 'start' });
   179→   179→  };
   180→   180→
   181→   181→  return (
   182→   182→    <Container className="space-y-8 py-8 sm:py-10">
   183→   183→      <PageHeader title={meta.title} subtitle={meta.subtitle}>
   184→   184→        {isGrid && !listRes.loading && listRes.data && (
   185→   185→          <p className="text-sm font-medium text-zinc-500">
   186→   186→            Showing {listRes.data.data.length} of {listRes.data.total}{' '}
   187→   187→            {listRes.data.total === 1 ? 'title' : 'titles'}
   188→   188→          </p>
   189→   189→        )}
   190→   190→      </PageHeader>
   191→   191→
   192→   192→      {isGrid && (
   193→   193→        <>
   194→   194→          {type === 'mined' && (
   195→   195→            <Alert className="border-rose-500/20 bg-rose-500/5">
   196→   196→              <Pickaxe className="h-4 w-4 text-rose-400" aria-hidden />
   197→   197→              <AlertTitle className="text-zinc-100">Extracted by the Animaxia Miner</AlertTitle>
   198→   198→              <AlertDescription className="text-zinc-400">
   199→   199→                <span>
   200→   200→                  Mined titles are extracted by the Animaxia Miner from any URL — open the Admin
   201→   201→                  panel to mine a new stream.
   202→   202→                </span>
   203→   203→                <Button
   204→   204→                  size="sm"
   205→   205→                  onClick={() => navigate('/admin')}
   206→   206→                  className="mt-2 bg-rose-600 font-semibold text-white hover:bg-rose-500"
   207→   207→                >
   208→   208→                  Open Admin panel
   209→   209→                </Button>
   210→   210→              </AlertDescription>
   211→   211→            </Alert>
   212→   212→          )}
   213→   213→
   214→   214→          {hasStrip &&
   215→   215→            (stripRes.error && !stripRes.loading && !stripRes.data ? (
   216→   216→              <ErrorState message={stripRes.error} onRetry={stripRes.refetch} compact />
   217→   217→            ) : (
   218→   218→              <ContentRow
   219→   219→                title={stripMeta.title}
   220→   220→                subtitle={stripMeta.subtitle}
   221→   221→                items={stripRes.data ?? []}
   222→   222→                loading={stripRes.loading}
   223→   223→                emptyMessage="Live provider data is unavailable right now."
   224→   224→              />
   225→   225→            ))}
   226→   226→
   227→   227→          <div ref={gridRef} className="scroll-mt-24 space-y-6">
   228→   228→            {listRes.error && !listRes.loading && !listRes.data ? (
   229→   229→              <ErrorState message={listRes.error} onRetry={listRes.refetch} />
   230→   230→            ) : (
   231→   231→              <ContentGrid
   232→   232→                items={items}
   233→   233→                loading={listRes.loading}
   234→   234→                skeletonCount={12}
   235→   235→                emptyTitle={type === 'all' ? 'Your library is empty' : `No ${meta.title.toLowerCase()} yet`}
   236→   236→                emptyMessage="Nothing has been added to this category so far — add titles from any detail page."
   237→   237→              />
   238→   238→            )}
   239→   239→            <Pagination page={page} totalPages={listRes.data?.totalPages ?? 1} onPage={goToPage} />
   240→   240→          </div>
   241→   241→        </>
   242→   242→      )}
   243→   243→
   244→   244→      {isSpecial && (
   245→   245→        <>
   246→   246→          {type === 'my-list' && <MyListPanel />}
   247→   247→          {type === 'my-collections' && <CollectionsPanel />}
   248→   248→          {type === 'manage' && <ManagePanel />}
   249→   249→          {type === 'tvshows' && <TvShowsPanel />}
   250→   250→        </>
   251→   251→      )}
   252→   252→
   253→   253→      {isVirtual && (
   254→   254→        <>
   255→   255→          {virtualRes.error && !virtualRes.loading && !virtualRes.data ? (
   256→   256→            <ErrorState message={virtualRes.error} onRetry={virtualRes.refetch} />
   257→   257→          ) : (
   258→   258→            <ContentGrid
   259→   259→              items={virtualItems}
   260→   260→              loading={virtualRes.loading}
   261→   261→              skeletonCount={9}
   262→   262→              emptyTitle={`Încă nimic în ${meta.title}`}
   263→   263→              emptyMessage={
   264→   264→                virtual?.emptyHint ??
   265→   265→                'Această categorie se umple automat cu tot ce încarci tu — adaugă conținut din orice sursă.'
   266→   266→              }
   267→   267→              emptyAction={
   268→   268→                <Button onClick={() => setAddOpen(true)} className="mt-4 bg-rose-600 font-bold hover:bg-rose-500">
   269→   269→                  <Plus className="mr-1.5 h-4 w-4" aria-hidden />
   270→   270→                  Adaugă conținut
   271→   271→                </Button>
   272→   272→              }
   273→   273→            />
   274→   274→          )}
   275→   275→          {!virtualRes.loading && virtualItems.length > 0 && (
   276→   276→            <p className="text-sm font-medium text-zinc-500">
   277→   277→              {virtualItems.length} {virtualItems.length === 1 ? 'titlu' : 'titluri'} în {meta.title}
   278→   278→            </p>
   279→   279→          )}
   280→   280→        </>
   281→   281→      )}
   282→   282→
   283→   283→      <AddContentDialog open={addOpen} onOpenChange={setAddOpen} />
   284→   284→
   285→   285→      {isNews && <NewsPanel source={newsSource} onSource={setNewsSource} res={newsRes} />}
   286→   286→
   287→   287→      {isSports && (
   288→   288→        <Tabs defaultValue="football" onValueChange={(value) => setSportTab(value as 'football' | 'f1')}>
   289→   289→          <TabsList className="h-10 w-full justify-start gap-1 rounded-xl border border-white/10 bg-white/[0.03] p-1 sm:w-auto">
   290→   290→            <TabsTrigger
   291→   291→              value="football"
   292→   292→              className="px-4 text-zinc-400 hover:text-zinc-100 data-[state=active]:bg-rose-600 data-[state=active]:text-white data-[state=active]:shadow-rose-900/40 sm:px-5"
   293→   293→            >
   294→   294→              Football
   295→   295→            </TabsTrigger>
   296→   296→            <TabsTrigger
   297→   297→              value="f1"
   298→   298→              className="px-4 text-zinc-400 hover:text-zinc-100 data-[state=active]:bg-rose-600 data-[state=active]:text-white data-[state=active]:shadow-rose-900/40 sm:px-5"
   299→   299→            >
   300→   300→              Formula 1
   301→   301→            </TabsTrigger>
   302→   302→          </TabsList>
   303→   303→          <TabsContent value="football" className="pt-6">
   304→   304→            <FootballPanel
   305→   305→              competition={competition}
   306→   306→              onCompetition={setCompetition}
   307→   307→              standingsRes={standingsRes}
   308→   308→              matchesRes={matchesRes}
   309→   309→            />
   310→   310→          </TabsContent>
   311→   311→          <TabsContent value="f1" className="pt-6">
   312→   312→            <F1Panel standingsRes={f1StandingsRes} scheduleRes={f1ScheduleRes} />
   313→   313→          </TabsContent>
   314→   314→        </Tabs>
   315→   315→      )}
   316→   316→    </Container>
   317→   317→  );
   318→   318→}

   321→   321→
   322→   322→function NewsPanel({
   323→   323→  source,
   324→   324→  onSource,
   325→   325→  res,
   326→   326→}: {
   327→   327→  source: string;
   328→   328→  onSource: (value: string) => void;
   329→   329→  res: UseApiResult<{ data: NewsItem[] }>;
   330→   330→}) {

   332→   332→  return (
   333→   333→    <div className="space-y-6">

   339→   339→        ))}
   340→   340→      </div>
   341→   341→      {res.error && !res.loading && !res.data ? (

   343→   343→      ) : (
   344→   344→        <ContentGrid
   345→   345→          items={items}
   346→   346→          loading={res.loading}
   347→   347→          skeletonCount={12}
   348→   348→          emptyTitle="No headlines right now"
   349→   349→          emptyMessage="This feed returned no recent articles — try another source."
   350→   350→        />
   351→   351→      )}
   352→   352→    </div>
   353→   353→  );
   354→   354→}

   358→   358→function FootballPanel({
   359→   359→  competition,
   360→   360→  onCompetition,
   361→   361→  standingsRes,
   362→   362→  matchesRes,
   363→   363→}: {
   364→   364→  competition: string;
   365→   365→  onCompetition: (code: string) => void;
   366→   366→  standingsRes: UseApiResult<{ data: { competition: string; table: FootballTableEntry[] } }>;
   367→   367→  matchesRes: UseApiResult<{ data: FootballMatch[] }>;
   368→   368→}) {
   369→   369→  const competitionName = COMPETITIONS.find((c) => c.code === competition)?.name ?? competition;
   370→   370→  return (
   371→   371→    <div className="space-y-8">
   372→   372→      <div className="space-y-3">
   373→   373→        <h2 className="text-lg font-bold text-zinc-100">Competition</h2>
   374→   374→        <div className="flex flex-wrap gap-2" role="group" aria-label="Competition filter">
   375→   375→          {COMPETITIONS.map((c) => (
   376→   376→            <FilterChip key={c.code} active={competition === c.code} onClick={() => onCompetition(c.code)}>
   377→   377→              {c.name}
   378→   378→            </FilterChip>
   379→   379→          ))}
   380→   380→        </div>
   381→   381→      </div>
   382→   382→
   383→   383→      <Section title="Standings" subtitle={`Current ${competitionName} table`}>
   384→   384→        {standingsRes.loading ? (
   385→   385→          <Skeleton className="h-80 w-full rounded-xl" />
   386→   386→        ) : standingsRes.error ? (
   387→   387→          <ErrorState message={standingsRes.error} onRetry={standingsRes.refetch} compact />
   388→   388→        ) : !standingsRes.data || standingsRes.data.data.table.length === 0 ? (
   389→   389→          <EmptyState title="No standings available" message="This competition has no table data right now." />
   390→   390→        ) : (
   391→   391→          <StandingsTable entries={standingsRes.data.data.table} />
   392→   392→        )}
   393→   393→      </Section>
   394→   394→
   395→   395→      <Section title="Fixtures & results" subtitle="Upcoming matches and recent scores">
   396→   396→        {matchesRes.loading ? (
   397→   397→          <Skeleton className="h-72 w-full rounded-xl" />
   398→   398→        ) : matchesRes.error ? (
   399→   399→          <ErrorState message={matchesRes.error} onRetry={matchesRes.refetch} compact />
   400→   400→        ) : !matchesRes.data || matchesRes.data.data.length === 0 ? (
   401→   401→          <EmptyState title="No matches scheduled" message="There are no upcoming fixtures for this competition." />
   402→   402→        ) : (
   403→   403→          <div className="overflow-hidden rounded-xl border border-white/10 bg-white/[0.02]">
   404→   404→            <div className="divide-y divide-white/5">
   405→   405→              {matchesRes.data.data.map((match) => (
   406→   406→                <MatchRow key={match.id} match={match} />
