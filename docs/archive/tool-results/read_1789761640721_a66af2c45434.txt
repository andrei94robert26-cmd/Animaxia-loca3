    82→function loadJson(file, fallback) {
    83→  try {
    84→    const raw = fs.readFileSync(file, 'utf8');
    85→    const parsed = JSON.parse(raw);
    86→    return parsed && typeof parsed === 'object' ? parsed : fallback;
    87→  } catch {
    88→    return fallback;
    89→  }
    90→}

    92→function saveJson(file, value) {
    93→  try {
    94→    if (!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR, { recursive: true });
    95→    const tmp = `${file}.tmp`;
    96→    fs.writeFileSync(tmp, JSON.stringify(value, null, 2), 'utf8');
    97→    fs.renameSync(tmp, file);
    98→    return true;
    99→  } catch (err) {
   100→    logger.warn(`[ai-home] failed to persist ${path.basename(file)}: ${err.message}`);
   101→    return false;
   102→  }
   103→}

   156→/**
   157→ * External source-page k: interleave TMDB trending movies + trending TV +
   158→ * Jikan top anime into one deterministic stream. Cached for CHUNK_TTL_MS so
   159→ * walking pages stays stable. Failures degrade to partial chunks.
   160→ */
   161→async function externalChunk(k) {
   162→  const key = `aihome:extchunk:${k}`;
   163→  const hit = cache.get(key);
   164→  if (hit) return hit;
   165→
   166→  const [mv, tv, an] = await Promise.allSettled([
   167→    (async () => {
   168→      const payload = await tmdbFetch('/trending/movie/week', { page: k });
   169→      const map = await getGenreMap('movie');
   170→      return (payload?.results || []).map((i) => normalizeTmdbListItem(i, 'movie', map)).filter(Boolean);
   171→    })(),
   172→    (async () => {
   173→      const payload = await tmdbFetch('/trending/tv/week', { page: k });
   174→      const map = await getGenreMap('tv');
   175→      return (payload?.results || []).map((i) => normalizeTmdbListItem(i, 'tv', map)).filter(Boolean);
   176→    })(),
   177→    (async () => {
   178→      const payload = await jikanFetch('/top/anime', { page: k, limit: 20 }, 3);
   179→      return (payload?.data || []).map(normalizeJikanListItem).filter(Boolean);
   180→    })(),
   181→  ]);
   182→
   183→  const movies = mv.status === 'fulfilled' ? mv.value : [];
   184→  const tvs = tv.status === 'fulfilled' ? tv.value : [];
   185→  const animes = an.status === 'fulfilled' ? an.value : [];
   186→  const maxLen = Math.max(movies.length, tvs.length, animes.length);
   187→  const interleaved = [];
   188→  for (let i = 0; i < maxLen; i += 1) {
   189→    if (movies[i]) interleaved.push(movies[i]);
   190→    if (tvs[i]) interleaved.push(tvs[i]);
   191→    if (animes[i]) interleaved.push(animes[i]);
   192→  }
   193→  cache.set(key, interleaved, CHUNK_TTL_MS);
   194→  return interleaved;
   195→}
   196→
   197→/**
   198→ * Walk the interleaved external stream starting at `offset`, collecting up to
   199→ * `count` items. Chunks are disjoint (each is a different TMDB/Jikan page),
   200→ * so pages stay deterministic within the chunk cache TTL.
   201→ */
   202→async function externalSlice(offset, count) {
   203→  const items = [];
   204→  let k = 1;
   205→  let pos = Math.max(0, offset);
   206→  while (items.length < count && k <= MAX_K) {
   207→    let chunk = [];
   208→    try {
   209→      chunk = await externalChunk(k);
   210→    } catch {
   211→      break;
   212→    }
   213→    if (pos >= chunk.length) {
   214→      pos -= chunk.length;
   215→      k += 1;
   216→      continue;
   217→    }
   218→    while (pos < chunk.length && items.length < count) {
   219→      items.push(chunk[pos]);
   220→      pos += 1;
   221→    }
   222→    if (pos >= chunk.length) {
   223→      pos = 0;
   224→      k += 1;
   225→    }
   226→  }
   227→  return items;
   228→}

   247→async function buildFeedPage(page, limit) {
   248→  const lib = await listContent({ page, limit, sort: 'createdat' });
   249→  const libTotal = lib.total || 0;
   250→  const libTotalPages = Math.max(1, Math.ceil(libTotal / limit));
   251→
   252→  let items = [];
   253→  let sourceMode = 'library';
   254→
   255→  if (page <= libTotalPages) {
   256→    items = (lib.data || []).map(libraryItem);
   257→    if (page === libTotalPages && items.length < limit) {
   258→      // Partial last library page — the head of the external stream fills it,
   259→      // so the external stream is pre-consumed by `limit - items.length`.
   260→      const fill = await externalSlice(0, limit - items.length);
   261→      items = items.concat(fill);
   262→      if (fill.length > 0) sourceMode = 'library+external';
   263→    }
   264→  } else {
   265→    // Purely external page. How much of the external stream was already
   266→    // consumed while filling the (possibly partial) last library page:
   267→    const onLastLibPage = libTotal === 0 ? 0 : libTotal % limit || limit;
   268→    const consumed = libTotal === 0 ? limit : limit - onLastLibPage === limit ? 0 : limit - onLastLibPage;
   269→    const offset = consumed + (page - libTotalPages - 1) * limit;
   270→    items = await externalSlice(offset, limit);
   271→    sourceMode = 'external';
   272→  }
   273→
   274→  const totalPages = Math.min(MAX_PAGES, libTotalPages + 60);
   275→  return {
   276→    ai: true,
   277→    engine: 'ai-infinite-feed',
   278→    page,
   279→    limit,
   280→    totalPages,
   281→    totalItems: totalPages * limit,
   282→    hasNextPage: page < totalPages && items.length > 0,
   283→    hasPrevPage: page > 1,
   284→    sourceMode,
   285→    items,
   286→    generatedAt: new Date().toISOString(),
   287→  };
   288→}

   306→async function taxonomyData() {
   307→  const key = 'aihome:taxonomy';
   308→  const hit = cache.get(key);
   309→  if (hit) return hit;
   310→
   311→  const [lib, extChunk1] = await Promise.all([
   312→    listContent({ page: 1, limit: 100, sort: 'createdat' }),
   313→    externalChunk(1).catch(() => []),
   314→  ]);
   315→
   316→  const genreCounts = new Map();
   317→  const categoryCounts = new Map();
   318→  const decadeCounts = new Map();
   319→  let analyzed = 0;
   320→
   321→  const account = (item, typeOverride) => {
   322→    analyzed += 1;
   323→    const type = typeOverride || item.mediaType || item.type;
   324→    const cat = categoryLabel(type);
   325→    categoryCounts.set(cat, (categoryCounts.get(cat) || 0) + 1);
   326→    for (const g of item.genres || []) {
   327→      const name = String(g).trim();
   328→      if (!name) continue;
   329→      genreCounts.set(name, (genreCounts.get(name) || 0) + 1);
   330→    }
   331→    const year = parseInt(item.year, 10);
   332→    if (Number.isInteger(year) && year > 1900) {
   333→      const decade = `${Math.floor(year / 10) * 10}s`;
   334→      decadeCounts.set(decade, (decadeCounts.get(decade) || 0) + 1);
   335→    }
   336→  };
   337→
   338→  for (const row of lib.data || []) {
   339→    account(libraryItem(row), row.type);
   340→  }
   341→  for (const item of extChunk1) account(item);
   342→
   343→  const payload = {
   344→    ai: true,
   345→    generatedAt: new Date().toISOString(),
   346→    analyzedItems: analyzed,
   347→    sourcesChecked: ['neon-library', 'tmdb-trending', 'jikan-top'],
   348→    genres: [...genreCounts.entries()]
   349→      .map(([name, count]) => ({ name, count }))
   350→      .sort((a, b) => b.count - a.count || a.name.localeCompare(b.name))
   351→      .slice(0, 48),
   352→    categories: [...categoryCounts.entries()]
   353→      .map(([name, count]) => ({ name, count }))
   354→      .sort((a, b) => b.count - a.count || a.name.localeCompare(b.name)),
   355→    decades: [...decadeCounts.entries()]
   356→      .map(([name, count]) => ({ name, count }))
   357→      .sort((a, b) => b.name.localeCompare(a.name)),
   358→  };
   359→  cache.set(key, payload, TAXONOMY_TTL_MS);
   360→  return payload;
   361→}

   367→async function externalStats(since) {
   368→  const key = `aihome:stats:${since || ''}`;
   369→  const hit = cache.get(key);
   370→  if (hit) return hit;
   371→
   372→  const lib = await listContent({ page: 1, limit: 100, sort: 'createdat' });
   373→  const rows = lib.data || [];
   374→
   375→  const bySource = {};
   376→  let postersFromExternal = 0;
   377→  let contentsFromExternal = 0;
   378→  let withGenres = 0;
   379→  let withYear = 0;
   380→  let withDescription = 0;
   381→  const recent = [];
   382→  let newSince = 0;
   383→
   384→  for (const row of rows) {
   385→    const md = row.metadata || {};
   386→    const source = String(md.source || (row.externalid || '').split(':')[0] || 'seed').toLowerCase();
   387→    const isExternal = EXTERNAL_SOURCE_RE.test(source);
   388→    bySource[source] = (bySource[source] || 0) + 1;
   389→    if (isExternal) contentsFromExternal += 1;
   390→    if (isExternal && (row.thumbnail || md.backdrop)) postersFromExternal += 1;
   391→    if (Array.isArray(md.genres) && md.genres.length) withGenres += 1;
   392→    if (md.year) withYear += 1;
   393→    if (md.overview || md.synopsis || md.description) withDescription += 1;
   394→    const created = row.createdat ? new Date(row.createdat) : null;
   395→    if (created && since && created.getTime() > new Date(since).getTime()) newSince += 1;
   396→  }
   397→
   398→  const recentRows = rows.slice(0, 12).map((row) => {
   399→    const md = row.metadata || {};
   400→    return {
   401→      id: row.id,
   402→      title: row.title,
   403→      type: row.type,
   404→      source: String(md.source || (row.externalid || '').split(':')[0] || 'seed'),
   405→      externalid: row.externalid || null,
   406→      hasPoster: Boolean(row.thumbnail || md.backdrop),
   407→      createdat: row.createdat || null,
   408→    };
   409→  });
   410→  recent.push(...recentRows);
   411→
   412→  // Live external catalog sizes (short-cached upstream calls, degrade to null).
   413→  const [trendingSize, tvSize, jikanSize] = await Promise.all([
   414→    tmdbFetch('/trending/movie/week', { page: 1 })
   415→      .then((p) => (typeof p?.total_results === 'number' ? p.total_results : null))
   416→      .catch(() => null),
   417→    tmdbFetch('/trending/tv/week', { page: 1 })
   418→      .then((p) => (typeof p?.total_results === 'number' ? p.total_results : null))
   419→      .catch(() => null),
   420→    jikanFetch('/top/anime', { page: 1, limit: 20 }, 1)
   421→      .then((p) => {
   422→        const items = Array.isArray(p?.data) ? p.data.length : 0;
   423→        return items > 0 ? items * 500 : null; // Jikan exposes ~500 pages
   424→      })
   425→      .catch(() => null),
   426→  ]);
   427→
   428→  const payload = {
   429→    ai: true,
   430→    live: true,
   431→    updatedAt: new Date().toISOString(),
   432→    library: {
   433→      total: lib.total,
   434→      withGenres,
   435→      withYear,
   436→      withDescription,
   437→      bySource,
   438→    },
   439→    posters: { totalInLibrary: postersFromExternal, fromExternal: postersFromExternal },
   440→    contents: { totalInLibrary: lib.total, fromExternal: contentsFromExternal },
   441→    externalCatalog: {
   442→      tmdbTrendingMovies: trendingSize,
   443→      tmdbTrendingTv: tvSize,
   444→      jikanTop: jikanSize,
   445→      checkedAt: new Date().toISOString(),
   446→    },
   447→    recent,
   448→    delta: since ? { since, newItems: newSince } : null,
   449→  };
   450→  cache.set(key, payload, STATS_TTL_MS);
   451→  return payload;
   452→}

   506→async function enrichJikan(id) {
   507→  const payload = await jikanFetch(`/anime/${id}`, {}, 2);
   508→  const item = Array.isArray(payload?.data) ? payload.data[0] : payload?.data;

   510→  const embed = item.trailer?.embed_url || item.trailer?.url || '';
   511→  const trailerKey = /embed\/([A-Za-z0-9_-]{6,})/.exec(embed)?.[1] || null;
   512→  const entry = {
   513→    key: `jikan:anime:${id}`,
   514→    kind: 'jikan',
   515→    mediaType: 'anime',
   516→    id: item.mal_id,
   517→    title: item.title || item.title_english || 'Untitled',
   518→    year: item.year || (item.aired?.from ? String(item.aired.from).slice(0, 4) : null),
   519→    genres: [
   520→      ...(item.genres || []),
   521→      ...(item.themes || []),
   522→      ...(item.demographics || []),
   523→    ]
   524→      .map((g) => g.name)
   525→      .filter(Boolean),
   526→    overview: item.synopsis || '',
   527→    rating: item.score ?? null,
   528→    cast: [], // Jikan does not expose voice-actor cast in the detail payload
   529→    studios: (item.studios || []).map((s) => s.name).filter(Boolean),
   530→    network: null,
   531→    collection: null,
   532→    franchise: item.franchise ? null : null, // franchise derived from relations at browse time
   533→    trailerKey,
   534→    image: item.images?.jpg?.large_image_url || null,
   535→    fetchedAt: new Date().toISOString(),
   536→  };
   537→  return entry;

   540→/** Extract a pending-enrichment queue from the library + trending head. */
   541→async function enrichmentQueue() {
   542→  const lib = await listContent({ page: 1, limit: 100, sort: 'createdat' });
   543→  const queue = [];
   544→  for (const row of lib.data || []) {
   545→    const ext = String(row.externalid || '');
   546→    let m;
   547→    if ((m = /^tmdb:(movie|tv):(\d+)$/.exec(ext))) queue.push({ kind: 'tmdb', mediaType: m[1], id: m[2] });
   548→    else if ((m = /^jikan:anime:(\d+)$/.exec(ext))) queue.push({ kind: 'jikan', id: m[1] });
   549→  }
   550→  // Rotating live pool: head of the trending chunk (cheap, already cached).
   551→  try {
   552→    const chunk = await externalChunk(1);
   553→    chunk.slice(0, 14).forEach((item) => {
   554→      if (item.mediaType === 'anime') queue.push({ kind: 'jikan', id: String(item.id) });
   555→      else queue.push({ kind: 'tmdb', mediaType: item.mediaType, id: String(item.id) });
   556→    });
   557→  } catch {
   558→    /* chunk unavailable — library-only pool */
   559→  }
   560→  // De-duplicate queue entries by their future store key.
   561→  const seen = new Set();
   562→  return queue.filter((q) => {
   563→    const key = q.kind === 'tmdb' ? `tmdb:${q.mediaType}:${q.id}` : `jikan:anime:${q.id}`;
   564→    if (seen.has(key)) return false;
   565→    seen.add(key);
   566→    return true;
   567→  });
   568→}

   570→let enrichRunning = false;
   571→
   572→/** Grow the enrichment pool by `batch` items (background-safe). */
   573→async function enrichNext(batch) {
   574→  if (enrichRunning) return { running: true, enriched: 0, skipped: 'already-running' };
   575→  enrichRunning = true;
   576→  let enriched = 0;
   577→  const failed = 0;
   578→  try {
   579→    const store = readEnrichment();
   580→    const now = Date.now();
   581→    const queue = await enrichmentQueue();
   582→    const pending = queue.filter((q) => {
   583→      const key = q.kind === 'tmdb' ? `tmdb:${q.mediaType}:${q.id}` : `jikan:anime:${q.id}`;
   584→      const entry = store.entries[key];
   585→      if (!entry) return true;
   586→      return now - new Date(entry.fetchedAt).getTime() > ENRICH_TTL_MS;
   587→    });
   588→    let slot = 0;
   589→    for (const q of pending) {
   590→      if (enriched >= batch) break;
   591→      try {
   592→        let entry;
   593→        if (q.kind === 'tmdb') entry = await enrichTmdb(q.mediaType, q.id);
   594→        else entry = await enrichJikan(q.id);
   595→        store.entries[entry.key] = entry;
   596→        // TMDB/Jikan trailers flow straight into the trailer store for free.
   597→        if (entry.trailerKey) rememberTrailer(entry.key, {
   598→          trailerKey: entry.trailerKey,
   599→          provider: entry.kind === 'tmdb' ? 'tmdb-videos' : 'jikan',
   600→          title: entry.title,
   601→        });
   602→        enriched += 1;
   603→        // Persist progress as we go (crash-safe).
   604→        if (enriched % 3 === 0) writeEnrichment(store);
   605→        await new Promise((r) => setTimeout(r, 250)); // gentle on the providers
   606→      } catch (err) {
   607→        logger.debug(`[ai-home] enrich failed ${q.kind}:${q.id}: ${err.message}`);
   608→      }
   609→      slot += 1;
   610→      if (slot > batch * 3) break; // don't churn on persistent failures
   611→    }
   612→    writeEnrichment(store);
   613→  } finally {
   614→    enrichRunning = false;
   615→  }
   616→  return { running: false, enriched, failed };
   617→}

   621→async function metadataData() {
   622→  const key = 'aihome:metadata';
   623→  const hit = cache.get(key);
   624→  if (hit) return hit;
   625→
   626→  const [lib, franchiseHub, trilogyHub] = await Promise.all([
   627→    listContent({ page: 1, limit: 100, sort: 'createdat' }),
   628→    getFranchiseHub().catch(() => ({ data: [] })),
   629→    getTrilogyHub().catch(() => ({ data: [] })),
   630→  ]);
   631→  const enrichment = readEnrichment().entries || {};
   632→  const rows = lib.data || [];
   633→
   634→  const dims = {
   635→    year: new Map(),
   636→    genre: new Map(),
   637→    category: new Map(),
   638→    studio: new Map(),
   639→    actor: new Map(),
   640→    collection: new Map(),
   641→  };
   642→
   643→  const bump = (map, value, inc = 1) => {
   644→    const name = String(value || '').trim();
   645→    if (!name) return;
   646→    map.set(name, (map.get(name) || 0) + inc);
   647→  };
   648→
   649→  let withYear = 0;
   650→  let withGenres = 0;
   651→  let withDescription = 0;
   652→  let withStudio = 0;
   653→  let withCast = 0;
   654→  let withCollection = 0;
   655→
   656→  for (const row of rows) {
   657→    const md = row.metadata || {};
   658→    const ext = String(row.externalid || '');
   659→    const entry = enrichment[ext]; // enrichment by exact externalid key
   660→    bump(dims.category, categoryLabel(row.type));
   661→    const year = md.year || entry?.year;
   662→    if (year) {
   663→      bump(dims.year, String(year));
   664→      withYear += 1;
   665→    }
   666→    const genres = Array.isArray(md.genres) && md.genres.length ? md.genres : entry?.genres || [];
   667→    if (genres.length) withGenres += 1;
   668→    for (const g of genres) bump(dims.genre, g);
   669→    if (md.overview || md.synopsis || md.description || entry?.overview) withDescription += 1;
   670→    const studios = entry?.studios || [];
   671→    if (studios.length) withStudio += 1;
   672→    for (const s of studios) bump(dims.studio, s);
   673→    const cast = entry?.cast || [];
   674→    if (cast.length) withCast += 1;
   675→    for (const a of cast) bump(dims.actor, a);
   676→    if (entry?.collection) {
   677→      bump(dims.collection, entry.collection);
   678→      withCollection += 1;
   679→    }
   680→  }
   681→
   682→  // Enrichment-only entries (trending pool) contribute studio/actor metadata.
   683→  for (const entry of Object.values(enrichment)) {
   684→    for (const s of entry.studios || []) bump(dims.studio, s, 0); // no count inflation
   685→    for (const a of entry.cast || []) bump(dims.actor, a, 0);
   686→    if (entry.collection) bump(dims.collection, entry.collection, 0);
   687→  }
   688→
   689→  const trailers = readTrailers();
   690→
   691→  const payload = {
   692→    ai: true,
   693→    generatedAt: new Date().toISOString(),
   694→    enriching: {
   695→      pool: Object.keys(enrichment).length,
   696→      enriched: Object.values(enrichment).filter((e) => e && e.fetchedAt).length,
   697→      ttlHours: ENRICH_TTL_MS / 3600000,
   698→    },
   699→    coverage: {
   700→      titles: lib.total,
   701→      withYear,
   702→      withGenres,
   703→      withDescription,
   704→      withStudio,
   705→      withCast,
   706→      withCollection,
   707→      withTrailer: Object.keys(trailers.entries).length,
   708→    },
   709→    dimensions: {
   710→      year: [...dims.year.entries()].map(([value, count]) => ({ value, count })).sort((a, b) => b.value.localeCompare(a.value)),
   711→      genre: [...dims.genre.entries()].map(([value, count]) => ({ value, count })).sort((a, b) => b.count - a.count || a.value.localeCompare(b.value)),
   712→      category: [...dims.category.entries()].map(([value, count]) => ({ value, count })).sort((a, b) => b.count - a.count || a.value.localeCompare(b.value)),
   713→      studio: [...dims.studio.entries()].map(([value, count]) => ({ value, count })).sort((a, b) => b.count - a.count || a.value.localeCompare(b.value)),
   714→      actor: [...dims.actor.entries()].map(([value, count]) => ({ value, count })).sort((a, b) => b.count - a.count || a.value.localeCompare(b.value)).slice(0, 60),
   715→      collection: [...dims.collection.entries()].map(([value, count]) => ({ value, count })).sort((a, b) => b.count - a.count || a.value.localeCompare(b.value)),
   716→      franchise: (franchiseHub.data || []).map((c) => ({ value: c.name, count: c.partCount })),
   717→      trilogy: (trilogyHub.data || [])
   718→        .filter((c) => c.partCount === 3)
   719→        .map((c) => ({ value: c.name, count: c.partCount })),
   720→    },
   721→  };
   722→  cache.set(key, payload, METADATA_TTL_MS);
   723→  return payload;
   724→}

   765→async function resolveTrailer(rawKey, title) {
   766→  const key = String(rawKey || '').trim();
   767→  if (!key) return { error: 'Missing key parameter' };
   768→
   769→  const store = readTrailers();
   770→  const cached = store.entries[key];
   771→  const fresh =
   772→    cached && Date.now() - new Date(cached.fetchedAt).getTime() < TRAILER_TTL_MS;
   773→  if (fresh) {
   774→    return {
   775→      key,
   776→      trailerKey: cached.trailerKey,
   777→      provider: cached.provider,
   778→      title: cached.title || title || null,
   779→      embedUrl: cached.trailerKey ? `https://www.youtube.com/embed/${cached.trailerKey}?autoplay=1&mute=1&controls=0&rel=0` : null,
   780→      cached: true,
   781→      fetchedAt: cached.fetchedAt,
   782→    };
   783→  }
   784→
   785→  let trailerKey = null;
   786→  let provider = null;
   787→  let resolvedTitle = title || null;
   788→
   789→  // 1. Enrichment store (enrichment runs ahead of the cursor).
   790→  const enrichment = readEnrichment().entries || {};
   791→  if (enrichment[key]?.trailerKey) {
   792→    trailerKey = enrichment[key].trailerKey;
   793→    provider = enrichment[key].kind === 'tmdb' ? 'tmdb-videos' : 'jikan';
   794→    resolvedTitle = resolvedTitle || enrichment[key].title;
   795→  }
   796→
   797→  // 2. Direct hub resolution.
   798→  if (!trailerKey) {
   799→    let m;
   800→    if ((m = /^tmdb:(movie|tv):(\d+)$/.exec(key))) {
   801→      try {
   802→        const videos = await tmdbFetch(`/${m[1]}/${m[2]}/videos`, {});
   803→        trailerKey = firstTrailerKeyOf(videos);
   804→        if (trailerKey) provider = 'tmdb-videos';
   805→      } catch (err) {
   806→        if (!(err instanceof ProviderError && err.status === 404)) logger.debug(`[ai-home] tmdb videos failed ${key}: ${err.message}`);
   807→      }
   808→    } else if ((m = /^jikan:anime:(\d+)$/.exec(key))) {
   809→      try {
   810→        const entry = await enrichJikan(m[1]);
   811→        trailerKey = entry.trailerKey;
   812→        if (trailerKey) {
   813→          provider = 'jikan';
   814→          resolvedTitle = resolvedTitle || entry.title;
   815→          const es = readEnrichment();
   816→          es.entries[entry.key] = entry;
   817→          writeEnrichment(es);
   818→        }
   819→      } catch (err) {
   820→        logger.debug(`[ai-home] jikan trailer failed ${key}: ${err.message}`);
   821→      }
   822→    } else if ((m = /^db:(\d+)$/.exec(key))) {
   823→      // Library row: follow its externalid, then fall back to a title search.
   824→      try {
   825→        const row = await getContentById(parseInt(m[1], 10));
   826→        resolvedTitle = resolvedTitle || row?.title || null;
   827→        const ext = String(row?.externalid || '');
   828→        if (/^tmdb:(movie|tv):\d+$/.test(ext)) {
   829→          const sub = await resolveTrailer(ext, row?.title);
   830→          if (sub?.trailerKey && !sub?.error) {
   831→            return { ...sub, key, cached: false };
   832→          }
   833→        }
   834→      } catch {
   835→        /* row missing — fall through to YouTube search */
   836→      }
   837→    }
   838→  }
   839→
   840→  // 3. YouTube Data API search fallback (works for any title).
   841→  if (!trailerKey && (title || resolvedTitle)) {
   842→    try {
   843→      trailerKey = await searchYouTubeId(resolvedTitle || title);
   844→      if (trailerKey) provider = 'youtube-search';
   845→    } catch (err) {
   846→      logger.debug(`[ai-home] youtube search failed: ${err.message}`);
   847→    }
   848→  }
   849→
   850→  if (!trailerKey) {
   851→    return {
   852→      key,
   853→      trailerKey: null,
   854→      provider: null,
   855→      title: resolvedTitle,
   856→      embedUrl: null,
   857→      cached: false,
   858→      fetchedAt: new Date().toISOString(),
   859→      note: 'No official trailer found in TMDB, Jikan or YouTube',
   860→    };
   861→  }
   862→
   863→  rememberTrailer(key, { trailerKey, provider, title: resolvedTitle });
   864→
   865→  return {
   866→    key,
   867→    trailerKey,
   868→    provider,
   869→    title: resolvedTitle,
   870→    embedUrl: `https://www.youtube.com/embed/${trailerKey}?autoplay=1&mute=1&controls=0&rel=0`,
   871→    cached: false,
   872→    fetchedAt: new Date().toISOString(),
   873→  };
   874→}

   912→async function browse(dim, value, page, limit) {
   913→  if (!BROWSE_DIMS.has(dim)) return { error: `Unknown dimension '${dim}'` };
   914→  const needle = String(value || '').trim().toLowerCase();
   915→  if (!needle) return { error: 'Missing value parameter' };
   916→
   917→  const items = [];
   918→  const seen = new Set();
   919→  const push = (item) => {
   920→    if (!item) return;
   921→    const key = `${item.mediaType}:${item.id}`;
   922→    if (seen.has(key)) return;
   923→    seen.add(key);
   924→    items.push(item);
   925→  };
   926→
   927→  const [lib, franchiseHub, trilogyHub] = await Promise.all([
   928→    listContent({ page: 1, limit: 100, sort: 'createdat' }),
   929→    dim === 'franchise' ? getFranchiseHub().catch(() => ({ data: [] })) : Promise.resolve({ data: [] }),
   930→    dim === 'trilogy' ? getTrilogyHub().catch(() => ({ data: [] })) : Promise.resolve({ data: [] }),
   931→  ]);
   932→  const enrichment = readEnrichment().entries || {};
   933→
   934→  // Franchise / trilogy browse: the real TMDB collection parts, in order.
   935→  if (dim === 'franchise' || dim === 'trilogy') {
   936→    const hub = dim === 'franchise' ? franchiseHub : trilogyHub;
   937→    const match = (hub.data || []).find(
   938→      (c) => c.name.toLowerCase() === needle || c.name.toLowerCase().startsWith(needle),
   939→    );
   940→    if (match) {
   941→      for (const part of match.parts || []) push(part);
   942→    }
   943→  }
   944→
   945→  for (const row of lib.data || []) {
   946→    const md = row.metadata || {};
   947→    const entry = enrichment[String(row.externalid || '')];
   948→    let hit = false;
   949→    switch (dim) {
   950→      case 'year':
   951→        hit = String(md.year ?? entry?.year ?? '') === needle;
   952→        break;
   953→      case 'genre':
   954→        hit = [...(md.genres || []), ...(entry?.genres || [])].some((g) => String(g).toLowerCase() === needle);
   955→        break;
   956→      case 'category':
   957→        hit = categoryLabel(row.type).toLowerCase() === needle || String(row.type).toLowerCase() === needle;
   958→        break;
   959→      case 'studio':
   960→        hit = (entry?.studios || []).some((s) => s.toLowerCase() === needle);
   961→        break;
   962→      case 'actor':
   963→        hit = (entry?.cast || []).some((a) => a.toLowerCase() === needle);
   964→        break;
   965→      case 'collection':
   966→        hit = String(entry?.collection || '').toLowerCase() === needle;
   967→        break;
   968→      default:
   969→        hit = false;
   970→    }
   971→    if (hit) push(libraryItem(row));
   972→  }
   973→
   974→  // Enrichment pool (library-linked + trending) — richer than row metadata.
   975→  if (dim !== 'franchise' && dim !== 'trilogy') {
   976→    for (const entry of Object.values(enrichment)) {
   977→      let hit = false;
   978→      switch (dim) {
   979→        case 'year':
   980→          hit = String(entry.year ?? '') === needle;
   981→          break;
   982→        case 'genre':
   983→          hit = (entry.genres || []).some((g) => g.toLowerCase() === needle);
   984→          break;
   985→        case 'category':
   986→          hit = categoryLabel(entry.mediaType).toLowerCase() === needle;
   987→          break;
   988→        case 'studio':
   989→          hit = (entry.studios || []).some((s) => s.toLowerCase() === needle);
   990→          break;
   991→        case 'actor':
   992→          hit = (entry.cast || []).some((a) => a.toLowerCase() === needle);
   993→          break;
   994→        case 'collection':
   995→          hit = String(entry.collection || '').toLowerCase() === needle;
   996→          break;
   997→        default:
   998→          hit = false;
   999→      }
  1000→      if (hit) push(enrichmentToItem(entry));
  1001→    }
  1002→  }
  1003→
  1004→  // Live external feeds for the crowd dimensions (year / genre / category).
  1005→  if (items.length < limit * 3 && (dim === 'year' || dim === 'genre' || dim === 'category')) {
  1006→    for (let k = 1; k <= 2; k += 1) {
  1007→      let chunk = [];
  1008→      try {
  1009→        chunk = await externalChunk(k);
  1010→      } catch {
  1011→        break;
  1012→      }
  1013→      for (const item of chunk) {
  1014→        let hit = false;
  1015→        if (dim === 'year') hit = String(item.year ?? '') === needle;
  1016→        else if (dim === 'genre') hit = (item.genres || []).some((g) => g.toLowerCase() === needle);
  1017→        else hit = categoryLabel(item.mediaType).toLowerCase() === needle;
  1018→        if (hit) push(item);
  1019→      }
  1020→    }
  1021→  }
  1022→
  1023→  const total = items.length;
  1024→  const totalPages = Math.max(1, Math.ceil(total / limit));
  1025→  const start = (page - 1) * limit;
  1026→  const sorted =
  1027→    dim === 'year'
  1028→      ? items.sort((a, b) => Number(b.rating || 0) - Number(a.rating || 0))
  1029→      : items;
  1030→

  1054→async function llmNarrate(prompt) {
  1055→  try {
  1056→    const zai = await getZai();
  1057→    const res = await Promise.race([
  1058→      zai.chat.completions.create({
  1059→        messages: [{ role: 'user', content: prompt }],
  1060→        thinking: { type: 'disabled' },
  1061→      }),
  1062→      new Promise((_, reject) => setTimeout(() => reject(new Error('llm-timeout')), 12000)),
  1063→    ]);
  1064→    return res?.choices?.[0]?.message?.content?.trim() || null;
  1065→  } catch (err) {
  1066→    logger.debug(`[ai-home] llm narration unavailable: ${err.message}`);
  1067→    return null;
  1068→  }
  1069→}

  1071→async function taxonomyNarration() {
  1072→  const key = 'aihome:narration';
  1073→  const hit = cache.get(key);
  1074→  if (hit) return hit;
  1075→  const tax = await taxonomyData();
  1076→  const top = tax.genres.slice(0, 8).map((g) => `${g.name} (${g.count})`).join(', ');
  1077→  const cats = tax.categories.slice(0, 5).map((c) => `${c.name} (${c.count})`).join(', ');
  1078→  const text = await llmNarrate(
  1079→    `Ești motorul AI de clasificare al unei platforme de streaming. Ai analizat ${tax.analyzedItems} conținuturi și ai detectat automat genurile: ${top}; categoriile: ${cats}. Răspunde în română, maxim 2 propoziții, fără listări, explicând pe scurt ce a detectat AI-ul și că taxonomia se actualizează automat la fiecare încărcare din surse externe.`,
  1080→  );
  1081→  const payload = { ai: true, provider: text ? 'z-ai' : 'heuristic', narration: text, generatedAt: new Date().toISOString() };
  1082→  if (text) cache.set(key, payload, NARRATION_TTL_MS);
  1083→  return payload;
  1084→}
