    89→function formatUptime(sec: number): string {
    90→  const h = Math.floor(sec / 3600);
    91→  const m = Math.floor((sec % 3600) / 60);
    92→  const s = Math.floor(sec % 60);
    93→  if (h > 0) return `${h}h ${m}m`;
    94→  if (m > 0) return `${m}m ${s}s`;
    95→  return `${s}s`;
    96→}
    97→
    98→function formatBytes(b?: number): string {
    99→  if (!b || b <= 0) return '0 B';
   100→  const units = ['B', 'KB', 'MB', 'GB'];
   101→  let v = b;
   102→  let i = 0;
   103→  while (v >= 1024 && i < units.length - 1) {
   104→    v /= 1024;
   105→    i += 1;
   106→  }
   107→  return `${v >= 100 ? Math.round(v) : v.toFixed(1)} ${units[i]}`;
   108→}
   109→
   110→function formatMs(ms?: number): string {
   111→  if (ms == null) return '—';
   112→  if (ms < 1000) return `${Math.round(ms)}ms`;
   113→  return `${(ms / 1000).toFixed(1)}s`;
   114→}
   115→
   116→function formatDate(iso?: string | null): string {
   117→  if (!iso) return '—';
   118→  const d = new Date(iso);
   119→  if (Number.isNaN(d.getTime())) return '—';
   120→  return d.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
   121→}
   122→
   123→function getErrorMessage(e: unknown): string {
   124→  return e instanceof Error ? e.message : 'Request failed';
   125→}

   127→/** True when an endpoint error means the caller is not an admin (403/401). */
   128→function isForbidden(message: string | null): boolean {
   129→  return Boolean(message && /admin access required|forbidden|status 40[13]/i.test(message));
   130→}
