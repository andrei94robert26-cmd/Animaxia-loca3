    19→    19→function platformUrl(path: string, params?: Record<string, string | number | boolean | undefined | null>): string {
    20→    20→  const search = new URLSearchParams();
    21→    21→  search.set('XTransformPort', String(BACKEND_PORT));
    22→    22→  if (params) {
    23→    23→    for (const [key, value] of Object.entries(params)) {
    24→    24→      if (value !== undefined && value !== null && value !== '') search.set(key, String(value));
    25→    25→    }
    26→    26→  }
    27→    27→  return `${path}?${search.toString()}`;
    28→    28→}

    46→    46→const REFRESH_KEY = 'animaxia.refreshToken';
    47→    47→
    48→    48→export const tokenStore = {
    49→    49→  get access(): string | null {
    50→    50→    try {
    51→    51→      return window.localStorage.getItem(ACCESS_KEY);
    52→    52→    } catch {
    53→    53→      return null;
    54→    54→    }
    55→    55→  },
    56→    56→  get refresh(): string | null {
    57→    57→    try {
    58→    58→      return window.localStorage.getItem(REFRESH_KEY);
    59→    59→    } catch {
    60→    60→      return null;
    61→    61→    }
    62→    62→  },
    63→    63→  set(access: string, refresh: string) {
    64→    64→    try {
    65→    65→      window.localStorage.setItem(ACCESS_KEY, access);
    66→    66→      window.localStorage.setItem(REFRESH_KEY, refresh);
    67→    67→    } catch {
    68→    68→      /* private mode — session will not persist */
    69→    69→    }
    70→    70→  },
    71→    71→  clear() {
    72→    72→    try {
    73→    73→      window.localStorage.removeItem(ACCESS_KEY);
    74→    74→      window.localStorage.removeItem(REFRESH_KEY);
    75→    75→    } catch {
    76→    76→      /* noop */
    77→    77→    }
    78→    78→  },
    79→    79→};

    82→    82→
    83→    83→async function rawRequest<T>(path: string, init?: RequestInit): Promise<T> {
    84→    84→  let res: Response;
    85→    85→  try {
    86→    86→    res = await fetch(path, {
    87→    87→      ...init,
    88→    88→      headers: {
    89→    89→        'Content-Type': 'application/json',
    90→    90→        ...(init?.headers as Record<string, string> | undefined),
    91→    91→      },
    92→    92→    });
    93→    93→  } catch {
    94→    94→    throw new PlatformError('Cannot reach the Animaxia backend — it may be restarting', 0);
    95→    95→  }
    96→    96→  let body: unknown = null;
    97→    97→  try {
    98→    98→    body = await res.json();
    99→    99→  } catch {
   100→   100→    body = null;
   101→   101→  }
   102→   102→  if (!res.ok) {
   103→   103→    const b = (body as { error?: string; upgrade?: boolean; code?: string } | null) ?? {};
   104→   104→    throw new PlatformError(b.error || `Request failed with status ${res.status}`, res.status, b.upgrade, b.code);
   105→   105→  }
   106→   106→  return body as T;
   107→   107→}

   109→   109→/** Authenticated request with one automatic token-refresh retry on 401. */
   110→   110→export async function authRequest<T>(path: string, init?: RequestInit): Promise<T> {
   111→   111→  const access = tokenStore.access;
   112→   112→  const doFetch = () =>
   113→   113→    rawRequest<T>(path, {
   114→   114→      ...init,
   115→   115→      headers: { ...(init?.headers as Record<string, string> | undefined), ...(access ? { Authorization: `Bearer ${access}` } : {}) },
   116→   116→    });
   117→   117→  try {
   118→   118→    return await doFetch();
   119→   119→  } catch (err) {
   120→   120→    if (err instanceof PlatformError && err.status === 401 && tokenStore.refresh) {
   121→   121→      const refreshed = await tryRefresh();
   122→   122→      if (refreshed) {
   123→   123→        return authRequest<T>(path, init); // retry with the new token
   124→   124→      }
   125→   125→    }
   126→   126→    throw err;
   127→   127→  }
   128→   128→}
   129→   129→
   130→   130→/** Attempt a refresh-token rotation. Returns true on success. */
   131→   131→export async function tryRefresh(): Promise<boolean> {
   132→   132→  const refresh = tokenStore.refresh;
   133→   133→  if (!refresh) return false;
   134→   134→  try {
   135→   135→    const res = await rawRequest<{ ok: true; accessToken: string; refreshToken: string }>('/api/auth/refresh?XTransformPort=4000', {
   136→   136→      method: 'POST',
   137→   137→      body: JSON.stringify({ refreshToken: refresh }),
   138→   138→    });
   139→   139→    tokenStore.set(res.accessToken, res.refreshToken);
   140→   140→    return true;
   141→   141→  } catch {
   142→   142→    tokenStore.clear();
   143→   143→    return false;
   144→   144→  }
   145→   145→}
