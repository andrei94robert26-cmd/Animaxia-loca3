   169→/* ----------------------------- usage bars --------------------------------- */
   170→
   171→function UsageBar({ metric, total, cap }: { metric: string; total: number; cap?: number }) {
   172→  const label = METRIC_LABELS[metric] ?? metric;
   173→  const pct = cap && cap > 0 ? Math.min(100, Math.round((total / cap) * 100)) : null;
   174→  return (
   175→    <div className="min-w-[150px] flex-1">
   176→      <div className="flex items-baseline justify-between gap-2">
   177→        <span className="text-[11px] font-medium text-zinc-400">{label}</span>
   178→        <span className="font-mono text-[11px] text-zinc-300">
   179→          {fmtNum(total)}
   180→          {pct != null && <span className="text-zinc-500"> / {fmtNum(cap)} ({pct}%)</span>}
   181→        </span>
   182→      </div>
   183→      <div className="mt-1 h-1.5 overflow-hidden rounded-full bg-white/10">
   184→        <div
   185→          className={cn('h-full rounded-full', (pct ?? 0) > 85 ? 'bg-rose-500' : (pct ?? 10) > 60 ? 'bg-amber-400' : 'bg-emerald-500')}
   186→          style={{ width: `${pct != null ? Math.max(3, pct) : Math.min(100, Math.max(4, Math.log10(total + 1) * 18))}%` }}
   187→        />
   188→      </div>
   189→    </div>
   190→  );
   191→}

   281→const LIMIT_FIELDS: { key: keyof TenantLimits; label: string }[] = [
   282→  { key: 'maxusers', label: 'Max users' },
   283→  { key: 'aiperday', label: 'AI / day' },
   284→  { key: 'ratelimitpermin', label: 'Rate / min' },
   285→  { key: 'vectorcollections', label: 'Vector collections' },
   286→  { key: 'storagegb', label: 'Storage GB' },
   287→];
   288→
   289→function LimitsEditor({ tenant, onSaved }: { tenant: Tenant; onSaved: () => void }) {
   290→  const [draft, setDraft] = useState<Record<string, string>>(
   291→    Object.fromEntries(LIMIT_FIELDS.map((f) => [f.key, String(tenant.limits[f.key] ?? 0)])),
   292→  );
   293→  const [busy, setBusy] = useState(false);
   294→
   295→  const save = useCallback(async () => {
   296→    setBusy(true);
   297→    try {
   298→      const body: Record<string, number> = {};
   299→      for (const f of LIMIT_FIELDS) {
   300→        const v = Number(draft[f.key]);
   301→        if (!Number.isFinite(v) || v < 0) throw new Error(`${f.label} must be a non-negative number`);
   302→        body[f.key] = Math.floor(v);
   303→      }
   304→      await tenantApi.updateLimits(tenant.slug, body as Partial<TenantLimits>);
   305→      toast.success(`Limits updated for '${tenant.slug}'`);
   306→      onSaved();
   307→    } catch (err) {
   308→      toast.error(err instanceof Error ? err.message : 'Failed to update limits');
   309→    } finally {
   310→      setBusy(false);
   311→    }
   312→  }, [draft, tenant.slug, onSaved]);
   313→
   314→  return (
   315→    <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-6">
   316→      {LIMIT_FIELDS.map((f) => (
   317→        <div key={f.key} className="grid gap-1.5">
   318→          <Label htmlFor={`limit-${f.key}`} className="text-[11px] text-zinc-400">{f.label}</Label>
   319→          <Input
   320→            id={`limit-${f.key}`}
   321→            inputMode="numeric"
   322→            value={draft[f.key] ?? '0'}
   323→            onChange={(e) => setDraft((d) => ({ ...d, [f.key]: e.target.value.replace(/[^0-9]/g, '') }))}
   324→            className="border-white/15 bg-white/5 font-mono text-sm"
   325→          />
   326→        </div>
   327→      ))}
   328→      <div className="flex items-end">
   329→        <Button size="sm" variant="outline" onClick={save} disabled={busy} className="border-white/15 bg-white/5 font-semibold hover:bg-white/10">
   330→          {busy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Check className="h-4 w-4" aria-hidden />}
   331→          Save limits
   332→        </Button>
   333→      </div>
   334→    </div>
   335→  );
   336→}

   340→function MemberList({ slug, members, onChanged }: { slug: string; members: TenantMember[]; onChanged: () => void }) {
   341→  const [email, setEmail] = useState('');
   342→  const [role, setRole] = useState('member');
   343→  const [busy, setBusy] = useState(false);
   344→
   345→  const add = useCallback(async () => {
   346→    if (!email.includes('@')) {
   347→      toast.error('Enter a valid account email');
   348→      return;
   349→    }
   350→    setBusy(true);
   351→    try {
   352→      await tenantApi.addMember(slug, email.trim().toLowerCase(), role);
   353→      toast.success(`${email} added to '${slug}' as ${role}`);
   354→      setEmail('');
   355→      onChanged();
   356→    } catch (err) {
   357→      toast.error(err instanceof Error ? err.message : 'Failed to add member');
   358→    } finally {
   359→      setBusy(false);
   360→    }
   361→  }, [slug, email, role, onChanged]);
   362→
   363→  const remove = useCallback(async (userid: number, memberEmail: string) => {
   364→    setBusy(true);
   365→    try {
   366→      await tenantApi.removeMember(slug, userid);
   367→      toast.success(`${memberEmail} removed from '${slug}'`);
   368→      onChanged();
   369→    } catch (err) {
   370→      toast.error(err instanceof Error ? err.message : 'Failed to remove member');
   371→    } finally {
   372→      setBusy(false);
   373→    }
   374→  }, [slug, onChanged]);
   375→
   376→  return (
   377→    <div className="space-y-3">
   378→      <div className="flex flex-wrap gap-2">
   379→        <Input
   380→          placeholder="account email…"
   381→          value={email}
   382→          onChange={(e) => setEmail(e.target.value)}
   383→          className="w-56 border-white/15 bg-white/5"
   384→          aria-label={`Add member to ${slug}`}
   385→        />
   386→        <Select value={role} onValueChange={setRole}>
   387→          <SelectTrigger className="w-[120px] border-white/15 bg-white/5" aria-label="Role">
   388→            <SelectValue />
   389→          </SelectTrigger>
   390→          <SelectContent>
   391→            <SelectItem value="owner">owner</SelectItem>
   392→            <SelectItem value="admin">admin</SelectItem>
   393→            <SelectItem value="member">member</SelectItem>
   394→          </SelectContent>
   395→        </Select>
   396→        <Button size="sm" onClick={add} disabled={busy} className="font-semibold">
   397→          {busy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Plus className="h-4 w-4" aria-hidden />}
   398→          Add
   399→        </Button>
   400→      </div>

   402→        <p className="py-3 text-sm text-zinc-500">No members yet — the tenant is accessible only via the X-Animaxia-Tenant header.</p>

   404→        <ul className="max-h-48 divide-y divide-white/5 overflow-y-auto rounded-xl border border-white/10">
   405→          {members.map((m) => (
   406→            <li key={m.userid} className="flex items-center justify-between gap-3 px-4 py-2.5">
   407→              <div className="min-w-0">
   408→                <p className="truncate text-sm text-zinc-200">{m.email}</p>

   411→              <div className="flex shrink-0 items-center gap-2">
   412→                <span className="rounded-full border border-white/10 bg-white/5 px-2 py-0.5 font-mono text-[10px] text-zinc-400">{m.role}</span>
   413→                <Button
   414→                  size="icon"
   415→                  variant="ghost"
   416→                  className="h-7 w-7 text-zinc-500 hover:bg-rose-500/10 hover:text-rose-400"
   417→                  onClick={() => remove(m.userid, m.email)}

   541→
   542→        {/* limits */}
   543→        <section aria-label="Tenant limits">
   544→          <h4 className="mb-2 text-[11px] font-semibold uppercase tracking-widest text-zinc-500">Enforceable limits</h4>

   546→        </section>
   547→
   548→        {/* members */}
   549→        <section aria-label="Tenant members">

   551→            <Users className="h-3.5 w-3.5" aria-hidden /> Members

   555→      </CardContent>
   556→    </Card>
   557→  );

   562→function TenantsTable({ tenants, loading, error, onRetry, onChanged }: {
   563→  tenants: Tenant[];
   564→  loading: boolean;
   565→  error: string | null;
   566→  onRetry: () => void;
   567→  onChanged: () => void;
   568→}) {
   569→  const [expanded, setExpanded] = useState<string | null>(null);
   570→
   571→  const rows = useMemo(() => tenants, [tenants]);
   572→
   573→  if (loading) {
   574→    return (
   575→      <div className="space-y-2">
   576→        {Array.from({ length: 3 }).map((_, i) => (
   577→          <Skeleton key={i} className="h-12 w-full rounded-xl" />
   578→        ))}
   579→      </div>
   580→    );
   581→  }
   582→  if (error && !tenants.length) {
   583→    return <ErrorState message={error} onRetry={onRetry} />;
   584→  }
   585→  if (!rows.length) {
   586→    return <EmptyState title="No tenants" message="Create the first tenant to start scoping AI, vectors and edge caches." />;
   587→  }
   588→
   589→  return (
   590→    <div className="space-y-3">
   591→      <div className="overflow-hidden rounded-xl border border-white/10">
   592→        <Table>
   593→          <TableHeader>
   594→            <TableRow className="hover:bg-transparent">
   595→              <TableHead className="w-8" />
   596→              <TableHead>Tenant</TableHead>
   597→              <TableHead>Plan</TableHead>
   598→              <TableHead className="text-right">Users</TableHead>
   599→              <TableHead className="text-right">AI (30d)</TableHead>
   600→              <TableHead className="hidden text-right sm:table-cell">Status</TableHead>
   601→            </TableRow>
   602→          </TableHeader>
   603→          <TableBody>
   604→            {rows.map((t) => {
   605→              const isOpen = expanded === t.slug;
   606→              const aiUsage = t.usage?.bymetric?.ai_call?.total ?? 0;
   607→              return (
   608→                <Fragment key={t.slug}>
   609→                  <TableRow
   610→                    className={cn('cursor-pointer', isOpen ? 'bg-white/[0.04]' : 'hover:bg-white/[0.03]')}
   611→                    onClick={() => setExpanded(isOpen ? null : t.slug)}
   612→                  >
   613→                    <TableCell className="w-8 text-zinc-500">
   614→                      {isOpen ? <ChevronDown className="h-4 w-4" aria-hidden /> : <ChevronRight className="h-4 w-4" aria-hidden />}
   615→                    </TableCell>
   616→                    <TableCell>
   617→                      <div className="flex items-center gap-2">
   618→                        <Server className="h-3.5 w-3.5 text-zinc-500" aria-hidden />
   619→                        <div>
   620→                          <p className="font-medium text-zinc-100">{t.name}</p>
   621→                          <p className="font-mono text-[11px] text-zinc-500">{t.slug}</p>
   622→                        </div>
   623→                      </div>
   624→                    </TableCell>
   625→                    <TableCell><PlanBadge plan={t.plan} /></TableCell>
   626→                    <TableCell className="text-right font-mono text-sm text-zinc-300">
   627→                      {fmtNum(t.usercount)}
   628→                      <span className="text-zinc-600"> / {fmtNum(t.limits.maxusers)}</span>
   629→                    </TableCell>
   630→                    <TableCell className="text-right font-mono text-sm text-zinc-300">
   631→                      {fmtNum(aiUsage)}
   632→                      <span className="text-zinc-600"> / {fmtNum(t.limits.aiperday * 30)}</span>
   633→                    </TableCell>
   634→                    <TableCell className="hidden sm:table-cell">
   635→                      <span className={cn('rounded-full border px-2 py-0.5 font-mono text-[10px]',
   636→                        t.status === 'active' ? 'border-emerald-500/30 bg-emerald-500/10 text-emerald-300' : 'border-zinc-500/30 bg-zinc-500/10 text-zinc-400')}>
   637→                        {t.status}
   638→                      </span>
   639→                    </TableCell>
   640→                  </TableRow>
   641→                  {isOpen && (
   642→                    <TableRow className="hover:bg-transparent">
   643→                      <TableCell colSpan={6} className="bg-black/20 p-4">
   644→                        <TenantDetailCard slug={t.slug} onChanged={onChanged} />
   645→                      </TableCell>
   646→                    </TableRow>
   647→                  )}
   648→                </Fragment>
   649→              );
   650→            })}
   651→          </TableBody>
   652→        </Table>
   653→      </div>
   654→    </div>
   655→  );
