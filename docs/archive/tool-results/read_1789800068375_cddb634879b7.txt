    57→    57→function formatBytes(bytes: number): string {
    58→    58→  if (!Number.isFinite(bytes) || bytes <= 0) return '0 B';
    59→    59→  const units = ['B', 'KB', 'MB', 'GB', 'TB'];
    60→    60→  let value = bytes;
    61→    61→  let unit = 0;
    62→    62→  while (value >= 1024 && unit < units.length - 1) {
    63→    63→    value /= 1024;
    64→    64→    unit += 1;
    65→    65→  }
    66→    66→  return `${value.toLocaleString('ro-RO', { maximumFractionDigits: 1 })} ${units[unit]}`;
    67→    67→}
    68→    68→
    69→    69→function formatSignedBytes(bytes: number): string {
    70→    70→  if (!Number.isFinite(bytes) || bytes === 0) return '±0 KB';
    71→    71→  const sign = bytes > 0 ? '+' : '-';
    72→    72→  return `${sign}${formatBytes(Math.abs(bytes))}`;
    73→    73→}
    74→    74→
    75→    75→function formatMs(ms: number): string {
    76→    76→  if (!Number.isFinite(ms)) return '—';
    77→    77→  return `${ms.toLocaleString('ro-RO', { maximumFractionDigits: 2 })}ms`;
    78→    78→}
    79→    79→
    80→    80→function formatUptime(sec: number): string {
    81→    81→  if (!Number.isFinite(sec) || sec < 0) return '—';
    82→    82→  const h = Math.floor(sec / 3600);
    83→    83→  const m = Math.floor((sec % 3600) / 60);
    84→    84→  const s = Math.floor(sec % 60);
    85→    85→  if (h > 0) return `${h}h ${m}m ${s}s`;
    86→    86→  if (m > 0) return `${m}m ${s}s`;
    87→    87→  return `${s}s`;
    88→    88→}
    89→    89→
    90→    90→function formatCount(n: number): string {
    91→    91→  return n.toLocaleString('ro-RO');
    92→    92→}
