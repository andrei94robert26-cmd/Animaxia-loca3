    33→function formatDate(iso?: string | null): string | null {
    34→  if (!iso) return null;
    35→  const d = new Date(iso);
    36→  if (Number.isNaN(d.getTime())) return null;
    37→  return d.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
    38→}
    39→
    40→function hostOf(url?: string): string | null {
    41→  if (!url) return null;
    42→  try {
    43→    return new URL(url).hostname.replace(/^www\./, '');
    44→  } catch {
    45→    return null;
    46→  }
    47→}

    49→function dbSourceLabel(c: DbContent): string {
    50→  const md = c.metadata || {};
    51→  if (typeof md.source === 'string' && md.source) return md.source;
    52→  if (typeof md.siteName === 'string' && md.siteName) return md.siteName;
    53→  const ext = c.externalid || '';
    54→  if (/^https?:\/\//.test(ext)) return hostOf(ext) || 'mined';
    55→  const prefix = ext.split(':')[0];
    56→  return prefix || 'animaxia';
    57→}

   149→export function ContentView({ route }: { route: AppRoute }) {
   150→  const ref = route.params[0] || '';
   151→  const isNumeric = /^\d+$/.test(ref);
   152→  const isExt = ref.startsWith('ext:');
   153→  const [, extType, extId] = isExt ? ref.split(':') : [];
   154→  const extTypeValid = extType === 'anime' || extType === 'movie' || extType === 'tv';
   155→
   156→  const db = useApi(`content-${ref}`, () => api.content(Number(ref)), { enabled: isNumeric });
   157→
   158→  const ext = useApi(
   159→    `ext-${ref}`,
   160→    () =>
   161→      extType === 'anime'
   162→        ? api.animeById(extId as string)
   163→        : extType === 'tv'
   164→          ? api.tvById(extId as string)
   165→          : api.movieById(extId as string),
   166→    { enabled: isExt && extTypeValid && Boolean(extId) },
   167→  );
   168→
   169→  if (isNumeric) {
   170→    /* useApi keeps the previous record while the key changes — only trust data matching the requested id. */
   171→    const current = db.data && db.data.id === Number(ref) ? db.data : null;
   172→    if (db.loading && !current) return <ContentSkeleton />;
   173→    if (db.error && !current) {
   174→      if (isNotFound(db.error)) return <NotFoundPanel message={db.error} />;
   175→      return (
   176→        <Container className="py-20">
   177→          <ErrorState message={db.error} onRetry={db.refetch} />
   178→        </Container>
   179→      );
   180→    }
   181→    if (current) return <DbContentDetail content={current} />;
   182→    return <ContentSkeleton />;
   183→  }
   184→
   185→  if (isExt) {
   186→    if (!extTypeValid || !extId) return <NotFoundPanel message="The external reference in the URL is malformed." />;
   187→    const instant = extItemCache.get(`ext:${extType}:${extId}`);
   188→    const live = ext.data && String(ext.data.data.id) === String(extId) ? ext.data.data : null;
   189→    if (ext.loading && !live && !instant) return <ContentSkeleton />;
   190→    if (ext.error && !live && !instant) {
   191→      if (isNotFound(ext.error)) return <NotFoundPanel message={ext.error} />;
   192→      return (
   193→        <Container className="py-20">
   194→          <ErrorState message={ext.error} onRetry={ext.refetch} />
   195→        </Container>
   196→      );
   197→    }
   198→    return <ExtContentDetail extType={extType} extId={extId} instant={instant} live={live} />;
   199→  }
   200→
   201→  return <NotFoundPanel message="The content reference in the URL is missing or malformed." />;
   202→}

   206→function DbContentDetail({ content }: { content: DbContent }) {
   207→  const md = content.metadata || {};
   208→  const poster = content.thumbnail || null;
   209→  const backdrop = typeof md.backdrop === 'string' && md.backdrop ? md.backdrop : null;

   211→    [md.overview, md.synopsis, md.description].find((v) => typeof v === 'string' && v.trim() !== '') || null;
   212→  const rating = typeof md.rating === 'number' ? md.rating : typeof md.score === 'number' ? md.score : null;
   213→  const year = md.year != null && md.year !== '' ? String(md.year) : null;
   214→  const genres = Array.isArray(md.genres) ? md.genres.map(String) : [];
   215→  const playable = Boolean(md.player?.sources?.length);
   216→  const sourceLabel = dbSourceLabel(content);
   217→  const mined = md.mined;
   218→  const added = formatDate(content.createdat);
   219→
   220→  const runtime = typeof md.runtime === 'number' ? md.runtime : null;
   221→  const episodes = typeof md.episodes === 'number' ? md.episodes : null;
   222→  const status = typeof md.status === 'string' && md.status ? md.status : null;
   223→  const studios = Array.isArray(md.studios) ? md.studios.map(String) : [];
   224→
   225→  const fetchMore = content.type !== 'featured';
   226→  const more = useApi(
   227→    `content-more-${content.type}-${content.id}`,
   228→    () => api.all({ type: content.type, limit: 13 }),
   229→    { enabled: fetchMore },
   230→  );
   231→  const moreItems = (more.data?.data || []).map(dbToItem).filter((i) => i.dbId !== content.id);
   232→
   233→  /* ---- platform: watchlist + view tracking (Task 10-b) ---- */
   234→
   235→  const { user } = useAuth();
   236→  const [wlBusy, setWlBusy] = useState(false);
   237→
   238→  /* ---- FAZA 10: watch party (creates a live room for this title) ---- */
   239→  const [partyBusy, setPartyBusy] = useState(false);
   240→
   241→  async function startWatchParty() {
   242→    if (!user) {
   243→      toast('Sign in to start a watch party');
   244→      navigate('/auth');
   245→      return;
   246→    }
   247→    if (partyBusy) return;
   248→    setPartyBusy(true);
   249→    try {
   250→      const res = await social.createParty({ contentId: content.id });
   251→      toast.success('Watch party started — share the invite code');
   252→      navigate(`/social/party/${res.party.id}`);
   253→    } catch (err) {
   254→      toast.error(err instanceof Error ? err.message : 'Could not start the watch party');
   255→    } finally {
   256→      setPartyBusy(false);
   257→    }
   258→  }
   259→  const watchlist = useApi(`watchlist-${user?.id ?? 'anon'}`, () => platform.watchlist(), {
   260→    enabled: !!user,
   261→  });
   262→  const inWatchlist = Boolean(watchlist.data?.data.some((entry) => entry.content.id === content.id));
   263→
   264→  async function toggleWatchlist() {
   265→    if (!user) {
   266→      toast('Sign in to use your watchlist');
   267→      navigate('/auth');
   268→      return;
   269→    }
   270→    if (wlBusy) return;
   271→    setWlBusy(true);
   272→    try {
   273→      if (!inWatchlist) {
   274→        await platform.addWatchlist(content.id);
   275→        toast.success('Added to watchlist');
   276→      } else {
   277→        await platform.removeWatchlist(content.id);
   278→        toast('Removed from watchlist');
   279→      }
   280→      await watchlist.refetch();
   281→    } catch (err) {
   282→      if (err instanceof PlatformError && err.upgrade) {
   283→        toast.error(err.message, { action: { label: 'Upgrade', onClick: () => navigate('/premium') } });
   284→      } else {
   285→        toast.error(err instanceof Error ? err.message : 'Could not update your watchlist');
   286→      }
   287→    } finally {
   288→      setWlBusy(false);
   289→    }
   290→  }
   291→
   292→  /* Fire-and-forget analytics view event — once per content id per mount. */
   293→  const viewedRef = useRef<number | null>(null);
   294→  useEffect(() => {
   295→    if (viewedRef.current === content.id) return;
   296→    viewedRef.current = content.id;
   297→    platform.recordView(content.id, 'view').catch(() => {
   298→      /* analytics is best-effort */
   299→    });
   300→  }, [content.id]);
   301→
   302→  const heroImage = backdrop || poster;
   303→
   304→  return (
   305→    <article>
   306→      {/* Backdrop hero */}
   307→      <div className="relative h-[38vh] min-h-[260px] w-full">
   308→        {heroImage ? (
   309→          <img src={heroImage} alt="" aria-hidden className="h-full w-full object-cover" />
   310→        ) : (
   311→          <div className="h-full w-full bg-gradient-to-br from-white/[0.08] via-transparent to-transparent" />
   312→        )}
   313→        <div aria-hidden className="absolute inset-0 bg-gradient-to-t from-[#0a0a10] via-[#0a0a10]/40 to-transparent" />
   314→        <div aria-hidden className="absolute inset-0 bg-gradient-to-r from-[#0a0a10]/70 via-[#0a0a10]/10 to-transparent" />
   315→      </div>
   316→
   317→      {/* Info panel */}
   318→      <Container className="relative z-10 -mt-20 sm:-mt-28">
   319→        <div className="flex gap-5 sm:gap-8">
   320→          <div className="w-28 shrink-0 sm:w-44">
   321→            <PosterFrame poster={poster} title={content.title} />
   322→          </div>
   323→          <div className="min-w-0 flex-1 space-y-3 pb-4 pt-1 sm:pt-6">
   324→            <div className="flex flex-wrap items-center gap-2">
   325→              <TypeBadge type={content.type} />
   326→              {rating != null && <RatingBadge value={rating} />}
   327→              {year && <span className="text-xs font-semibold text-zinc-400">{year}</span>}
   328→              <SourceChip label={sourceLabel} />
   329→              {playable && (
   330→                <span className="inline-flex items-center gap-1 rounded-full border border-emerald-500/30 bg-emerald-500/10 px-2.5 py-0.5 text-[11px] font-semibold text-emerald-300">
   331→                  <Play className="h-3 w-3 fill-emerald-300" aria-hidden /> Playable
   332→                </span>
   333→              )}
   334→            </div>
   335→            <h1 className="text-2xl font-black leading-tight tracking-tight text-white sm:text-4xl">
   336→              {content.title}
   337→            </h1>
   338→            {genres.length > 0 && (
   339→              <div className="flex flex-wrap items-center gap-1.5">
   340→                {genres.slice(0, 4).map((g) => (
   341→                  <Chip key={g}>{g}</Chip>
   342→                ))}
   343→                {genres.length > 4 && <Chip>+{genres.length - 4}</Chip>}
   344→              </div>
   345→            )}
   346→            <div className="flex flex-wrap gap-3 pt-1">
   347→              <Button
   348→                size="lg"
   349→                className="bg-rose-600 font-bold text-white hover:bg-rose-500"
   350→                onClick={() => navigate(`/player/${content.id}`)}
   351→              >
   352→                <Play className="fill-white" aria-hidden /> Watch now
   353→              </Button>
   354→              <Button
   355→                size="lg"
   356→                variant="outline"
   357→                aria-pressed={inWatchlist}
   358→                onClick={toggleWatchlist}
   359→                disabled={wlBusy}

   527→function ExtContentDetail({
   528→  extType,
   529→  extId,
   530→  instant,
   531→  live,
   532→}: {
   533→  extType: 'anime' | 'movie' | 'tv';
   534→  extId: string;
   535→  instant?: NormalizedItem;
   536→  live: AnimeItem | MediaItem | null;
   537→}) {
   538→  const [adding, setAdding] = useState(false);
   539→
   540→  const isAnime = extType === 'anime';
   541→  const liveAnime = isAnime && live ? (live as AnimeItem) : null;
   542→  const liveMedia = !isAnime && live ? (live as MediaItem) : null;
   543→
   544→  const title = live?.title || instant?.title || 'Unknown title';
   545→  const poster =
   546→    (isAnime ? liveAnime?.image : liveMedia?.image || liveMedia?.backdrop || null) || instant?.image || null;
   547→  const backdrop = liveMedia?.backdrop || instant?.backdrop || null;
   548→  const overview = (isAnime ? liveAnime?.synopsis : liveMedia?.overview) || instant?.overview || null;
   549→  const rating = (isAnime ? liveAnime?.score : liveMedia?.rating) ?? instant?.rating ?? null;
   550→  const year = live?.year ?? instant?.year ?? null;
   551→  const genres = live?.genres || instant?.genres || [];
   552→  const source = isAnime ? 'jikan' : 'tmdb';
   553→  const externalId = isAnime ? `jikan:anime:${extId}` : `tmdb:${extType}:${extId}`;
   554→
   555→  const status = liveAnime?.status || null;
   556→  const episodes = typeof liveAnime?.episodes === 'number' ? liveAnime.episodes : null;
   557→  const studios = liveAnime?.studios || [];
   558→  const runtime = typeof liveMedia?.runtime === 'number' ? liveMedia.runtime : null;
   559→  const trailerKey = liveMedia?.trailerKey || null;
   560→  const heroImage = backdrop || poster;
   561→
   562→  async function handleAdd() {
   563→    if (adding) return;
   564→    setAdding(true);
   565→    try {
   566→      const metadata: Record<string, unknown> = {
   567→        overview: overview ?? undefined,
   568→        rating: rating ?? undefined,
   569→        year: year != null && year !== '' ? String(year) : undefined,
   570→        genres: genres.length > 0 ? genres : undefined,
   571→        source,
   572→        mediaType: extType,
   573→      };
   574→      if (backdrop) metadata.backdrop = backdrop;
   575→      if (trailerKey) {
   576→        metadata.player = {
   577→          sources: [
   578→            {
   579→              type: 'youtube',
   580→              url: `https://www.youtube.com/embed/${trailerKey}`,
   581→              label: 'Official trailer',
   582→            },
   583→          ],
   584→        };
   585→      }
   586→      const res = await api.upsertContent({
   587→        externalid: externalId,
   588→        type: extType,
   589→        title,
   590→        thumbnail: poster ?? null,
   591→        metadata,
   592→      });
   593→      toast.success('Added to your Animaxia library');
   594→      navigate(`/content/${res.record.id}`);
   595→    } catch (e) {
   596→      toast.error(e instanceof Error ? e.message : 'Could not add this title to the library');
   597→    } finally {
   598→      setAdding(false);
   599→    }
   600→  }
   601→
   602→  return (
   603→    <article>
   604→      {/* Backdrop hero */}
   605→      <div className="relative h-[38vh] min-h-[260px] w-full">
   606→        {heroImage ? (
   607→          <img src={heroImage} alt="" aria-hidden className="h-full w-full object-cover" />
   608→        ) : (
   609→          <div className="h-full w-full bg-gradient-to-br from-white/[0.08] via-transparent to-transparent" />
   610→        )}
   611→        <div aria-hidden className="absolute inset-0 bg-gradient-to-t from-[#0a0a10] via-[#0a0a10]/40 to-transparent" />
   612→        <div aria-hidden className="absolute inset-0 bg-gradient-to-r from-[#0a0a10]/70 via-[#0a0a10]/10 to-transparent" />
   613→      </div>
   614→
   615→      {/* Info panel */}
   616→      <Container className="relative z-10 -mt-20 sm:-mt-28">
   617→        <div className="flex gap-5 sm:gap-8">
   618→          <div className="w-28 shrink-0 sm:w-44">
   619→            <PosterFrame poster={poster} title={title} />
   620→          </div>
   621→          <div className="min-w-0 flex-1 space-y-3 pb-4 pt-1 sm:pt-6">
   622→            <div className="flex flex-wrap items-center gap-2">
   623→              <TypeBadge type={extType} />
   624→              {rating != null && <RatingBadge value={rating} />}
   625→              {year != null && year !== '' && (
   626→                <span className="text-xs font-semibold text-zinc-400">{String(year)}</span>
   627→              )}
   628→              <SourceChip label={source} />
   629→              <span className="inline-flex items-center rounded-full border border-white/10 bg-white/[0.04] px-2.5 py-0.5 text-[11px] font-medium text-zinc-400">
   630→                Not in library
   631→              </span>
   632→            </div>
   633→            <h1 className="text-2xl font-black leading-tight tracking-tight text-white sm:text-4xl">{title}</h1>
   634→            {genres.length > 0 && (
   635→              <div className="flex flex-wrap items-center gap-1.5">
   636→                {genres.slice(0, 4).map((g) => (
   637→                  <Chip key={g}>{g}</Chip>
   638→                ))}
   639→                {genres.length > 4 && <Chip>+{genres.length - 4}</Chip>}
   640→              </div>
   641→            )}
   642→            <div className="flex flex-wrap gap-3 pt-1">
   643→              <Button
   644→                size="lg"
   645→                className="bg-rose-600 font-bold text-white hover:bg-rose-500"
   646→                onClick={handleAdd}
   647→                disabled={adding}
   648→              >
   649→                {adding ? (
   650→                  <Loader2 className="animate-spin" aria-hidden />
   651→                ) : (
   652→                  <Plus aria-hidden />
   653→                )}
   654→                {adding ? 'Adding…' : 'Add to Animaxia'}

   664→            </div>
   665→          </div>
   666→        </div>
   667→      </Container>

   669→      {/* Body */}
   670→      <Container className="py-8 sm:py-10">

   672→          <div className="space-y-8 lg:col-span-2">

   694→          <Card className="h-fit border-white/10 bg-white/[0.03]">
