    57→function formatBytes(bytes: number): string {
    58→  if (!Number.isFinite(bytes) || bytes <= 0) return '0 B';
    59→  const units = ['B', 'KB', 'MB', 'GB', 'TB'];
    60→  let value = bytes;
    61→  let unit = 0;
    62→  while (value >= 1024 && unit < units.length - 1) {
    63→    value /= 1024;
    64→    unit += 1;
    65→  }
    66→  return `${value.toLocaleString('ro-RO', { maximumFractionDigits: 1 })} ${units[unit]}`;
    67→}
    68→
    69→function formatSignedBytes(bytes: number): string {
    70→  if (!Number.isFinite(bytes) || bytes === 0) return '±0 KB';
    71→  const sign = bytes > 0 ? '+' : '-';
    72→  return `${sign}${formatBytes(Math.abs(bytes))}`;
    73→}
    74→
    75→function formatMs(ms: number): string {
    76→  if (!Number.isFinite(ms)) return '—';
    77→  return `${ms.toLocaleString('ro-RO', { maximumFractionDigits: 2 })}ms`;
    78→}
    79→
    80→function formatUptime(sec: number): string {
    81→  if (!Number.isFinite(sec) || sec < 0) return '—';
    82→  const h = Math.floor(sec / 3600);
    83→  const m = Math.floor((sec % 3600) / 60);
    84→  const s = Math.floor(sec % 60);
    85→  if (h > 0) return `${h}h ${m}m ${s}s`;
    86→  if (m > 0) return `${m}m ${s}s`;
    87→  return `${s}s`;
    88→}
    89→
    90→function formatCount(n: number): string {
    91→  return n.toLocaleString('ro-RO');
    92→}

   338→function StressSection() {
   339→  const [suite, setSuite] = useState<SuiteKey>('virtual30b');
   340→  const [durationSec, setDurationSec] = useState(5);
   341→  const [concurrency, setConcurrency] = useState(8);
   342→  const [q, setQ] = useState('dune');
   343→  const [running, setRunning] = useState(false);
   344→  const [result, setResult] = useState<StressRunResponse | null>(null);
   345→
   346→  const suiteMeta = SUITES.find((x) => x.key === suite) ?? SUITES[0];
   347→
   348→  const run = async () => {
   349→    setRunning(true);
   350→    setResult(null);
   351→    try {
   352→      const res = await api.platform.stressRun({
   353→        suite,
   354→        durationSec,
   355→        concurrency,
   356→        q: q.trim() || 'dune',
   357→      });
   358→      setResult(res);
   359→      toast.success(`Test finalizat — ${formatCount(res.requests)} cereri, ${res.errorCount} erori`);

   361→      if (e instanceof ApiError) {
   362→        if (e.status === 409) {
   363→          toast.error('Un test rulează deja', {
   364→            description: 'Nodul acceptă un singur test de stres simultan — așteaptă finalizarea lui.',
   365→          });

   368→            description: 'Limita backend este de 2 teste de stres pe minut.',
   369→          });
   370→        } else {
   371→          toast.error(e.message);
   372→        }
   373→      } else {
   374→        toast.error(e instanceof Error ? e.message : 'Testul de stres a eșuat');
   375→      }
   376→    } finally {
   377→      setRunning(false);
   378→    }

   628→function VirtualBrowserSection() {
   629→  const [seed, setSeed] = useState<string>('animaxia');
   630→  const [cursor, setCursor] = useState('');
   631→  const [pageInput, setPageInput] = useState('1');
   632→
   633→  const res = useApi(`virtual-30b-${seed}-${cursor || 'start'}`, () =>
   634→    api.platform.virtualCatalog({ cursor: cursor || undefined, seed, limit: VIRTUAL_LIMIT }),
   635→  );
   636→
   637→  const goToPage = (page: number) => {
   638→    if (!Number.isFinite(page)) return;
   639→    const p = Math.min(Math.max(1, Math.floor(page)), VIRTUAL_MAX_PAGE);
   640→    setCursor(buildVirtualCursor((p - 1) * VIRTUAL_LIMIT, seed));
   641→    setPageInput(String(p));
   642→  };
   643→
   644→  const jump = () => {
   645→    const p = Number.parseInt(pageInput, 10);
   646→    if (!Number.isFinite(p) || p < 1 || p > VIRTUAL_MAX_PAGE) {
   647→      toast.error(`Pagina trebuie să fie între 1 și ${formatCount(VIRTUAL_MAX_PAGE)}`);
   648→      return;
   649→    }
   650→    goToPage(p);
   651→  };
   652→
   653→  const randomDeep = () => goToPage(Math.floor(Math.random() * (VIRTUAL_MAX_PAGE - 1)) + 1);
   654→
   655→  const data = res.data;
   656→  const totalPages = data ? Math.floor(data.addressSpace / VIRTUAL_LIMIT) : VIRTUAL_MAX_PAGE;
   657→
   658→  return (
   659→    <Section
   660→      title="Browser virtual 30B"
   661→      subtitle="Catalog sintetic determinist — simulează cele 30 de miliarde de conținuturi ca adresă; fiecare pagină se generează O(1), nimic nu se scanează."
   662→    >
   663→      {/* controls */}
   664→      <div className="flex flex-wrap items-end gap-2.5 rounded-2xl border border-white/10 bg-white/[0.03] p-3.5">
   665→        <div className="w-40">
   666→          <Label htmlFor="virtual-page" className="text-[11px] font-semibold text-zinc-400">
   667→            Pagina (1 – {formatCount(VIRTUAL_MAX_PAGE)})
   668→          </Label>
   669→          <Input
   670→            id="virtual-page"
   671→            type="number"
   672→            min={1}
   673→            max={VIRTUAL_MAX_PAGE}
   674→            value={pageInput}
   675→            onChange={(e) => setPageInput(e.target.value)}
   676→            onKeyDown={(e) => {
   677→              if (e.key === 'Enter') jump();
   678→            }}
   679→            className="mt-1 border-white/15 bg-black/30 text-zinc-100"
   680→          />
   681→        </div>
   682→        <Button
   683→          onClick={jump}
   684→          className="h-9 bg-rose-600 font-bold hover:bg-rose-500"
   685→          disabled={res.loading && !data}
   686→        >
   687→          Sari
   688→        </Button>
   689→        <Button
   690→          onClick={randomDeep}
   691→          variant="outline"
   692→          className="h-9 border-white/15 bg-white/5 font-semibold text-zinc-200 hover:bg-white/10 hover:text-white"
   693→        >
   694→          <Dices className="mr-1.5 h-4 w-4" aria-hidden />
   695→          Pagină aleatoare adâncă
   696→        </Button>
   697→        <Button
   698→          onClick={() => {
   699→            if (!data?.nextCursor) return;
   700→            /* nextCursor = indexul următor → pagina curentă + 1 (determinist). */
   701→            setPageInput(String(data.page + 1));
   702→            setCursor(data.nextCursor);
   703→          }}
   704→          disabled={!data?.nextCursor}
   705→          variant="outline"
   706→          className="h-9 border-white/15 bg-white/5 font-semibold text-zinc-200 hover:bg-white/10 hover:text-white"
   707→        >
   708→          Pagina următoare
   709→        </Button>
   710→        <Button
   711→          onClick={() => {
   712→            setCursor('');
   713→            setPageInput('1');
   714→          }}
   715→          variant="outline"
   716→          className="h-9 border-white/15 bg-white/5 font-semibold text-zinc-200 hover:bg-white/10 hover:text-white"
   717→        >
   718→          De la început
   719→        </Button>
   720→        <div className="ml-auto flex items-end gap-2">
   721→          <div className="w-40">
   722→            <Label htmlFor="virtual-seed" className="text-[11px] font-semibold text-zinc-400">
   723→              Seed determinist
   724→            </Label>
   725→            <Select
   726→              value={seed}
   727→              onValueChange={(v) => {
   728→                setSeed(v);
   729→                setCursor('');
   730→                setPageInput('1');
   731→              }}
   732→            >
   733→              <SelectTrigger id="virtual-seed" className="mt-1 w-full border-white/15 bg-black/30">
   734→                <SelectValue />
   735→              </SelectTrigger>
   736→              <SelectContent>
   737→                {VIRTUAL_SEEDS.map((s) => (
   738→                  <SelectItem key={s} value={s}>
   739→                    «{s}»
   740→                  </SelectItem>
   741→                ))}
   742→              </SelectContent>
   743→            </Select>
   744→          </div>
   745→        </div>
   746→      </div>
   747→
   748→      {/* grid */}
   749→      {res.error && !res.loading && !data ? (
   750→        <ErrorState message={res.error} onRetry={() => void res.refetch()} />
   751→      ) : !data ? (
   752→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5">
   753→          {Array.from({ length: VIRTUAL_LIMIT }).map((_, i) => (
   754→            <div key={i} className="space-y-2">
   755→              <div className="aspect-[2/3] animate-pulse rounded-xl border border-white/10 bg-white/[0.03]" />
   756→              <div className="h-4 w-4/5 animate-pulse rounded bg-white/[0.05]" />
   757→            </div>
   758→          ))}
   759→        </div>
   760→      ) : (
   761→        <>
   762→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5">
   763→            {data.items.map((item) => (
   764→              <VirtualCard key={item.index} item={item} />
   765→            ))}
   766→          </div>
   767→          <p className="text-[11px] leading-relaxed text-zinc-500">
   768→            Pagina {formatCount(data.page)} din {formatCount(totalPages)} · spațiu de adresare 30.000.000.000
   769→            conținuturi · generată în {formatCount(data.tookMs)}ms · cheie deterministă seed «{seed}» · O(1)
   770→            {res.loading ? (
   771→              <span className="ml-2 inline-flex items-center gap-1 text-rose-300">
   772→                <Loader2 className="h-3 w-3 animate-spin" aria-hidden /> se încarcă…
   773→              </span>
   774→            ) : null}
   775→          </p>
   776→        </>
   777→      )}
   778→    </Section>
   779→  );
   780→}
