    92→function formatUptime(sec: number): string {
    93→  const h = Math.floor(sec / 3600);
    94→  const m = Math.floor((sec % 3600) / 60);
    95→  const s = Math.floor(sec % 60);
    96→  if (h > 0) return `${h}h ${m}m`;
    97→  if (m > 0) return `${m}m ${s}s`;
    98→  return `${s}s`;
    99→}
   100→
   101→function formatBytes(b?: number): string {
   102→  if (!b || b <= 0) return '0 B';
   103→  const units = ['B', 'KB', 'MB', 'GB'];
   104→  let v = b;
   105→  let i = 0;
   106→  while (v >= 1024 && i < units.length - 1) {
   107→    v /= 1024;
   108→    i += 1;
   109→  }
   110→  return `${v >= 100 ? Math.round(v) : v.toFixed(1)} ${units[i]}`;
   111→}
   112→
   113→function formatMs(ms?: number): string {
   114→  if (ms == null) return '—';
   115→  if (ms < 1000) return `${Math.round(ms)}ms`;
   116→  return `${(ms / 1000).toFixed(1)}s`;
   117→}
   118→
   119→function formatDate(iso?: string | null): string {
   120→  if (!iso) return '—';
   121→  const d = new Date(iso);
   122→  if (Number.isNaN(d.getTime())) return '—';
   123→  return d.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
   124→}

   126→function getErrorMessage(e: unknown): string {
   127→  return e instanceof Error ? e.message : 'Request failed';
   128→}

   132→  return Boolean(message && /admin access required|forbidden|status 40[13]/i.test(message));
   133→}
