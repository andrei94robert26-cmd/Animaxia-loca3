    96→/** base64url encode/decode (no padding issues, URL-safe). */
    97→function b64urlEncode(obj) {
    98→  return Buffer.from(JSON.stringify(obj), 'utf8')
    99→    .toString('base64')
   100→    .replace(/\+/g, '-')
   101→    .replace(/\//g, '_')
   102→    .replace(/=+$/, '');
   103→}

   105→function b64urlDecode(text) {
   106→  try {
   107→    const padded = String(text).replace(/-/g, '+').replace(/_/g, '/');
   108→    return JSON.parse(Buffer.from(padded, 'base64').toString('utf8'));
   109→  } catch {
   110→    return null;
   111→  }
   112→}

   114→/**
   115→ * Build the next cursor from the LAST row of the current page, per sort.
   116→ * Filter-only + no q → effective sort is contentid (see effectiveSort).
   117→ */
   118→function buildNextCursor({ sort, lastRow, pageNumber }) {
   119→  if (!lastRow) return null;
   120→  const id = Number(lastRow.id);
   121→  if (!Number.isFinite(id)) return null;
   122→  let payload = null;
   123→  if (sort === 'year') payload = { y: lastRow._year ?? null, id };
   124→  else if (sort === 'rating') payload = { r: lastRow._rating ?? null, id };
   125→  else if (sort === 'title') payload = { t: lastRow._title ?? null, id };
   126→  else if (sort === 'score') payload = { s: Number(lastRow._score ?? 0), id };
   127→  else payload = { id }; // contentid keyset (filter-only)
   128→  payload.p = pageNumber; // page estimate — incremented on each cursor hop
   129→  return b64urlEncode(payload);
   130→}

   132→/**
   133→ * Decode + validate the cursor payload for a given effective sort.
   134→ * @returns {{ payload: object, where: string, params: any[] } | { error: string }}
   135→ */
   136→function cursorContinuation(effectiveSort, cursorText) {
   137→  const payload = b64urlDecode(cursorText);
   138→  if (!payload || typeof payload !== 'object' || !Number.isFinite(Number(payload.id))) {
   139→    return { error: 'Invalid cursor' };
   140→  }
   141→  const id = Number(payload.id);
   142→  const params = [];
   143→  const P = (v) => {
   144→    params.push(v);
   145→    return `$${params.length}`;
   146→  };
   147→
   148→  if (effectiveSort === 'score') {
   149→    const s = Number(payload.s);
   150→    if (!Number.isFinite(s)) return { error: 'Invalid cursor (missing s)' };
   151→    const sp = P(s);
   152→    const ip = P(id);
   153→    return { payload, where: `(t.score < ${sp} OR (t.score = ${sp} AND t.contentid > ${ip}))`, params };
   154→  }
   155→  if (effectiveSort === 'year') {
   156→    const y = Number(payload.y);
   157→    if (!Number.isFinite(y)) return { error: 'Invalid cursor (missing y)' };
   158→    const yp = P(y);
   159→    const ip = P(id);
   160→    return { payload, where: `(t.year < ${yp} OR (t.year = ${yp} AND t.contentid > ${ip}))`, params };
   161→  }
   162→  if (effectiveSort === 'rating') {
   163→    const r = Number(payload.r);
   164→    if (!Number.isFinite(r)) return { error: 'Invalid cursor (missing r)' };
   165→    const rp = P(r);
   166→    const ip = P(id);
   167→    return { payload, where: `(t.rating < ${rp} OR (t.rating = ${rp} AND t.contentid > ${ip}))`, params };
   168→  }
   169→  if (effectiveSort === 'title') {
   170→    const t = payload.t;
   171→    if (typeof t !== 'string') return { error: 'Invalid cursor (missing t)' };
   172→    const tp = P(t);
   173→    const ip = P(id);
   174→    return { payload, where: `(t.title > ${tp} OR (t.title = ${tp} AND t.contentid > ${ip}))`, params };
   175→  }
   176→  // contentid keyset (filter-only / no q)
   177→  const ip = P(id);
   178→  return { payload, where: `t.contentid > ${ip}`, params };
   179→}

   183→/** Coerce a metadata value to a clean string array (strings or {name} objects). */
   184→function toNameArray(value) {
   185→  if (value == null) return [];
   186→  const list = Array.isArray(value) ? value : [value];
   187→  const out = [];
   188→  for (const item of list) {
   189→    if (typeof item === 'string') {
   190→      const name = item.trim();
   191→      if (name) out.push(name);
   192→    } else if (item && typeof item === 'object') {
   193→      const name = String(item.name ?? '').trim();
   194→      if (name) out.push(name);
   195→    }
   196→  }
   197→  return out;
   198→}

   210→/** Build the FTS document text: title + genres + first 500 chars of overview. */
   211→function buildTsvText(title, genres, meta) {
   212→  const overview = String(meta.overview || meta.synopsis || meta.description || '');

   218→let syncPromise = null; // async lock — concurrent ensureSynced() calls coalesce
   219→let lastSync = null; // ISO timestamp of the last successful sync

   222→async function syncIndex() {
   223→  const raw = await rows(
   224→    'SELECT id, externalid, type, title, metadata, thumbnail, createdat FROM content',
   225→  );
   226→
   227→  let synced = 0;
   228→  for (let i = 0; i < raw.length; i += SYNC_CHUNK) {
   229→    const slice = raw.slice(i, i + SYNC_CHUNK);
   230→    const values = [];
   231→    const tuples = slice.map((row, j) => {
   232→      const meta = parseJsonb(row.metadata, {});
   233→      const genres = toNameArray(meta.genres);
   234→      const actors = toNameArray(meta.actors ?? meta.cast);
   235→      const studios = toNameArray(meta.studios ?? meta.studio);
   236→      const tsvText = buildTsvText(row.title, genres, meta);
   237→      // FAZA 13 (sharding): searchdocs carries its shard id = the catalog
   238→      // RANGE shard of the contentid it indexes (ids < 1B → 0, legacy-safe).
   239→      const shardid = computeShardIdFromContentId(row.id);
   240→      const base = j * 12;
   241→      values.push(
   242→        Number(row.id),
   243→        row.externalid ?? null,
   244→        row.type ?? null,
   245→        row.title ?? 'untitled',
   246→        tsvText,
   247→        toIntOrNull(meta.year),
   248→        toFloatOrNull(meta.rating ?? meta.score),
   249→        genres,
   250→        actors,
   251→        studios,
   252→        row.type ?? null,
   253→        shardid,
   254→      );
   255→      const b = (k) => `$${base + k}`;
   256→      return `(${b(1)},${b(2)},${b(3)},${b(4)},to_tsvector('simple', ${b(5)}),${b(6)},${b(7)},${b(8)}::text[],${b(9)}::text[],${b(10)}::text[],${b(11)},${b(12)},NOW())`;
   257→    });
   258→    await rows(
   259→      `INSERT INTO searchdocs
   260→         (contentid, externalid, type, title, tsv, year, rating, genres, actors, studios, category, shardid, syncedat)
   261→       VALUES ${tuples.join(', ')}
   262→       ON CONFLICT (contentid) DO UPDATE SET
   263→         externalid = EXCLUDED.externalid,
   264→         type = EXCLUDED.type,
   265→         title = EXCLUDED.title,
   266→         tsv = EXCLUDED.tsv,
   267→         year = EXCLUDED.year,
   268→         rating = EXCLUDED.rating,
   269→         genres = EXCLUDED.genres,
   270→         actors = EXCLUDED.actors,
   271→         studios = EXCLUDED.studios,
   272→         category = EXCLUDED.category,
   273→         shardid = EXCLUDED.shardid,
   274→         syncedat = NOW()`,
   275→      values,
   276→    );
   277→    synced += slice.length;
   278→  }
   279→
   280→  // Drop stale docs (content rows deleted since the last sync).
   281→  try {
   282→    await rows('DELETE FROM searchdocs WHERE contentid NOT IN (SELECT id FROM content)');
   283→  } catch (err) {
   284→    logger.warn(`[search] stale-doc cleanup skipped: ${err.message}`);
   285→  }
   286→
   287→  lastSync = new Date().toISOString();
   288→  dirty = false;
   289→  return synced;
   290→}

   292→/** Rebuild once under an async lock; concurrent callers await the same run. */
   293→async function ensureSynced() {
   294→  if (!dirty && lastSync) return;
   295→  if (!syncPromise) {
   296→    syncPromise = syncIndex().finally(() => {
   297→      syncPromise = null;
   298→    });
   299→  }
   300→  await syncPromise;
   301→}
   302→
   303→/** Forced full resync (POST /index): waits for any in-flight sync, then runs one. */
   304→async function forceResync() {
   305→  while (syncPromise) {
   306→    await syncPromise.catch(() => {});
   307→  }
   308→  syncPromise = syncIndex().finally(() => {
   309→    syncPromise = null;
   310→  });
   311→  return syncPromise;
   312→}

   317→  dirty = true;

   323→  if (typeof resyncTimer.unref === 'function') resyncTimer.unref();
   324→});

   371→/**
   372→ * Run one advanced query against searchdocs (joined back to content).
   373→ * Filter fragments are built once and shared by the count + data statements.
   374→ * FAZA 17-a: `cursor` (base64url payload) switches to keyset pagination —
   375→ * the ORDER BY becomes the cursor continuation WHERE and OFFSET is dropped;
   376→ * `effectiveSort` is what actually orders the result (contentid keyset when
   377→ * there is no text query and the caller asked for the default 'score' sort,
   378→ * because score would be constant 0 for every row).
   379→ */
   380→async function runAdvancedQuery({
   381→  q,
   382→  type,
   383→  genre,
   384→  year,
   385→  minRating,
   386→  actor,
   387→  studio,
   388→  category,
   389→  sort,
   390→  fuzzyEnabled,
   391→  limit,
   392→  page,
   393→  cursor = null,
   394→}) {
   395→  const params = [];
   396→  const P = (v) => {
   397→    params.push(v);
   398→    return `$${params.length}`;
   399→  };
   400→
   401→  // Effective sort: filter-only (no q) with default 'score' → contentid keyset.
   402→  const effectiveSort = !q && sort === 'score' ? 'contentid' : sort;
   403→
   404→  // Keyset continuation (cursor present) — validated per effective sort.
   405→  let cursorPayload = null;
   406→  let cursorWhere = null;
   407→  let cursorParams = [];
   408→  if (cursor) {
   409→    const decoded = cursorContinuation(effectiveSort, cursor);
   410→    if (decoded.error) {
   411→      const err = new Error(decoded.error);
   412→      err.status = 400;
   413→      throw err;
   414→    }
   415→    cursorPayload = decoded.payload;
   416→    cursorWhere = decoded.where;
   417→    cursorParams = decoded.params;
   418→  }
   419→
   420→  // Field filters (SQL-side, case-insensitive where array containment).
   421→  const filters = [];
   422→  if (type) filters.push(`s.type = ${P(type)}`);
   423→  if (genre) filters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${P(genre)})`);
   424→  if (year != null) filters.push(`s.year = ${P(year)}`);
   425→  if (actor) {
   426→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${P(actor)} || '%')`);
   427→  }
   428→  if (studio) {
   429→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${P(studio)} || '%')`);
   430→  }
   431→  if (category) filters.push(`s.category = ${P(category)}`);
   432→  if (minRating != null) filters.push(`s.rating >= ${P(minRating)}`);
   433→  const filterSql = filters.length ? ` AND ${filters.join(' AND ')}` : '';
   434→
   435→  // Text score expression (combined: ts_rank * 2 + similarity + prefix bonus).
   436→  let scoreExpr = '0';
   437→  let textMatch = '';
   438→  if (q) {
   439→    const qp = P(q);
   440→    scoreExpr =
   441→      `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qp})) * 2` +
   442→      ` + similarity(s.title, ${qp})` +
   443→      ` + CASE WHEN s.title ILIKE ${qp} || '%' THEN 0.5 ELSE 0 END`;
   444→    textMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qp})`;
   445→  }
   446→
   447→  const whereFor = (matchSql) => `WHERE (${matchSql || 'TRUE'})${filterSql}`;
   448→
   449→  // 1. FTS pass (or filter-only when q is empty).
   450→  let matchSql = textMatch;
   451→  let countRow = await one(
   452→    `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   453→    params,
   454→  );
   455→
   456→  // 2. Trigram fuzzy fallback when the FTS pass matched nothing.
   457→  //    `title % q` (pg_trgm, default 0.3 threshold) plus a looser
   458→  //    similarity() > 0.1 guard so short typo'd queries still reach long
   459→  //    titles (e.g. 'dunee' → 'Dune: Part Two - Wikipedia', sim ≈ 0.15).
   460→  if (q && fuzzyEnabled && Number(countRow?.total ?? 0) === 0) {
   461→    const words = q.split(/\s+/).filter(Boolean);
   462→    const trigramClauses = [];
   463→    for (const candidate of [q, ...words]) {
   464→      const ph = P(`%${candidate}%`);
   465→      trigramClauses.push(`s.title ILIKE ${ph}`);
   466→    }
   467→    const qSim = P(q);
   468→    trigramClauses.push(`(s.title % ${qSim} OR similarity(s.title, ${qSim}) > 0.1)`);
   469→    matchSql = `(${trigramClauses.join(' OR ')})`;
   470→    countRow = await one(
   471→      `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   472→      params,
   473→    );
   474→  }
   475→
   476→  const total = Number(countRow?.total ?? 0);
   477→  const maxScore = Number(Number(countRow?.maxscore ?? 0).toFixed(6));
   478→
   479→  // 3. Data page — subquery keeps the score expression, then joins content.
   480→  //    FAZA 17-a: expose year/rating/title to JS for nextCursor construction;
   481→  //    cursor mode replaces OFFSET with the keyset continuation predicate.
   482→  const orderExpr =
   483→    effectiveSort === 'year'
   484→      ? 't.year DESC NULLS LAST, t.contentid ASC'
   485→      : effectiveSort === 'rating'
   486→        ? 't.rating DESC NULLS LAST, t.score DESC, t.contentid ASC'
   487→        : effectiveSort === 'title'
   488→          ? 't.title ASC, t.contentid ASC'
   489→          : effectiveSort === 'contentid'
   490→            ? 't.contentid ASC'
   491→            : 't.score DESC, t.contentid ASC';
   492→
   493→  const limitPh = P(limit);
   494→  const offsetPh = P((page - 1) * limit);
   495→  // Cursor mode: continuation params come FIRST ($1..$n), then LIMIT; the
   496→  // filter/score params are re-bound only inside the subquery — so build a
   497→  // dedicated statement for each mode instead of renumbering dynamically.
   498→  let dataSql;
   499→  let dataParams;
   500→  if (cursorWhere) {
   501→    // Rebuild the subquery with a fresh parameter namespace for the cursor run
   502→    // (filters + score expression are re-parameterized after the cursor slots).
   503→    const subParams = [...cursorParams, limit];
   504→    const sub = (v) => {
   505→      subParams.push(v);
   506→      return `$${subParams.length}`;
   507→    };
   508→    const cFilters = [];
   509→    if (type) cFilters.push(`s.type = ${sub(type)}`);
   510→    if (genre) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${sub(genre)})`);
   511→    if (year != null) cFilters.push(`s.year = ${sub(year)}`);
   512→    if (actor) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${sub(actor)} || '%')`);
   513→    if (studio) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${sub(studio)} || '%')`);
   514→    if (category) cFilters.push(`s.category = ${sub(category)}`);
   515→    if (minRating != null) cFilters.push(`s.rating >= ${sub(minRating)}`);
   516→    const cFilterSql = cFilters.length ? ` AND ${cFilters.join(' AND ')}` : '';
   517→
   518→    let cScoreExpr = '0';
   519→    let cMatch = 'TRUE';
   520→    // Single q slot, bound LAST (after cursor params + LIMIT) so every
   521→    // placeholder references a parameter that exists in the bind array.
   522→    let qParam = null;
   523→    if (q) {
   524→      qParam = sub(String(q));
   525→      cScoreExpr =
   526→        `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qParam})) * 2` +
   527→        ` + similarity(s.title, ${qParam})` +
   528→        ` + CASE WHEN s.title ILIKE ${qParam} || '%' THEN 0.5 ELSE 0 END`;
   529→      cMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qParam})`;
   530→    }
   531→    // Fuzzy fallback consistency: when the FTS pass matched nothing the offset
   532→    // run switched matchSql — mirror that for the cursor run using `total`.
   533→    if (q && fuzzyEnabled && total === 0) {
   534→      const words = q.split(/\s+/).filter(Boolean);
   535→      const clauses = [q, ...words].map((candidate) => `s.title ILIKE ${sub(`%${candidate}%`)}`);
   536→      clauses.push(`(s.title % ${qParam} OR similarity(s.title, ${qParam}) > 0.1)`);
   537→      cMatch = `(${clauses.join(' OR ')})`;
   538→    }
   539→
   540→    dataSql =
   541→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   542→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   543→       FROM (
   544→         SELECT s.contentid, s.year, s.rating, s.title, ${cScoreExpr} AS score
   545→         FROM searchdocs s
   546→         WHERE (${cMatch})${cFilterSql}
   547→       ) t
   548→       JOIN content c ON c.id = t.contentid
   549→       WHERE ${cursorWhere}
   550→       ORDER BY ${orderExpr}
   551→       LIMIT $${cursorParams.length + 1}::int`;
   552→    // LIMIT is the placeholder right after the cursor params (::int — a bare
   553→    // parameter in LIMIT cannot be type-inferred by Postgres); the q/filter
   554→    // slots follow it so every bound parameter is referenced by the statement.
   555→    dataParams = subParams;
   556→  } else {
   557→    dataSql =
   558→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   559→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   560→       FROM (
   561→         SELECT s.contentid, s.year, s.rating, s.title, ${scoreExpr} AS score
   562→         FROM searchdocs s
   563→         ${whereFor(matchSql)}
   564→       ) t
   565→       JOIN content c ON c.id = t.contentid
   566→       ORDER BY ${orderExpr}
   567→       LIMIT ${limitPh} OFFSET ${offsetPh}`;
   568→    dataParams = params;
   569→  }
   570→  const dataRows = await rows(dataSql, dataParams);
   571→
   572→  // Keyset paging estimates the page number by incrementing the cursor's `p`.
   573→  const resultPage = cursor ? (Number(cursorPayload?.p) || 1) + 1 : page;
   574→  const normalized = dataRows.map(normalizeHit).filter(Boolean);
   575→  const lastRaw = dataRows.length ? dataRows[dataRows.length - 1] : null;
   576→
   577→  return {
   578→    total,
   579→    max_score: maxScore,
   580→    data: normalized,
   581→    page: resultPage,
   582→    nextCursor: buildNextCursor({ sort: effectiveSort, lastRow: lastRaw, pageNumber: resultPage }),
   583→    paginationMode: cursor ? 'cursor' : 'offset',
   584→  };
   585→}

   593→const inflightSearch = new Map();

   595→function coalesceAdvanced(normalizedKey, run) {
   596→  const existing = inflightSearch.get(normalizedKey);
   597→  if (existing) {
   598→    searchOps.coalesced += 1;
   599→    return existing;
   600→  }
   601→  const promise = run().finally(() => {
   602→    inflightSearch.delete(normalizedKey);
   603→  });
   604→  inflightSearch.set(normalizedKey, promise);
   605→  return promise;
   606→}

   608→/** Race a query promise against the 8s budget (loser is logged, not awaited). */
   609→function withQueryTimeout(promise, describe) {
   610→  let timer = null;
   611→  const budget = new Promise((_, reject) => {
   612→    timer = setTimeout(() => {
   613→      const err = new Error('Search query timeout');
   614→      err.status = 504;
   615→      reject(err);
   616→    }, QUERY_TIMEOUT_MS);
   617→    if (typeof timer.unref === 'function') timer.unref();
   618→  });
   619→  return promise.then(
   620→    (value) => {
   621→      clearTimeout(timer);
   622→      return value;
   623→    },
   624→    (err) => {
   625→      clearTimeout(timer);
   626→      throw err;
   627→    },
   628→  ).finally(() => {
   629→    clearTimeout(timer);
   630→    promise.catch((loser) => {
   631→      // The lost race (if it also failed) must not become an unhandledRejection.
   632→      logger.warn(`[search] query lost the timeout race (${describe}): ${loser?.message ?? loser}`);
   633→    });
   634→  });
   635→}
