    72→function fmtRoNum(v: number | null | undefined): string {

    76→function fmtMs(v: number | null | undefined): string {
    77→  const n = Number(v ?? 0);
    78→  return Number.isFinite(n) && n > 0 ? `${RO_NUMBER.format(Math.round(n))} ms` : '—';
    79→}

    81→/** HH:MM:SS clock label for the history X axis (samples are 1 s apart). */
    82→function fmtClock(iso: string): string {
    83→  const d = new Date(iso);
    84→  if (Number.isNaN(d.getTime())) return iso;
    85→  const p = (n: number) => String(n).padStart(2, '0');
    86→  return `${p(d.getHours())}:${p(d.getMinutes())}:${p(d.getSeconds())}`;
    87→}

   224→/* ---------------------------- settings form ------------------------------ */
   225→
   226→const SETTINGS_FIELDS = [
   227→  { key: 'maxConcurrent', label: 'Max concurrent', hint: '1 – 2.000', min: 1, max: 2000 },
   228→  { key: 'intervalMs', label: 'Interval fereastră', hint: '10 – 2.000 ms', min: 10, max: 2000 },
   229→  { key: 'batchSize', label: 'Batch size', hint: '1 – 1.000', min: 1, max: 1000 },
   230→  { key: 'queueCap', label: 'Queue cap', hint: '100 – 500.000', min: 100, max: 500000 },
   231→] as const;
   232→
   233→type SettingsKey = (typeof SETTINGS_FIELDS)[number]['key'];
   234→type SettingsForm = Record<SettingsKey, string>;
   235→
   236→function FlowSettingsCard({
   237→  config,
   238→  isAdmin,
   239→  onSaved,
   240→}: {
   241→  config: SearchFlowConfig;
   242→  isAdmin: boolean;
   243→  onSaved: () => void;
   244→}) {
   245→  const [form, setForm] = useState<SettingsForm>({
   246→    maxConcurrent: String(config.maxConcurrent),
   247→    intervalMs: String(config.intervalMs),
   248→    batchSize: String(config.batchSize),
   249→    queueCap: String(config.queueCap),
   250→  });
   251→  const [dirty, setDirty] = useState(false);
   252→  const [saving, setSaving] = useState(false);
   253→
   254→  /* Re-sync the pre-filled inputs from the polled config while the user has
   255→     not edited anything (and right after a successful save). */
   256→  useEffect(() => {
   257→    if (dirty) return;
   258→    setForm((prev) => {
   259→      const next: SettingsForm = {
   260→        maxConcurrent: String(config.maxConcurrent),
   261→        intervalMs: String(config.intervalMs),
   262→        batchSize: String(config.batchSize),
   263→        queueCap: String(config.queueCap),
   264→      };
   265→      return (Object.keys(next) as SettingsKey[]).every((k) => prev[k] === next[k]) ? prev : next;
   266→    });
   267→  }, [config, dirty]);
   268→
   269→  async function handleSave() {
   270→    if (saving) return;
   271→    const parsed: Record<SettingsKey, number> = {
   272→      maxConcurrent: Number(form.maxConcurrent),
   273→      intervalMs: Number(form.intervalMs),
   274→      batchSize: Number(form.batchSize),
   275→      queueCap: Number(form.queueCap),
   276→    };
   277→    for (const f of SETTINGS_FIELDS) {
   278→      const v = parsed[f.key];
   279→      if (!Number.isFinite(v) || v < f.min || v > f.max) {
   280→        toast.error(`${f.label} trebuie să fie între ${RO_NUMBER.format(f.min)} și ${RO_NUMBER.format(f.max)}`);
   281→        return;
   282→      }
   283→    }
   284→    setSaving(true);
   285→    try {
   286→      await authRequest<{ ok: boolean; config: SearchFlowConfig }>(
   287→        `/api/search/flow/settings?XTransformPort=${BACKEND_PORT}`,
   288→        { method: 'POST', body: JSON.stringify(parsed) },
   289→      );
   290→      toast.success('Setări salvate');
   291→      setDirty(false);
   292→      onSaved();
   293→    } catch (e) {
   294→      if (e instanceof PlatformError && (e.status === 401 || e.status === 403)) {
   295→        toast.error('Setările fluxului necesită cont de administrator');
   296→      } else {
   297→        toast.error(e instanceof Error ? e.message : 'Salvarea setărilor a eșuat');
   298→      }
   299→    } finally {
   300→      setSaving(false);
   301→    }
   302→  }
   303→
   304→  return (
   305→    <Card className="border-white/10 bg-white/[0.03]">
   306→      <CardHeader className="pb-3">
   307→        <CardTitle className="flex items-center gap-2 text-base">
   308→          <ShieldCheck className="h-4 w-4 text-rose-400" aria-hidden />
   309→          Setări runtime (administrator)
   310→        </CardTitle>
   311→        <CardDescription>
   312→          Reglare live a cozii și a ferestrelor de dispatch — se aplică imediat, fără restart.
   313→        </CardDescription>
   314→      </CardHeader>
   315→      <CardContent className="space-y-4">
   316→        {isAdmin ? (
   317→          <>
   318→            <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
   319→              {SETTINGS_FIELDS.map((f) => (
   320→                <div key={f.key} className="space-y-1.5">
   321→                  <Label htmlFor={`flow-setting-${f.key}`} className="text-xs text-zinc-400">
   322→                    {f.label} <span className="font-normal text-zinc-600">({f.hint})</span>
   323→                  </Label>
   324→                  <Input
   325→                    id={`flow-setting-${f.key}`}
   326→                    type="number"
   327→                    inputMode="numeric"
   328→                    min={f.min}
   329→                    max={f.max}
   330→                    step={1}
   331→                    value={form[f.key]}
   332→                    onChange={(e) => {
   333→                      setDirty(true);
   334→                      setForm((prev) => ({ ...prev, [f.key]: e.target.value }));
   335→                    }}
   336→                    className="border-white/10 bg-white/[0.04] font-mono text-zinc-100"
   337→                    aria-label={`${f.label}, interval acceptat ${f.hint}`}
   338→                  />
   339→                </div>
   340→              ))}
   341→            </div>
   342→            <div className="flex flex-wrap items-center justify-between gap-3">
   343→              <p className="max-w-xl text-[11px] leading-snug text-zinc-600">
   344→                Validarea și limitarea valorilor (clamping) se fac pe server — valorile finale sunt cele
   345→                returnate după salvare și reflectate mai sus în cardurile de configurare.
   346→              </p>
   347→              <Button
   348→                size="sm"
   349→                onClick={() => void handleSave()}
   350→                disabled={saving || !dirty}
   351→                className="bg-rose-600 font-bold hover:bg-rose-500"
   352→              >
   353→                {saving ? <RefreshCw className="animate-spin" aria-hidden /> : <Save aria-hidden />}
   354→                Salvează
   355→              </Button>
   356→            </div>
   357→          </>
   358→        ) : (
   359→          <p className="flex items-center gap-2 rounded-xl border border-amber-500/25 bg-amber-500/[0.06] px-4 py-2.5 text-xs text-amber-200">
   360→            <Lock className="h-3.5 w-3.5 shrink-0" aria-hidden />
   361→            Setările fluxului necesită cont de administrator — valorile curente sunt read-only.
   362→          </p>
   363→        )}
   364→      </CardContent>
   365→    </Card>
   366→  );
   367→}
