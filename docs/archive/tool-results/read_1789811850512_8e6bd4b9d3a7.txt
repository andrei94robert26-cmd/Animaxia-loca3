   100→ * FAZA 16 admission helper — runs the rate gate for one search request.

   102→ * already sent — the caller just returns). Blocked requests never reach the

   106→function rejectIfRateLimited(req, res, kind, describe) {
   107→  const gate = searchRateGate(req, kind);
   108→  if (gate.ok) return false;
   109→  const retryAfterSec = Math.max(1, Math.ceil(gate.body.retryInMs / 1000));
   110→  res.set('Retry-After', String(retryAfterSec));

   112→    `[search] FAZA 16 gate rejected ${kind} ${describe} → ${gate.body.error} (retryInMs=${gate.body.retryInMs}, identity=${gate.identity}, shard=${gate.shardId})`,
   113→  );
   114→  res.status(429).json(gate.body);
   115→  return true;
   116→}

   138→/** base64url encode/decode (no padding issues, URL-safe). */
   139→function b64urlEncode(obj) {
   140→  return Buffer.from(JSON.stringify(obj), 'utf8')
   141→    .toString('base64')
   142→    .replace(/\+/g, '-')
   143→    .replace(/\//g, '_')
   144→    .replace(/=+$/, '');
   145→}

   147→function b64urlDecode(text) {
   148→  try {
   149→    const padded = String(text).replace(/-/g, '+').replace(/_/g, '/');
   150→    return JSON.parse(Buffer.from(padded, 'base64').toString('utf8'));
   151→  } catch {
   152→    return null;
   153→  }
   154→}

   156→/**
   157→ * Build the next cursor from the LAST row of the current page, per sort.
   158→ * Filter-only + no q → effective sort is contentid (see effectiveSort).
   159→ */
   160→function buildNextCursor({ sort, lastRow, pageNumber }) {
   161→  if (!lastRow) return null;
   162→  const id = Number(lastRow.id);
   163→  if (!Number.isFinite(id)) return null;
   164→  let payload = null;
   165→  if (sort === 'year') payload = { y: lastRow._year ?? null, id };
   166→  else if (sort === 'rating') payload = { r: lastRow._rating ?? null, id };
   167→  else if (sort === 'title') payload = { t: lastRow._title ?? null, id };
   168→  else if (sort === 'score') payload = { s: Number(lastRow._score ?? 0), id };
   169→  else payload = { id }; // contentid keyset (filter-only)
   170→  payload.p = pageNumber; // page estimate — incremented on each cursor hop
   171→  return b64urlEncode(payload);
   172→}

   174→/**
   175→ * Decode + validate the cursor payload for a given effective sort.
   176→ * @returns {{ payload: object, where: string, params: any[] } | { error: string }}
   177→ */
   178→function cursorContinuation(effectiveSort, cursorText) {
   179→  const payload = b64urlDecode(cursorText);
   180→  if (!payload || typeof payload !== 'object' || !Number.isFinite(Number(payload.id))) {
   181→    return { error: 'Invalid cursor' };
   182→  }
   183→  const id = Number(payload.id);
   184→  const params = [];
   185→  const P = (v) => {
   186→    params.push(v);
   187→    return `$${params.length}`;
   188→  };
   189→
   190→  if (effectiveSort === 'score') {
   191→    const s = Number(payload.s);
   192→    if (!Number.isFinite(s)) return { error: 'Invalid cursor (missing s)' };
   193→    const sp = P(s);
   194→    const ip = P(id);
   195→    return { payload, where: `(t.score < ${sp} OR (t.score = ${sp} AND t.contentid > ${ip}))`, params };
   196→  }
   197→  if (effectiveSort === 'year') {
   198→    const y = Number(payload.y);
   199→    if (!Number.isFinite(y)) return { error: 'Invalid cursor (missing y)' };
   200→    const yp = P(y);
   201→    const ip = P(id);
   202→    return { payload, where: `(t.year < ${yp} OR (t.year = ${yp} AND t.contentid > ${ip}))`, params };
   203→  }
   204→  if (effectiveSort === 'rating') {
   205→    const r = Number(payload.r);
   206→    if (!Number.isFinite(r)) return { error: 'Invalid cursor (missing r)' };
   207→    const rp = P(r);
   208→    const ip = P(id);
   209→    return { payload, where: `(t.rating < ${rp} OR (t.rating = ${rp} AND t.contentid > ${ip}))`, params };
   210→  }
   211→  if (effectiveSort === 'title') {
   212→    const t = payload.t;
   213→    if (typeof t !== 'string') return { error: 'Invalid cursor (missing t)' };
   214→    const tp = P(t);
   215→    const ip = P(id);
   216→    return { payload, where: `(t.title > ${tp} OR (t.title = ${tp} AND t.contentid > ${ip}))`, params };
   217→  }
   218→  // contentid keyset (filter-only / no q)
   219→  const ip = P(id);
   220→  return { payload, where: `t.contentid > ${ip}`, params };
   221→}

   225→/** Coerce a metadata value to a clean string array (strings or {name} objects). */
   226→function toNameArray(value) {
   227→  if (value == null) return [];
   228→  const list = Array.isArray(value) ? value : [value];
   229→  const out = [];
   230→  for (const item of list) {
   231→    if (typeof item === 'string') {
   232→      const name = item.trim();
   233→      if (name) out.push(name);
   234→    } else if (item && typeof item === 'object') {
   235→      const name = String(item.name ?? '').trim();
   236→      if (name) out.push(name);
   237→    }
   238→  }
   239→  return out;
   240→}

   264→async function syncIndex() {
   265→  const raw = await rows(
   266→    'SELECT id, externalid, type, title, metadata, thumbnail, createdat FROM content',
   267→  );
   268→
   269→  let synced = 0;
   270→  for (let i = 0; i < raw.length; i += SYNC_CHUNK) {
   271→    const slice = raw.slice(i, i + SYNC_CHUNK);
   272→    const values = [];
   273→    const tuples = slice.map((row, j) => {
   274→      const meta = parseJsonb(row.metadata, {});
   275→      const genres = toNameArray(meta.genres);
   276→      const actors = toNameArray(meta.actors ?? meta.cast);
   277→      const studios = toNameArray(meta.studios ?? meta.studio);
   278→      const tsvText = buildTsvText(row.title, genres, meta);
   279→      // FAZA 13 (sharding): searchdocs carries its shard id = the catalog
   280→      // RANGE shard of the contentid it indexes (ids < 1B → 0, legacy-safe).
   281→      const shardid = computeShardIdFromContentId(row.id);
   282→      const base = j * 12;
   283→      values.push(
   284→        Number(row.id),
   285→        row.externalid ?? null,
   286→        row.type ?? null,
   287→        row.title ?? 'untitled',
   288→        tsvText,
   289→        toIntOrNull(meta.year),
   290→        toFloatOrNull(meta.rating ?? meta.score),
   291→        genres,
   292→        actors,
   293→        studios,
   294→        row.type ?? null,
   295→        shardid,
   296→      );
   297→      const b = (k) => `$${base + k}`;
   298→      return `(${b(1)},${b(2)},${b(3)},${b(4)},to_tsvector('simple', ${b(5)}),${b(6)},${b(7)},${b(8)}::text[],${b(9)}::text[],${b(10)}::text[],${b(11)},${b(12)},NOW())`;
   299→    });
   300→    await rows(
   301→      `INSERT INTO searchdocs
   302→         (contentid, externalid, type, title, tsv, year, rating, genres, actors, studios, category, shardid, syncedat)
   303→       VALUES ${tuples.join(', ')}
   304→       ON CONFLICT (contentid) DO UPDATE SET
   305→         externalid = EXCLUDED.externalid,
   306→         type = EXCLUDED.type,
   307→         title = EXCLUDED.title,
   308→         tsv = EXCLUDED.tsv,
   309→         year = EXCLUDED.year,
   310→         rating = EXCLUDED.rating,
   311→         genres = EXCLUDED.genres,
   312→         actors = EXCLUDED.actors,
   313→         studios = EXCLUDED.studios,
   314→         category = EXCLUDED.category,
   315→         shardid = EXCLUDED.shardid,
   316→         syncedat = NOW()`,
   317→      values,
   318→    );
   319→    synced += slice.length;
   320→  }
   321→
   322→  // Drop stale docs (content rows deleted since the last sync).
   323→  try {
   324→    await rows('DELETE FROM searchdocs WHERE contentid NOT IN (SELECT id FROM content)');
   325→  } catch (err) {
   326→    logger.warn(`[search] stale-doc cleanup skipped: ${err.message}`);
   327→  }
   328→
   329→  lastSync = new Date().toISOString();
   330→  dirty = false;
   331→  return synced;
   332→}

   334→/** Rebuild once under an async lock; concurrent callers await the same run. */
   335→async function ensureSynced() {
   336→  if (!dirty && lastSync) return;
   337→  if (!syncPromise) {
   338→    syncPromise = syncIndex().finally(() => {
   339→      syncPromise = null;
   340→    });
   341→  }
   342→  await syncPromise;
   343→}

   345→/** Forced full resync (POST /index): waits for any in-flight sync, then runs one. */
   346→async function forceResync() {
   347→  while (syncPromise) {
   348→    await syncPromise.catch(() => {});
   349→  }
   350→  syncPromise = syncIndex().finally(() => {
   351→    syncPromise = null;
   352→  });
   353→  return syncPromise;
   354→}

   413→/**
   414→ * Run one advanced query against searchdocs (joined back to content).
   415→ * Filter fragments are built once and shared by the count + data statements.
   416→ * FAZA 17-a: `cursor` (base64url payload) switches to keyset pagination —
   417→ * the ORDER BY becomes the cursor continuation WHERE and OFFSET is dropped;
   418→ * `effectiveSort` is what actually orders the result (contentid keyset when
   419→ * there is no text query and the caller asked for the default 'score' sort,
   420→ * because score would be constant 0 for every row).
   421→ */
   422→async function runAdvancedQuery({
   423→  q,
   424→  type,
   425→  genre,
   426→  year,
   427→  minRating,
   428→  actor,
   429→  studio,
   430→  category,
   431→  sort,
   432→  fuzzyEnabled,
   433→  limit,
   434→  page,
   435→  cursor = null,
   436→}) {
   437→  const params = [];
   438→  const P = (v) => {
   439→    params.push(v);
   440→    return `$${params.length}`;
   441→  };
   442→
   443→  // Effective sort: filter-only (no q) with default 'score' → contentid keyset.
   444→  const effectiveSort = !q && sort === 'score' ? 'contentid' : sort;
   445→
   446→  // Keyset continuation (cursor present) — validated per effective sort.
   447→  let cursorPayload = null;
   448→  let cursorWhere = null;
   449→  let cursorParams = [];
   450→  if (cursor) {
   451→    const decoded = cursorContinuation(effectiveSort, cursor);
   452→    if (decoded.error) {
   453→      const err = new Error(decoded.error);
   454→      err.status = 400;
   455→      throw err;
   456→    }
   457→    cursorPayload = decoded.payload;
   458→    cursorWhere = decoded.where;
   459→    cursorParams = decoded.params;
   460→  }
   461→
   462→  // Field filters (SQL-side, case-insensitive where array containment).
   463→  const filters = [];
   464→  if (type) filters.push(`s.type = ${P(type)}`);
   465→  if (genre) filters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${P(genre)})`);
   466→  if (year != null) filters.push(`s.year = ${P(year)}`);
   467→  if (actor) {
   468→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${P(actor)} || '%')`);
   469→  }
   470→  if (studio) {
   471→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${P(studio)} || '%')`);
   472→  }
   473→  if (category) filters.push(`s.category = ${P(category)}`);
   474→  if (minRating != null) filters.push(`s.rating >= ${P(minRating)}`);
   475→  const filterSql = filters.length ? ` AND ${filters.join(' AND ')}` : '';
   476→
   477→  // Text score expression (combined: ts_rank * 2 + similarity + prefix bonus).
   478→  let scoreExpr = '0';
   479→  let textMatch = '';
   480→  if (q) {
   481→    const qp = P(q);
   482→    scoreExpr =
   483→      `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qp})) * 2` +
   484→      ` + similarity(s.title, ${qp})` +
   485→      ` + CASE WHEN s.title ILIKE ${qp} || '%' THEN 0.5 ELSE 0 END`;
   486→    textMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qp})`;
   487→  }
   488→
   489→  const whereFor = (matchSql) => `WHERE (${matchSql || 'TRUE'})${filterSql}`;
   490→
   491→  // 1. FTS pass (or filter-only when q is empty).
   492→  let matchSql = textMatch;
   493→  let countRow = await one(
   494→    `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   495→    params,
   496→  );
   497→
   498→  // 2. Trigram fuzzy fallback when the FTS pass matched nothing.
   499→  //    `title % q` (pg_trgm, default 0.3 threshold) plus a looser
   500→  //    similarity() > 0.1 guard so short typo'd queries still reach long
   501→  //    titles (e.g. 'dunee' → 'Dune: Part Two - Wikipedia', sim ≈ 0.15).
   502→  if (q && fuzzyEnabled && Number(countRow?.total ?? 0) === 0) {
   503→    const words = q.split(/\s+/).filter(Boolean);
   504→    const trigramClauses = [];
   505→    for (const candidate of [q, ...words]) {
   506→      const ph = P(`%${candidate}%`);
   507→      trigramClauses.push(`s.title ILIKE ${ph}`);
   508→    }
   509→    const qSim = P(q);
   510→    trigramClauses.push(`(s.title % ${qSim} OR similarity(s.title, ${qSim}) > 0.1)`);
   511→    matchSql = `(${trigramClauses.join(' OR ')})`;
   512→    countRow = await one(
   513→      `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   514→      params,
   515→    );
   516→  }
   517→
   518→  const total = Number(countRow?.total ?? 0);
   519→  const maxScore = Number(Number(countRow?.maxscore ?? 0).toFixed(6));
   520→
   521→  // 3. Data page — subquery keeps the score expression, then joins content.
   522→  //    FAZA 17-a: expose year/rating/title to JS for nextCursor construction;
   523→  //    cursor mode replaces OFFSET with the keyset continuation predicate.
   524→  const orderExpr =
   525→    effectiveSort === 'year'
   526→      ? 't.year DESC NULLS LAST, t.contentid ASC'
   527→      : effectiveSort === 'rating'
   528→        ? 't.rating DESC NULLS LAST, t.score DESC, t.contentid ASC'
   529→        : effectiveSort === 'title'
   530→          ? 't.title ASC, t.contentid ASC'
   531→          : effectiveSort === 'contentid'
   532→            ? 't.contentid ASC'
   533→            : 't.score DESC, t.contentid ASC';
   534→
   535→  const limitPh = P(limit);
   536→  const offsetPh = P((page - 1) * limit);
   537→  // Cursor mode: continuation params come FIRST ($1..$n), then LIMIT; the
   538→  // filter/score params are re-bound only inside the subquery — so build a
   539→  // dedicated statement for each mode instead of renumbering dynamically.
   540→  let dataSql;
   541→  let dataParams;
   542→  if (cursorWhere) {
   543→    // Rebuild the subquery with a fresh parameter namespace for the cursor run
   544→    // (filters + score expression are re-parameterized after the cursor slots).
   545→    const subParams = [...cursorParams, limit];
   546→    const sub = (v) => {
   547→      subParams.push(v);
   548→      return `$${subParams.length}`;
   549→    };
   550→    const cFilters = [];
   551→    if (type) cFilters.push(`s.type = ${sub(type)}`);
   552→    if (genre) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${sub(genre)})`);
   553→    if (year != null) cFilters.push(`s.year = ${sub(year)}`);
   554→    if (actor) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${sub(actor)} || '%')`);
   555→    if (studio) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${sub(studio)} || '%')`);
   556→    if (category) cFilters.push(`s.category = ${sub(category)}`);
   557→    if (minRating != null) cFilters.push(`s.rating >= ${sub(minRating)}`);
   558→    const cFilterSql = cFilters.length ? ` AND ${cFilters.join(' AND ')}` : '';
   559→
   560→    let cScoreExpr = '0';
   561→    let cMatch = 'TRUE';
   562→    // Single q slot, bound LAST (after cursor params + LIMIT) so every
   563→    // placeholder references a parameter that exists in the bind array.
   564→    let qParam = null;
   565→    if (q) {
   566→      qParam = sub(String(q));
   567→      cScoreExpr =
   568→        `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qParam})) * 2` +
   569→        ` + similarity(s.title, ${qParam})` +
   570→        ` + CASE WHEN s.title ILIKE ${qParam} || '%' THEN 0.5 ELSE 0 END`;
   571→      cMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qParam})`;
   572→    }
   573→    // Fuzzy fallback consistency: when the FTS pass matched nothing the offset
   574→    // run switched matchSql — mirror that for the cursor run using `total`.
   575→    if (q && fuzzyEnabled && total === 0) {
   576→      const words = q.split(/\s+/).filter(Boolean);
   577→      const clauses = [q, ...words].map((candidate) => `s.title ILIKE ${sub(`%${candidate}%`)}`);
   578→      clauses.push(`(s.title % ${qParam} OR similarity(s.title, ${qParam}) > 0.1)`);
   579→      cMatch = `(${clauses.join(' OR ')})`;
   580→    }
   581→
   582→    dataSql =
   583→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   584→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   585→       FROM (
   586→         SELECT s.contentid, s.year, s.rating, s.title, ${cScoreExpr} AS score
   587→         FROM searchdocs s
   588→         WHERE (${cMatch})${cFilterSql}
   589→       ) t
   590→       JOIN content c ON c.id = t.contentid
   591→       WHERE ${cursorWhere}
   592→       ORDER BY ${orderExpr}
   593→       LIMIT $${cursorParams.length + 1}::int`;
   594→    // LIMIT is the placeholder right after the cursor params (::int — a bare
   595→    // parameter in LIMIT cannot be type-inferred by Postgres); the q/filter
   596→    // slots follow it so every bound parameter is referenced by the statement.
   597→    dataParams = subParams;
   598→  } else {
   599→    dataSql =
   600→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   601→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   602→       FROM (
   603→         SELECT s.contentid, s.year, s.rating, s.title, ${scoreExpr} AS score
   604→         FROM searchdocs s
   605→         ${whereFor(matchSql)}
   606→       ) t
   607→       JOIN content c ON c.id = t.contentid
   608→       ORDER BY ${orderExpr}
   609→       LIMIT ${limitPh} OFFSET ${offsetPh}`;
   610→    dataParams = params;
   611→  }
   612→  const dataRows = await rows(dataSql, dataParams);
   613→
   614→  // Keyset paging estimates the page number by incrementing the cursor's `p`.
   615→  const resultPage = cursor ? (Number(cursorPayload?.p) || 1) + 1 : page;
   616→  const normalized = dataRows.map(normalizeHit).filter(Boolean);
   617→  const lastRaw = dataRows.length ? dataRows[dataRows.length - 1] : null;
   618→
   619→  return {
   620→    total,
   621→    max_score: maxScore,
   622→    data: normalized,
   623→    page: resultPage,
   624→    nextCursor: buildNextCursor({ sort: effectiveSort, lastRow: lastRaw, pageNumber: resultPage }),
   625→    paginationMode: cursor ? 'cursor' : 'offset',
   626→  };
   627→}

   632→ * FAZA 17-a: identical concurrent /advanced requests share one promise.
   633→ * Key = normalized query params; the map entry is removed on settle.
   634→ */
   635→const inflightSearch = new Map();
   636→
   637→function coalesceAdvanced(normalizedKey, run) {
   638→  const existing = inflightSearch.get(normalizedKey);
   639→  if (existing) {
   640→    searchOps.coalesced += 1;
   641→    return existing;
   642→  }
   643→  const promise = run().finally(() => {
   644→    inflightSearch.delete(normalizedKey);
   645→  });
   646→  inflightSearch.set(normalizedKey, promise);
   647→  return promise;
   648→}

   650→/** Race a query promise against the 8s budget (loser is logged, not awaited). */
   651→function withQueryTimeout(promise, describe) {
   652→  let timer = null;
   653→  const budget = new Promise((_, reject) => {
   654→    timer = setTimeout(() => {
   655→      const err = new Error('Search query timeout');
   656→      err.status = 504;
   657→      reject(err);
   658→    }, QUERY_TIMEOUT_MS);
   659→    if (typeof timer.unref === 'function') timer.unref();
   660→  });
   661→  return promise.then(
   662→    (value) => {
   663→      clearTimeout(timer);
   664→      return value;
   665→    },
   666→    (err) => {
   667→      clearTimeout(timer);
   668→      throw err;
   669→    },
   670→  ).finally(() => {
   671→    clearTimeout(timer);
   672→    promise.catch((loser) => {
   673→      // The lost race (if it also failed) must not become an unhandledRejection.
   674→      logger.warn(`[search] query lost the timeout race (${describe}): ${loser?.message ?? loser}`);
   675→    });
   676→  });
   677→}
