   138→   138→/** base64url encode/decode (no padding issues, URL-safe). */
   139→   139→function b64urlEncode(obj) {
   140→   140→  return Buffer.from(JSON.stringify(obj), 'utf8')
   141→   141→    .toString('base64')
   142→   142→    .replace(/\+/g, '-')
   143→   143→    .replace(/\//g, '_')
   144→   144→    .replace(/=+$/, '');
   145→   145→}

   147→   147→function b64urlDecode(text) {
   148→   148→  try {
   149→   149→    const padded = String(text).replace(/-/g, '+').replace(/_/g, '/');
   150→   150→    return JSON.parse(Buffer.from(padded, 'base64').toString('utf8'));
   151→   151→  } catch {
   152→   152→    return null;
   153→   153→  }
   154→   154→}

   156→   156→/**
   157→   157→ * Build the next cursor from the LAST row of the current page, per sort.
   158→   158→ * Filter-only + no q → effective sort is contentid (see effectiveSort).
   159→   159→ */
   160→   160→function buildNextCursor({ sort, lastRow, pageNumber }) {
   161→   161→  if (!lastRow) return null;
   162→   162→  const id = Number(lastRow.id);
   163→   163→  if (!Number.isFinite(id)) return null;
   164→   164→  let payload = null;
   165→   165→  if (sort === 'year') payload = { y: lastRow._year ?? null, id };
   166→   166→  else if (sort === 'rating') payload = { r: lastRow._rating ?? null, id };
   167→   167→  else if (sort === 'title') payload = { t: lastRow._title ?? null, id };
   168→   168→  else if (sort === 'score') payload = { s: Number(lastRow._score ?? 0), id };
   169→   169→  else payload = { id }; // contentid keyset (filter-only)
   170→   170→  payload.p = pageNumber; // page estimate — incremented on each cursor hop
   171→   171→  return b64urlEncode(payload);
   172→   172→}

   174→   174→/**
   175→   175→ * Decode + validate the cursor payload for a given effective sort.
   176→   176→ * @returns {{ payload: object, where: string, params: any[] } | { error: string }}
   177→   177→ */
   178→   178→function cursorContinuation(effectiveSort, cursorText) {
   179→   179→  const payload = b64urlDecode(cursorText);
   180→   180→  if (!payload || typeof payload !== 'object' || !Number.isFinite(Number(payload.id))) {
   181→   181→    return { error: 'Invalid cursor' };
   182→   182→  }
   183→   183→  const id = Number(payload.id);
   184→   184→  const params = [];
   185→   185→  const P = (v) => {
   186→   186→    params.push(v);
   187→   187→    return `$${params.length}`;
   188→   188→  };
   189→   189→
   190→   190→  if (effectiveSort === 'score') {
   191→   191→    const s = Number(payload.s);
   192→   192→    if (!Number.isFinite(s)) return { error: 'Invalid cursor (missing s)' };
   193→   193→    const sp = P(s);
   194→   194→    const ip = P(id);
   195→   195→    return { payload, where: `(t.score < ${sp} OR (t.score = ${sp} AND t.contentid > ${ip}))`, params };
   196→   196→  }
   197→   197→  if (effectiveSort === 'year') {
   198→   198→    const y = Number(payload.y);
   199→   199→    if (!Number.isFinite(y)) return { error: 'Invalid cursor (missing y)' };
   200→   200→    const yp = P(y);
   201→   201→    const ip = P(id);
   202→   202→    return { payload, where: `(t.year < ${yp} OR (t.year = ${yp} AND t.contentid > ${ip}))`, params };
   203→   203→  }
   204→   204→  if (effectiveSort === 'rating') {
   205→   205→    const r = Number(payload.r);
   206→   206→    if (!Number.isFinite(r)) return { error: 'Invalid cursor (missing r)' };
   207→   207→    const rp = P(r);
   208→   208→    const ip = P(id);
   209→   209→    return { payload, where: `(t.rating < ${rp} OR (t.rating = ${rp} AND t.contentid > ${ip}))`, params };
   210→   210→  }
   211→   211→  if (effectiveSort === 'title') {
   212→   212→    const t = payload.t;
   213→   213→    if (typeof t !== 'string') return { error: 'Invalid cursor (missing t)' };
   214→   214→    const tp = P(t);
   215→   215→    const ip = P(id);
   216→   216→    return { payload, where: `(t.title > ${tp} OR (t.title = ${tp} AND t.contentid > ${ip}))`, params };
   217→   217→  }
   218→   218→  // contentid keyset (filter-only / no q)
   219→   219→  const ip = P(id);
   220→   220→  return { payload, where: `t.contentid > ${ip}`, params };
   221→   221→}

   281→   281→      const shardid = computeShardIdFromContentId(row.id);
   282→   282→      const base = j * 12;

   286→   286→        row.type ?? null,

   291→   291→        genres,
   292→   292→        actors,
   293→   293→        studios,
   294→   294→        row.type ?? null,
   295→   295→        shardid,
   296→   296→      );
   297→   297→      const b = (k) => `$${base + k}`;
   298→   298→      return `(${b(1)},${b(2)},${b(3)},${b(4)},to_tsvector('simple', ${b(5)}),${b(6)},${b(7)},${b(8)}::text[],${b(9)}::text[],${b(10)}::text[],${b(11)},${b(12)},NOW())`;

   318→   318→    );

   320→   320→  }

   341→   341→  }

   418→   418→ * `effectiveSort` is what actually orders the result (contentid keyset when
   419→   419→ * there is no text query and the caller asked for the default 'score' sort,
   420→   420→ * because score would be constant 0 for every row).
   421→   421→ */
   422→   422→async function runAdvancedQuery({
   423→   423→  q,
   424→   424→  type,
   425→   425→  genre,
   426→   426→  year,
   427→   427→  minRating,
   428→   428→  actor,
   429→   429→  studio,
   430→   430→  category,
   431→   431→  sort,
   432→   432→  fuzzyEnabled,
   433→   433→  limit,
   434→   434→  page,
   435→   435→  cursor = null,
   436→   436→}) {
   437→   437→  const params = [];
   438→   438→  const P = (v) => {
   439→   439→    params.push(v);
   440→   440→    return `$${params.length}`;
   441→   441→  };
   442→   442→
   443→   443→  // Effective sort: filter-only (no q) with default 'score' → contentid keyset.
   444→   444→  const effectiveSort = !q && sort === 'score' ? 'contentid' : sort;
   445→   445→
   446→   446→  // Keyset continuation (cursor present) — validated per effective sort.
   447→   447→  let cursorPayload = null;
   448→   448→  let cursorWhere = null;
   449→   449→  let cursorParams = [];
   450→   450→  if (cursor) {
   451→   451→    const decoded = cursorContinuation(effectiveSort, cursor);
   452→   452→    if (decoded.error) {
   453→   453→      const err = new Error(decoded.error);
   454→   454→      err.status = 400;
   455→   455→      throw err;
   456→   456→    }
   457→   457→    cursorPayload = decoded.payload;
   458→   458→    cursorWhere = decoded.where;
   459→   459→    cursorParams = decoded.params;
   460→   460→  }
   461→   461→
   462→   462→  // Field filters (SQL-side, case-insensitive where array containment).
   463→   463→  const filters = [];
   464→   464→  if (type) filters.push(`s.type = ${P(type)}`);
   465→   465→  if (genre) filters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${P(genre)})`);
   466→   466→  if (year != null) filters.push(`s.year = ${P(year)}`);
   467→   467→  if (actor) {
   468→   468→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${P(actor)} || '%')`);
   469→   469→  }
   470→   470→  if (studio) {
   471→   471→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${P(studio)} || '%')`);
   472→   472→  }
   473→   473→  if (category) filters.push(`s.category = ${P(category)}`);
   474→   474→  if (minRating != null) filters.push(`s.rating >= ${P(minRating)}`);
   475→   475→  const filterSql = filters.length ? ` AND ${filters.join(' AND ')}` : '';
   476→   476→
   477→   477→  // Text score expression (combined: ts_rank * 2 + similarity + prefix bonus).
   478→   478→  let scoreExpr = '0';
   479→   479→  let textMatch = '';
   480→   480→  if (q) {
   481→   481→    const qp = P(q);
   482→   482→    scoreExpr =
   483→   483→      `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qp})) * 2` +
   484→   484→      ` + similarity(s.title, ${qp})` +
   485→   485→      ` + CASE WHEN s.title ILIKE ${qp} || '%' THEN 0.5 ELSE 0 END`;
   486→   486→    textMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qp})`;
   487→   487→  }
   488→   488→
   489→   489→  const whereFor = (matchSql) => `WHERE (${matchSql || 'TRUE'})${filterSql}`;
   490→   490→
   491→   491→  // 1. FTS pass (or filter-only when q is empty).
   492→   492→  let matchSql = textMatch;
   493→   493→  let countRow = await one(
   494→   494→    `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   495→   495→    params,
   496→   496→  );
   497→   497→
   498→   498→  // 2. Trigram fuzzy fallback when the FTS pass matched nothing.
   499→   499→  //    `title % q` (pg_trgm, default 0.3 threshold) plus a looser
   500→   500→  //    similarity() > 0.1 guard so short typo'd queries still reach long
   501→   501→  //    titles (e.g. 'dunee' → 'Dune: Part Two - Wikipedia', sim ≈ 0.15).
   502→   502→  if (q && fuzzyEnabled && Number(countRow?.total ?? 0) === 0) {
   503→   503→    const words = q.split(/\s+/).filter(Boolean);
   504→   504→    const trigramClauses = [];
   505→   505→    for (const candidate of [q, ...words]) {
   506→   506→      const ph = P(`%${candidate}%`);
   507→   507→      trigramClauses.push(`s.title ILIKE ${ph}`);
   508→   508→    }
   509→   509→    const qSim = P(q);
   510→   510→    trigramClauses.push(`(s.title % ${qSim} OR similarity(s.title, ${qSim}) > 0.1)`);
   511→   511→    matchSql = `(${trigramClauses.join(' OR ')})`;
   512→   512→    countRow = await one(
   513→   513→      `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   514→   514→      params,
   515→   515→    );
   516→   516→  }
   517→   517→
   518→   518→  const total = Number(countRow?.total ?? 0);
   519→   519→  const maxScore = Number(Number(countRow?.maxscore ?? 0).toFixed(6));
   520→   520→
   521→   521→  // 3. Data page — subquery keeps the score expression, then joins content.
   522→   522→  //    FAZA 17-a: expose year/rating/title to JS for nextCursor construction;
   523→   523→  //    cursor mode replaces OFFSET with the keyset continuation predicate.
   524→   524→  const orderExpr =
   525→   525→    effectiveSort === 'year'
   526→   526→      ? 't.year DESC NULLS LAST, t.contentid ASC'
   527→   527→      : effectiveSort === 'rating'
   528→   528→        ? 't.rating DESC NULLS LAST, t.score DESC, t.contentid ASC'
   529→   529→        : effectiveSort === 'title'
   530→   530→          ? 't.title ASC, t.contentid ASC'
   531→   531→          : effectiveSort === 'contentid'
   532→   532→            ? 't.contentid ASC'
   533→   533→            : 't.score DESC, t.contentid ASC';
   534→   534→
   535→   535→  const limitPh = P(limit);
   536→   536→  const offsetPh = P((page - 1) * limit);
   537→   537→  // Cursor mode: continuation params come FIRST ($1..$n), then LIMIT; the
   538→   538→  // filter/score params are re-bound only inside the subquery — so build a
   539→   539→  // dedicated statement for each mode instead of renumbering dynamically.
   540→   540→  let dataSql;
   541→   541→  let dataParams;
   542→   542→  if (cursorWhere) {
   543→   543→    // Rebuild the subquery with a fresh parameter namespace for the cursor run
   544→   544→    // (filters + score expression are re-parameterized after the cursor slots).
   545→   545→    const subParams = [...cursorParams, limit];
   546→   546→    const sub = (v) => {
   547→   547→      subParams.push(v);
   548→   548→      return `$${subParams.length}`;
   549→   549→    };
   550→   550→    const cFilters = [];
   551→   551→    if (type) cFilters.push(`s.type = ${sub(type)}`);
   552→   552→    if (genre) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${sub(genre)})`);
   553→   553→    if (year != null) cFilters.push(`s.year = ${sub(year)}`);
   554→   554→    if (actor) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${sub(actor)} || '%')`);
   555→   555→    if (studio) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${sub(studio)} || '%')`);
   556→   556→    if (category) cFilters.push(`s.category = ${sub(category)}`);
   557→   557→    if (minRating != null) cFilters.push(`s.rating >= ${sub(minRating)}`);
   558→   558→    const cFilterSql = cFilters.length ? ` AND ${cFilters.join(' AND ')}` : '';
   559→   559→
   560→   560→    let cScoreExpr = '0';
   561→   561→    let cMatch = 'TRUE';
   562→   562→    // Single q slot, bound LAST (after cursor params + LIMIT) so every
   563→   563→    // placeholder references a parameter that exists in the bind array.
   564→   564→    let qParam = null;
   565→   565→    if (q) {
   566→   566→      qParam = sub(String(q));
   567→   567→      cScoreExpr =
   568→   568→        `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qParam})) * 2` +
   569→   569→        ` + similarity(s.title, ${qParam})` +
   570→   570→        ` + CASE WHEN s.title ILIKE ${qParam} || '%' THEN 0.5 ELSE 0 END`;
   571→   571→      cMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qParam})`;
   572→   572→    }
   573→   573→    // Fuzzy fallback consistency: when the FTS pass matched nothing the offset
   574→   574→    // run switched matchSql — mirror that for the cursor run using `total`.
   575→   575→    if (q && fuzzyEnabled && total === 0) {
   576→   576→      const words = q.split(/\s+/).filter(Boolean);
   577→   577→      const clauses = [q, ...words].map((candidate) => `s.title ILIKE ${sub(`%${candidate}%`)}`);
   578→   578→      clauses.push(`(s.title % ${qParam} OR similarity(s.title, ${qParam}) > 0.1)`);
   579→   579→      cMatch = `(${clauses.join(' OR ')})`;
   580→   580→    }
   581→   581→
   582→   582→    dataSql =
   583→   583→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   584→   584→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   585→   585→       FROM (
   586→   586→         SELECT s.contentid, s.year, s.rating, s.title, ${cScoreExpr} AS score
   587→   587→         FROM searchdocs s
   588→   588→         WHERE (${cMatch})${cFilterSql}
   589→   589→       ) t
   590→   590→       JOIN content c ON c.id = t.contentid
   591→   591→       WHERE ${cursorWhere}
   592→   592→       ORDER BY ${orderExpr}
   593→   593→       LIMIT $${cursorParams.length + 1}::int`;
   594→   594→    // LIMIT is the placeholder right after the cursor params (::int — a bare
   595→   595→    // parameter in LIMIT cannot be type-inferred by Postgres); the q/filter
   596→   596→    // slots follow it so every bound parameter is referenced by the statement.
   597→   597→    dataParams = subParams;
   598→   598→  } else {
   599→   599→    dataSql =
   600→   600→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   601→   601→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   602→   602→       FROM (
   603→   603→         SELECT s.contentid, s.year, s.rating, s.title, ${scoreExpr} AS score
   604→   604→         FROM searchdocs s
   605→   605→         ${whereFor(matchSql)}
   606→   606→       ) t
   607→   607→       JOIN content c ON c.id = t.contentid
   608→   608→       ORDER BY ${orderExpr}
   609→   609→       LIMIT ${limitPh} OFFSET ${offsetPh}`;
   610→   610→    dataParams = params;
   611→   611→  }
   612→   612→  const dataRows = await rows(dataSql, dataParams);
   613→   613→
   614→   614→  // Keyset paging estimates the page number by incrementing the cursor's `p`.
   615→   615→  const resultPage = cursor ? (Number(cursorPayload?.p) || 1) + 1 : page;
   616→   616→  const normalized = dataRows.map(normalizeHit).filter(Boolean);
   617→   617→  const lastRaw = dataRows.length ? dataRows[dataRows.length - 1] : null;
   618→   618→
   619→   619→  return {
   620→   620→    total,
   621→   621→    max_score: maxScore,
   622→   622→    data: normalized,
   623→   623→    page: resultPage,
   624→   624→    nextCursor: buildNextCursor({ sort: effectiveSort, lastRow: lastRaw, pageNumber: resultPage }),
   625→   625→    paginationMode: cursor ? 'cursor' : 'offset',
   626→   626→  };
   627→   627→}

   652→   652→  let timer = null;
   653→   653→  const budget = new Promise((_, reject) => {

   660→   660→  });
   661→   661→  return promise.then(
   662→   662→    (value) => {

   675→   675→    });
   676→   676→  });
   677→   677→}
