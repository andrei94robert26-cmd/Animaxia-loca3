    68→function fmtRoNum(v: number | null | undefined): string {

    72→function fmtMs(v: number | null | undefined): string {
    73→  const n = Number(v ?? 0);
    74→  return Number.isFinite(n) && n > 0 ? `${RO_NUMBER.format(Math.round(n))} ms` : '—';
    75→}

    77→/** HH:MM:SS clock label for the history X axis (samples are 1 s apart). */
    78→function fmtClock(iso: string): string {
    79→  const d = new Date(iso);
    80→  if (Number.isNaN(d.getTime())) return iso;
    81→  const p = (n: number) => String(n).padStart(2, '0');
    82→  return `${p(d.getHours())}:${p(d.getMinutes())}:${p(d.getSeconds())}`;
    83→}

   150→/* ---------------------------- settings form ------------------------------ */
   151→
   152→const SETTINGS_FIELDS = [
   153→  { key: 'perUserMs', label: 'Limită / utilizator', hint: '100 – 5.000 ms', min: 100, max: 5000 },
   154→  { key: 'cooldownMs', label: 'Cooldown după rezultat', hint: '250 – 10.000 ms', min: 250, max: 10000 },
   155→  { key: 'spamThreshold', label: 'Prag anti-spam (ENTER)', hint: '2 – 10 apăsări', min: 2, max: 10 },
   156→  { key: 'spamBlockMs', label: 'Blocare anti-spam', hint: '100 – 10.000 ms', min: 100, max: 10000 },
   157→  { key: 'burstThreshold', label: 'Prag burst / fereastră', hint: '10 – 100.000', min: 10, max: 100000 },
   158→] as const;
   159→
   160→type SettingsKey = (typeof SETTINGS_FIELDS)[number]['key'];
   161→type SettingsForm = Record<SettingsKey, string>;
   162→
   163→function RateSettingsCard({
   164→  config,
   165→  isAdmin,
   166→  onSaved,
   167→}: {
   168→  config: SearchRateConfig;
   169→  isAdmin: boolean;
   170→  onSaved: () => void;
   171→}) {
   172→  const [form, setForm] = useState<SettingsForm>({
   173→    perUserMs: String(config.perUserMs),
   174→    cooldownMs: String(config.cooldownMs),
   175→    spamThreshold: String(config.spamThreshold),
   176→    spamBlockMs: String(config.spamBlockMs),
   177→    burstThreshold: String(config.burstThreshold),
   178→  });
   179→  const [dirty, setDirty] = useState(false);
   180→  const [saving, setSaving] = useState(false);
   181→
   182→  /* Re-sync the pre-filled inputs from the polled config while the user has
   183→     not edited anything (and right after a successful save). */
   184→  useEffect(() => {
   185→    if (dirty) return;
   186→    setForm((prev) => {
   187→      const next: SettingsForm = {
   188→        perUserMs: String(config.perUserMs),
   189→        cooldownMs: String(config.cooldownMs),
   190→        spamThreshold: String(config.spamThreshold),
   191→        spamBlockMs: String(config.spamBlockMs),
   192→        burstThreshold: String(config.burstThreshold),
   193→      };
   194→      return (Object.keys(next) as SettingsKey[]).every((k) => prev[k] === next[k]) ? prev : next;
   195→    });
   196→  }, [config, dirty]);
   197→
   198→  async function handleSave() {
   199→    if (saving) return;
   200→    const parsed: Record<SettingsKey, number> = {
   201→      perUserMs: Number(form.perUserMs),
   202→      cooldownMs: Number(form.cooldownMs),
   203→      spamThreshold: Number(form.spamThreshold),
   204→      spamBlockMs: Number(form.spamBlockMs),
   205→      burstThreshold: Number(form.burstThreshold),
   206→    };
   207→    for (const f of SETTINGS_FIELDS) {
   208→      const v = parsed[f.key];
   209→      if (!Number.isFinite(v) || v < f.min || v > f.max) {
   210→        toast.error(`${f.label} trebuie să fie între ${RO_NUMBER.format(f.min)} și ${RO_NUMBER.format(f.max)}`);
   211→        return;
   212→      }
   213→    }
   214→    setSaving(true);
   215→    try {
   216→      await authRequest<{ ok: boolean; config: SearchRateConfig }>(
   217→        `/api/search/rate/settings?XTransformPort=${BACKEND_PORT}`,
   218→        { method: 'POST', body: JSON.stringify(parsed) },
   219→      );
   220→      toast.success('Setări salvate');
   221→      setDirty(false);
   222→      onSaved();
   223→    } catch (e) {
   224→      if (e instanceof PlatformError && (e.status === 401 || e.status === 403)) {
   225→        toast.error('Setările ritmului necesită cont de administrator');
   226→      } else {
   227→        toast.error(e instanceof Error ? e.message : 'Salvarea setărilor a eșuat');
   228→      }
   229→    } finally {
   230→      setSaving(false);
   231→    }
   232→  }
   233→
   234→  return (
   235→    <Card className="border-white/10 bg-white/[0.03]">
   236→      <CardHeader className="pb-3">
   237→        <CardTitle className="flex items-center gap-2 text-base">
   238→          <ShieldCheck className="h-4 w-4 text-rose-400" aria-hidden />
   239→          Setări runtime (administrator)
   240→        </CardTitle>
   241→        <CardDescription>
   242→          Reglare live a ritmului de căutare — se aplică imediat, fără restart.
   243→        </CardDescription>
   244→      </CardHeader>
   245→      <CardContent className="space-y-4">
   246→        {isAdmin ? (
   247→          <>
   248→            <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-5">
   249→              {SETTINGS_FIELDS.map((f) => (
   250→                <div key={f.key} className="space-y-1.5">
   251→                  <Label htmlFor={`rate-setting-${f.key}`} className="text-xs text-zinc-400">
   252→                    {f.label} <span className="font-normal text-zinc-600">({f.hint})</span>
   253→                  </Label>
   254→                  <Input
   255→                    id={`rate-setting-${f.key}`}
   256→                    type="number"
   257→                    inputMode="numeric"
   258→                    min={f.min}
   259→                    max={f.max}
   260→                    step={1}
   261→                    value={form[f.key]}
   262→                    onChange={(e) => {
   263→                      setDirty(true);
   264→                      setForm((prev) => ({ ...prev, [f.key]: e.target.value }));
   265→                    }}
   266→                    className="border-white/10 bg-white/[0.04] font-mono text-zinc-100"
   267→                    aria-label={`${f.label}, interval acceptat ${f.hint}`}
   268→                  />
   269→                </div>
   270→              ))}
   271→            </div>
   272→            <div className="flex flex-wrap items-center justify-between gap-3">
   273→              <p className="max-w-xl text-[11px] leading-snug text-zinc-600">
   274→                Validarea și limitarea valorilor (clamping) se fac pe server — valorile finale sunt cele
   275→                returnate după salvare și reflectate mai sus în cardurile de configurare.
   276→              </p>
   277→              <Button
   278→                size="sm"
   279→                onClick={() => void handleSave()}
   280→                disabled={saving || !dirty}
   281→                className="bg-rose-600 font-bold hover:bg-rose-500"
   282→              >
   283→                {saving ? <RefreshCw className="animate-spin" aria-hidden /> : <Save aria-hidden />}
   284→                Salvează
   285→              </Button>
   286→            </div>
   287→          </>
   288→        ) : (
   289→          <p className="flex items-center gap-2 rounded-xl border border-amber-500/25 bg-amber-500/[0.06] px-4 py-2.5 text-xs text-amber-200">
   290→            <Lock className="h-3.5 w-3.5 shrink-0" aria-hidden />
   291→            Setările ritmului necesită cont de administrator — valorile curente sunt read-only.
   292→          </p>
   293→        )}
   294→      </CardContent>
   295→    </Card>
   296→  );
   297→}

   480→              {
   481→                label: 'Limită / utilizator',
   482→                value: fmtMs(config.perUserMs),
   483→                sub: 'interval minim între căutări (per tip)',
   484→              },
   485→              {
   486→                label: 'Cooldown',
   487→                value: fmtMs(config.cooldownMs),
   488→                sub: 'după primirea unui rezultat',
   489→              },
   490→              {
   491→                label: 'Anti-spam',
   492→                value: `${fmtRoNum(config.spamThreshold)} / ${fmtMs(config.spamWindowMs)}`,
   493→                sub: `apăsări ENTER → blocare ${fmtMs(config.spamBlockMs)}`,
   494→              },
   495→              {
   496→                label: 'Burst prag',
   497→                value: fmtRoNum(config.burstThreshold),
   498→                sub: `sosiri / ${fmtMs(config.burstWindowMs)} = eveniment burst`,
   499→              },
   500→              {
   501→                label: 'Strat FAZA 16',
   502→                value: config.enabled ? 'ACTIV' : 'DEZACTIVAT',
   503→                sub: config.enabled ? 'rate-limit + cooldown + anti-spam' : 'doar coada FAZA 15 rămâne',
   504→              },
   505→            ].map((c) => (
   506→              <Card key={c.label} className="border-white/10 bg-white/[0.03]">
   507→                <CardContent className="p-4">
   508→                  <p className="text-lg font-black leading-tight text-white">{c.value}</p>
   509→                  <p className="mt-0.5 text-[11px] font-semibold uppercase tracking-widest text-zinc-500">{c.label}</p>
   510→                  <p className="mt-1 text-[11px] leading-snug text-zinc-600">{c.sub}</p>
   511→                </CardContent>
   512→              </Card>
   513→            ))}
   514→          </div>
   515→        </section>
   516→      ) : null}
   517→
   518→      {/* Chart: arrivals + blocked over the rolling history */}
   519→      <section aria-label="Istoricul ritmului de căutare">
   520→        <Card className="border-white/10 bg-white/[0.03]">
   521→          <CardHeader className="pb-2">
   522→            <CardTitle className="flex items-center gap-2 text-base">
   523→              <LineChart className="h-4 w-4 text-rose-400" aria-hidden />
   524→              Sosiri &amp; respingeri (ultimele 3 minute)
   525→            </CardTitle>
   526→            <CardDescription>
   527→              180 eșantioane de 1 s — sosiri căutări (emerald) și cereri respinse explicit (rose).
   528→            </CardDescription>
   529→          </CardHeader>
   530→          <CardContent>
   531→            {loadingFirst ? (
   532→              <Skeleton className="h-56 w-full rounded-xl sm:h-64" />
   533→            ) : chartData.length === 0 ? (
   534→              <EmptyState
   535→                title="Fără activitate încă"
   536→                message="Istoricul se populează odată cu primele căutări trecute prin stratul de ritm."
   537→              />
   538→            ) : (
   539→              <ChartContainer config={RATE_CHART_CONFIG} className="h-56 w-full sm:h-64">
   540→                <AreaChart data={chartData} margin={{ left: 4, right: 8, top: 8, bottom: 0 }}>
   541→                  <defs>
   542→                    <linearGradient id="fillRateArrivals" x1="0" y1="0" x2="0" y2="1">
   543→                      <stop offset="5%" stopColor="#10b981" stopOpacity={0.45} />
   544→                      <stop offset="95%" stopColor="#10b981" stopOpacity={0.02} />
   545→                    </linearGradient>
   546→                    <linearGradient id="fillRateBlocked" x1="0" y1="0" x2="0" y2="1">
   547→                      <stop offset="5%" stopColor="#f43f5e" stopOpacity={0.45} />
   548→                      <stop offset="95%" stopColor="#f43f5e" stopOpacity={0.02} />
   549→                    </linearGradient>
   550→                  </defs>
   551→                  <CartesianGrid vertical={false} stroke="rgba(255,255,255,0.08)" />
   552→                  <XAxis
   553→                    dataKey="clock"
   554→                    tickLine={false}
   555→                    axisLine={false}
   556→                    minTickGap={32}
   557→                    tick={{ fill: '#71717a', fontSize: 11 }}
   558→                  />
   559→                  <YAxis
   560→                    allowDecimals={false}
   561→                    tickLine={false}
   562→                    axisLine={false}
   563→                    width={30}
   564→                    tick={{ fill: '#71717a', fontSize: 11 }}
   565→                  />
   566→                  <ChartTooltip
   567→                    cursor={{ stroke: 'rgba(16,185,129,0.35)' }}
   568→                    content={
   569→                      <ChartTooltipContent
   570→                        className="border-white/10 bg-[#1a1a24] text-zinc-200"
   571→                        indicator="line"
   572→                      />
   573→                    }
   574→                  />
   575→                  <Area
   576→                    dataKey="sosiri"
   577→                    type="monotone"
   578→                    stroke="#10b981"
   579→                    strokeWidth={2}
   580→                    fill="url(#fillRateArrivals)"
   581→                    dot={false}
   582→                    activeDot={{ r: 4, fill: '#10b981' }}
   583→                  />
   584→                  <Area
   585→                    dataKey="blocate"
   586→                    type="monotone"
   587→                    stroke="#f43f5e"
   588→                    strokeWidth={2}
   589→                    fill="url(#fillRateBlocked)"
   590→                    dot={false}
   591→                    activeDot={{ r: 4, fill: '#f43f5e' }}
   592→                  />
   593→                </AreaChart>
   594→              </ChartContainer>
   595→            )}
   596→          </CardContent>
   597→        </Card>
   598→      </section>
   599→
   600→      {/* Per-shard attribution */}
   601→      {loadingFirst ? (
   602→        <div className="space-y-2">
   603→          {Array.from({ length: 4 }).map((_, i) => (
   604→            <Skeleton key={i} className="h-12 w-full rounded-xl" />
   605→          ))}
   606→        </div>
   607→      ) : rate.error && !data ? (
   608→        <ErrorState message={rate.error} onRetry={rate.refetch} compact />
   609→      ) : shards.length === 0 ? (
   610→        <EmptyState
   611→          title="Fără sharduri"
   612→          message="Backend-ul nu a raportat încă atribuiri per-shard pentru ritmul de căutare."
   613→        />

   615→        <section aria-label="Atribuirea ritmului de căutare pe sharduri" className="space-y-3">
   616→          <div className="flex flex-wrap items-center justify-between gap-3">
   617→            <h3 className="flex items-center gap-2 text-sm font-semibold text-zinc-200">
   618→              <Network className="h-4 w-4 text-rose-400" aria-hidden />
   619→              Atribuire pe sharduri ({shards.length})
   620→            </h3>
   621→            <span className="font-mono text-[11px] text-zinc-600">
   622→              rândurile cu activitate sunt evidențiate în amber · respingerile în roșu
   623→            </span>
   624→          </div>
   625→          <div className="custom-scrollbar max-h-[32rem] overflow-y-auto rounded-xl border border-white/10">
   626→            <Table aria-label="Atribuirea ritmului de căutare pe sharduri — un rând per shard logic">

   630→                  <TableHead scope="col">Admise</TableHead>

   637→                {shards.map((row) => {

   640→                  return (

   642→                      key={row.shardid}
   643→                      className={cn(
   644→                        'hover:bg-white/[0.03]',

   647→                      )}
   648→                    >
   649→                      <TableCell>
   650→                        <div className="flex items-center gap-2">
   651→                          <Server className="h-3.5 w-3.5 shrink-0 text-zinc-500" aria-hidden />
   652→                          <p className="font-mono text-sm font-medium text-zinc-100">#{row.shardid}</p>
   653→                        </div>
   654→                      </TableCell>
   655→                      <TableCell className="font-mono text-xs text-zinc-300">{fmtRoNum(row.admitted)}</TableCell>
   656→                      <TableCell
   657→                        className={cn('font-mono text-xs', row.rateLimited > 0 ? 'text-red-300' : 'text-zinc-600')}
   658→                      >
   659→                        {fmtRoNum(row.rateLimited)}
   660→                      </TableCell>
   661→                      <TableCell
   662→                        className={cn('font-mono text-xs', row.cooldownActive > 0 ? 'text-red-300' : 'text-zinc-600')}
   663→                      >
   664→                        {fmtRoNum(row.cooldownActive)}
   665→                      </TableCell>
   666→                      <TableCell
   667→                        className={cn('font-mono text-xs', row.spamBlocked > 0 ? 'text-red-300' : 'text-zinc-600')}
   668→                      >
   669→                        {fmtRoNum(row.spamBlocked)}
   670→                      </TableCell>
   671→                    </TableRow>
   672→                  );
   673→                })}
   674→              </TableBody>
   675→            </Table>
   676→          </div>
   677→        </section>
   678→      )}
   679→
   680→      {/* Settings (admin only) */}
   681→      {config ? (
   682→        <RateSettingsCard
   683→          config={config}
   684→          isAdmin={!initializing && isAdmin}
   685→          onSaved={() => void rate.refetch()}
   686→        />
   687→      ) : loadingFirst ? (
   688→        <Skeleton className="h-56 w-full rounded-xl" />
   689→      ) : null}
   690→    </div>
   691→  );
   692→}
