   161→   161→/** base64url encode/decode (no padding issues, URL-safe). */
   162→   162→function b64urlEncode(obj) {
   163→   163→  return Buffer.from(JSON.stringify(obj), 'utf8')
   164→   164→    .toString('base64')
   165→   165→    .replace(/\+/g, '-')
   166→   166→    .replace(/\//g, '_')
   167→   167→    .replace(/=+$/, '');
   168→   168→}
   169→   169→
   170→   170→function b64urlDecode(text) {
   171→   171→  try {
   172→   172→    const padded = String(text).replace(/-/g, '+').replace(/_/g, '/');
   173→   173→    return JSON.parse(Buffer.from(padded, 'base64').toString('utf8'));
   174→   174→  } catch {
   175→   175→    return null;
   176→   176→  }
   177→   177→}

   179→   179→/**
   180→   180→ * Build the next cursor from the LAST row of the current page, per sort.
   181→   181→ * Filter-only + no q → effective sort is contentid (see effectiveSort).
   182→   182→ */
   183→   183→function buildNextCursor({ sort, lastRow, pageNumber }) {
   184→   184→  if (!lastRow) return null;
   185→   185→  const id = Number(lastRow.id);
   186→   186→  if (!Number.isFinite(id)) return null;
   187→   187→  let payload = null;
   188→   188→  if (sort === 'year') payload = { y: lastRow._year ?? null, id };
   189→   189→  else if (sort === 'rating') payload = { r: lastRow._rating ?? null, id };
   190→   190→  else if (sort === 'title') payload = { t: lastRow._title ?? null, id };
   191→   191→  else if (sort === 'score') payload = { s: Number(lastRow._score ?? 0), id };
   192→   192→  else payload = { id }; // contentid keyset (filter-only)
   193→   193→  payload.p = pageNumber; // page estimate — incremented on each cursor hop
   194→   194→  return b64urlEncode(payload);
   195→   195→}

   197→   197→/**
   198→   198→ * Decode + validate the cursor payload for a given effective sort.
   199→   199→ * @returns {{ payload: object, where: string, params: any[] } | { error: string }}
   200→   200→ */
   201→   201→function cursorContinuation(effectiveSort, cursorText) {
   202→   202→  const payload = b64urlDecode(cursorText);
   203→   203→  if (!payload || typeof payload !== 'object' || !Number.isFinite(Number(payload.id))) {
   204→   204→    return { error: 'Invalid cursor' };
   205→   205→  }
   206→   206→  const id = Number(payload.id);
   207→   207→  const params = [];
   208→   208→  const P = (v) => {
   209→   209→    params.push(v);
   210→   210→    return `$${params.length}`;
   211→   211→  };
   212→   212→
   213→   213→  if (effectiveSort === 'score') {
   214→   214→    const s = Number(payload.s);
   215→   215→    if (!Number.isFinite(s)) return { error: 'Invalid cursor (missing s)' };
   216→   216→    const sp = P(s);
   217→   217→    const ip = P(id);
   218→   218→    return { payload, where: `(t.score < ${sp} OR (t.score = ${sp} AND t.contentid > ${ip}))`, params };
   219→   219→  }
   220→   220→  if (effectiveSort === 'year') {
   221→   221→    const y = Number(payload.y);
   222→   222→    if (!Number.isFinite(y)) return { error: 'Invalid cursor (missing y)' };
   223→   223→    const yp = P(y);
   224→   224→    const ip = P(id);
   225→   225→    return { payload, where: `(t.year < ${yp} OR (t.year = ${yp} AND t.contentid > ${ip}))`, params };
   226→   226→  }
   227→   227→  if (effectiveSort === 'rating') {
   228→   228→    const r = Number(payload.r);
   229→   229→    if (!Number.isFinite(r)) return { error: 'Invalid cursor (missing r)' };
   230→   230→    const rp = P(r);
   231→   231→    const ip = P(id);
   232→   232→    return { payload, where: `(t.rating < ${rp} OR (t.rating = ${rp} AND t.contentid > ${ip}))`, params };
   233→   233→  }
   234→   234→  if (effectiveSort === 'title') {
   235→   235→    const t = payload.t;
   236→   236→    if (typeof t !== 'string') return { error: 'Invalid cursor (missing t)' };
   237→   237→    const tp = P(t);
   238→   238→    const ip = P(id);
   239→   239→    return { payload, where: `(t.title > ${tp} OR (t.title = ${tp} AND t.contentid > ${ip}))`, params };
   240→   240→  }
   241→   241→  // contentid keyset (filter-only / no q)
   242→   242→  const ip = P(id);
   243→   243→  return { payload, where: `t.contentid > ${ip}`, params };
   244→   244→}

   248→   248→/** Coerce a metadata value to a clean string array (strings or {name} objects). */
   249→   249→function toNameArray(value) {
   250→   250→  if (value == null) return [];
   251→   251→  const list = Array.isArray(value) ? value : [value];
   252→   252→  const out = [];
   253→   253→  for (const item of list) {
   254→   254→    if (typeof item === 'string') {
   255→   255→      const name = item.trim();
   256→   256→      if (name) out.push(name);
   257→   257→    } else if (item && typeof item === 'object') {
   258→   258→      const name = String(item.name ?? '').trim();
   259→   259→      if (name) out.push(name);
   260→   260→    }
   261→   261→  }
   262→   262→  return out;
   263→   263→}

   287→   287→async function syncIndex() {
   288→   288→  const raw = await rows(
   289→   289→    'SELECT id, externalid, type, title, metadata, thumbnail, createdat FROM content',
   290→   290→  );
   291→   291→
   292→   292→  let synced = 0;
   293→   293→  for (let i = 0; i < raw.length; i += SYNC_CHUNK) {
   294→   294→    const slice = raw.slice(i, i + SYNC_CHUNK);
   295→   295→    const values = [];
   296→   296→    const tuples = slice.map((row, j) => {
   297→   297→      const meta = parseJsonb(row.metadata, {});
   298→   298→      const genres = toNameArray(meta.genres);
   299→   299→      const actors = toNameArray(meta.actors ?? meta.cast);
   300→   300→      const studios = toNameArray(meta.studios ?? meta.studio);
   301→   301→      const tsvText = buildTsvText(row.title, genres, meta);
   302→   302→      // FAZA 13 (sharding): searchdocs carries its shard id = the catalog
   303→   303→      // RANGE shard of the contentid it indexes (ids < 1B → 0, legacy-safe).
   304→   304→      const shardid = computeShardIdFromContentId(row.id);
   305→   305→      const base = j * 12;
   306→   306→      values.push(
   307→   307→        Number(row.id),
   308→   308→        row.externalid ?? null,
   309→   309→        row.type ?? null,
   310→   310→        row.title ?? 'untitled',
   311→   311→        tsvText,
   312→   312→        toIntOrNull(meta.year),
   313→   313→        toFloatOrNull(meta.rating ?? meta.score),
   314→   314→        genres,
   315→   315→        actors,
   316→   316→        studios,
   317→   317→        row.type ?? null,
   318→   318→        shardid,
   319→   319→      );
   320→   320→      const b = (k) => `$${base + k}`;
   321→   321→      return `(${b(1)},${b(2)},${b(3)},${b(4)},to_tsvector('simple', ${b(5)}),${b(6)},${b(7)},${b(8)}::text[],${b(9)}::text[],${b(10)}::text[],${b(11)},${b(12)},NOW())`;
   322→   322→    });
   323→   323→    await rows(
   324→   324→      `INSERT INTO searchdocs
   325→   325→         (contentid, externalid, type, title, tsv, year, rating, genres, actors, studios, category, shardid, syncedat)
   326→   326→       VALUES ${tuples.join(', ')}
   327→   327→       ON CONFLICT (contentid) DO UPDATE SET
   328→   328→         externalid = EXCLUDED.externalid,
   329→   329→         type = EXCLUDED.type,
   330→   330→         title = EXCLUDED.title,
   331→   331→         tsv = EXCLUDED.tsv,
   332→   332→         year = EXCLUDED.year,
   333→   333→         rating = EXCLUDED.rating,
   334→   334→         genres = EXCLUDED.genres,
   335→   335→         actors = EXCLUDED.actors,
   336→   336→         studios = EXCLUDED.studios,
   337→   337→         category = EXCLUDED.category,
   338→   338→         shardid = EXCLUDED.shardid,
   339→   339→         syncedat = NOW()`,
   340→   340→      values,
   341→   341→    );
   342→   342→    synced += slice.length;
   343→   343→  }
   344→   344→
   345→   345→  // Drop stale docs (content rows deleted since the last sync).
   346→   346→  try {
   347→   347→    await rows('DELETE FROM searchdocs WHERE contentid NOT IN (SELECT id FROM content)');
   348→   348→  } catch (err) {
   349→   349→    logger.warn(`[search] stale-doc cleanup skipped: ${err.message}`);
   350→   350→  }
   351→   351→
   352→   352→  lastSync = new Date().toISOString();
   353→   353→  dirty = false;
   354→   354→  return synced;
   355→   355→}

   357→   357→/** Rebuild once under an async lock; concurrent callers await the same run. */
   358→   358→async function ensureSynced() {
   359→   359→  if (!dirty && lastSync) return;
   360→   360→  if (!syncPromise) {
   361→   361→    syncPromise = syncIndex().finally(() => {
   362→   362→      syncPromise = null;
   363→   363→    });
   364→   364→  }
   365→   365→  await syncPromise;
   366→   366→}

   368→   368→/** Forced full resync (POST /index): waits for any in-flight sync, then runs one. */
   369→   369→async function forceResync() {
   370→   370→  while (syncPromise) {
   371→   371→    await syncPromise.catch(() => {});
   372→   372→  }
   373→   373→  syncPromise = syncIndex().finally(() => {
   374→   374→    syncPromise = null;
   375→   375→  });
   376→   376→  return syncPromise;
   377→   377→}

   400→   400→function percentile(sorted, p) {
   401→   401→  if (sorted.length === 0) return 0;
   402→   402→  const idx = Math.min(sorted.length - 1, Math.max(0, Math.ceil(p * sorted.length) - 1));

   404→   404→}

   414→   414→}

   416→   416→/* ----------------------------- query construction ---------------------------- */

   418→   418→const SORTS = new Set(['score', 'year', 'rating', 'title']);
   419→   419→
   420→   420→/** Normalize a joined row to the canonical content shape + _score. */
   421→   421→function normalizeHit(row) {
   422→   422→  if (!row) return null;
   423→   423→  return {
   424→   424→    id: Number(row.id),
   425→   425→    externalid: row.externalid ?? null,
   426→   426→    type: row.type ?? null,
   427→   427→    title: row.title ?? null,
   428→   428→    metadata: parseJsonb(row.metadata, {}),
   429→   429→    thumbnail: row.thumbnail === undefined ? null : row.thumbnail,
   430→   430→    createdat:
   431→   431→      row.createdat instanceof Date ? row.createdat.toISOString() : String(row.createdat ?? ''),
   432→   432→    _score: row._score == null ? 0 : Number(Number(row._score).toFixed(6)),
   433→   433→  };
   434→   434→}
   435→   435→
   436→   436→/**
   437→   437→ * Run one advanced query against searchdocs (joined back to content).
   438→   438→ * Filter fragments are built once and shared by the count + data statements.
   439→   439→ * FAZA 17-a: `cursor` (base64url payload) switches to keyset pagination —
   440→   440→ * the ORDER BY becomes the cursor continuation WHERE and OFFSET is dropped;
   441→   441→ * `effectiveSort` is what actually orders the result (contentid keyset when
   442→   442→ * there is no text query and the caller asked for the default 'score' sort,
   443→   443→ * because score would be constant 0 for every row).
   444→   444→ */
   445→   445→async function runAdvancedQuery({
   446→   446→  q,
   447→   447→  type,
   448→   448→  genre,
   449→   449→  year,
   450→   450→  minRating,
   451→   451→  actor,
   452→   452→  studio,
   453→   453→  category,
   454→   454→  sort,
   455→   455→  fuzzyEnabled,
   456→   456→  limit,
   457→   457→  page,
   458→   458→  cursor = null,
   459→   459→}) {
   460→   460→  const params = [];
   461→   461→  const P = (v) => {
   462→   462→    params.push(v);
   463→   463→    return `$${params.length}`;
   464→   464→  };
   465→   465→
   466→   466→  // Effective sort: filter-only (no q) with default 'score' → contentid keyset.
   467→   467→  const effectiveSort = !q && sort === 'score' ? 'contentid' : sort;
   468→   468→
   469→   469→  // Keyset continuation (cursor present) — validated per effective sort.
   470→   470→  let cursorPayload = null;
   471→   471→  let cursorWhere = null;
   472→   472→  let cursorParams = [];
   473→   473→  if (cursor) {
   474→   474→    const decoded = cursorContinuation(effectiveSort, cursor);
   475→   475→    if (decoded.error) {
   476→   476→      const err = new Error(decoded.error);
   477→   477→      err.status = 400;
   478→   478→      throw err;
   479→   479→    }
   480→   480→    cursorPayload = decoded.payload;
   481→   481→    cursorWhere = decoded.where;
   482→   482→    cursorParams = decoded.params;
   483→   483→  }
   484→   484→
   485→   485→  // Field filters (SQL-side, case-insensitive where array containment).
   486→   486→  const filters = [];
   487→   487→  if (type) filters.push(`s.type = ${P(type)}`);
   488→   488→  if (genre) filters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${P(genre)})`);
   489→   489→  if (year != null) filters.push(`s.year = ${P(year)}`);
   490→   490→  if (actor) {
   491→   491→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${P(actor)} || '%')`);
   492→   492→  }
   493→   493→  if (studio) {
   494→   494→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${P(studio)} || '%')`);
   495→   495→  }
   496→   496→  if (category) filters.push(`s.category = ${P(category)}`);
   497→   497→  if (minRating != null) filters.push(`s.rating >= ${P(minRating)}`);
   498→   498→  const filterSql = filters.length ? ` AND ${filters.join(' AND ')}` : '';
   499→   499→
   500→   500→  // Text score expression (combined: ts_rank * 2 + similarity + prefix bonus).
   501→   501→  let scoreExpr = '0';
   502→   502→  let textMatch = '';
   503→   503→  if (q) {
   504→   504→    const qp = P(q);
   505→   505→    scoreExpr =
   506→   506→      `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qp})) * 2` +
   507→   507→      ` + similarity(s.title, ${qp})` +
   508→   508→      ` + CASE WHEN s.title ILIKE ${qp} || '%' THEN 0.5 ELSE 0 END`;
   509→   509→    textMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qp})`;
   510→   510→  }
   511→   511→
   512→   512→  const whereFor = (matchSql) => `WHERE (${matchSql || 'TRUE'})${filterSql}`;
   513→   513→
   514→   514→  // 1. FTS pass (or filter-only when q is empty).
   515→   515→  let matchSql = textMatch;
   516→   516→  let countRow = await one(
   517→   517→    `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   518→   518→    params,
   519→   519→  );
   520→   520→
   521→   521→  // 2. Trigram fuzzy fallback when the FTS pass matched nothing.
   522→   522→  //    `title % q` (pg_trgm, default 0.3 threshold) plus a looser
   523→   523→  //    similarity() > 0.1 guard so short typo'd queries still reach long
   524→   524→  //    titles (e.g. 'dunee' → 'Dune: Part Two - Wikipedia', sim ≈ 0.15).
   525→   525→  if (q && fuzzyEnabled && Number(countRow?.total ?? 0) === 0) {
   526→   526→    const words = q.split(/\s+/).filter(Boolean);
   527→   527→    const trigramClauses = [];
   528→   528→    for (const candidate of [q, ...words]) {
   529→   529→      const ph = P(`%${candidate}%`);
   530→   530→      trigramClauses.push(`s.title ILIKE ${ph}`);
   531→   531→    }
   532→   532→    const qSim = P(q);
   533→   533→    trigramClauses.push(`(s.title % ${qSim} OR similarity(s.title, ${qSim}) > 0.1)`);
   534→   534→    matchSql = `(${trigramClauses.join(' OR ')})`;
   535→   535→    countRow = await one(
   536→   536→      `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   537→   537→      params,
   538→   538→    );
   539→   539→  }
   540→   540→
   541→   541→  const total = Number(countRow?.total ?? 0);
   542→   542→  const maxScore = Number(Number(countRow?.maxscore ?? 0).toFixed(6));
   543→   543→
   544→   544→  // 3. Data page — subquery keeps the score expression, then joins content.
   545→   545→  //    FAZA 17-a: expose year/rating/title to JS for nextCursor construction;
   546→   546→  //    cursor mode replaces OFFSET with the keyset continuation predicate.
   547→   547→  const orderExpr =
   548→   548→    effectiveSort === 'year'
   549→   549→      ? 't.year DESC NULLS LAST, t.contentid ASC'
   550→   550→      : effectiveSort === 'rating'
   551→   551→        ? 't.rating DESC NULLS LAST, t.score DESC, t.contentid ASC'
   552→   552→        : effectiveSort === 'title'
   553→   553→          ? 't.title ASC, t.contentid ASC'
   554→   554→          : effectiveSort === 'contentid'
   555→   555→            ? 't.contentid ASC'
   556→   556→            : 't.score DESC, t.contentid ASC';
   557→   557→
   558→   558→  const limitPh = P(limit);
   559→   559→  const offsetPh = P((page - 1) * limit);
   560→   560→  // Cursor mode: continuation params come FIRST ($1..$n), then LIMIT; the
   561→   561→  // filter/score params are re-bound only inside the subquery — so build a
   562→   562→  // dedicated statement for each mode instead of renumbering dynamically.
   563→   563→  let dataSql;
   564→   564→  let dataParams;
   565→   565→  if (cursorWhere) {
   566→   566→    // Rebuild the subquery with a fresh parameter namespace for the cursor run
   567→   567→    // (filters + score expression are re-parameterized after the cursor slots).
   568→   568→    const subParams = [...cursorParams, limit];
   569→   569→    const sub = (v) => {
   570→   570→      subParams.push(v);
   571→   571→      return `$${subParams.length}`;
   572→   572→    };
   573→   573→    const cFilters = [];
   574→   574→    if (type) cFilters.push(`s.type = ${sub(type)}`);
   575→   575→    if (genre) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${sub(genre)})`);
   576→   576→    if (year != null) cFilters.push(`s.year = ${sub(year)}`);
   577→   577→    if (actor) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${sub(actor)} || '%')`);
   578→   578→    if (studio) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${sub(studio)} || '%')`);
   579→   579→    if (category) cFilters.push(`s.category = ${sub(category)}`);
   580→   580→    if (minRating != null) cFilters.push(`s.rating >= ${sub(minRating)}`);
   581→   581→    const cFilterSql = cFilters.length ? ` AND ${cFilters.join(' AND ')}` : '';
   582→   582→
   583→   583→    let cScoreExpr = '0';
   584→   584→    let cMatch = 'TRUE';
   585→   585→    // Single q slot, bound LAST (after cursor params + LIMIT) so every
   586→   586→    // placeholder references a parameter that exists in the bind array.
   587→   587→    let qParam = null;
   588→   588→    if (q) {
   589→   589→      qParam = sub(String(q));
   590→   590→      cScoreExpr =
   591→   591→        `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qParam})) * 2` +
   592→   592→        ` + similarity(s.title, ${qParam})` +
   593→   593→        ` + CASE WHEN s.title ILIKE ${qParam} || '%' THEN 0.5 ELSE 0 END`;
   594→   594→      cMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qParam})`;
   595→   595→    }
   596→   596→    // Fuzzy fallback consistency: when the FTS pass matched nothing the offset
   597→   597→    // run switched matchSql — mirror that for the cursor run using `total`.
   598→   598→    if (q && fuzzyEnabled && total === 0) {
   599→   599→      const words = q.split(/\s+/).filter(Boolean);
   600→   600→      const clauses = [q, ...words].map((candidate) => `s.title ILIKE ${sub(`%${candidate}%`)}`);
   601→   601→      clauses.push(`(s.title % ${qParam} OR similarity(s.title, ${qParam}) > 0.1)`);
   602→   602→      cMatch = `(${clauses.join(' OR ')})`;
   603→   603→    }
   604→   604→
   605→   605→    dataSql =
   606→   606→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   607→   607→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   608→   608→       FROM (
   609→   609→         SELECT s.contentid, s.year, s.rating, s.title, ${cScoreExpr} AS score
   610→   610→         FROM searchdocs s
   611→   611→         WHERE (${cMatch})${cFilterSql}
   612→   612→       ) t
   613→   613→       JOIN content c ON c.id = t.contentid
   614→   614→       WHERE ${cursorWhere}
   615→   615→       ORDER BY ${orderExpr}
   616→   616→       LIMIT $${cursorParams.length + 1}::int`;
   617→   617→    // LIMIT is the placeholder right after the cursor params (::int — a bare
   618→   618→    // parameter in LIMIT cannot be type-inferred by Postgres); the q/filter
   619→   619→    // slots follow it so every bound parameter is referenced by the statement.
   620→   620→    dataParams = subParams;
   621→   621→  } else {
   622→   622→    dataSql =
   623→   623→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   624→   624→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   625→   625→       FROM (
   626→   626→         SELECT s.contentid, s.year, s.rating, s.title, ${scoreExpr} AS score
   627→   627→         FROM searchdocs s
   628→   628→         ${whereFor(matchSql)}
   629→   629→       ) t
   630→   630→       JOIN content c ON c.id = t.contentid
   631→   631→       ORDER BY ${orderExpr}
   632→   632→       LIMIT ${limitPh} OFFSET ${offsetPh}`;
   633→   633→    dataParams = params;
   634→   634→  }
   635→   635→  const dataRows = await rows(dataSql, dataParams);
   636→   636→
   637→   637→  // Keyset paging estimates the page number by incrementing the cursor's `p`.
   638→   638→  const resultPage = cursor ? (Number(cursorPayload?.p) || 1) + 1 : page;
   639→   639→  const normalized = dataRows.map(normalizeHit).filter(Boolean);
   640→   640→  const lastRaw = dataRows.length ? dataRows[dataRows.length - 1] : null;
   641→   641→
   642→   642→  return {
   643→   643→    total,
   644→   644→    max_score: maxScore,
   645→   645→    data: normalized,
   646→   646→    page: resultPage,
   647→   647→    nextCursor: buildNextCursor({ sort: effectiveSort, lastRow: lastRaw, pageNumber: resultPage }),
   648→   648→    paginationMode: cursor ? 'cursor' : 'offset',
   649→   649→  };
   650→   650→}

   670→   670→  return promise;
   671→   671→}
