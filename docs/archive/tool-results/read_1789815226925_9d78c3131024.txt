   179→/**
   180→ * Build the next cursor from the LAST row of the current page, per sort.
   181→ * Filter-only + no q → effective sort is contentid (see effectiveSort).
   182→ */
   183→function buildNextCursor({ sort, lastRow, pageNumber }) {
   184→  if (!lastRow) return null;
   185→  const id = Number(lastRow.id);
   186→  if (!Number.isFinite(id)) return null;
   187→  let payload = null;
   188→  if (sort === 'year') payload = { y: lastRow._year ?? null, id };
   189→  else if (sort === 'rating') payload = { r: lastRow._rating ?? null, id };
   190→  else if (sort === 'title') payload = { t: lastRow._title ?? null, id };
   191→  else if (sort === 'score') payload = { s: Number(lastRow._score ?? 0), id };
   192→  else payload = { id }; // contentid keyset (filter-only)
   193→  payload.p = pageNumber; // page estimate — incremented on each cursor hop
   194→  return b64urlEncode(payload);
   195→}

   197→/**
   198→ * Decode + validate the cursor payload for a given effective sort.
   199→ * @returns {{ payload: object, where: string, params: any[] } | { error: string }}
   200→ */
   201→function cursorContinuation(effectiveSort, cursorText) {
   202→  const payload = b64urlDecode(cursorText);
   203→  if (!payload || typeof payload !== 'object' || !Number.isFinite(Number(payload.id))) {
   204→    return { error: 'Invalid cursor' };
   205→  }
   206→  const id = Number(payload.id);
   207→  const params = [];
   208→  const P = (v) => {
   209→    params.push(v);
   210→    return `$${params.length}`;
   211→  };
   212→
   213→  if (effectiveSort === 'score') {
   214→    const s = Number(payload.s);
   215→    if (!Number.isFinite(s)) return { error: 'Invalid cursor (missing s)' };
   216→    const sp = P(s);
   217→    const ip = P(id);
   218→    return { payload, where: `(t.score < ${sp} OR (t.score = ${sp} AND t.contentid > ${ip}))`, params };
   219→  }
   220→  if (effectiveSort === 'year') {
   221→    const y = Number(payload.y);
   222→    if (!Number.isFinite(y)) return { error: 'Invalid cursor (missing y)' };
   223→    const yp = P(y);
   224→    const ip = P(id);
   225→    return { payload, where: `(t.year < ${yp} OR (t.year = ${yp} AND t.contentid > ${ip}))`, params };
   226→  }
   227→  if (effectiveSort === 'rating') {
   228→    const r = Number(payload.r);
   229→    if (!Number.isFinite(r)) return { error: 'Invalid cursor (missing r)' };
   230→    const rp = P(r);
   231→    const ip = P(id);
   232→    return { payload, where: `(t.rating < ${rp} OR (t.rating = ${rp} AND t.contentid > ${ip}))`, params };
   233→  }
   234→  if (effectiveSort === 'title') {
   235→    const t = payload.t;
   236→    if (typeof t !== 'string') return { error: 'Invalid cursor (missing t)' };
   237→    const tp = P(t);
   238→    const ip = P(id);
   239→    return { payload, where: `(t.title > ${tp} OR (t.title = ${tp} AND t.contentid > ${ip}))`, params };
   240→  }
   241→  // contentid keyset (filter-only / no q)
   242→  const ip = P(id);
   243→  return { payload, where: `t.contentid > ${ip}`, params };
   244→}

   322→    });

   329→         type = EXCLUDED.type,
   330→         title = EXCLUDED.title,

   348→  } catch (err) {
   349→    logger.warn(`[search] stale-doc cleanup skipped: ${err.message}`);
   350→  }
   351→
   352→  lastSync = new Date().toISOString();

   354→  return synced;
   355→}

   357→/** Rebuild once under an async lock; concurrent callers await the same run. */
   358→async function ensureSynced() {
   359→  if (!dirty && lastSync) return;
   360→  if (!syncPromise) {
   361→    syncPromise = syncIndex().finally(() => {
   362→      syncPromise = null;
   363→    });
   364→  }
   365→  await syncPromise;
   366→}

   436→/**
   437→ * Run one advanced query against searchdocs (joined back to content).
   438→ * Filter fragments are built once and shared by the count + data statements.
   439→ * FAZA 17-a: `cursor` (base64url payload) switches to keyset pagination —
   440→ * the ORDER BY becomes the cursor continuation WHERE and OFFSET is dropped;
   441→ * `effectiveSort` is what actually orders the result (contentid keyset when
   442→ * there is no text query and the caller asked for the default 'score' sort,
   443→ * because score would be constant 0 for every row).
   444→ */
   445→async function runAdvancedQuery({
   446→  q,
   447→  type,
   448→  genre,
   449→  year,
   450→  minRating,
   451→  actor,
   452→  studio,
   453→  category,
   454→  sort,
   455→  fuzzyEnabled,
   456→  limit,
   457→  page,
   458→  cursor = null,
   459→}) {
   460→  const params = [];
   461→  const P = (v) => {
   462→    params.push(v);
   463→    return `$${params.length}`;
   464→  };
   465→
   466→  // Effective sort: filter-only (no q) with default 'score' → contentid keyset.
   467→  const effectiveSort = !q && sort === 'score' ? 'contentid' : sort;
   468→
   469→  // Keyset continuation (cursor present) — validated per effective sort.
   470→  let cursorPayload = null;
   471→  let cursorWhere = null;
   472→  let cursorParams = [];
   473→  if (cursor) {
   474→    const decoded = cursorContinuation(effectiveSort, cursor);
   475→    if (decoded.error) {
   476→      const err = new Error(decoded.error);
   477→      err.status = 400;
   478→      throw err;
   479→    }
   480→    cursorPayload = decoded.payload;
   481→    cursorWhere = decoded.where;
   482→    cursorParams = decoded.params;
   483→  }
   484→
   485→  // Field filters (SQL-side, case-insensitive where array containment).
   486→  const filters = [];
   487→  if (type) filters.push(`s.type = ${P(type)}`);
   488→  if (genre) filters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${P(genre)})`);
   489→  if (year != null) filters.push(`s.year = ${P(year)}`);
   490→  if (actor) {
   491→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${P(actor)} || '%')`);
   492→  }
   493→  if (studio) {
   494→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${P(studio)} || '%')`);
   495→  }
   496→  if (category) filters.push(`s.category = ${P(category)}`);
   497→  if (minRating != null) filters.push(`s.rating >= ${P(minRating)}`);
   498→  const filterSql = filters.length ? ` AND ${filters.join(' AND ')}` : '';
   499→
   500→  // Text score expression (combined: ts_rank * 2 + similarity + prefix bonus).
   501→  let scoreExpr = '0';
   502→  let textMatch = '';
   503→  if (q) {
   504→    const qp = P(q);
   505→    scoreExpr =
   506→      `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qp})) * 2` +
   507→      ` + similarity(s.title, ${qp})` +
   508→      ` + CASE WHEN s.title ILIKE ${qp} || '%' THEN 0.5 ELSE 0 END`;
   509→    textMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qp})`;
   510→  }
   511→
   512→  const whereFor = (matchSql) => `WHERE (${matchSql || 'TRUE'})${filterSql}`;
   513→
   514→  // 1. FTS pass (or filter-only when q is empty).
   515→  let matchSql = textMatch;
   516→  let countRow = await one(
   517→    `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   518→    params,
   519→  );
   520→
   521→  // 2. Trigram fuzzy fallback when the FTS pass matched nothing.
   522→  //    `title % q` (pg_trgm, default 0.3 threshold) plus a looser
   523→  //    similarity() > 0.1 guard so short typo'd queries still reach long
   524→  //    titles (e.g. 'dunee' → 'Dune: Part Two - Wikipedia', sim ≈ 0.15).
   525→  if (q && fuzzyEnabled && Number(countRow?.total ?? 0) === 0) {
   526→    const words = q.split(/\s+/).filter(Boolean);
   527→    const trigramClauses = [];
   528→    for (const candidate of [q, ...words]) {
   529→      const ph = P(`%${candidate}%`);
   530→      trigramClauses.push(`s.title ILIKE ${ph}`);
   531→    }
   532→    const qSim = P(q);
   533→    trigramClauses.push(`(s.title % ${qSim} OR similarity(s.title, ${qSim}) > 0.1)`);
   534→    matchSql = `(${trigramClauses.join(' OR ')})`;
   535→    countRow = await one(
   536→      `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   537→      params,
   538→    );
   539→  }
   540→
   541→  const total = Number(countRow?.total ?? 0);
   542→  const maxScore = Number(Number(countRow?.maxscore ?? 0).toFixed(6));
   543→

   545→  //    FAZA 17-a: expose year/rating/title to JS for nextCursor construction;
   546→  //    cursor mode replaces OFFSET with the keyset continuation predicate.
   547→  const orderExpr =
   548→    effectiveSort === 'year'
   549→      ? 't.year DESC NULLS LAST, t.contentid ASC'
   550→      : effectiveSort === 'rating'
   551→        ? 't.rating DESC NULLS LAST, t.score DESC, t.contentid ASC'
   552→        : effectiveSort === 'title'
   553→          ? 't.title ASC, t.contentid ASC'
   554→          : effectiveSort === 'contentid'
   555→            ? 't.contentid ASC'
   556→            : 't.score DESC, t.contentid ASC';
   557→
   558→  const limitPh = P(limit);
   559→  const offsetPh = P((page - 1) * limit);
   560→  // Cursor mode: continuation params come FIRST ($1..$n), then LIMIT; the
   561→  // filter/score params are re-bound only inside the subquery — so build a
   562→  // dedicated statement for each mode instead of renumbering dynamically.
   563→  let dataSql;
   564→  let dataParams;
   565→  if (cursorWhere) {
   566→    // Rebuild the subquery with a fresh parameter namespace for the cursor run
   567→    // (filters + score expression are re-parameterized after the cursor slots).
   568→    const subParams = [...cursorParams, limit];
   569→    const sub = (v) => {
   570→      subParams.push(v);
   571→      return `$${subParams.length}`;
   572→    };
   573→    const cFilters = [];
   574→    if (type) cFilters.push(`s.type = ${sub(type)}`);
   575→    if (genre) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${sub(genre)})`);
   576→    if (year != null) cFilters.push(`s.year = ${sub(year)}`);
   577→    if (actor) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${sub(actor)} || '%')`);
   578→    if (studio) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${sub(studio)} || '%')`);
   579→    if (category) cFilters.push(`s.category = ${sub(category)}`);
   580→    if (minRating != null) cFilters.push(`s.rating >= ${sub(minRating)}`);
   581→    const cFilterSql = cFilters.length ? ` AND ${cFilters.join(' AND ')}` : '';
   582→
   583→    let cScoreExpr = '0';
   584→    let cMatch = 'TRUE';
   585→    // Single q slot, bound LAST (after cursor params + LIMIT) so every
   586→    // placeholder references a parameter that exists in the bind array.
   587→    let qParam = null;
   588→    if (q) {
   589→      qParam = sub(String(q));
   590→      cScoreExpr =
   591→        `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qParam})) * 2` +
   592→        ` + similarity(s.title, ${qParam})` +
   593→        ` + CASE WHEN s.title ILIKE ${qParam} || '%' THEN 0.5 ELSE 0 END`;
   594→      cMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qParam})`;
   595→    }

   610→         FROM searchdocs s
   611→         WHERE (${cMatch})${cFilterSql}

   618→    // parameter in LIMIT cannot be type-inferred by Postgres); the q/filter

   623→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,

   625→       FROM (
   626→         SELECT s.contentid, s.year, s.rating, s.title, ${scoreExpr} AS score
   627→         FROM searchdocs s
   628→         ${whereFor(matchSql)}
   629→       ) t
   630→       JOIN content c ON c.id = t.contentid
   631→       ORDER BY ${orderExpr}
   632→       LIMIT ${limitPh} OFFSET ${offsetPh}`;
   633→    dataParams = params;
   634→  }
   635→  const dataRows = await rows(dataSql, dataParams);
   636→
   637→  // Keyset paging estimates the page number by incrementing the cursor's `p`.
   638→  const resultPage = cursor ? (Number(cursorPayload?.p) || 1) + 1 : page;
   639→  const normalized = dataRows.map(normalizeHit).filter(Boolean);
   640→  const lastRaw = dataRows.length ? dataRows[dataRows.length - 1] : null;
   641→
   642→  return {
   643→    total,
   644→    max_score: maxScore,
   645→    data: normalized,
   646→    page: resultPage,
   647→    nextCursor: buildNextCursor({ sort: effectiveSort, lastRow: lastRaw, pageNumber: resultPage }),
   648→    paginationMode: cursor ? 'cursor' : 'offset',
   649→  };

   655→ * FAZA 17-a: identical concurrent /advanced requests share one promise.
   656→ * Key = normalized query params; the map entry is removed on settle.

   658→const inflightSearch = new Map();

   660→function coalesceAdvanced(normalizedKey, run) {
   661→  const existing = inflightSearch.get(normalizedKey);
   662→  if (existing) {
   663→    searchOps.coalesced += 1;
   664→    return existing;
   665→  }
   666→  const promise = run().finally(() => {
   667→    inflightSearch.delete(normalizedKey);
   668→  });
   669→  inflightSearch.set(normalizedKey, promise);
   670→  return promise;
   671→}

   673→/** Race a query promise against the 8s budget (loser is logged, not awaited). */
   674→function withQueryTimeout(promise, describe) {
   675→  let timer = null;
   676→  const budget = new Promise((_, reject) => {
   677→    timer = setTimeout(() => {
   678→      const err = new Error('Search query timeout');
   679→      err.status = 504;
   680→      reject(err);
   681→    }, QUERY_TIMEOUT_MS);
   682→    if (typeof timer.unref === 'function') timer.unref();
   683→  });
   684→  return promise.then(
   685→    (value) => {
   686→      clearTimeout(timer);
   687→      return value;
   688→    },
   689→    (err) => {
   690→      clearTimeout(timer);
   691→      throw err;
   692→    },
   693→  ).finally(() => {
   694→    clearTimeout(timer);
   695→    promise.catch((loser) => {
   696→      // The lost race (if it also failed) must not become an unhandledRejection.
   697→      logger.warn(`[search] query lost the timeout race (${describe}): ${loser?.message ?? loser}`);
   698→    });
   699→  });
   700→}
