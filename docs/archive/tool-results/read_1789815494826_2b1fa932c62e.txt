    68→const RO_NUMBER = new Intl.NumberFormat('ro-RO');
    69→const RO_RATE = new Intl.NumberFormat('ro-RO', { maximumFractionDigits: 2 });
    70→
    71→function fmtRoNum(v: number | null | undefined): string {
    72→  return RO_NUMBER.format(Number(v ?? 0));
    73→}
    74→
    75→function fmtMs(v: number | null | undefined): string {
    76→  const n = Number(v ?? 0);
    77→  return Number.isFinite(n) && n > 0 ? `${RO_NUMBER.format(Math.round(n))} ms` : '—';
    78→}
    79→
    80→function fmtFactor(v: number | null | undefined): string {
    81→  const n = Number(v ?? 0);
    82→  return Number.isFinite(n) && n >= 1 ? `${RO_RATE.format(n)}×` : '—';
    83→}
    84→
    85→/** HH:MM:SS clock label for the history X axis (samples are 1 s apart). */
    86→function fmtClock(iso: string): string {
    87→  const d = new Date(iso);
    88→  if (Number.isNaN(d.getTime())) return iso;
    89→  const p = (n: number) => String(n).padStart(2, '0');
    90→  return `${p(d.getHours())}:${p(d.getMinutes())}:${p(d.getSeconds())}`;
    91→}

   215→const SETTINGS_FIELDS = [
   216→  { key: 'cacheTtlMs', label: 'Cache TTL', hint: '1.000 – 5.000 ms', min: 1000, max: 5000 },
   217→  { key: 'cacheMax', label: 'Intrări cache max', hint: '100 – 100.000', min: 100, max: 100000 },

   221→type SettingsKey = (typeof SETTINGS_FIELDS)[number]['key'];
   222→type SettingsForm = Record<SettingsKey, string>;
   223→
   224→function OptSettingsCard({
   225→  config,
   226→  isAdmin,
   227→  onSaved,
   228→}: {
   229→  config: SearchOptConfig;
   230→  isAdmin: boolean;
   231→  onSaved: () => void;
   232→}) {
   233→  const [form, setForm] = useState<SettingsForm>({
   234→    cacheTtlMs: String(config.cacheTtlMs),
   235→    cacheMax: String(config.cacheMax),
   236→    batchMax: String(config.batchMax),
   237→  });
   238→  const [enabled, setEnabled] = useState<boolean>(config.enabled);
   239→  const [dirty, setDirty] = useState(false);
   240→  const [saving, setSaving] = useState(false);
   241→
   242→  /* Re-sync the pre-filled inputs from the polled config while the user has
   243→     not edited anything (and right after a successful save). */
   244→  useEffect(() => {
   245→    if (dirty) return;
   246→    setForm((prev) => {
   247→      const next: SettingsForm = {
   248→        cacheTtlMs: String(config.cacheTtlMs),
   249→        cacheMax: String(config.cacheMax),
   250→        batchMax: String(config.batchMax),
   251→      };
   252→      return (Object.keys(next) as SettingsKey[]).every((k) => prev[k] === next[k]) ? prev : next;
   253→    });
   254→    setEnabled((prev) => (prev === config.enabled ? prev : config.enabled));
   255→  }, [config, dirty]);
   256→
   257→  async function handleSave() {
   258→    if (saving) return;
   259→    const parsed: Record<SettingsKey, number> = {
   260→      cacheTtlMs: Number(form.cacheTtlMs),
   261→      cacheMax: Number(form.cacheMax),
   262→      batchMax: Number(form.batchMax),
   263→    };
   264→    for (const f of SETTINGS_FIELDS) {
   265→      const v = parsed[f.key];
   266→      if (!Number.isFinite(v) || v < f.min || v > f.max) {
   267→        toast.error(`${f.label} trebuie să fie între ${RO_NUMBER.format(f.min)} și ${RO_NUMBER.format(f.max)}`);
   268→        return;
   269→      }
   270→    }
   271→    setSaving(true);
   272→    try {
   273→      await authRequest<{ ok: boolean; config: SearchOptConfig }>(
   274→        `/api/search/opt/settings?XTransformPort=${BACKEND_PORT}`,
   275→        { method: 'POST', body: JSON.stringify({ ...parsed, enabled }) },
   276→      );
   277→      toast.success('Setări salvate');
   278→      setDirty(false);
   279→      onSaved();
   280→    } catch (e) {
   281→      if (e instanceof PlatformError && (e.status === 401 || e.status === 403)) {
   282→        toast.error('Setările de optimizare necesită cont de administrator');
   283→      } else {
   284→        toast.error(e instanceof Error ? e.message : 'Salvarea setărilor a eșuat');
   285→      }
   286→    } finally {
   287→      setSaving(false);
   288→    }
   289→  }
   290→
   291→  return (
   292→    <Card className="border-white/10 bg-white/[0.03]">
   293→      <CardHeader className="pb-3">
   294→        <CardTitle className="flex items-center gap-2 text-base">
   295→          <ShieldCheck className="h-4 w-4 text-emerald-400" aria-hidden />
   296→          Setări runtime (administrator)
   297→        </CardTitle>
   298→        <CardDescription>
   299→          Reglare live a stratului de optimizare — se aplică imediat, fără restart.
   300→        </CardDescription>
   301→      </CardHeader>
   302→      <CardContent className="space-y-4">
   303→        {isAdmin ? (
   304→          <>
   305→            <div className="flex flex-wrap items-center justify-between gap-3 rounded-xl border border-white/10 bg-white/[0.02] px-4 py-3">
   306→              <div>
   307→                <p className="text-xs font-semibold text-zinc-200">Stratul de optimizare</p>
   308→                <p className="mt-0.5 text-[11px] leading-snug text-zinc-600">
   309→                  {enabled
   310→                    ? 'Dedup + batching + cache ACTIVE — cererile identice partajează interogări reale.'
   311→                    : 'DEZACTIVAT — fiecare cerere execută propriul FTS (comportament FAZA 16).'}
   312→                </p>
   313→              </div>
   314→              <div className="flex items-center gap-2">
   315→                <Switch
   316→                  id="opt-enabled"
   317→                  checked={enabled}
   318→                  onCheckedChange={(v) => {
   319→                    setDirty(true);
   320→                    setEnabled(v);
   321→                  }}
   322→                  aria-label="Activează stratul de optimizare a căutării"
   323→                />
   324→                <Label htmlFor="opt-enabled" className="text-xs text-zinc-400">
   325→                  {enabled ? 'Activ' : 'Dezactivat'}
   326→                </Label>
   327→              </div>
   328→            </div>
   329→            <div className="grid gap-3 sm:grid-cols-3">
   330→              {SETTINGS_FIELDS.map((f) => (
   331→                <div key={f.key} className="space-y-1.5">
   332→                  <Label htmlFor={`opt-setting-${f.key}`} className="text-xs text-zinc-400">
   333→                    {f.label} <span className="font-normal text-zinc-600">({f.hint})</span>
   334→                  </Label>
   335→                  <Input
   336→                    id={`opt-setting-${f.key}`}
   337→                    type="number"
   338→                    inputMode="numeric"
   339→                    min={f.min}
   340→                    max={f.max}
   341→                    step={1}
   342→                    value={form[f.key]}
   343→                    onChange={(e) => {
   344→                      setDirty(true);
   345→                      setForm((prev) => ({ ...prev, [f.key]: e.target.value }));
   346→                    }}
   347→                    className="border-white/10 bg-white/[0.04] font-mono text-zinc-100"
   348→                    aria-label={`${f.label}, interval acceptat ${f.hint}`}
   349→                  />
   350→                </div>
   351→              ))}
   352→            </div>
   353→            <div className="flex flex-wrap items-center justify-between gap-3">
   354→              <p className="max-w-xl text-[11px] leading-snug text-zinc-600">
   355→                Validarea și limitarea valorilor (clamping) se fac pe server — TTL-ul rămâne în plaja
   356→                pinned 1.000–5.000 ms din specificația FAZA 17.
   357→              </p>
   358→              <Button
   359→                size="sm"
   360→                onClick={() => void handleSave()}
   361→                disabled={saving || !dirty}
   362→                className="bg-emerald-600 font-bold hover:bg-emerald-500"
   363→              >
   364→                {saving ? <RefreshCw className="animate-spin" aria-hidden /> : <Save aria-hidden />}
   365→                Salvează
   366→              </Button>
   367→            </div>
   368→          </>
   369→        ) : (
   370→          <p className="flex items-center gap-2 rounded-xl border border-amber-500/25 bg-amber-500/[0.06] px-4 py-2.5 text-xs text-amber-200">
   371→            <Lock className="h-3.5 w-3.5 shrink-0" aria-hidden />
   372→            Setările de optimizare necesită cont de administrator — valorile curente sunt read-only.
   373→          </p>
   374→        )}
   375→      </CardContent>
   376→    </Card>
   377→  );
   378→}
