   427→async function handleMine(req, res) {
   428→  const url = req.method === 'GET' ? req.query.url : req.body?.url;
   429→  if (!url || typeof url !== 'string') {
   430→    return res.status(400).json({ error: 'Missing url' });
   431→  }
   432→  const options =
   433→    req.method === 'GET'
   434→      ? { type: req.query.type, externalId: req.query.externalId }
   435→      : { type: req.body?.type, externalId: req.body?.externalId };
   436→  try {
   437→    const { extraction, record } = await mineUrl(url, options);
   438→    invalidateHttpCache();
   439→    await auditLog({ userid: req.user?.id ?? null, action: 'miner.run', ip: clientIpOf(req), meta: { url, id: record?.id } });
   440→    return res.json({ ok: true, extraction, record });
   441→  } catch (err) {
   442→    return res.status(err.status || 502).json({ error: err.message || 'Mining failed' });
   443→  }
   444→}
