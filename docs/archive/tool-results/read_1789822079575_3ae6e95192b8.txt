    69→function ExternalSourceSurface({ source, title }: { source: PlayerSource; title: string }) {
    70→  const videoRef = useRef<HTMLVideoElement>(null);
    71→
    72→  /* HLS via hls.js — same dynamic-import pattern as the main surface. */
    73→  useEffect(() => {
    74→    const url = source.url;
    75→    const el = videoRef.current;
    76→    if (source.type !== 'm3u8' || !url || !el) return undefined;
    77→    let hls: { destroy: () => void } | null = null;
    78→    let cancelled = false;
    79→    (async () => {
    80→      try {
    81→        const mod = await import('hls.js');
    82→        const Hls = mod.default;
    83→        if (cancelled) return;
    84→        if (Hls.isSupported()) {
    85→          const instance = new Hls({ enableWorker: true });
    86→          instance.loadSource(url);
    87→          instance.attachMedia(el);
    88→          hls = instance;
    89→        } else if (el.canPlayType('application/vnd.apple.mpegurl')) {
    90→          el.src = url;
    91→        }
    92→      } catch {
    93→        /* hls.js failed to init — the <video> element surfaces its own error */
    94→      }
    95→    })();
    96→    return () => {
    97→      cancelled = true;
    98→      hls?.destroy();
    99→    };
   100→  }, [source.type, source.url]);
   101→
   102→  return (
   103→    <div className="aspect-video w-full overflow-hidden rounded-2xl border border-white/10 bg-black">
   104→      {source.type === 'mp4' && (
   105→        <video
   106→          key={`ext-mp4-${source.url}`}
   107→          controls
   108→          autoPlay
   109→          playsInline
   110→          className="h-full w-full"
   111→          src={source.url}
   112→        />
   113→      )}
   114→      {source.type === 'm3u8' && (
   115→        <video
   116→          key={`ext-m3u8-${source.url}`}
   117→          ref={videoRef}
   118→          controls
   119→          autoPlay
   120→          playsInline
   121→          className="h-full w-full"
   122→        />
   123→      )}
   124→      {source.type === 'youtube' && (
   125→        <iframe
   126→          key={`ext-youtube-${source.url}`}
   127→          src={`${source.url}${source.url.includes('?') ? '&' : '?'}autoplay=1&rel=0`}
   128→          title={title}
   129→          className="h-full w-full"
   130→          allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
   131→          allowFullScreen
   132→        />
   133→      )}
   134→      {(source.type === 'vimeo' || source.type === 'rumble') && (
   135→        <iframe
   136→          key={`ext-${source.type}-${source.url}`}
   137→          src={`${source.url}${source.url.includes('?') ? '&' : '?'}autoplay=1`}
   138→          title={title}
   139→          className="h-full w-full"
   140→          allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
   141→          allowFullScreen
   142→        />
   143→      )}
   144→      {source.type === 'iframe' && (
   145→        <iframe
   146→          key={`ext-iframe-${source.url}`}
   147→          src={source.url}
   148→          title={title}
   149→          className="h-full w-full"
   150→          sandbox="allow-scripts; allow-same-origin; allow-presentation; allow-popups; allow-forms"
   151→          allowFullScreen
   152→          referrerPolicy="no-referrer"

   280→export function PlayerView({ route }: { route: AppRoute }) {
   281→  const { user } = useAuth();
   282→  const id = Number(route.params[0]);
   283→  const validId = Number.isInteger(id) && id > 0;
   284→  const player = useApi(`player-${id}`, () => api.player(id), { enabled: validId });
   285→
   286→  /* FAZA 17 — source fallback chain. `sourceIndex` walks data.sources on any
   287→     load failure (mp4/HLS error, 12s load timeout). `failedState` marks the
   288→     friendly end-of-chain card and is scoped to the content id so it can
   289→     never leak across titles. `reloadNonce` forces a fresh <video> mount when
   290→     retrying the same source. */
   291→  const [sourceIndex, setSourceIndex] = useState(0);
   292→  const [failedState, setFailedState] = useState<{ contentId: number; reason: string } | null>(null);
   293→  const [reloadNonce, setReloadNonce] = useState(0);
   294→  const [mineUrl, setMineUrl] = useState('');
   295→  const [mining, setMining] = useState(false);
   296→
   297→  /* FAZA 14 — external URL playback (rule 4): the resolved source plays in
   298→     the main player surface as the «Extern» chip; on branches without a
   299→     library player (bad id / error) it plays in the standalone surface. */
   300→  const [externalUrl, setExternalUrl] = useState('');
   301→  const [externalResolving, setExternalResolving] = useState(false);
   302→  const [externalError, setExternalError] = useState<string | null>(null);
   303→  const [externalSource, setExternalSource] = useState<PlayerSource | null>(null);
   304→  const [externalInfo, setExternalInfo] = useState<PlayerExternalResponse | null>(null);
   305→  const [usingExternal, setUsingExternal] = useState(false);
   306→
   307→  const videoRef = useRef<HTMLVideoElement>(null);
   308→  const sourcesRef = useRef<PlayerSource[]>([]);
   309→  const sourceIndexRef = useRef(0);
   310→  const loadTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   311→  const lastAdvanceRef = useRef<{ key: string; at: number } | null>(null);
   312→
   313→  /* Only trust data matching the requested id — useApi keeps the previous
   314→     record alive while the key changes (stale data across titles). */
   315→  const current = player.data && player.data.content.id === id ? player.data : null;
   316→  const sources = current?.sources ?? [];
   317→  const exhausted = failedState != null && failedState.contentId === id && sources.length > 0;
   318→  /* FAZA 14 — while the «Extern» chip is selected, the resolved external
   319→     source takes over the main surface (bypasses the end-of-chain card). */
   320→  const externalActive = usingExternal && externalSource != null;
   321→  const activeSource: PlayerSource | null = externalActive
   322→    ? externalSource
   323→    : exhausted
   324→      ? null
   325→      : sources[sourceIndex] ?? current?.primary ?? null;
   326→
   327→  /** Clears the 12s load watchdog — any progress event calls this. */
   328→  const clearLoadTimer = useCallback(() => {
   329→    if (loadTimerRef.current) {
   330→      clearTimeout(loadTimerRef.current);
   331→      loadTimerRef.current = null;
   332→    }
   333→  }, []);
   334→
   335→  /** Advance to the next source on failure; when none remain, switch to the
   336→      friendly end-of-chain card instead of crashing. Duplicate signals for the
   337→      same attempt (e.g. hls.js error + element error) are coalesced. */
   338→  const advanceSource = useCallback(
   339→    (reason: string) => {
   340→      clearLoadTimer();
   341→      const list = sourcesRef.current;
   342→      const i = sourceIndexRef.current;
   343→      const key = `${id}:${i}:${reason}`;
   344→      const now = Date.now();
   345→      if (lastAdvanceRef.current && lastAdvanceRef.current.key === key && now - lastAdvanceRef.current.at < 1500) {
   346→        return;
   347→      }
   348→      lastAdvanceRef.current = { key, at: now };
   349→      if (i + 1 < list.length) {
   350→        sourceIndexRef.current = i + 1;
   351→        setSourceIndex(i + 1);
   352→        toast.info(`Sursa ${i + 1} nu răspunde (${reason}) — trec la următoarea…`);
   353→      } else {
   354→        setFailedState({ contentId: id, reason });
   355→      }
   356→    },
   357→    [id, clearLoadTimer],
   358→  );
   359→
   360→  /** Manual source chip click — clears the chain failure; clicking the chip of
   361→      the current source forces a clean reload of that source. */
   362→  const selectSource = (i: number) => {
   363→    if (i < 0 || i >= sources.length) return;
   364→    setFailedState(null);
   365→    if (i === sourceIndex) {
   366→      setReloadNonce((n) => n + 1);
   367→    } else {
   368→      sourceIndexRef.current = i;
   369→      setSourceIndex(i);
   370→    }
   371→  };
   372→
   373→  /** "Reîncearcă de la prima sursă" — reset the chain and remount the video. */
   374→  const retryFromFirst = () => {
   375→    setFailedState(null);
   376→    sourceIndexRef.current = 0;
   377→    setSourceIndex(0);
   378→    setReloadNonce((n) => n + 1);
   379→  };
   380→
   381→  /* New title (route change) → reset the fallback chain before the new data
   382→     ever renders, so no stale index/failure can flash. The resolved external
   383→     source survives (it is library-independent); only the selection resets so
   384→     the new title's own sources play first. */
   385→  useEffect(() => {
   386→    sourceIndexRef.current = 0;
   387→    setSourceIndex(0);
   388→    setFailedState(null);
   389→    setUsingExternal(false);
   390→  }, [id]);
   391→
   392→  /* Fresh sources (first load or refetch after mining) → start at the primary
   393→     source and force a clean reload of the player surface. */
   394→  useEffect(() => {
   395→    const d = player.data;
   396→    if (!d || d.content.id !== id) return;
   397→    sourcesRef.current = d.sources;
   398→    const primary = d.primary;
   399→    const primaryIndex = primary
   400→      ? d.sources.findIndex((s) => s.url === primary.url && s.type === primary.type)
   401→      : -1;
   402→    const start = primaryIndex >= 0 ? primaryIndex : 0;
   403→    sourceIndexRef.current = start;
   404→    setSourceIndex(start);
   405→    setFailedState(null);
   406→    setReloadNonce((n) => n + 1);
   407→  }, [player.data, id]);
   408→
   409→  /* ------------------------- analytics tracking ------------------------- */
   410→
   411→  const trackedViewRef = useRef<number | null>(null);
   412→  const trackedPlayRef = useRef<number | null>(null);
   413→
   414→  /* Fire-and-forget 'view' event, once per content id per mount. */
   415→  useEffect(() => {
   416→    if (!player.data || player.data.content.id !== id) return;
   417→    if (trackedViewRef.current === id) return;
   418→    trackedViewRef.current = id;
   419→    platform.recordView(id, 'view').catch(() => {});
   420→  }, [player.data, id]);
   421→
   422→  /* 'play' event for embed renderers (youtube/iframe), once per content id. */
   423→  useEffect(() => {
   424→    if (!player.data || player.data.content.id !== id) return;
   425→    const src = activeSource;
   426→    if (!src || (src.type !== 'youtube' && src.type !== 'iframe')) return;
   427→    if (trackedPlayRef.current === id) return;
   428→    trackedPlayRef.current = id;
   429→    platform.recordView(id, 'play').catch(() => {});
   430→  }, [player.data, activeSource, id]);
   431→
   432→  const trackPlay = () => {
   433→    if (trackedPlayRef.current === id) return;
   434→    trackedPlayRef.current = id;
   435→    platform.recordView(id, 'play').catch(() => {});
   436→  };
   437→
   438→  const trackComplete = () => {
   439→    platform.recordView(id, 'complete').catch(() => {});
   440→  };
   441→
   442→  useEffect(() => {
   443→    const url = activeSource?.url;
   444→    const el = videoRef.current;
   445→    if (activeSource?.type !== 'm3u8' || !url || !el) return;
   446→    let hls: { destroy: () => void } | null = null;
   447→    let cancelled = false;
   448→    (async () => {
   449→      try {
   450→        const mod = await import('hls.js');
   451→        const Hls = mod.default;
   452→        if (cancelled) return;
   453→        if (Hls.isSupported()) {
   454→          const instance = new Hls({ enableWorker: true });
   455→          /* FAZA 17 — fatal HLS errors destroy the instance and advance the
   456→             source chain; non-fatal ones get a light startLoad() retry. */
   457→          instance.on(Hls.Events.ERROR, (_event, data) => {
   458→            if (cancelled) return;
   459→            if (data?.fatal) {
   460→              try {
   461→                instance.destroy();
   462→              } catch {
   463→                /* already destroyed */
   464→              }
   465→              advanceSource('eroare HLS');
   466→            } else {
   467→              try {
   468→                instance.startLoad();
   469→              } catch {
   470→                /* retry not possible — the load watchdog will advance */
   471→              }
   472→            }
   473→          });
   474→          instance.loadSource(url);
   475→          instance.attachMedia(el);
   476→          hls = instance;
   477→        } else if (el.canPlayType('application/vnd.apple.mpegurl')) {
   478→          el.src = url;
   479→        }
   480→      } catch {
   481→        /* hls.js failed to init — hand over to the fallback chain. */
   482→        if (!cancelled) advanceSource('eroare HLS');
   483→      }
   484→    })();
   485→    return () => {
   486→      cancelled = true;
   487→      hls?.destroy();
   488→    };
   489→  }, [activeSource?.type, activeSource?.url, sourceIndex, reloadNonce, advanceSource]);
   490→
   491→  /* FAZA 17 — 12s load watchdog: if a video source produces no loadeddata /
   492→     canplay within the window, advance to the next source. */
   493→  useEffect(() => {
   494→    const t = activeSource?.type;
   495→    if (!activeSource?.url || exhausted) return undefined;
   496→    if (externalActive) return undefined; // external sources are user-resolved — no chain advance
   497→    if (t !== 'mp4' && t !== 'm3u8') return undefined;
   498→    clearLoadTimer();
   499→    loadTimerRef.current = setTimeout(() => {
   500→      loadTimerRef.current = null;
   501→      advanceSource('timeout sursă');
   502→    }, 12_000);
   503→    return clearLoadTimer;
   504→  }, [activeSource?.type, activeSource?.url, sourceIndex, reloadNonce, exhausted, externalActive, advanceSource, clearLoadTimer]);
   505→
   506→  /* FAZA 17 — release the decoder/buffer whenever the current video element
   507→     goes away (source change, retry or unmount): pause, drop the src and run
   508→     load() so the browser frees the media pipeline. */
   509→  useEffect(() => {
   510→    const el = videoRef.current;
   511→    if (!el) return undefined;
   512→    return () => {
   513→      clearLoadTimer();
   514→      try {
   515→        el.pause();
   516→        el.removeAttribute('src');
   517→        el.load();
   518→      } catch {
   519→        /* element already detached */
   520→      }
   521→    };
