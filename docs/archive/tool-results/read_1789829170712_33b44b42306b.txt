     1→     1→/**
     2→     2→ * modules/player/hub.js — FAZA 19 (Task 19-1): Player Hub API.
     3→     3→ *
     4→     4→ * The server-side backbone of the advanced player: user playlists (ordered,
     5→     5→ * deduped, capped), unified watch progress/resume (REUSES the Phase-2
     6→     6→ * watchhistory table — the very same rows served by /api/users/me/history),
     7→     7→ * continue-watching, WebVTT subtitle tracks (public read, admin write) and
     8→     8→ * public platform stats. 100% REAL data in Neon (playlists, playlistitems,
     9→     9→ * subtitletracks tables — see modules/platform/migrations.js).
    10→    10→ *
    11→    11→ * modules/player/player.js (the REAL external playback resolver) is NEVER
    12→    12→ * touched — this module only sits next to it under /api/playerhub.
    13→    13→ *
    14→    14→ * Mount: app.use('/api/playerhub', playerHubRouter) — attachUser is global,
    15→    15→ * so req.user is present whenever a valid Bearer token is supplied (null
    16→    16→ * otherwise). /status, the subtitle listing/tracks and /stats stay public;
    17→    17→ * user routes enforce auth per-route via requireAuthLocal.
    18→    18→ *
    19→    19→ * Routes:
    20→    20→ *   GET    /status                              → hub contract + live totals (public)
    21→    21→ *   GET    /continue-watching                   → last 12 watchhistory entries + resume (auth)
    22→    22→ *   GET    /playlists                           → user playlists + itemsCount + cover (auth)
    23→    23→ *   POST   /playlists          { title, description? } → create (max 20/user) (auth)
    24→    24→ *   GET    /playlists/:id                       → playlist detail + ordered items (auth, owner)
    25→    25→ *   PATCH  /playlists/:id      { title?, description? } → update (auth, owner)
    26→    26→ *   DELETE /playlists/:id                       → delete (auth, owner)
    27→    27→ *   POST   /playlists/:id/items { contentId }   → add item (max 200, dedupe) (auth, owner)
    28→    28→ *   DELETE /playlists/:id/items/:contentId      → remove item + renumber (auth, owner)
    29→    29→ *   POST   /playlists/:id/reorder { contentIds } → full reorder (auth, owner)
    30→    30→ *   GET    /progress/:contentId                 → { contentId, progress, watchedat } (auth)
    31→    31→ *   POST   /progress           { contentId, seconds, duration? } → upsert (auth)
    32→    32→ *   DELETE /progress/:contentId                 → reset resume (auth)
    33→    33→ *   GET    /subtitles/:contentId                → tracks [{ id, label, lang }] (public)
    34→    34→ *   GET    /subtitles/:contentId/:trackId       → raw VTT (public, cached 1h, CORS *)
    35→    35→ *   POST   /subtitles/:contentId { label, lang, vtt } → attach (admin, max 8)
    36→    36→ *   DELETE /subtitles/:contentId/:trackId       → detach (admin)
    37→    37→ *   GET    /stats                               → totals + top continue-watching (public)
    38→    38→ *
    39→    39→ * Mutations emit 'playlist:changed' { userid, playlistid? } on the platform
    40→    40→ * bus; admin subtitle writes are audit-logged ('playerhub.subtitle.*').
    41→    41→ */

    53→    53→/** Per-route auth guard (the router is mounted WITHOUT requireAuth so the
    54→    54→ *  public endpoints — /status, subtitle reads, /stats — stay anonymous). */
    55→    55→const requireAuthLocal = (req, res, next) => {
    56→    56→  if (!req.user) return res.status(401).json({ error: 'Authentication required' });
    57→    57→  next();
    58→    58→};
    59→    59→
    60→    60→/** Per-route admin guard (401 anonymous / 403 authenticated non-admin). */
    61→    61→const requireAdminLocal = (req, res, next) => {
    62→    62→  if (!req.user) return res.status(401).json({ error: 'Authentication required' });
    63→    63→  if (req.user.role !== 'admin') return res.status(403).json({ error: 'Admin access required' });
    64→    64→  next();
    65→    65→};

    67→    67→/** Parse a positive integer route/body param; null when invalid. */
    68→    68→function intParam(raw) {
    69→    69→  const n = Number(raw);
    70→    70→  return Number.isInteger(n) && n > 0 ? n : null;
    71→    71→}

    73→    73→/** Normalize a raw content row (from a JOIN) into the public content shape
    74→    74→ *  (same contract as modules/users/profile.js contentRow). */
    75→    75→function contentRow(row) {
    76→    76→  if (!row) return null;
    77→    77→  return {
    78→    78→    id: num(row.id),
    79→    79→    externalid: row.externalid ?? null,
    80→    80→    type: row.type ?? null,
    81→    81→    title: row.title ?? null,
    82→    82→    metadata: parseJsonb(row.metadata, {}),
    83→    83→    thumbnail: row.thumbnail ?? null,
    84→    84→    createdat: isoOrNull(row.createdat),
    85→    85→  };
    86→    86→}
    87→    87→
    88→    88→/** Number of playable sources on a content row (metadata.player.sources,
    89→    89→ *  falling back to metadata.sources — same lookup as resolvePlayer). */
    90→    90→function sourcesCountOf(metadata) {
    91→    91→  const sources = metadata?.player?.sources || metadata?.sources;
    92→    92→  return Array.isArray(sources) ? sources.length : 0;
    93→    93→}

    95→    95→/** Serialize a playlist row (+ optional aggregated extras). */
    96→    96→function playlistRow(row, extra = {}) {
    97→    97→  return {
    98→    98→    id: num(row.id),
    99→    99→    title: row.title ?? '',
   100→   100→    description: row.description ?? null,
   101→   101→    createdat: isoOrNull(row.createdat),
   102→   102→    updatedat: isoOrNull(row.updatedat),
   103→   103→    ...extra,
   104→   104→  };
   105→   105→}

   107→   107→/** Load one playlist owned by the caller; responds (400/404) and returns
   108→   108→ *  null when the id is invalid, missing or owned by someone else. */
   109→   109→async function ownPlaylistOrRespond(req, res) {
   110→   110→  const id = intParam(req.params.id);
   111→   111→  if (!id) {
   112→   112→    res.status(400).json({ error: 'Invalid playlist id' });
   113→   113→    return null;
   114→   114→  }
   115→   115→  const playlist = await one(
   116→   116→    `SELECT id, userid, title, description, createdat, updatedat
   117→   117→     FROM playlists WHERE id = $1 AND userid = $2`,
   118→   118→    [id, req.user.id],
   119→   119→  );
   120→   120→  if (!playlist) {
   121→   121→    // 404 both for missing and foreign playlists (no ownership leak).
   122→   122→    res.status(404).json({ error: 'Playlist not found' });
   123→   123→    return null;
   124→   124→  }
   125→   125→  return playlist;
   126→   126→}
   127→   127→
   128→   128→/** Validate an optional playlist description; returns { error } or { value }. */
   129→   129→function validateDescription(raw) {
   130→   130→  if (raw === undefined || raw === null) return { value: null };
   131→   131→  if (typeof raw !== 'string') return { error: 'description must be a string (max 500 characters)' };
   132→   132→  const trimmed = raw.trim();
   133→   133→  if (!trimmed) return { value: null };
   134→   134→  if (trimmed.length > 500) return { error: 'description must be at most 500 characters' };
   135→   135→  return { value: trimmed };
   136→   136→}

   251→   251→/** POST /playlists — create a playlist (max 20 per user). */
   252→   252→playerHubRouter.post('/playlists', requireAuthLocal, wrap(async (req, res) => {
   253→   253→  const body = req.body || {};
   254→   254→  const title = typeof body.title === 'string' ? body.title.trim() : '';
   255→   255→  if (title.length < 2 || title.length > 120) {
   256→   256→    return res.status(400).json({ error: 'title must be 2-120 characters (non-blank)' });
   257→   257→  }
   258→   258→  const description = validateDescription(body.description);
   259→   259→  if (description.error) return res.status(400).json({ error: description.error });
   260→   260→
   261→   261→  const countRow = await one(`SELECT COUNT(*)::int AS count FROM playlists WHERE userid = $1`, [req.user.id]);
   262→   262→  if (num(countRow?.count) >= MAX_PLAYLISTS_PER_USER) {
   263→   263→    return res.status(409).json({
   264→   264→      error: `Playlist limit reached — max ${MAX_PLAYLISTS_PER_USER} playlists per user`,

   266→   266→    });
   267→   267→  }
   268→   268→
   269→   269→  const created = await one(
   270→   270→    `INSERT INTO playlists (userid, title, description) VALUES ($1, $2, $3) RETURNING *`,
   271→   271→    [req.user.id, title, description.value],
   272→   272→  );
   273→   273→  emitPlatform('playlist:changed', { userid: req.user.id, playlistid: num(created?.id) });
   274→   274→  return res.status(201).json({ ok: true, playlist: playlistRow(created, { itemsCount: 0 }) });
   275→   275→}));

   301→   301→/** PATCH /playlists/:id — partial update (owner only); bumps updatedat. */
   302→   302→playerHubRouter.patch('/playlists/:id', requireAuthLocal, wrap(async (req, res) => {
   303→   303→  const playlist = await ownPlaylistOrRespond(req, res);
   304→   304→  if (!playlist) return undefined;
   305→   305→
   306→   306→  const body = req.body || {};
   307→   307→  const sets = [];
   308→   308→  const params = [];
   309→   309→
   310→   310→  if (Object.prototype.hasOwnProperty.call(body, 'title')) {
   311→   311→    const title = typeof body.title === 'string' ? body.title.trim() : '';
   312→   312→    if (title.length < 2 || title.length > 120) {
   313→   313→      return res.status(400).json({ error: 'title must be 2-120 characters (non-blank)' });
   314→   314→    }
   315→   315→    params.push(title);
   316→   316→    sets.push(`title = $${params.length}`);
   317→   317→  }
   318→   318→
   319→   319→  if (Object.prototype.hasOwnProperty.call(body, 'description')) {
   320→   320→    const description = validateDescription(body.description);
   321→   321→    if (description.error) return res.status(400).json({ error: description.error });
   322→   322→    params.push(description.value);
   323→   323→    sets.push(`description = $${params.length}`);
   324→   324→  }
   325→   325→
   326→   326→  if (!sets.length) {
   327→   327→    return res.json({ ok: true, playlist: playlistRow(playlist) }); // idempotent no-op
   328→   328→  }
   329→   329→
   330→   330→  sets.push(`updatedat = NOW()`);
   331→   331→  params.push(playlist.id);
   332→   332→  const updated = await one(
   333→   333→    `UPDATE playlists SET ${sets.join(', ')} WHERE id = $${params.length} RETURNING *`,
   334→   334→    params,
   335→   335→  );
   336→   336→  emitPlatform('playlist:changed', { userid: req.user.id, playlistid: playlist.id });
   337→   337→  return res.json({ ok: true, playlist: playlistRow(updated) });
   338→   338→}));

   344→   344→

   350→   350→/** POST /playlists/:id/items — append a content item (dedupe, max 200). */
   351→   351→playerHubRouter.post('/playlists/:id/items', requireAuthLocal, wrap(async (req, res) => {
   352→   352→  const playlist = await ownPlaylistOrRespond(req, res);
   353→   353→  if (!playlist) return undefined;
   354→   354→
   355→   355→  const contentId = intParam(req.body?.contentId);
   356→   356→  if (!contentId) return res.status(400).json({ error: 'Invalid content id' });
   357→   357→
   358→   358→  const content = await one(`SELECT id, title FROM content WHERE id = $1`, [contentId]);
   359→   359→  if (!content) return res.status(404).json({ error: 'Content not found' });
   360→   360→
   361→   361→  // Dedupe on (playlistid, contentid) — a duplicate is a 200 no-op.
   362→   362→  const existing = await one(
   363→   363→    `SELECT id, position, addedat FROM playlistitems WHERE playlistid = $1 AND contentid = $2`,
   364→   364→    [playlist.id, contentId],
   365→   365→  );
   366→   366→  if (existing) {
   367→   367→    return res.json({ ok: true, added: false, duplicate: true });
   368→   368→  }
   369→   369→
   370→   370→  const countRow = await one(
   371→   371→    `SELECT COUNT(*)::int AS count FROM playlistitems WHERE playlistid = $1`,
   372→   372→    [playlist.id],
   373→   373→  );
   374→   374→  if (num(countRow?.count) >= MAX_ITEMS_PER_PLAYLIST) {
   375→   375→    return res.status(409).json({
   376→   376→      error: `Playlist item limit reached — max ${MAX_ITEMS_PER_PLAYLIST} items per playlist`,
   377→   377→      limit: MAX_ITEMS_PER_PLAYLIST,
   378→   378→    });
   379→   379→  }
   380→   380→
   381→   381→  // position = max(position)+1; ON CONFLICT DO NOTHING keeps concurrent adds race-safe.
   382→   382→  const inserted = await one(
   383→   383→    `INSERT INTO playlistitems (playlistid, contentid, position)
   384→   384→     VALUES ($1, $2, (SELECT COALESCE(MAX(position), -1) + 1 FROM playlistitems WHERE playlistid = $1))
   385→   385→     ON CONFLICT (playlistid, contentid) DO NOTHING
   386→   386→     RETURNING id, position, addedat`,
   387→   387→    [playlist.id, contentId],
   388→   388→  );
   389→   389→  if (!inserted) {
   390→   390→    return res.json({ ok: true, added: false, duplicate: true });
   391→   391→  }
   392→   392→
   393→   393→  await one(`UPDATE playlists SET updatedat = NOW() WHERE id = $1`, [playlist.id]);
   394→   394→  emitPlatform('playlist:changed', { userid: req.user.id, playlistid: playlist.id });
   395→   395→  return res.status(201).json({
   396→   396→    ok: true,
   397→   397→    added: true,
   398→   398→    duplicate: false,
   399→   399→    item: {
   400→   400→      contentId,
   401→   401→      position: num(inserted.position),
   402→   402→      addedat: isoOrNull(inserted.addedat),
   403→   403→    },
   404→   404→  });
   405→   405→}));
   406→   406→
   407→   407→/** DELETE /playlists/:id/items/:contentId — remove an item and renumber the
   408→   408→ *  remaining positions 0..n-1 (owner only). */
   409→   409→playerHubRouter.delete('/playlists/:id/items/:contentId', requireAuthLocal, wrap(async (req, res) => {
   410→   410→  const playlist = await ownPlaylistOrRespond(req, res);
   411→   411→  if (!playlist) return undefined;
   412→   412→
   413→   413→  const contentId = intParam(req.params.contentId);
   414→   414→  if (!contentId) return res.status(400).json({ error: 'Invalid content id' });
   415→   415→
   416→   416→  const removed = await one(
   417→   417→    `DELETE FROM playlistitems WHERE playlistid = $1 AND contentid = $2 RETURNING id`,
   418→   418→    [playlist.id, contentId],
   419→   419→  );
   420→   420→  if (!removed) return res.status(404).json({ error: 'Item not found in this playlist' });
   421→   421→
   422→   422→  // Renumber the remaining items 0..n-1 (single statement, parameterized).
   423→   423→  await rows(
   424→   424→    `WITH ordered AS (
   425→   425→       SELECT id, (ROW_NUMBER() OVER (ORDER BY position ASC, id ASC) - 1) AS newpos
   426→   426→       FROM playlistitems
   427→   427→       WHERE playlistid = $1
   428→   428→     )
   429→   429→     UPDATE playlistitems pi
   430→   430→     SET position = ordered.newpos
   431→   431→     FROM ordered
   432→   432→     WHERE pi.id = ordered.id AND pi.position <> ordered.newpos`,
   433→   433→    [playlist.id],
   434→   434→  );
   435→   435→  await one(`UPDATE playlists SET updatedat = NOW() WHERE id = $1`, [playlist.id]);
   436→   436→  emitPlatform('playlist:changed', { userid: req.user.id, playlistid: playlist.id });
   437→   437→  return res.json({ ok: true, removed: true });
   438→   438→}));

   440→   440→/** POST /playlists/:id/reorder — full reorder: the submitted contentIds must
   441→   441→ *  be EXACTLY the set of current items; each becomes position = array index. */
   442→   442→playerHubRouter.post('/playlists/:id/reorder', requireAuthLocal, wrap(async (req, res) => {
   443→   443→  const playlist = await ownPlaylistOrRespond(req, res);
   444→   444→  if (!playlist) return undefined;
   445→   445→
   446→   446→  const contentIds = req.body?.contentIds;
   447→   447→  if (!Array.isArray(contentIds) || !contentIds.every((v) => intParam(v) !== null)) {
   448→   448→    return res.status(400).json({ error: 'contentIds must be an array of positive integer content ids' });
   449→   449→  }
   450→   450→  const ids = contentIds.map((v) => Number(v));
   451→   451→
   452→   452→  const current = await rows(
   453→   453→    `SELECT contentid FROM playlistitems WHERE playlistid = $1`,
   454→   454→    [playlist.id],
   455→   455→  );
   456→   456→  const currentSet = new Set(current.map((r) => num(r.contentid)));
   457→   457→  const requestedSet = new Set(ids);
   458→   458→
   459→   459→  const duplicates = ids.filter((id, i) => ids.indexOf(id) !== i);
   460→   460→  const missing = [...currentSet].filter((id) => !requestedSet.has(id)); // in DB, not sent
   461→   461→  const unexpected = [...requestedSet].filter((id) => !currentSet.has(id)); // sent, not in DB
   462→   462→  if (duplicates.length || missing.length || unexpected.length) {
   463→   463→    return res.status(400).json({
   464→   464→      error: 'Reorder payload does not match the playlist items exactly',
   465→   465→      duplicateIds: duplicates,
   466→   466→      missingIds: missing,
   467→   467→      unexpectedIds: unexpected,
   468→   468→    });
   469→   469→  }
   470→   470→
   471→   471→  if (ids.length > 0) {
   472→   472→    // position = index in the submitted array (unnest keeps the pairs aligned).
   473→   473→    await rows(
   474→   474→      `UPDATE playlistitems pi
   475→   475→         SET position = u.pos
   476→   476→       FROM unnest($2::int[], $3::int[]) AS u(cid, pos)
   477→   477→       WHERE pi.playlistid = $1 AND pi.contentid = u.cid`,
   478→   478→      [playlist.id, ids, ids.map((_, i) => i)],
   479→   479→    );
   480→   480→  }
   481→   481→  await one(`UPDATE playlists SET updatedat = NOW() WHERE id = $1`, [playlist.id]);
   482→   482→  emitPlatform('playlist:changed', { userid: req.user.id, playlistid: playlist.id });
   483→   483→  return res.json({ ok: true, reordered: true, count: ids.length });
   484→   484→}));

   508→   508→/** POST /progress — upsert playback progress into the SAME watchhistory table
   509→   509→ *  as /api/users/me/history (unified history). The player pings every ~10s;
   510→   510→ *  no extra throttle by design. `duration` is validated but intentionally
   511→   511→ *  NOT persisted (no new columns on Phase-2 schemas — golden rule). */
   512→   512→playerHubRouter.post('/progress', requireAuthLocal, wrap(async (req, res) => {
   513→   513→  const body = req.body || {};
   514→   514→  const contentId = intParam(body.contentId);
   515→   515→  if (!contentId) return res.status(400).json({ error: 'Invalid content id' });
   516→   516→
   517→   517→  const seconds = Number(body.seconds);
   518→   518→  if (!Number.isInteger(seconds) || seconds < 0 || seconds > MAX_PROGRESS_SECONDS) {
   519→   519→    return res.status(400).json({
   520→   520→      error: `seconds must be an integer between 0 and ${MAX_PROGRESS_SECONDS} (24h)`,
   521→   521→    });
   522→   522→  }
   523→   523→  if (body.duration !== undefined && body.duration !== null) {
   524→   524→    const duration = Number(body.duration);
   525→   525→    if (!Number.isInteger(duration) || duration <= 0) {
   526→   526→      return res.status(400).json({ error: 'duration must be a positive integer (seconds)' });
   527→   527→    }
   528→   528→  }
   529→   529→
   530→   530→  const content = await one(`SELECT id FROM content WHERE id = $1`, [contentId]);
   531→   531→  if (!content) return res.status(404).json({ error: 'Content not found' });
   532→   532→
   533→   533→  await one(
   534→   534→    `INSERT INTO watchhistory (userid, contentid, progress, watchedat)
   535→   535→     VALUES ($1, $2, $3, NOW())
   536→   536→     ON CONFLICT (userid, contentid)
   537→   537→     DO UPDATE SET progress = EXCLUDED.progress, watchedat = NOW()`,
   538→   538→    [req.user.id, contentId, seconds],
   539→   539→  );
   540→   540→
   541→   541→  // Rolling-window eviction, identical to POST /api/users/me/history — keeps
   542→   542→  // the unified history bounded by the caller's plan (-1 = unlimited → skip).
   543→   543→  const limit = planLimits(req.user.plan).history;
   544→   544→  if (limit >= 0) {
   545→   545→    await rows(
   546→   546→      `DELETE FROM watchhistory
   547→   547→       WHERE userid = $1 AND id NOT IN (
   548→   548→         SELECT id FROM watchhistory WHERE userid = $1
   549→   549→         ORDER BY watchedat DESC, id DESC
   550→   550→         LIMIT $2
   551→   551→       )`,
   552→   552→      [req.user.id, limit],
   553→   553→    );
   554→   554→  }
   555→   555→
   556→   556→  return res.json({ ok: true, saved: true, progress: seconds });
   557→   557→}));

   609→   609→/** POST /subtitles/:contentId — attach a WebVTT track (admin only, max 8). */
   610→   610→playerHubRouter.post('/subtitles/:contentId', requireAdminLocal, wrap(async (req, res) => {
   611→   611→  const contentId = intParam(req.params.contentId);
   612→   612→  if (!contentId) return res.status(400).json({ error: 'Invalid content id' });
   613→   613→
   614→   614→  const body = req.body || {};
   615→   615→  const label = typeof body.label === 'string' ? body.label.trim() : '';
   616→   616→  if (label.length < 1 || label.length > 120) {
   617→   617→    return res.status(400).json({ error: 'label must be 1-120 characters (non-blank)' });
   618→   618→  }
   619→   619→  const lang = typeof body.lang === 'string' ? body.lang.trim() : '';
   620→   620→  if (!LANG_RE.test(lang)) {
   621→   621→    return res.status(400).json({ error: 'lang must be 2-10 letters/hyphens (e.g. "en", "ro", "pt-BR")' });
   622→   622→  }
   623→   623→  if (typeof body.vtt !== 'string' || !body.vtt.trim().startsWith('WEBVTT')) {
   624→   624→    return res.status(400).json({ error: 'vtt must be a WebVTT document starting with "WEBVTT"' });
   625→   625→  }
   626→   626→  if (body.vtt.length > MAX_VTT_CHARS) {
   627→   627→    return res.status(400).json({
   628→   628→      error: `vtt too large — ${Math.round(body.vtt.length / 1024)}KB, limit is 256KB`,
   629→   629→    });
   630→   630→  }
   631→   631→
   632→   632→  const content = await one(`SELECT id, title FROM content WHERE id = $1`, [contentId]);
   633→   633→  if (!content) return res.status(404).json({ error: 'Content not found' });
   634→   634→
   635→   635→  const countRow = await one(
   636→   636→    `SELECT COUNT(*)::int AS count FROM subtitletracks WHERE contentid = $1`,

   639→   639→  if (num(countRow?.count) >= MAX_SUBTITLE_TRACKS_PER_CONTENT) {

   641→   641→      error: `Subtitle track limit reached — max ${MAX_SUBTITLE_TRACKS_PER_CONTENT} tracks per content`,
   642→   642→      limit: MAX_SUBTITLE_TRACKS_PER_CONTENT,
   643→   643→    });
   644→   644→  }
   645→   645→
   646→   646→  const created = await one(
   647→   647→    `INSERT INTO subtitletracks (contentid, label, lang, vtt) VALUES ($1, $2, $3, $4) RETURNING id, label, lang`,

   649→   649→  );

   651→   651→    userid: req.user.id,
   652→   652→    action: 'playerhub.subtitle.attach',
   653→   653→    ip: clientIpOf(req),
   654→   654→    meta: { contentid: contentId, trackid: num(created?.id), label, lang },
   655→   655→  });
   656→   656→  return res.status(201).json({
   657→   657→    ok: true,
   658→   658→    track: { id: num(created?.id), label: created?.label, lang: created?.lang },
   659→   659→  });
   660→   660→}));
