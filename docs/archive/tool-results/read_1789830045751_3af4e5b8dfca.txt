    74→/** Content slug for the browse `type` filter (matches the real content.type
    75→ *  values: movie, tv, anime, mined, video, featured, page — lower-case). */
    76→const TYPE_RE = /^[a-z0-9-]{1,24}$/;
    77→
    78→/** Genres exist in metadata as a jsonb ARRAY of strings (verified live:
    79→ *  81/100 rows array, 19 rows missing the key) — the guard below keeps the
    80→ *  jsonb_array_elements_text lateral join total-safe even if a future writer
    81→ *  stores a non-array value. */
    82→const GENRES_ARRAY_SAFE = `CASE WHEN jsonb_typeof(c.metadata->'genres') = 'array' THEN c.metadata->'genres' ELSE '[]'::jsonb END`;
    83→/** Same shape-guard for arbitrary string-array metadata keys (actors/cast/
    84→ *  studios — verified live: 2 rows each with `actors`/`studios`, none with
    85→ *  `cast`/`studio`/`network`). */
    86→const ARRAY_KEY_SAFE = (key) => `CASE WHEN jsonb_typeof(c.metadata->'${key}') = 'array' THEN c.metadata->'${key}' ELSE '[]'::jsonb END`;
    87→
    88→/** Sort key for metadata.rating: NULLS LAST + regex-guarded cast (a row with
    89→ *  a non-numeric rating string sorts last instead of erroring the query). */
    90→const RATING_SORT = `CASE WHEN (c.metadata->>'rating') ~ '^[0-9]+(\\.[0-9]+)?$' THEN (c.metadata->>'rating')::numeric ELSE NULL END`;
    91→
    92→/** year filter: metadata.year exists on 93/100 rows as number OR string —
    93→ *  `->>` normalizes both to text; the regex guards the ::int cast. */
    94→const YEAR_TEXT_GUARD = `(c.metadata->>'year') ~ '^[0-9]{4}$'`;
    95→
    96→/** Escape LIKE wildcards so a user `q` of "100%" matches literally. */
    97→function escapeLike(value) {
    98→  return value.replace(/[\\%_]/g, (ch) => `\\${ch}`);
    99→}

   101→/** Strict integer query param: { value } when valid, { error } otherwise. */
   102→function intQuery(raw, { min, max, label }) {
   103→  const n = Number(raw);
   104→  if (!Number.isInteger(n) || n < min || n > max) {
   105→    return { error: `${label} must be an integer between ${min} and ${max}` };
   106→  }
   107→  return { value: n };
   108→}

   110→/** Normalize a raw content row into the public content shape (same contract
   111→ *  as modules/users/profile.js contentRow / modules/player/hub.js contentRow). */
   112→function contentRow(row) {
   113→  if (!row) return null;
   114→  return {
   115→    id: num(row.id),
   116→    externalid: row.externalid ?? null,
   117→    type: row.type ?? null,
   118→    title: row.title ?? null,
   119→    metadata: parseJsonb(row.metadata, {}),
   120→    thumbnail: row.thumbnail ?? null,
   121→    createdat: isoOrNull(row.createdat),
   122→  };
   123→}

   154→/* ----------------------- A. /content — faceted browse ---------------------- */
   155→
   156→/** GET /content — REAL faceted browse over `content`.
   157→ *
   158→ * Honest scope: `q` is a simple parametrized ILIKE on title (browse aid),
   159→ * NOT the full-text search — the real search stays in /api/search/* (FAZA
   160→ * 15–18 pipeline, untouched). Facets are computed on the FILTERED set
   161→ * (without pagination): types = GROUP BY c.type; genres = lateral
   162→ * jsonb_array_elements_text + GROUP BY; years = text year (regex-guarded)
   163→ * + GROUP BY. The `popular` sort aggregates viewevents with a LEFT JOIN
   164→ * (NULLS LAST + createdat tiebreak). */
   165→hubRouter.get('/content', wrap(async (req, res) => {
   166→  /* ---- parse + strict validation (invalid → 400) ---- */
   167→  const limit = intQuery(req.query.limit ?? CONTENT_PAGE_DEFAULT, { min: 1, max: CONTENT_PAGE_MAX, label: 'limit' });
   168→  if (limit.error) return res.status(400).json({ error: limit.error });
   169→  const page = intQuery(req.query.page ?? 1, { min: 1, max: 100000, label: 'page' });
   170→  if (page.error) return res.status(400).json({ error: page.error });
   171→
   172→  const sort = req.query.sort === undefined ? 'recent' : String(req.query.sort);
   173→  if (!Object.prototype.hasOwnProperty.call(SORTS, sort)) {
   174→    return res.status(400).json({ error: `sort must be one of: ${Object.keys(SORTS).join(', ')}` });
   175→  }
   176→
   177→  const type = req.query.type === undefined ? null : String(req.query.type);
   178→  if (type !== null && !TYPE_RE.test(type)) {
   179→    return res.status(400).json({ error: 'type must be a lower-case slug (e.g. movie, tv, anime, mined, video)' });
   180→  }
   181→
   182→  const genre = req.query.genre === undefined ? null : String(req.query.genre).trim();
   183→  if (genre !== null && (genre.length < 1 || genre.length > MAX_GENRE_CHARS)) {
   184→    return res.status(400).json({ error: `genre must be 1-${MAX_GENRE_CHARS} characters (non-blank)` });
   185→  }
   186→
   187→  const year = req.query.year === undefined ? null : intQuery(req.query.year, { min: 1000, max: 3000, label: 'year' });
   188→  if (year !== null && year.error) return res.status(400).json({ error: year.error });
   189→
   190→  const q = req.query.q === undefined ? null : String(req.query.q).trim().slice(0, MAX_Q_CHARS + 1);
   191→  if (q !== null && q.length > MAX_Q_CHARS) {
   192→    return res.status(400).json({ error: `q must be at most ${MAX_Q_CHARS} characters` });
   193→  }
   194→
   195→  /* ---- parametrized WHERE (shared by data, count and facets) ---- */
   196→  const conds = ['TRUE'];
   197→  const params = [];
   198→  if (type !== null) {
   199→    params.push(type);
   200→    conds.push(`c.type = $${params.length}`);
   201→  }
   202→  if (genre !== null) {
   203→    // genres is a jsonb ARRAY of strings → exact containment beats ILIKE on
   204→    // the serialized text (verified live: metadata->'genres' @> '"Action"').
   205→    params.push(JSON.stringify(genre));
   206→    conds.push(`c.metadata->'genres' @> $${params.length}::jsonb`);
   207→  }
   208→  if (year !== null) {
   209→    params.push(year.value);
   210→    conds.push(`${YEAR_TEXT_GUARD} AND (c.metadata->>'year')::int = $${params.length}`);
   211→  }
   212→  if (q !== null && q !== '') {
   213→    params.push(`%${escapeLike(q)}%`);
   214→    conds.push(`c.title ILIKE '%' || $${params.length} || '%' ESCAPE '\\'`);
   215→  }
   216→  const where = conds.join(' AND ');
   217→
   218→  const fromPopular = sort === 'popular'
   219→    ? 'LEFT JOIN (SELECT contentid, COUNT(*)::int AS views FROM viewevents GROUP BY contentid) ve ON ve.contentid = c.id'
   220→    : '';
   221→
   222→  const filterParams = params.slice(); // shared by count + facets
   223→  params.push(limit.value, (page.value - 1) * limit.value);
   224→  const limitIdx = params.length - 1;
   225→  const offsetIdx = params.length;
   226→
   227→  /* ---- data page + total + facets (all over the SAME filtered set) ---- */
   228→  const [dataRows, totalRow, typeFacetRows, genreFacetRows, yearFacetRows] = await Promise.all([
   229→    rows(
   230→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat
   231→       FROM content c ${fromPopular}
   232→       WHERE ${where}
   233→       ORDER BY ${SORTS[sort]}
   234→       LIMIT $${limitIdx} OFFSET $${offsetIdx}`,
   235→      params,
   236→    ),
   237→    one(`SELECT COUNT(*)::int AS total FROM content c WHERE ${where}`, filterParams),
   238→    rows(
   239→      `SELECT c.type AS value, COUNT(*)::int AS count
   240→       FROM content c WHERE ${where}
   241→       GROUP BY c.type ORDER BY count DESC, value ASC`,
   242→      filterParams,
   243→    ),
   244→    rows(
   245→      `SELECT g AS value, COUNT(*)::int AS count
   246→       FROM content c, jsonb_array_elements_text(${GENRES_ARRAY_SAFE}) g
   247→       WHERE ${where} AND g <> ''
   248→       GROUP BY g ORDER BY count DESC, value ASC
   249→       LIMIT ${FACET_TOP}`,
   250→      filterParams,
   251→    ),
   252→    rows(
   253→      `SELECT (c.metadata->>'year') AS value, COUNT(*)::int AS count
   254→       FROM content c
   255→       WHERE ${where} AND ${YEAR_TEXT_GUARD}
   256→       GROUP BY 1 ORDER BY value DESC
   257→       LIMIT ${FACET_TOP}`,
   258→      filterParams,
   259→    ),
   260→  ]);
   261→
   262→  const total = num(totalRow?.total);
   263→  return res.json({
   264→    data: dataRows.map(contentRow),
   265→    page: page.value,
   266→    limit: limit.value,
   267→    total,
   268→    totalPages: total === 0 ? 0 : Math.ceil(total / limit.value),
   269→    facets: {
   270→      types: typeFacetRows.map((r) => ({ value: r.value, count: num(r.count) })),
   271→      genres: genreFacetRows.map((r) => ({ value: r.value, count: num(r.count) })),
   272→      years: yearFacetRows.map((r) => ({ value: num(r.value), count: num(r.count) })),
   273→    },
   274→  });
   275→}));

   398→/** Shared strict ?limit validation for the facet hubs (1..100, default 50).
   399→ *  Responds 400 and returns null when invalid. */
   400→function facetHubLimit(req, res) {
   401→  const limit = intQuery(req.query.limit ?? FACET_HUB_LIMIT_DEFAULT, { min: 1, max: FACET_HUB_LIMIT_MAX, label: 'limit' });
   402→  if (limit.error) {

   406→  return limit.value;
   407→}

   409→/** GET /years — real release years (DISTINCT metadata->>'year', numeric only,

   411→hubRouter.get('/years', wrap(async (req, res) => {
   412→  const limit = facetHubLimit(req, res);
   413→  if (limit === null) return undefined;
   414→  const data = await rows(
   415→    `SELECT (c.metadata->>'year')::int AS year, COUNT(*)::int AS count,
   416→            COALESCE((array_agg(c.thumbnail ORDER BY c.id) FILTER (WHERE c.thumbnail IS NOT NULL))[1:3], ARRAY[]::text[]) AS sample
   417→     FROM content c
   418→     WHERE ${YEAR_TEXT_GUARD}
   419→     GROUP BY 1
   420→     ORDER BY 1 DESC
   421→     LIMIT $1`,
   422→    [limit],
   423→  );

   585→/** GET /recommendations — REAL public recommendations (no auth): the globally
   586→ *  most-watched genre (viewevents JOIN content → genres lateral) drives a
   587→ *  top-rated selection from that genre; when the genre holds fewer than 12
   588→ *  real titles the list is BACKFILLED with the global top rating (distinct)
   589→ *  and the split is reported honestly (fromGenre / backfilled). With zero
   590→ *  viewevents it falls back to the global top rating. Honest scope: full
   591→ *  per-user personalization already lives in /api/ai/* (FAZA 15–16,
   592→ *  untouched). */
   593→hubRouter.get('/recommendations', wrap(async (req, res) => {
   594→  const topGenre = await one(
   595→    `SELECT g AS genre, COUNT(*)::int AS views
   596→     FROM viewevents v
   597→     JOIN content c ON c.id = v.contentid,
   598→          jsonb_array_elements_text(${GENRES_ARRAY_SAFE}) g
   599→     WHERE g <> ''
   600→     GROUP BY g
   601→     ORDER BY views DESC, genre ASC
   602→     LIMIT 1`,
   603→  );
   604→
   605→  const note = 'recomandări publice reale (genul cel mai vizitat global) — personalizarea completă per user există în /api/ai/*';
   606→  if (topGenre) {
   607→    const genreRows = await rows(
   608→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat
   609→       FROM content c
   610→       WHERE c.metadata->'genres' @> $1::jsonb
   611→       ORDER BY ${RATING_SORT} DESC NULLS LAST, c.createdat DESC, c.id DESC
   612→       LIMIT ${RECOMMENDATIONS_SIZE}`,
   613→      [JSON.stringify(topGenre.genre)],
   614→    );
   615→
   616→    if (genreRows.length < RECOMMENDATIONS_SIZE) {
   617→      // Honest backfill: the rest of the 12 comes from the GLOBAL top rating
   618→      // (distinct ids) — reported via fromGenre/backfilled below.
   619→      const seen = genreRows.map((r) => num(r.id));
   620→      const fill = await rows(
   621→        `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat
   622→         FROM content c
   623→         WHERE c.id <> ALL($1::int[])
   624→         ORDER BY ${RATING_SORT} DESC NULLS LAST, c.createdat DESC, c.id DESC
   625→         LIMIT ${RECOMMENDATIONS_SIZE - genreRows.length}`,
   626→        [seen],
   627→      );
   628→      return res.json({
   629→        ok: true,
   630→        strategy: 'most-watched-genre',
   631→        genre: topGenre.genre,
   632→        basedOnViews: num(topGenre.views),
   633→        fromGenre: genreRows.length,
   634→        backfilled: fill.length,
   635→        data: [...genreRows, ...fill].map(contentRow),
   636→        note,
   637→      });
   638→    }
   639→
   640→    return res.json({
   641→      ok: true,
   642→      strategy: 'most-watched-genre',
   643→      genre: topGenre.genre,
   644→      basedOnViews: num(topGenre.views),
   645→      fromGenre: genreRows.length,
   646→      backfilled: 0,
   647→      data: genreRows.map(contentRow),
   648→      note,
   649→    });
   650→  }
   651→
   652→  // No viewevents at all → honest global fallback (top rating, any genre).
   653→  const data = await rows(
   654→    `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat
   655→     FROM content c
   656→     ORDER BY ${RATING_SORT} DESC NULLS LAST, c.createdat DESC, c.id DESC
   657→     LIMIT ${RECOMMENDATIONS_SIZE}`,
   658→  );
   659→  return res.json({
   660→    ok: true,
   661→    strategy: 'top-rated-fallback',
   662→    genre: null,
   663→    fromGenre: 0,
   664→    backfilled: data.length,
   665→    data: data.map(contentRow),
   666→    note,
   667→  });
   668→}));
