   105→   105→    return { error: `${label} must be an integer between ${min} and ${max}` };
   106→   106→  }
   107→   107→  return { value: n };
   108→   108→}

   154→   154→/* ----------------------- A. /content — faceted browse ---------------------- */
   155→   155→
   156→   156→/** GET /content — REAL faceted browse over `content`.
   157→   157→ *
   158→   158→ * Honest scope: `q` is a simple parametrized ILIKE on title (browse aid),
   159→   159→ * NOT the full-text search — the real search stays in /api/search/* (FAZA
   160→   160→ * 15–18 pipeline, untouched). Facets are computed on the FILTERED set
   161→   161→ * (without pagination): types = GROUP BY c.type; genres = lateral
   162→   162→ * jsonb_array_elements_text + GROUP BY; years = text year (regex-guarded)
   163→   163→ * + GROUP BY. The `popular` sort aggregates viewevents with a LEFT JOIN
   164→   164→ * (NULLS LAST + createdat tiebreak). */
   165→   165→hubRouter.get('/content', wrap(async (req, res) => {
   166→   166→  /* ---- parse + strict validation (invalid → 400) ---- */
   167→   167→  const limit = intQuery(req.query.limit ?? CONTENT_PAGE_DEFAULT, { min: 1, max: CONTENT_PAGE_MAX, label: 'limit' });
   168→   168→  if (limit.error) return res.status(400).json({ error: limit.error });
   169→   169→  const page = intQuery(req.query.page ?? 1, { min: 1, max: 100000, label: 'page' });
   170→   170→  if (page.error) return res.status(400).json({ error: page.error });
   171→   171→
   172→   172→  const sort = req.query.sort === undefined ? 'recent' : String(req.query.sort);
   173→   173→  if (!Object.prototype.hasOwnProperty.call(SORTS, sort)) {
   174→   174→    return res.status(400).json({ error: `sort must be one of: ${Object.keys(SORTS).join(', ')}` });
   175→   175→  }
   176→   176→
   177→   177→  const type = req.query.type === undefined ? null : String(req.query.type);
   178→   178→  if (type !== null && !TYPE_RE.test(type)) {
   179→   179→    return res.status(400).json({ error: 'type must be a lower-case slug (e.g. movie, tv, anime, mined, video)' });
   180→   180→  }
   181→   181→
   182→   182→  const genre = req.query.genre === undefined ? null : String(req.query.genre).trim();
   183→   183→  if (genre !== null && (genre.length < 1 || genre.length > MAX_GENRE_CHARS)) {
   184→   184→    return res.status(400).json({ error: `genre must be 1-${MAX_GENRE_CHARS} characters (non-blank)` });
   185→   185→  }
   186→   186→
   187→   187→  const year = req.query.year === undefined ? null : intQuery(req.query.year, { min: 1000, max: 3000, label: 'year' });
   188→   188→  if (year !== null && year.error) return res.status(400).json({ error: year.error });
   189→   189→
   190→   190→  const q = req.query.q === undefined ? null : String(req.query.q).trim().slice(0, MAX_Q_CHARS + 1);
   191→   191→  if (q !== null && q.length > MAX_Q_CHARS) {
   192→   192→    return res.status(400).json({ error: `q must be at most ${MAX_Q_CHARS} characters` });
   193→   193→  }
   194→   194→
   195→   195→  /* ---- parametrized WHERE (shared by data, count and facets) ---- */
   196→   196→  const conds = ['TRUE'];
   197→   197→  const params = [];
   198→   198→  if (type !== null) {
   199→   199→    params.push(type);
   200→   200→    conds.push(`c.type = $${params.length}`);
   201→   201→  }
   202→   202→  if (genre !== null) {
   203→   203→    // genres is a jsonb ARRAY of strings → exact containment beats ILIKE on
   204→   204→    // the serialized text (verified live: metadata->'genres' @> '"Action"').
   205→   205→    params.push(JSON.stringify(genre));
   206→   206→    conds.push(`c.metadata->'genres' @> $${params.length}::jsonb`);
   207→   207→  }
   208→   208→  if (year !== null) {
   209→   209→    params.push(year.value);
   210→   210→    conds.push(`${YEAR_TEXT_GUARD} AND (c.metadata->>'year')::int = $${params.length}`);
   211→   211→  }
   212→   212→  if (q !== null && q !== '') {
   213→   213→    params.push(`%${escapeLike(q)}%`);
   214→   214→    conds.push(`c.title ILIKE '%' || $${params.length} || '%' ESCAPE '\\'`);
   215→   215→  }
   216→   216→  const where = conds.join(' AND ');
   217→   217→
   218→   218→  const fromPopular = sort === 'popular'
   219→   219→    ? 'LEFT JOIN (SELECT contentid, COUNT(*)::int AS views FROM viewevents GROUP BY contentid) ve ON ve.contentid = c.id'
   220→   220→    : '';
   221→   221→
   222→   222→  const filterParams = params.slice(); // shared by count + facets
   223→   223→  params.push(limit.value, (page.value - 1) * limit.value);
   224→   224→  const limitIdx = params.length - 1;
   225→   225→  const offsetIdx = params.length;
   226→   226→
   227→   227→  /* ---- data page + total + facets (all over the SAME filtered set) ---- */
   228→   228→  const [dataRows, totalRow, typeFacetRows, genreFacetRows, yearFacetRows] = await Promise.all([
   229→   229→    rows(
   230→   230→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat
   231→   231→       FROM content c ${fromPopular}
   232→   232→       WHERE ${where}
   233→   233→       ORDER BY ${SORTS[sort]}
   234→   234→       LIMIT $${limitIdx} OFFSET $${offsetIdx}`,
   235→   235→      params,
   236→   236→    ),
   237→   237→    one(`SELECT COUNT(*)::int AS total FROM content c WHERE ${where}`, filterParams),
   238→   238→    rows(
   239→   239→      `SELECT c.type AS value, COUNT(*)::int AS count
   240→   240→       FROM content c WHERE ${where}
   241→   241→       GROUP BY c.type ORDER BY count DESC, value ASC`,
   242→   242→      filterParams,
   243→   243→    ),
   244→   244→    rows(
   245→   245→      `SELECT g AS value, COUNT(*)::int AS count
   246→   246→       FROM content c, jsonb_array_elements_text(${GENRES_ARRAY_SAFE}) g
   247→   247→       WHERE ${where} AND g <> ''
   248→   248→       GROUP BY g ORDER BY count DESC, value ASC
   249→   249→       LIMIT ${FACET_TOP}`,
   250→   250→      filterParams,
   251→   251→    ),
   252→   252→    rows(
   253→   253→      `SELECT (c.metadata->>'year') AS value, COUNT(*)::int AS count
   254→   254→       FROM content c
   255→   255→       WHERE ${where} AND ${YEAR_TEXT_GUARD}
   256→   256→       GROUP BY 1 ORDER BY value DESC
   257→   257→       LIMIT ${FACET_TOP}`,
   258→   258→      filterParams,
   259→   259→    ),
   260→   260→  ]);
   261→   261→
   262→   262→  const total = num(totalRow?.total);
   263→   263→  return res.json({
   264→   264→    data: dataRows.map(contentRow),
   265→   265→    page: page.value,
   266→   266→    limit: limit.value,
   267→   267→    total,
   268→   268→    totalPages: total === 0 ? 0 : Math.ceil(total / limit.value),
   269→   269→    facets: {
   270→   270→      types: typeFacetRows.map((r) => ({ value: r.value, count: num(r.count) })),
   271→   271→      genres: genreFacetRows.map((r) => ({ value: r.value, count: num(r.count) })),
   272→   272→      years: yearFacetRows.map((r) => ({ value: num(r.value), count: num(r.count) })),
   273→   273→    },
   274→   274→  });
   275→   275→}));
