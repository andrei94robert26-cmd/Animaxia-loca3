    83→    83→/** API error with the parsed JSON body attached (FAZA 16: 429 pacing
    84→    84→ *  responses carry { error, code, retryInMs } the UI needs for its
    85→    85→ *  „Așteaptă un moment…” affordance). */
    86→    86→export class ApiError extends Error {
    87→    87→  status: number;
    88→    88→  body: Record<string, unknown> | null;
    89→    89→  constructor(message: string, status: number, body: Record<string, unknown> | null = null) {
    90→    90→    super(message);
    91→    91→    this.name = 'ApiError';
    92→    92→    this.status = status;
    93→    93→    this.body = body;
    94→    94→  }
    95→    95→}

    97→    97→type ParamValue = string | number | boolean | undefined | null;

    99→    99→export function apiUrl(path: string, params?: Record<string, ParamValue>): string {
   100→   100→  const search = new URLSearchParams();
   101→   101→  search.set('XTransformPort', String(BACKEND_PORT));
   102→   102→  if (params) {
   103→   103→    for (const [key, value] of Object.entries(params)) {
   104→   104→      if (value !== undefined && value !== null && value !== '') search.set(key, String(value));
   105→   105→    }
   106→   106→  }
   107→   107→  return `${path}?${search.toString()}`;
   108→   108→}

   110→   110→/** FAZA 19 — direct `<track src>` URL for a WebVTT subtitle track. The URL
   111→   111→ *  is relative (same origin) with the gateway port marker, exactly like every
   112→   112→ *  other request: the browser fetches it through Caddy, which forwards to the
   113→   113→ *  backend where the raw VTT is served public with a 1h cache. */
   114→   114→export function playerHubSubtitleUrl(contentId: number, trackId: number): string {
   115→   115→  return apiUrl(`/api/playerhub/subtitles/${contentId}/${trackId}`);
   116→   116→}

   118→   118→async function request<T>(path: string, init?: RequestInit): Promise<T> {
   119→   119→  let res: Response;
   120→   120→  try {
   121→   121→    res = await fetch(path, {
   122→   122→      ...init,
   123→   123→      headers: { 'Content-Type': 'application/json', ...(init?.headers as Record<string, string> | undefined) },
   124→   124→    });
   125→   125→  } catch {
   126→   126→    throw new ApiError('Cannot reach the Animaxia backend — it may be restarting', 0);
   127→   127→  }
   128→   128→  let body: unknown = null;
   129→   129→  try {
   130→   130→    body = await res.json();
   131→   131→  } catch {
   132→   132→    body = null;
   133→   133→  }
   134→   134→  if (!res.ok) {
   135→   135→    const message =
   136→   136→      (body as { error?: string } | null)?.error || `Request failed with status ${res.status}`;
   137→   137→    throw new ApiError(message, res.status, body as Record<string, unknown> | null);
   138→   138→  }
   139→   139→  return body as T;
   140→   140→}

   142→   142→/** Bearer header for FAZA 16 auth endpoints (same storage key as platform.ts
   143→   143→ *  tokenStore — read directly here to avoid a circular import). */
   144→   144→function authHeaders(): Record<string, string> {
   145→   145→  try {
   146→   146→    const token = window.localStorage.getItem('animaxia.accessToken');
   147→   147→    return token ? { Authorization: `Bearer ${token}` } : {};
   148→   148→  } catch {
   149→   149→    return {};
   150→   150→  }
   151→   151→}

   486→   486→/** FAZA 17 — build a virtual-catalog cursor client-side, byte-for-byte like
   487→   487→ *  the backend: base64url(JSON {i:<index>, s:<seed>}). Enables O(1) jumps to
   488→   488→ *  ANY page without server round-trips (verified live against :4000). */
   489→   489→export function buildVirtualCursor(index: number, seed: string): string {
   490→   490→  const json = JSON.stringify({ i: index, s: seed });
   491→   491→  const b64 =
   492→   492→    typeof btoa === 'function'
   493→   493→      ? btoa(json)
   494→   494→      : Buffer.from(json, 'utf8').toString('base64');
   495→   495→  return b64.replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
   496→   496→}
