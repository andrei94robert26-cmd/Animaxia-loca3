    99→function formatUptime(sec: number): string {
   100→  const h = Math.floor(sec / 3600);
   101→  const m = Math.floor((sec % 3600) / 60);
   102→  const s = Math.floor(sec % 60);
   103→  if (h > 0) return `${h}h ${m}m`;
   104→  if (m > 0) return `${m}m ${s}s`;
   105→  return `${s}s`;
   106→}
   107→
   108→function formatBytes(b?: number): string {
   109→  if (!b || b <= 0) return '0 B';
   110→  const units = ['B', 'KB', 'MB', 'GB'];
   111→  let v = b;
   112→  let i = 0;
   113→  while (v >= 1024 && i < units.length - 1) {
   114→    v /= 1024;
   115→    i += 1;
   116→  }
   117→  return `${v >= 100 ? Math.round(v) : v.toFixed(1)} ${units[i]}`;
   118→}
   119→
   120→function formatMs(ms?: number): string {
   121→  if (ms == null) return '—';
   122→  if (ms < 1000) return `${Math.round(ms)}ms`;
   123→  return `${(ms / 1000).toFixed(1)}s`;
   124→}
   125→
   126→function formatDate(iso?: string | null): string {
   127→  if (!iso) return '—';
   128→  const d = new Date(iso);
   129→  if (Number.isNaN(d.getTime())) return '—';
   130→  return d.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
   131→}
