    87→export const BACKEND_PORT = 4000;
    88→
    89→/** API error with the parsed JSON body attached (FAZA 16: 429 pacing
    90→ *  responses carry { error, code, retryInMs } the UI needs for its
    91→ *  „Așteaptă un moment…” affordance). */
    92→export class ApiError extends Error {
    93→  status: number;
    94→  body: Record<string, unknown> | null;
    95→  constructor(message: string, status: number, body: Record<string, unknown> | null = null) {
    96→    super(message);
    97→    this.name = 'ApiError';
    98→    this.status = status;
    99→    this.body = body;
   100→  }
   101→}
   102→
   103→type ParamValue = string | number | boolean | undefined | null;
   104→
   105→export function apiUrl(path: string, params?: Record<string, ParamValue>): string {
   106→  const search = new URLSearchParams();
   107→  search.set('XTransformPort', String(BACKEND_PORT));
   108→  if (params) {
   109→    for (const [key, value] of Object.entries(params)) {
   110→      if (value !== undefined && value !== null && value !== '') search.set(key, String(value));
   111→    }
   112→  }
   113→  return `${path}?${search.toString()}`;
   114→}

   116→/** FAZA 19 — direct `<track src>` URL for a WebVTT subtitle track. The URL
   117→ *  is relative (same origin) with the gateway port marker, exactly like every
   118→ *  other request: the browser fetches it through Caddy, which forwards to the
   119→ *  backend where the raw VTT is served public with a 1h cache. */
   120→export function playerHubSubtitleUrl(contentId: number, trackId: number): string {
   121→  return apiUrl(`/api/playerhub/subtitles/${contentId}/${trackId}`);
   122→}
   123→
   124→async function request<T>(path: string, init?: RequestInit): Promise<T> {
   125→  let res: Response;
   126→  try {
   127→    res = await fetch(path, {
   128→      ...init,
   129→      headers: { 'Content-Type': 'application/json', ...(init?.headers as Record<string, string> | undefined) },
   130→    });
   131→  } catch {
   132→    throw new ApiError('Cannot reach the Animaxia backend — it may be restarting', 0);
   133→  }
   134→  let body: unknown = null;
   135→  try {
   136→    body = await res.json();
   137→  } catch {
   138→    body = null;
   139→  }
   140→  if (!res.ok) {
   141→    const message =
   142→      (body as { error?: string } | null)?.error || `Request failed with status ${res.status}`;
   143→    throw new ApiError(message, res.status, body as Record<string, unknown> | null);
   144→  }
   145→  return body as T;
   146→}

   148→/** Bearer header for FAZA 16 auth endpoints (same storage key as platform.ts
   149→ *  tokenStore — read directly here to avoid a circular import). */
   150→function authHeaders(): Record<string, string> {
   151→  try {
   152→    const token = window.localStorage.getItem('animaxia.accessToken');
   153→    return token ? { Authorization: `Bearer ${token}` } : {};
   154→  } catch {
   155→    return {};
   156→  }
   157→}

   546→/** FAZA 17 — build a virtual-catalog cursor client-side, byte-for-byte like
   547→ *  the backend: base64url(JSON {i:<index>, s:<seed>}). Enables O(1) jumps to
   548→ *  ANY page without server round-trips (verified live against :4000). */
   549→export function buildVirtualCursor(index: number, seed: string): string {
   550→  const json = JSON.stringify({ i: index, s: seed });
   551→  const b64 =
   552→    typeof btoa === 'function'
   553→      ? btoa(json)
   554→      : Buffer.from(json, 'utf8').toString('base64');
   555→  return b64.replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
   556→}

   560→export function dbToItem(c: DbContent): NormalizedItem {
   561→  const md = c.metadata || {};
   562→  const sources = md.player?.sources || md.sources;
   563→  return {
   564→    key: `db-${c.id}`,
   565→    kind: 'db',
   566→    dbId: c.id,
   567→    title: c.title,
   568→    image: c.thumbnail || md.backdrop || null,
   569→    backdrop: md.backdrop || null,
   570→    overview: md.overview || md.synopsis || md.description || undefined,
   571→    rating: typeof md.rating === 'number' ? md.rating : typeof md.score === 'number' ? md.score : null,
   572→    year: (md.year as string | number | undefined) ?? null,
   573→    type: c.type || 'mined',
   574→    genres: Array.isArray(md.genres) ? md.genres.map(String) : undefined,
   575→    playable: Boolean(sources && sources.length > 0),
   576→    source: typeof md.source === 'string' ? md.source : undefined,
   577→  };
   578→}
