    87→    87→class LocalStore {
    88→    88→  constructor(file) {
    89→    89→    this.file = file;
    90→    90→    this.rows = [];
    91→    91→    this.load();
    92→    92→  }
    93→    93→
    94→    94→  load() {
    95→    95→    try {
    96→    96→      if (fs.existsSync(this.file)) {
    97→    97→        const parsed = JSON.parse(fs.readFileSync(this.file, 'utf8'));
    98→    98→        this.rows = Array.isArray(parsed) ? parsed : [];
    99→    99→      }
   100→   100→    } catch (err) {
   101→   101→      logger.error(`[DB] Failed to read local store: ${err.message} — starting empty`);
   102→   102→      this.rows = [];
   103→   103→    }
   104→   104→  }
   105→   105→
   106→   106→  persist() {
   107→   107→    fs.mkdirSync(path.dirname(this.file), { recursive: true });
   108→   108→    const tmp = `${this.file}.tmp`;
   109→   109→    fs.writeFileSync(tmp, JSON.stringify(this.rows, null, 2), 'utf8');
   110→   110→    fs.renameSync(tmp, this.file);
   111→   111→  }
   112→   112→
   113→   113→  nextId() {
   114→   114→    return this.rows.reduce((max, r) => Math.max(max, Number(r.id) || 0), 0) + 1;
   115→   115→  }
   116→   116→
   117→   117→  all() {
   118→   118→    return this.rows;
   119→   119→  }
   120→   120→
   121→   121→  list({ page = 1, limit = 20, type, search, sort = 'createdat' }) {
   122→   122→    let filtered = this.rows;
   123→   123→    if (type) filtered = filtered.filter((r) => r.type === type);
   124→   124→    if (search) {
   125→   125→      const needle = String(search).toLowerCase();
   126→   126→      filtered = filtered.filter((r) => String(r.title || '').toLowerCase().includes(needle));
   127→   127→    }
   128→   128→    const sorted = [...filtered].sort((a, b) => {
   129→   129→      if (sort === 'title') {
   130→   130→        return String(a.title || '').localeCompare(String(b.title || '')) || a.id - b.id;
   131→   131→      }
   132→   132→      const at = Date.parse(a.createdat) || 0;
   133→   133→      const bt = Date.parse(b.createdat) || 0;
   134→   134→      if (bt !== at) return bt - at;
   135→   135→      return b.id - a.id;
   136→   136→    });
   137→   137→    const total = sorted.length;
   138→   138→    const safeLimit = Math.min(100, Math.max(1, limit));
   139→   139→    const safePage = Math.max(1, page);
   140→   140→    const start = (safePage - 1) * safeLimit;
   141→   141→    return {
   142→   142→      data: sorted.slice(start, start + safeLimit).map(normalizeRow),
   143→   143→      total,
   144→   144→      page: safePage,
   145→   145→      limit: safeLimit,
   146→   146→      totalPages: Math.max(1, Math.ceil(total / safeLimit)),
   147→   147→    };
   148→   148→  }
   149→   149→
   150→   150→  byId(id) {
   151→   151→    const numeric = Number(id);
   152→   152→    return this.rows.find((r) => Number(r.id) === numeric) || null;
   153→   153→  }
   154→   154→
   155→   155→  byExternalId(externalid) {
   156→   156→    return this.rows.find((r) => r.externalid === externalid) || null;
   157→   157→  }
   158→   158→
   159→   159→  upsert({ externalid, type, title, metadata, thumbnail }) {
   160→   160→    const existing = this.byExternalId(externalid);
   161→   161→    if (existing) {
   162→   162→      existing.type = type === undefined || type === null ? existing.type : type;
   163→   163→      existing.title = title ?? existing.title;
   164→   164→      existing.metadata = safeParseJson(metadata, {}) ?? {};
   165→   165→      existing.thumbnail = thumbnail === undefined ? existing.thumbnail : thumbnail;
   166→   166→      this.persist();
   167→   167→      return { record: normalizeRow(existing), created: false };
   168→   168→    }
   169→   169→    const row = {
   170→   170→      id: this.nextId(),
   171→   171→      externalid,
   172→   172→      type: type ?? null,
   173→   173→      title,
   174→   174→      metadata: safeParseJson(metadata, {}) ?? {},
   175→   175→      thumbnail: thumbnail ?? null,
   176→   176→      createdat: new Date().toISOString(),
   177→   177→    };
   178→   178→    this.rows.push(row);
   179→   179→    this.persist();
   180→   180→    return { record: normalizeRow(row), created: true };
   181→   181→  }
   182→   182→
   183→   183→  remove(id) {
   184→   184→    const existing = this.byId(id);
   185→   185→    if (!existing) return null;
   186→   186→    this.rows = this.rows.filter((r) => r !== existing);
   187→   187→    this.persist();
   188→   188→    return normalizeRow(existing);
   189→   189→  }
   190→   190→
   191→   191→  count() {
   192→   192→    return this.rows.length;
   193→   193→  }
   194→   194→
   195→   195→  countsByType() {
   196→   196→    return this.rows.reduce((acc, r) => {
   197→   197→      const key = r.type || 'unknown';
   198→   198→      acc[key] = (acc[key] || 0) + 1;
   199→   199→      return acc;
   200→   200→    }, {});
   201→   201→  }

   221→   221→ * One-time migration of local-fallback rows into PostgreSQL (idempotent upserts).
   222→   222→ * After a successful migration the local file is archived as content.json.migrated
   223→   223→ * so subsequent boots do not repeat the work.

   225→   225→async function migrateLocalToPostgres() {
   226→   226→  if (!fs.existsSync(LOCAL_FILE)) return 0;
   227→   227→  let rows = [];
   228→   228→  try {
   229→   229→    const parsed = JSON.parse(fs.readFileSync(LOCAL_FILE, 'utf8'));
   230→   230→    rows = Array.isArray(parsed) ? parsed : [];
   231→   231→  } catch (err) {
   232→   232→    logger.warn(`[DB] Local store unreadable, skipping migration: ${err.message}`);
   233→   233→    return 0;
   234→   234→  }
   235→   235→  if (!rows.length) return 0;
   236→   236→  let migrated = 0;
   237→   237→  for (const row of rows) {
   238→   238→    try {
   239→   239→      await q(UPSERT_SQL, [
   240→   240→        row.externalid,
   241→   241→        row.type ?? null,
   242→   242→        row.title ?? 'untitled',
   243→   243→        JSON.stringify(safeParseJson(row.metadata, {})),
   244→   244→        row.thumbnail ?? null,
   245→   245→      ]);
   246→   246→      migrated += 1;
   247→   247→    } catch (err) {
   248→   248→      logger.warn(`[DB] Migration skipped row ${row.externalid}: ${err.message}`);
   249→   249→    }
   250→   250→  }
   251→   251→  try {
   252→   252→    fs.renameSync(LOCAL_FILE, `${LOCAL_FILE}.migrated`);
   253→   253→  } catch (err) {
   254→   254→    logger.warn(`[DB] Could not archive local store: ${err.message}`);
   255→   255→  }
   256→   256→  logger.info(`[DB] Migrated ${migrated}/${rows.length} local rows into PostgreSQL`);
   257→   257→  return migrated;
   258→   258→}

   261→   261→  if (reconnectTimer || !pool) return;

   310→   310→
   311→   311→  try {
   312→   312→    await withTimeout(pool.query('SELECT 1'), 8000);

   314→   314→    // Adopt any rows written while running on the local fallback store.
   315→   315→    await migrateLocalToPostgres();

   317→   317→    lastError = null;
   318→   318→    stopReconnectProbe();
   319→   319→    logger.info('[DB] Connected to PostgreSQL (Neon) — schema ready');
   320→   320→  } catch (err) {
   321→   321→    lastError = err.message;
   322→   322→    logger.error(`[DB] PostgreSQL unavailable (${err.message}) — switching to local fallback store`);
   323→   323→    switchToLocal();
   324→   324→  }
   325→   325→  return driverInfo();
   326→   326→}

   395→   395→/**
   396→   396→ * FAZA 16-a central dedup hook: after a successful postgres upsert, extract
   397→   397→ * every source embedded in the content metadata (url, finalUrl, player,
   398→   398→ * sources[] {src,kind}, trailer, youtube) and register each one in the global

   400→   400→ * (miner, POST /api/content, discover add, AI enrichment, external frontend

   404→   404→  try {
   405→   405→    if (driver !== 'postgres' || !record?.id) return;
   406→   406→    const { registerSource } = await import('./modules/ingest/dedup.js');
   407→   407→    const meta = safeParseJson(record.metadata, {}) ?? {};
   408→   408→    const candidates = [];
   409→   409→    const pushStr = (v) => {
   410→   410→      if (typeof v === 'string' && v.trim()) candidates.push(v);
   411→   411→    };
   412→   412→    pushStr(meta.url);
   413→   413→    pushStr(meta.finalUrl);
   414→   414→    pushStr(meta.trailer);
   415→   415→    pushStr(meta.youtube);
   416→   416→    if (typeof meta.player === 'string') {
   417→   417→      pushStr(meta.player);
   418→   418→    } else if (meta.player && typeof meta.player === 'object') {
   419→   419→      pushStr(meta.player.url);
   420→   420→      pushStr(meta.player.embed);
   421→   421→      pushStr(meta.player.src);
   422→   422→    }
   423→   423→    if (Array.isArray(meta.sources)) {
   424→   424→      for (const s of meta.sources) {
   425→   425→        if (typeof s === 'string') pushStr(s);
   426→   426→        else if (s && typeof s === 'object') pushStr(s.src);
   427→   427→      }
   428→   428→    }
   429→   429→    for (const raw of candidates) {
   430→   430→      try {
   431→   431→        await registerSource({ url: raw, contentId: record.id, externalId: record.externalid, addedBy: 'upsert' });
   432→   432→      } catch {
   433→   433→        /* best-effort per source */
   434→   434→      }
   435→   435→    }
   436→   436→  } catch {
   437→   437→    /* never throw from the hook */
   438→   438→  }

   516→   516→  if (driver !== 'postgres' || !pool) {
   517→   517→    const err = new Error('SQL layer unavailable — PostgreSQL driver inactive (local fallback store in use)');
   518→   518→    err.status = 503;
   519→   519→    throw err;
   520→   520→  }
   521→   521→  return q(text, params);
   522→   522→}
   523→   523→
   524→   524→export async function ping() {
   525→   525→  if (driver === 'postgres') {
   526→   526→    try {
   527→   527→      await withTimeout(pool.query('SELECT 1'), 3000);
   528→   528→      return true;
   529→   529→    } catch {
   530→   530→      return false;
   531→   531→    }
   532→   532→  }
   533→   533→  return true;
   534→   534→}

   536→   536→export async function closeDb() {
   537→   537→  stopReconnectProbe();
   538→   538→  if (pool) {
   539→   539→    try {
   540→   540→      await pool.end();
   541→   541→      logger.info('[DB] PostgreSQL pool closed');
   542→   542→    } catch (err) {
   543→   543→      logger.warn(`[DB] pool close error: ${err.message}`);
   544→   544→    }
   545→   545→  }
   546→   546→}

   548→   548→/* ====================================================================
   549→   549→ * FAZA 13 (sharding — Task 13-2): Neon multi-branch shard pool registry.
   550→   550→ *
   551→   551→ * Every shard id in the generated shard map (see modules/sharding/model.js)
   552→   552→ * gets a registry entry. Shard 0 and any shard N ≥ 1 WITHOUT a configured
   553→   553→ * SHARD{N}URL resolve to the MAIN pool (logical sharding mode: shardid
   554→   554→ * columns, routing, metrics and health are real, one physical database). A
   555→   555→ * shard N ≥ 1 WITH a non-empty SHARD{N}URL gets a dedicated pg.Pool on that
   556→   556→ * Neon branch (physical mode, activated purely by configuration). queryOnShard
   557→   557→ * falls back to the main pool when a dedicated branch pool fails at the
   558→   558→ * connection level — a down branch never breaks reads.
   559→   559→ * ==================================================================== */
   560→   560→
   561→   561→const shardPools = new Map(); // shardId -> { shardid, branch, url, pool, ownPool, healthy, lastError, stats }
   562→   562→
   563→   563→/** Create a standalone pg.Pool for a Neon branch (same tuning as the main
   564→   564→ *  pool). Does NOT register it — registration is by shardId via ensureShardPool. */
   565→   565→export function createPoolForBranch(name, url) {
   566→   566→  const branchPool = new pg.Pool({
   567→   567→    connectionString: url,
   568→   568→    ssl: process.env.PGSSL === 'true' ? { rejectUnauthorized: false } : undefined,
   569→   569→    max: parseInt(process.env.PGPOOLMAX || '10', 10),
   570→   570→    idleTimeoutMillis: 30000,
   571→   571→    connectionTimeoutMillis: 8000,
   572→   572→  });
   573→   573→  branchPool.on('error', (err) => logger.error(`[DB] shard pool ${name} idle client error: ${err.message}`));
   574→   574→  return branchPool;
   575→   575→}
   576→   576→
   577→   577→/**
   578→   578→ * Idempotent per-shard registry entry. Shard 0 / shards without SHARD{N}URL →
   579→   579→ * { ownPool: false, pool: null } where the MAIN pool is resolved lazily at
   580→   580→ * query time (initDb() may not have run yet when this is first called).
   581→   581→ * Shard N ≥ 1 with a non-empty SHARD{N}URL → dedicated branch pool
   582→   582→ * (ownPool: true) created once and cached.
   583→   583→ */
   584→   584→export function ensureShardPool(shardId) {
   585→   585→  const n = Number(shardId);
   586→   586→  const id = Number.isInteger(n) && n >= 0 ? n : 0;
   587→   587→  const existing = shardPools.get(id);
   588→   588→  if (existing) return existing;
   589→   589→
   590→   590→  const branch = shardBranchName(id);
   591→   591→  const url = id >= 1 ? String(process.env[`SHARD${id}URL`] || '').trim() : '';
   592→   592→  const stats = { queries: 0, errors: 0 };
   593→   593→
   594→   594→  if (id >= 1 && url) {
   595→   595→    const entry = {
   596→   596→      shardid: id,
   597→   597→      branch,
   598→   598→      url,
   599→   599→      ownPool: true,
   600→   600→      pool: createPoolForBranch(branch, url),
   601→   601→      healthy: true,
   602→   602→      lastError: null,
   603→   603→      stats,
   604→   604→    };
   605→   605→    shardPools.set(id, entry);
   606→   606→    logger.info(`[DB] shard ${id} → dedicated Neon branch pool (${branch})`);
   607→   607→    return entry;
   608→   608→  }
   609→   609→
   610→   610→  // Logical mode: main-pool-backed. `pool: null` on purpose — the main pool
   611→   611→  // is resolved at query time (it may not exist yet pre-boot).
   612→   612→  const entry = {
   613→   613→    shardid: id,
   614→   614→    branch,
   615→   615→    url: null,
   616→   616→    ownPool: false,
   617→   617→    pool: null,
   618→   618→    healthy: null, // mirrors driver === 'postgres' at read time
   619→   619→    lastError: null,
   620→   620→    stats,
   621→   621→  };
   622→   622→  shardPools.set(id, entry);
   623→   623→  return entry;
   624→   624→}
   625→   625→
   626→   626→/** Ensure + return the underlying pg pool for a shard (null pre-boot for
   627→   627→ *  main-backed entries — the main pool is initialised by initDb()). */
   628→   628→export function getPoolForShard(shardId) {
   629→   629→  const entry = ensureShardPool(shardId);
   630→   630→  if (entry.ownPool) return entry.pool;
   631→   631→  return pool; // module-scope main pool (null until initDb runs)
   632→   632→}

   634→   634→/** Connection-level failure detection for the branch → main fallback. */
   635→   635→function isConnectionError(err) {
   636→   636→  const code = String(err?.code || '');
   637→   637→  if (
   638→   638→    code.startsWith('08') || // PostgreSQL class 08 — connection_exception
   639→   639→    ['57P01', '57P02', '57P03'].includes(code) || // admin/crash shutdown, cannot connect now
   640→   640→    ['ECONNREFUSED', 'ETIMEDOUT', 'ENOTFOUND', 'ECONNRESET', 'EPIPE', 'EAI_AGAIN'].includes(code)
   641→   641→  ) {
   642→   642→    return true;
   643→   643→  }
   644→   644→  const message = String(err?.message || '');
   645→   645→  return (
   646→   646→    message.includes('timeout exceeded when trying to connect') ||
   647→   647→    message.includes('Connection terminated') ||
   648→   648→    message.includes('connection terminated')
   649→   649→  );
   650→   650→}
   651→   651→
   652→   652→/**
   653→   653→ * Run a query on a specific shard's pool. Dedicated branch pools fall back to
   654→   654→ * the MAIN pool on connection-level failures (resilience — a down branch never
   655→   655→ * breaks reads; logged as a warning). Throws the same 503-tagged error as
   656→   656→ * query() when the PostgreSQL driver is inactive (local fallback store).
   657→   657→ */
   658→   658→export async function queryOnShard(shardId, text, params = []) {
   659→   659→  if (driver !== 'postgres' || !pool) {
   660→   660→    const err = new Error('SQL layer unavailable — PostgreSQL driver inactive (local fallback store in use)');
   661→   661→    err.status = 503;
   662→   662→    throw err;
   663→   663→  }
   664→   664→  const entry = ensureShardPool(shardId);
   665→   665→  entry.stats.queries += 1;
   666→   666→  const target = entry.ownPool ? entry.pool : pool;
   667→   667→  try {
   668→   668→    if (!target) throw new Error('shard pool not initialised');
   669→   669→    const res = await target.query(text, params);
   670→   670→    if (entry.ownPool) {
   671→   671→      entry.healthy = true;
   672→   672→      entry.lastError = null;
   673→   673→    }
   674→   674→    return res;
   675→   675→  } catch (err) {
   676→   676→    entry.stats.errors += 1;
   677→   677→    if (entry.ownPool && (err.message === 'shard pool not initialised' || isConnectionError(err))) {
   678→   678→      entry.healthy = false;
   679→   679→      entry.lastError = err.message;
   680→   680→      logger.warn(
   681→   681→        `[DB] shard ${entry.shardid} (${entry.branch}) query failed (${err.message}) — falling back to main pool`,
   682→   682→      );
   683→   683→      return pool.query(text, params); // main pool is guaranteed non-null here
   684→   684→    }
   685→   685→    throw err;
   686→   686→  }
   687→   687→}

   689→   689→/** Registry snapshot for metrics/admin — one entry per shard in the shard map. */
   690→   690→export function listShardPools() {
   691→   691→  return Object.keys(shardMap())
   692→   692→    .map(Number)
   693→   693→    .sort((a, b) => a - b)
   694→   694→    .map((id) => {
   695→   695→      const entry = ensureShardPool(id);
   696→   696→      const poolStatsOf = (p) => ({
   697→   697→        total: typeof p?.totalCount === 'number' ? p.totalCount : 0,
   698→   698→        idle: typeof p?.idleCount === 'number' ? p.idleCount : 0,
   699→   699→        waiting: typeof p?.waitingCount === 'number' ? p.waitingCount : 0,
   700→   700→      });
   701→   701→      if (entry.ownPool) {
   702→   702→        return {
   703→   703→          shardid: entry.shardid,
   704→   704→          branch: entry.branch,
   705→   705→          mode: 'branch',
   706→   706→          urlConfigured: true,
   707→   707→          connected: entry.healthy === true,
   708→   708→          healthy: entry.healthy,
   709→   709→          lastError: entry.lastError,
   710→   710→          pool: poolStatsOf(entry.pool),
   711→   711→          stats: { ...entry.stats },
   712→   712→        };
   713→   713→      }
   714→   714→      return {
   715→   715→        shardid: entry.shardid,
   716→   716→        branch: entry.branch,
   717→   717→        mode: 'main',
   718→   718→        urlConfigured: id === 0 ? Boolean(String(process.env.DATABASEURL || '').trim()) : false,
   719→   719→        connected: driver === 'postgres',
   720→   720→        lastError: entry.lastError,
   721→   721→        pool: poolStats(),
   722→   722→        stats: { ...entry.stats },
   723→   723→      };
   724→   724→    });
   725→   725→}

   727→   727→/** End every dedicated branch pool (ownPool). The main pool is closed by
   728→   728→ *  closeDb() — never double-closed here. */
   729→   729→export async function closeShardPools() {
   730→   730→  const closed = [];
   731→   731→  for (const [id, entry] of shardPools) {
   732→   732→    if (!entry.ownPool || !entry.pool) continue;
   733→   733→    try {
   734→   734→      await entry.pool.end();
   735→   735→      closed.push(id);
   736→   736→      entry.healthy = false;
   737→   737→      entry.lastError = 'pool closed';
   738→   738→    } catch (err) {
   739→   739→      logger.warn(`[DB] shard ${id} pool close error: ${err.message}`);
   740→   740→    }
   741→   741→  }
   742→   742→  if (closed.length) logger.info(`[DB] shard branch pools closed (${closed.join(', ')})`);
   743→   743→  return closed;
   744→   744→}
