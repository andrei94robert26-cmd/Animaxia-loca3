    94→export function CategoryView({ route }: { route: AppRoute }) {
    95→  const type = route.params[0] || 'all';
    96→
    97→  /* Reset pagination whenever the category changes (adjust during render). */
    98→  const [page, setPage] = useState(1);
    99→  const [lastType, setLastType] = useState(type);
   100→  if (lastType !== type) {
   101→    setLastType(type);
   102→    setPage(1);
   103→  }
   104→
   105→  const [newsSource, setNewsSource] = useState('all');
   106→  const [sportTab, setSportTab] = useState<'football' | 'f1'>('football');
   107→  const [competition, setCompetition] = useState('PL');
   108→  const [addOpen, setAddOpen] = useState(false);
   109→
   110→  const isNews = type === 'news';
   111→  const isSports = type === 'sports';
   112→  const isSpecial = SPECIAL_CATEGORIES.has(type);
   113→  const virtual = VIRTUAL_CATEGORIES[type];
   114→  const isLive = Boolean(virtual?.live) && !isSpecial;
   115→  const isVirtual = Boolean(virtual) && !isSpecial;
   116→  const isGrid = !isNews && !isSports && !isSpecial && !isVirtual;
   117→  const hasStrip = type === 'movie' || type === 'tv' || type === 'anime';
   118→
   119→  const meta =
   120→    CATEGORY_META[type] ??
   121→    (virtual
   122→      ? { title: virtual.title, subtitle: virtual.subtitle }
   123→      : {
   124→          title: capitalize(type),
   125→          subtitle: 'Browse every title Animaxia has filed under this category.',
   126→        });
   127→
   128→  const listRes = useApi(
   129→    `cat-${type}-${page}`,
   130→    () => api.all({ type: type === 'all' ? undefined : type, page, limit: 24 }),
   131→    { enabled: isGrid },
   132→  );
   133→
   134→  /* Virtual sidebar categories: one full-catalog fetch + client-side filter. */
   135→  const virtualRes = useApi(`cat-virtual-${type}`, () => api.all({ limit: 100 }), { enabled: isVirtual });
   136→  const virtualItems = useMemo<NormalizedItem[]>(() => {
   137→    if (!virtual || !virtualRes.data) return [];
   138→    const rows = virtualRes.data.data.filter(virtual.filter);
   139→    const sorted = virtual.sort ? [...rows].sort(virtual.sort) : rows;
   140→    return sorted.map(dbToItem);
   141→  }, [virtual, virtualRes.data]);
   142→
   143→  /* FAZA 14: live category feeds (real TMDB discover) for every sidebar menu. */
   144→  const liveRes = useApi(`cat-live-${type}-${page}`, () => api.discover(type, page), { enabled: isLive });
   145→  const liveItems = useMemo<NormalizedItem[]>(() => {
   146→    if (!liveRes.data) return [];
   147→    return liveRes.data.data.map(mediaToItem);
   148→  }, [liveRes.data]);
   149→
   150→  const stripRes = useApi(
   151→    `cat-strip-${type}`,
   152→    async (): Promise<NormalizedItem[]> => {
   153→      if (type === 'movie') return (await api.moviesTrending('day')).data.map(mediaToItem);
   154→      if (type === 'tv') return (await api.tvPopular()).data.map(mediaToItem);
   155→      return (await api.animeSeason()).data.map(animeToItem);
   156→    },
   157→    { enabled: hasStrip },
   158→  );
   159→
   160→  const newsRes = useApi(`cat-news-${newsSource}`, () => api.news(newsSource, 24), { enabled: isNews });
   161→
   162→  const standingsRes = useApi(`fb-standings-${competition}`, () => api.footballStandings(competition), {
   163→    enabled: isSports,
   164→  });
   165→  const matchesRes = useApi(`fb-matches-${competition}`, () => api.footballMatches(competition, 10), {
   166→    enabled: isSports,
   167→  });
   168→  const f1StandingsRes = useApi('f1-standings', () => api.f1Standings(), {
   169→    enabled: isSports && sportTab === 'f1',
   170→  });
   171→  const f1ScheduleRes = useApi('f1-schedule', () => api.f1Schedule(), {
   172→    enabled: isSports && sportTab === 'f1',
   173→  });
   174→
   175→  const items = useMemo(() => (listRes.data?.data ?? []).map(dbToItem), [listRes.data]);
   176→  const gridRef = useRef<HTMLDivElement>(null);
   177→
   178→  const stripMeta =
   179→    type === 'movie'
   180→      ? { title: 'Trending on TMDB', subtitle: 'Today’s most-watched films worldwide' }
   181→      : type === 'tv'
   182→        ? { title: 'Popular on TMDB', subtitle: 'Series people cannot stop watching' }
   183→        : { title: 'Airing this season', subtitle: 'Currently broadcasting in Japan' };
   184→
   185→  const goToPage = (next: number) => {
   186→    setPage(next);
   187→    gridRef.current?.scrollIntoView({ behavior: 'smooth', block: 'start' });
   188→  };
   189→
   190→  return (
   191→    <Container className="space-y-8 py-8 sm:py-10">
   192→      <PageHeader title={meta.title} subtitle={meta.subtitle}>
   193→        {isGrid && !listRes.loading && listRes.data && (
   194→          <p className="text-sm font-medium text-zinc-500">
   195→            Showing {listRes.data.data.length} of {listRes.data.total}{' '}
   196→            {listRes.data.total === 1 ? 'title' : 'titles'}
   197→          </p>
   198→        )}
   199→        {isLive && virtual?.liveSource && (
   200→          <span className="inline-flex items-center gap-1.5 rounded-full border border-rose-500/30 bg-rose-600/10 px-3 py-1 text-xs font-bold text-rose-200">
   201→            <Zap className="h-3 w-3" aria-hidden />
   202→            LIVE · {virtual.liveSource}
   203→          </span>
   204→        )}
   205→      </PageHeader>
   206→
   207→      {isGrid && (
   208→        <>
   209→          {type === 'mined' && (
   210→            <Alert className="border-rose-500/20 bg-rose-500/5">
   211→              <Pickaxe className="h-4 w-4 text-rose-400" aria-hidden />
   212→              <AlertTitle className="text-zinc-100">Extracted by the Animaxia Miner</AlertTitle>
   213→              <AlertDescription className="text-zinc-400">
   214→                <span>
   215→                  Mined titles are extracted by the Animaxia Miner from any URL — open the Admin
   216→                  panel to mine a new stream.
   217→                </span>
   218→                <Button
   219→                  size="sm"
   220→                  onClick={() => navigate('/admin')}
   221→                  className="mt-2 bg-rose-600 font-semibold text-white hover:bg-rose-500"
   222→                >
   223→                  Open Admin panel
   224→                </Button>
   225→              </AlertDescription>
   226→            </Alert>
   227→          )}
   228→
   229→          {hasStrip &&
   230→            (stripRes.error && !stripRes.loading && !stripRes.data ? (
   231→              <ErrorState message={stripRes.error} onRetry={stripRes.refetch} compact />
   232→            ) : (
   233→              <ContentRow
   234→                title={stripMeta.title}
   235→                subtitle={stripMeta.subtitle}
   236→                items={stripRes.data ?? []}
   237→                loading={stripRes.loading}
   238→                emptyMessage="Live provider data is unavailable right now."
   239→              />
   240→            ))}
   241→
   242→          <div ref={gridRef} className="scroll-mt-24 space-y-6">
   243→            {listRes.error && !listRes.loading && !listRes.data ? (
   244→              <ErrorState message={listRes.error} onRetry={listRes.refetch} />
   245→            ) : (
   246→              <ContentGrid
   247→                items={items}
   248→                loading={listRes.loading}
   249→                skeletonCount={12}
   250→                emptyTitle={type === 'all' ? 'Your library is empty' : `No ${meta.title.toLowerCase()} yet`}
   251→                emptyMessage="Nothing has been added to this category so far — add titles from any detail page."
   252→              />
   253→            )}
   254→            <Pagination page={page} totalPages={listRes.data?.totalPages ?? 1} onPage={goToPage} />
   255→          </div>
   256→        </>
   257→      )}
   258→
   259→      {isSpecial && (
   260→        <>
   261→          {type === 'my-list' && <MyListPanel />}
   262→          {type === 'my-collections' && <CollectionsPanel />}
   263→          {type === 'manage' && <ManagePanel />}
   264→          {type === 'tvshows' && <TvShowsPanel />}
   265→          {type === 'radio' && <RadioPanel />}
   266→        </>
   267→      )}
   268→
   269→      {isVirtual && !isLive && (
   270→        <>
   271→          {virtualRes.error && !virtualRes.loading && !virtualRes.data ? (
   272→            <ErrorState message={virtualRes.error} onRetry={virtualRes.refetch} />
   273→          ) : (
   274→            <ContentGrid
   275→              items={virtualItems}
   276→              loading={virtualRes.loading}
   277→              skeletonCount={9}
   278→              emptyTitle={`Încă nimic în ${meta.title}`}
   279→              emptyMessage={
   280→                virtual?.emptyHint ??
   281→                'Această categorie se umple automat cu tot ce încarci tu — adaugă conținut din orice sursă.'
   282→              }
   283→              emptyAction={
   284→                <Button onClick={() => setAddOpen(true)} className="mt-4 bg-rose-600 font-bold hover:bg-rose-500">
   285→                  <Plus className="mr-1.5 h-4 w-4" aria-hidden />
   286→                  Adaugă conținut
   287→                </Button>
   288→              }
   289→            />
   290→          )}
   291→          {!virtualRes.loading && virtualItems.length > 0 && (
   292→            <p className="text-sm font-medium text-zinc-500">
   293→              {virtualItems.length} {virtualItems.length === 1 ? 'titlu' : 'titluri'} în {meta.title}
   294→            </p>
   295→          )}
   296→        </>
   297→      )}
   298→
   299→      {isLive && (
   300→        <>
   301→          {/* Titles already in the Neon library that match this category */}
   302→          {!virtualRes.loading && virtualItems.length > 0 && (
   303→            <ContentRow
   304→              title="În biblioteca ta"
   305→              subtitle={`${virtualItems.length} ${virtualItems.length === 1 ? 'titlu' : 'titluri'} salvate în Neon apar și aici`}
   306→              items={virtualItems}
   307→              emptyMessage=""
   308→            />
   309→          )}
   310→
   311→          {/* Live TMDB feed */}
   312→          <div ref={gridRef} className="scroll-mt-24 space-y-6">
   313→            {liveRes.error && !liveRes.loading && !liveRes.data ? (
   314→              <ErrorState message={liveRes.error} onRetry={liveRes.refetch} />
   315→            ) : (
   316→              <ContentGrid
   317→                items={liveItems}
   318→                loading={liveRes.loading && !liveRes.data}
   319→                skeletonCount={12}
   320→                emptyTitle="Feed-ul live nu a returnat nimic"
   321→                emptyMessage="Sursa live este momentan indisponibilă — încearcă din nou sau adaugă conținut propriu."
   322→                emptyAction={
   323→                  <Button onClick={() => setAddOpen(true)} className="mt-4 bg-rose-600 font-bold hover:bg-rose-500">
   324→                    <Plus className="mr-1.5 h-4 w-4" aria-hidden />
   325→                    Adaugă conținut
   326→                  </Button>
   327→                }
   328→              />
   329→            )}
   330→            <div className="flex items-center justify-between gap-4">
   331→              {liveRes.data && liveRes.data.total > 0 && (
   332→                <p className="text-sm font-medium text-zinc-500">
   333→                  {liveRes.data.total.toLocaleString('ro-RO')}{' '}
   334→                  {liveRes.data.total === 1 ? 'titlu' : 'titluri'} în feed-ul live
   335→                  {liveRes.data.legsOk < liveRes.data.legs && (
   336→                    <span className="text-amber-400"> · {liveRes.data.legsOk}/{liveRes.data.legs} surse active</span>
   337→                  )}
   338→                </p>
   339→              )}
   340→            </div>
   341→            <Pagination page={page} totalPages={Math.min(liveRes.data?.totalPages ?? 1, 50)} onPage={goToPage} />
   342→          </div>
   343→        </>
   344→      )}
   345→
   346→      <AddContentDialog open={addOpen} onOpenChange={setAddOpen} />
   347→
   348→      {isNews && <NewsPanel source={newsSource} onSource={setNewsSource} res={newsRes} />}
   349→
   350→      {isSports && (
   351→        <Tabs defaultValue="football" onValueChange={(value) => setSportTab(value as 'football' | 'f1')}>
   352→          <TabsList className="h-10 w-full justify-start gap-1 rounded-xl border border-white/10 bg-white/[0.03] p-1 sm:w-auto">
   353→            <TabsTrigger
   354→              value="football"
   355→              className="px-4 text-zinc-400 hover:text-zinc-100 data-[state=active]:bg-rose-600 data-[state=active]:text-white data-[state=active]:shadow-rose-900/40 sm:px-5"
   356→            >
   357→              Football
   358→            </TabsTrigger>
   359→            <TabsTrigger
   360→              value="f1"
   361→              className="px-4 text-zinc-400 hover:text-zinc-100 data-[state=active]:bg-rose-600 data-[state=active]:text-white data-[state=active]:shadow-rose-900/40 sm:px-5"
   362→            >
   363→              Formula 1
   364→            </TabsTrigger>
   365→          </TabsList>
   366→          <TabsContent value="football" className="pt-6">
   367→            <FootballPanel
   368→              competition={competition}
   369→              onCompetition={setCompetition}
   370→              standingsRes={standingsRes}
   371→              matchesRes={matchesRes}
   372→            />
   373→          </TabsContent>
   374→          <TabsContent value="f1" className="pt-6">
   375→            <F1Panel standingsRes={f1StandingsRes} scheduleRes={f1ScheduleRes} />
   376→          </TabsContent>
   377→        </Tabs>
   378→      )}
   379→    </Container>
   380→  );
   381→}

   383→/* ------------------------------ news panel ------------------------------- */
   384→
   385→function NewsPanel({
   386→  source,
   387→  onSource,
   388→  res,
   389→}: {
   390→  source: string;
   391→  onSource: (value: string) => void;
   392→  res: UseApiResult<{ data: NewsItem[] }>;
   393→}) {
   394→  const items = useMemo(() => (res.data?.data ?? []).map(newsToItem), [res.data]);
   395→  return (
   396→    <div className="space-y-6">
   397→      <div className="flex flex-wrap items-center gap-2" role="group" aria-label="News source filter">
   398→        {NEWS_SOURCES.map((s) => (
   399→          <FilterChip key={s.value} active={source === s.value} onClick={() => onSource(s.value)}>
   400→            {s.label}
   401→          </FilterChip>
   402→        ))}
   403→      </div>
   404→      {res.error && !res.loading && !res.data ? (
   405→        <ErrorState message={res.error} onRetry={res.refetch} />
   406→      ) : (
   407→        <ContentGrid
   408→          items={items}
   409→          loading={res.loading}
   410→          skeletonCount={12}
   411→          emptyTitle="No headlines right now"
   412→          emptyMessage="This feed returned no recent articles — try another source."
   413→        />
   414→      )}
   415→    </div>
   416→  );
   417→}

   419→/* ---------------------------- football panel ----------------------------- */

   421→function FootballPanel({
   422→  competition,
   423→  onCompetition,
   424→  standingsRes,
   425→  matchesRes,
   426→}: {
   427→  competition: string;
   428→  onCompetition: (code: string) => void;
   429→  standingsRes: UseApiResult<{ data: { competition: string; table: FootballTableEntry[] } }>;
   430→  matchesRes: UseApiResult<{ data: FootballMatch[] }>;
   431→}) {
   432→  const competitionName = COMPETITIONS.find((c) => c.code === competition)?.name ?? competition;
   433→  return (
   434→    <div className="space-y-8">
   435→      <div className="space-y-3">
   436→        <h2 className="text-lg font-bold text-zinc-100">Competition</h2>
   437→        <div className="flex flex-wrap gap-2" role="group" aria-label="Competition filter">
   438→          {COMPETITIONS.map((c) => (
   439→            <FilterChip key={c.code} active={competition === c.code} onClick={() => onCompetition(c.code)}>
   440→              {c.name}
   441→            </FilterChip>
   442→          ))}
   443→        </div>
   444→      </div>
   445→
   446→      <Section title="Standings" subtitle={`Current ${competitionName} table`}>
   447→        {standingsRes.loading ? (
   448→          <Skeleton className="h-80 w-full rounded-xl" />
   449→        ) : standingsRes.error ? (
   450→          <ErrorState message={standingsRes.error} onRetry={standingsRes.refetch} compact />
   451→        ) : !standingsRes.data || standingsRes.data.data.table.length === 0 ? (
   452→          <EmptyState title="No standings available" message="This competition has no table data right now." />
   453→        ) : (
   454→          <StandingsTable entries={standingsRes.data.data.table} />
   455→        )}
   456→      </Section>
   457→
   458→      <Section title="Fixtures & results" subtitle="Upcoming matches and recent scores">
   459→        {matchesRes.loading ? (
   460→          <Skeleton className="h-72 w-full rounded-xl" />
   461→        ) : matchesRes.error ? (
   462→          <ErrorState message={matchesRes.error} onRetry={matchesRes.refetch} compact />
   463→        ) : !matchesRes.data || matchesRes.data.data.length === 0 ? (
   464→          <EmptyState title="No matches scheduled" message="There are no upcoming fixtures for this competition." />
   465→        ) : (
   466→          <div className="overflow-hidden rounded-xl border border-white/10 bg-white/[0.02]">
   467→            <div className="divide-y divide-white/5">
   468→              {matchesRes.data.data.map((match) => (
   469→                <MatchRow key={match.id} match={match} />
   470→              ))}
   471→            </div>
   472→          </div>
   473→        )}
   474→      </Section>
   475→    </div>
   476→  );
   477→}

   536→function MatchRow({ match }: { match: FootballMatch }) {
   537→  const finished =
   538→    match.status === 'FINISHED' && match.score.home != null && match.score.away != null;
   539→  const date = new Date(match.utcDate);
   540→  const dateLabel = date.toLocaleDateString('en-US', {
   541→    weekday: 'short',
   542→    month: 'short',
   543→    day: 'numeric',
   544→    timeZone: 'UTC',
   545→  });
   546→  const timeLabel = date.toLocaleTimeString('en-US', {
   547→    hour: '2-digit',
   548→    minute: '2-digit',
   549→    hour12: false,
   550→    timeZone: 'UTC',
   551→  });
   552→  return (
   553→    <div className="flex flex-col gap-3 px-4 py-4 sm:flex-row sm:items-center sm:justify-between">
   554→      <div className="grid min-w-0 flex-1 grid-cols-[1fr_auto_1fr] items-center gap-3">
   555→        <div className="flex min-w-0 items-center justify-end gap-2">
   556→          <span className="truncate text-right text-sm font-medium text-zinc-200">{match.home}</span>
   557→          {match.homeCrest ? (
   558→            <img src={match.homeCrest} alt="" aria-hidden loading="lazy" className="h-6 w-6 shrink-0 object-contain" />
   559→          ) : null}
   560→        </div>
   561→        <div className="w-14 text-center text-sm font-bold">
   562→          {finished ? (
   563→            <span className="text-emerald-400">
   564→              {match.score.home} – {match.score.away}
   565→            </span>
   566→          ) : (
   567→            <span className="text-zinc-500">vs</span>
   568→          )}
   569→        </div>
   570→        <div className="flex min-w-0 items-center gap-2">
   571→          {match.awayCrest ? (
   572→            <img src={match.awayCrest} alt="" aria-hidden loading="lazy" className="h-6 w-6 shrink-0 object-contain" />
   573→          ) : null}
   574→          <span className="truncate text-sm font-medium text-zinc-200">{match.away}</span>
   575→        </div>
   576→      </div>
   577→      <div className="flex items-center gap-3 sm:flex-col sm:items-end sm:gap-1">
   578→        <span className="text-xs text-zinc-500">
   579→          {dateLabel} · {timeLabel} UTC
   580→        </span>
   581→        <StatusBadge status={match.status} />
   582→      </div>
   583→    </div>
   584→  );
