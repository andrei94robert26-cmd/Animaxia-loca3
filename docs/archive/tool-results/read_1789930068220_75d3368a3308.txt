    74→    74→function b64urlEncode(obj) {
    75→    75→  return Buffer.from(JSON.stringify(obj), 'utf8')
    76→    76→    .toString('base64')
    77→    77→    .replace(/\+/g, '-')
    78→    78→    .replace(/\//g, '_')
    79→    79→    .replace(/=+$/, '');
    80→    80→}

    82→    82→function b64urlDecode(text) {
    83→    83→  try {
    84→    84→    const padded = String(text).replace(/-/g, '+').replace(/_/g, '/');
    85→    85→    return JSON.parse(Buffer.from(padded, 'base64').toString('utf8'));
    86→    86→  } catch {
    87→    87→    return null;
    88→    88→  }

    91→    91→/**
    92→    92→ * Build the next cursor from the LAST row of the current page, per sort.
    93→    93→ * Filter-only + no q → effective sort is contentid (see effectiveSort).
    94→    94→ */
    95→    95→function buildNextCursor({ sort, lastRow, pageNumber }) {
    96→    96→  if (!lastRow) return null;
    97→    97→  const id = Number(lastRow.id);
    98→    98→  if (!Number.isFinite(id)) return null;
    99→    99→  let payload = null;
   100→   100→  if (sort === 'year') payload = { y: lastRow._year ?? null, id };
   101→   101→  else if (sort === 'rating') payload = { r: lastRow._rating ?? null, id };
   102→   102→  else if (sort === 'title') payload = { t: lastRow._title ?? null, id };
   103→   103→  else if (sort === 'score') payload = { s: Number(lastRow._score ?? 0), id };
   104→   104→  else payload = { id }; // contentid keyset (filter-only)
   105→   105→  payload.p = pageNumber; // page estimate — incremented on each cursor hop
   106→   106→  return b64urlEncode(payload);
   107→   107→}

   109→   109→/**
   110→   110→ * Decode + validate the cursor payload for a given effective sort.
   111→   111→ * @returns {{ payload: object, where: string, params: any[] } | { error: string }}
   112→   112→ */
   113→   113→function cursorContinuation(effectiveSort, cursorText) {
   114→   114→  const payload = b64urlDecode(cursorText);
   115→   115→  if (!payload || typeof payload !== 'object' || !Number.isFinite(Number(payload.id))) {
   116→   116→    return { error: 'Invalid cursor' };
   117→   117→  }
   118→   118→  const id = Number(payload.id);
   119→   119→  const params = [];
   120→   120→  const P = (v) => {
   121→   121→    params.push(v);
   122→   122→    return `$${params.length}`;
   123→   123→  };
   124→   124→
   125→   125→  if (effectiveSort === 'score') {
   126→   126→    const s = Number(payload.s);
   127→   127→    if (!Number.isFinite(s)) return { error: 'Invalid cursor (missing s)' };
   128→   128→    const sp = P(s);
   129→   129→    const ip = P(id);
   130→   130→    return { payload, where: `(t.score < ${sp} OR (t.score = ${sp} AND t.contentid > ${ip}))`, params };
   131→   131→  }
   132→   132→  if (effectiveSort === 'year') {
   133→   133→    const y = Number(payload.y);
   134→   134→    if (!Number.isFinite(y)) return { error: 'Invalid cursor (missing y)' };
   135→   135→    const yp = P(y);
   136→   136→    const ip = P(id);
   137→   137→    return { payload, where: `(t.year < ${yp} OR (t.year = ${yp} AND t.contentid > ${ip}))`, params };
   138→   138→  }
   139→   139→  if (effectiveSort === 'rating') {
   140→   140→    const r = Number(payload.r);
   141→   141→    if (!Number.isFinite(r)) return { error: 'Invalid cursor (missing r)' };
   142→   142→    const rp = P(r);
   143→   143→    const ip = P(id);
   144→   144→    return { payload, where: `(t.rating < ${rp} OR (t.rating = ${rp} AND t.contentid > ${ip}))`, params };
   145→   145→  }
   146→   146→  if (effectiveSort === 'title') {
   147→   147→    const t = payload.t;
   148→   148→    if (typeof t !== 'string') return { error: 'Invalid cursor (missing t)' };
   149→   149→    const tp = P(t);
   150→   150→    const ip = P(id);
   151→   151→    return { payload, where: `(t.title > ${tp} OR (t.title = ${tp} AND t.contentid > ${ip}))`, params };
   152→   152→  }
   153→   153→  // contentid keyset (filter-only / no q)
   154→   154→  const ip = P(id);
   155→   155→  return { payload, where: `t.contentid > ${ip}`, params };
   156→   156→}

   161→   161→function toNameArray(value) {

   164→   164→  const out = [];
   165→   165→  for (const item of list) {
   166→   166→    if (typeof item === 'string') {
   167→   167→      const name = item.trim();

   170→   170→      const name = String(item.name ?? '').trim();
   171→   171→      if (name) out.push(name);
   172→   172→    }
   173→   173→  }

   199→   199→async function syncIndex() {
   200→   200→  const raw = await rows(
   201→   201→    'SELECT id, externalid, type, title, metadata, thumbnail, createdat FROM content',
   202→   202→  );
   203→   203→
   204→   204→  let synced = 0;
   205→   205→  for (let i = 0; i < raw.length; i += SYNC_CHUNK) {
   206→   206→    const slice = raw.slice(i, i + SYNC_CHUNK);
   207→   207→    const values = [];
   208→   208→    const tuples = slice.map((row, j) => {
   209→   209→      const meta = parseJsonb(row.metadata, {});
   210→   210→      const genres = toNameArray(meta.genres);
   211→   211→      const actors = toNameArray(meta.actors ?? meta.cast);
   212→   212→      const studios = toNameArray(meta.studios ?? meta.studio);
   213→   213→      const tsvText = buildTsvText(row.title, genres, meta);
   214→   214→      // FAZA 13 (sharding): searchdocs carries its shard id = the catalog
   215→   215→      // RANGE shard of the contentid it indexes (ids < 1B → 0, legacy-safe).
   216→   216→      const shardid = computeShardIdFromContentId(row.id);
   217→   217→      const base = j * 12;
   218→   218→      values.push(
   219→   219→        Number(row.id),
   220→   220→        row.externalid ?? null,
   221→   221→        row.type ?? null,
   222→   222→        row.title ?? 'untitled',
   223→   223→        tsvText,
   224→   224→        toIntOrNull(meta.year),
   225→   225→        toFloatOrNull(meta.rating ?? meta.score),
   226→   226→        genres,
   227→   227→        actors,
   228→   228→        studios,
   229→   229→        row.type ?? null,
   230→   230→        shardid,
   231→   231→      );
   232→   232→      const b = (k) => `$${base + k}`;
   233→   233→      return `(${b(1)},${b(2)},${b(3)},${b(4)},to_tsvector('simple', ${b(5)}),${b(6)},${b(7)},${b(8)}::text[],${b(9)}::text[],${b(10)}::text[],${b(11)},${b(12)},NOW())`;
   234→   234→    });
   235→   235→    await rows(
   236→   236→      `INSERT INTO searchdocs
   237→   237→         (contentid, externalid, type, title, tsv, year, rating, genres, actors, studios, category, shardid, syncedat)
   238→   238→       VALUES ${tuples.join(', ')}
   239→   239→       ON CONFLICT (contentid) DO UPDATE SET
   240→   240→         externalid = EXCLUDED.externalid,
   241→   241→         type = EXCLUDED.type,
   242→   242→         title = EXCLUDED.title,
   243→   243→         tsv = EXCLUDED.tsv,
   244→   244→         year = EXCLUDED.year,
   245→   245→         rating = EXCLUDED.rating,
   246→   246→         genres = EXCLUDED.genres,
   247→   247→         actors = EXCLUDED.actors,
   248→   248→         studios = EXCLUDED.studios,
   249→   249→         category = EXCLUDED.category,
   250→   250→         shardid = EXCLUDED.shardid,
   251→   251→         syncedat = NOW()`,
   252→   252→      values,
   253→   253→    );
   254→   254→    synced += slice.length;
   255→   255→  }
   256→   256→
   257→   257→  // Drop stale docs (content rows deleted since the last sync).
   258→   258→  try {
   259→   259→    await rows('DELETE FROM searchdocs WHERE contentid NOT IN (SELECT id FROM content)');
   260→   260→  } catch (err) {
   261→   261→    logger.warn(`[search] stale-doc cleanup skipped: ${err.message}`);
   262→   262→  }
   263→   263→
   264→   264→  lastSync = new Date().toISOString();
   265→   265→  dirty = false;
   266→   266→  return synced;
   267→   267→}

   269→   269→/** Rebuild once under an async lock; concurrent callers await the same run. */
   270→   270→async function ensureSynced() {
   271→   271→  if (!dirty && lastSync) return;
   272→   272→  if (!syncPromise) {
   273→   273→    syncPromise = syncIndex().finally(() => {
   274→   274→      syncPromise = null;
   275→   275→    });
   276→   276→  }
   277→   277→  await syncPromise;
   278→   278→}

   285→   285→  syncPromise = syncIndex().finally(() => {
   286→   286→    syncPromise = null;
   287→   287→  });
   288→   288→  return syncPromise;
   289→   289→}

   348→   348→/**
   349→   349→ * Run one advanced query against searchdocs (joined back to content).
   350→   350→ * Filter fragments are built once and shared by the count + data statements.
   351→   351→ * FAZA 17-a: `cursor` (base64url payload) switches to keyset pagination —
   352→   352→ * the ORDER BY becomes the cursor continuation WHERE and OFFSET is dropped;
   353→   353→ * `effectiveSort` is what actually orders the result (contentid keyset when
   354→   354→ * there is no text query and the caller asked for the default 'score' sort,
   355→   355→ * because score would be constant 0 for every row).
   356→   356→ */
   357→   357→async function runAdvancedQuery({
   358→   358→  q,
   359→   359→  type,
   360→   360→  genre,
   361→   361→  year,
   362→   362→  minRating,
   363→   363→  actor,
   364→   364→  studio,
   365→   365→  category,
   366→   366→  sort,
   367→   367→  fuzzyEnabled,
   368→   368→  limit,
   369→   369→  page,
   370→   370→  cursor = null,
   371→   371→}) {
   372→   372→  const params = [];
   373→   373→  const P = (v) => {
   374→   374→    params.push(v);
   375→   375→    return `$${params.length}`;
   376→   376→  };
   377→   377→
   378→   378→  // Effective sort: filter-only (no q) with default 'score' → contentid keyset.
   379→   379→  const effectiveSort = !q && sort === 'score' ? 'contentid' : sort;
   380→   380→
   381→   381→  // Keyset continuation (cursor present) — validated per effective sort.
   382→   382→  let cursorPayload = null;
   383→   383→  let cursorWhere = null;
   384→   384→  let cursorParams = [];
   385→   385→  if (cursor) {
   386→   386→    const decoded = cursorContinuation(effectiveSort, cursor);
   387→   387→    if (decoded.error) {
   388→   388→      const err = new Error(decoded.error);
   389→   389→      err.status = 400;
   390→   390→      throw err;
   391→   391→    }
   392→   392→    cursorPayload = decoded.payload;
   393→   393→    cursorWhere = decoded.where;
   394→   394→    cursorParams = decoded.params;
   395→   395→  }
   396→   396→
   397→   397→  // Field filters (SQL-side, case-insensitive where array containment).
   398→   398→  const filters = [];
   399→   399→  if (type) filters.push(`s.type = ${P(type)}`);
   400→   400→  if (genre) filters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${P(genre)})`);
   401→   401→  if (year != null) filters.push(`s.year = ${P(year)}`);
   402→   402→  if (actor) {
   403→   403→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${P(actor)} || '%')`);
   404→   404→  }
   405→   405→  if (studio) {
   406→   406→    filters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${P(studio)} || '%')`);
   407→   407→  }
   408→   408→  if (category) filters.push(`s.category = ${P(category)}`);
   409→   409→  if (minRating != null) filters.push(`s.rating >= ${P(minRating)}`);
   410→   410→  const filterSql = filters.length ? ` AND ${filters.join(' AND ')}` : '';
   411→   411→
   412→   412→  // Text score expression (combined: ts_rank * 2 + similarity + prefix bonus).
   413→   413→  let scoreExpr = '0';
   414→   414→  let textMatch = '';
   415→   415→  if (q) {
   416→   416→    const qp = P(q);
   417→   417→    scoreExpr =
   418→   418→      `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qp})) * 2` +
   419→   419→      ` + similarity(s.title, ${qp})` +
   420→   420→      ` + CASE WHEN s.title ILIKE ${qp} || '%' THEN 0.5 ELSE 0 END`;
   421→   421→    textMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qp})`;
   422→   422→  }
   423→   423→
   424→   424→  const whereFor = (matchSql) => `WHERE (${matchSql || 'TRUE'})${filterSql}`;
   425→   425→
   426→   426→  // 1. FTS pass (or filter-only when q is empty).
   427→   427→  let matchSql = textMatch;
   428→   428→  let countRow = await one(
   429→   429→    `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   430→   430→    params,
   431→   431→  );
   432→   432→
   433→   433→  // 2. Trigram fuzzy fallback when the FTS pass matched nothing.
   434→   434→  //    `title % q` (pg_trgm, default 0.3 threshold) plus a looser
   435→   435→  //    similarity() > 0.1 guard so short typo'd queries still reach long
   436→   436→  //    titles (e.g. 'dunee' → 'Dune: Part Two - Wikipedia', sim ≈ 0.15).
   437→   437→  if (q && fuzzyEnabled && Number(countRow?.total ?? 0) === 0) {
   438→   438→    const words = q.split(/\s+/).filter(Boolean);
   439→   439→    const trigramClauses = [];
   440→   440→    for (const candidate of [q, ...words]) {
   441→   441→      const ph = P(`%${candidate}%`);
   442→   442→      trigramClauses.push(`s.title ILIKE ${ph}`);
   443→   443→    }
   444→   444→    const qSim = P(q);
   445→   445→    trigramClauses.push(`(s.title % ${qSim} OR similarity(s.title, ${qSim}) > 0.1)`);
   446→   446→    matchSql = `(${trigramClauses.join(' OR ')})`;
   447→   447→    countRow = await one(
   448→   448→      `SELECT COUNT(*)::int AS total, COALESCE(MAX(${scoreExpr}), 0) AS maxscore FROM searchdocs s ${whereFor(matchSql)}`,
   449→   449→      params,
   450→   450→    );
   451→   451→  }
   452→   452→
   453→   453→  const total = Number(countRow?.total ?? 0);
   454→   454→  const maxScore = Number(Number(countRow?.maxscore ?? 0).toFixed(6));
   455→   455→
   456→   456→  // 3. Data page — subquery keeps the score expression, then joins content.
   457→   457→  //    FAZA 17-a: expose year/rating/title to JS for nextCursor construction;
   458→   458→  //    cursor mode replaces OFFSET with the keyset continuation predicate.
   459→   459→  const orderExpr =
   460→   460→    effectiveSort === 'year'
   461→   461→      ? 't.year DESC NULLS LAST, t.contentid ASC'
   462→   462→      : effectiveSort === 'rating'
   463→   463→        ? 't.rating DESC NULLS LAST, t.score DESC, t.contentid ASC'
   464→   464→        : effectiveSort === 'title'
   465→   465→          ? 't.title ASC, t.contentid ASC'
   466→   466→          : effectiveSort === 'contentid'
   467→   467→            ? 't.contentid ASC'
   468→   468→            : 't.score DESC, t.contentid ASC';
   469→   469→
   470→   470→  const limitPh = P(limit);
   471→   471→  const offsetPh = P((page - 1) * limit);
   472→   472→  // Cursor mode: continuation params come FIRST ($1..$n), then LIMIT; the
   473→   473→  // filter/score params are re-bound only inside the subquery — so build a
   474→   474→  // dedicated statement for each mode instead of renumbering dynamically.
   475→   475→  let dataSql;
   476→   476→  let dataParams;
   477→   477→  if (cursorWhere) {
   478→   478→    // Rebuild the subquery with a fresh parameter namespace for the cursor run
   479→   479→    // (filters + score expression are re-parameterized after the cursor slots).
   480→   480→    const subParams = [...cursorParams, limit];
   481→   481→    const sub = (v) => {
   482→   482→      subParams.push(v);
   483→   483→      return `$${subParams.length}`;
   484→   484→    };
   485→   485→    const cFilters = [];
   486→   486→    if (type) cFilters.push(`s.type = ${sub(type)}`);
   487→   487→    if (genre) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.genres) g WHERE g ILIKE ${sub(genre)})`);
   488→   488→    if (year != null) cFilters.push(`s.year = ${sub(year)}`);
   489→   489→    if (actor) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.actors) a WHERE a ILIKE '%' || ${sub(actor)} || '%')`);
   490→   490→    if (studio) cFilters.push(`EXISTS (SELECT 1 FROM unnest(s.studios) st WHERE st ILIKE '%' || ${sub(studio)} || '%')`);
   491→   491→    if (category) cFilters.push(`s.category = ${sub(category)}`);
   492→   492→    if (minRating != null) cFilters.push(`s.rating >= ${sub(minRating)}`);
   493→   493→    const cFilterSql = cFilters.length ? ` AND ${cFilters.join(' AND ')}` : '';
   494→   494→
   495→   495→    let cScoreExpr = '0';
   496→   496→    let cMatch = 'TRUE';
   497→   497→    // Single q slot, bound LAST (after cursor params + LIMIT) so every
   498→   498→    // placeholder references a parameter that exists in the bind array.
   499→   499→    let qParam = null;
   500→   500→    if (q) {
   501→   501→      qParam = sub(String(q));
   502→   502→      cScoreExpr =
   503→   503→        `ts_rank(s.tsv, websearch_to_tsquery('simple', ${qParam})) * 2` +
   504→   504→        ` + similarity(s.title, ${qParam})` +
   505→   505→        ` + CASE WHEN s.title ILIKE ${qParam} || '%' THEN 0.5 ELSE 0 END`;
   506→   506→      cMatch = `s.tsv @@ websearch_to_tsquery('simple', ${qParam})`;
   507→   507→    }
   508→   508→    // Fuzzy fallback consistency: when the FTS pass matched nothing the offset
   509→   509→    // run switched matchSql — mirror that for the cursor run using `total`.
   510→   510→    if (q && fuzzyEnabled && total === 0) {
   511→   511→      const words = q.split(/\s+/).filter(Boolean);
   512→   512→      const clauses = [q, ...words].map((candidate) => `s.title ILIKE ${sub(`%${candidate}%`)}`);
   513→   513→      clauses.push(`(s.title % ${qParam} OR similarity(s.title, ${qParam}) > 0.1)`);
   514→   514→      cMatch = `(${clauses.join(' OR ')})`;
   515→   515→    }
   516→   516→
   517→   517→    dataSql =
   518→   518→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   519→   519→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   520→   520→       FROM (
   521→   521→         SELECT s.contentid, s.year, s.rating, s.title, ${cScoreExpr} AS score
   522→   522→         FROM searchdocs s
   523→   523→         WHERE (${cMatch})${cFilterSql}
   524→   524→       ) t
   525→   525→       JOIN content c ON c.id = t.contentid
   526→   526→       WHERE ${cursorWhere}
   527→   527→       ORDER BY ${orderExpr}
   528→   528→       LIMIT $${cursorParams.length + 1}::int`;
   529→   529→    // LIMIT is the placeholder right after the cursor params (::int — a bare
   530→   530→    // parameter in LIMIT cannot be type-inferred by Postgres); the q/filter
   531→   531→    // slots follow it so every bound parameter is referenced by the statement.
   532→   532→    dataParams = subParams;
   533→   533→  } else {
   534→   534→    dataSql =
   535→   535→      `SELECT c.id, c.externalid, c.type, c.title, c.metadata, c.thumbnail, c.createdat,
   536→   536→              t.score AS _score, t.year AS _year, t.rating AS _rating, t.title AS _title
   537→   537→       FROM (
   538→   538→         SELECT s.contentid, s.year, s.rating, s.title, ${scoreExpr} AS score
   539→   539→         FROM searchdocs s
   540→   540→         ${whereFor(matchSql)}
   541→   541→       ) t
   542→   542→       JOIN content c ON c.id = t.contentid
   543→   543→       ORDER BY ${orderExpr}
   544→   544→       LIMIT ${limitPh} OFFSET ${offsetPh}`;
   545→   545→    dataParams = params;
   546→   546→  }
   547→   547→  const dataRows = await rows(dataSql, dataParams);
   548→   548→
   549→   549→  // Keyset paging estimates the page number by incrementing the cursor's `p`.
   550→   550→  const resultPage = cursor ? (Number(cursorPayload?.p) || 1) + 1 : page;
   551→   551→  const normalized = dataRows.map(normalizeHit).filter(Boolean);
   552→   552→  const lastRaw = dataRows.length ? dataRows[dataRows.length - 1] : null;
   553→   553→
   554→   554→  return {
   555→   555→    total,
   556→   556→    max_score: maxScore,
   557→   557→    data: normalized,
   558→   558→    page: resultPage,
   559→   559→    nextCursor: buildNextCursor({ sort: effectiveSort, lastRow: lastRaw, pageNumber: resultPage }),
   560→   560→    paginationMode: cursor ? 'cursor' : 'offset',
   561→   561→  };
   562→   562→}

   572→   572→function coalesceAdvanced(normalizedKey, run) {
   573→   573→  const existing = inflightSearch.get(normalizedKey);
   574→   574→  if (existing) {
   575→   575→    searchOps.coalesced += 1;
   576→   576→    return existing;
   577→   577→  }

   579→   579→    inflightSearch.delete(normalizedKey);
   580→   580→  });
   581→   581→  inflightSearch.set(normalizedKey, promise);
   582→   582→  return promise;
   583→   583→}

   585→   585→/** Race a query promise against the 8s budget (loser is logged, not awaited). */
   586→   586→function withQueryTimeout(promise, describe) {
   587→   587→  let timer = null;
   588→   588→  const budget = new Promise((_, reject) => {
   589→   589→    timer = setTimeout(() => {
   590→   590→      const err = new Error('Search query timeout');
   591→   591→      err.status = 504;
   592→   592→      reject(err);
   593→   593→    }, QUERY_TIMEOUT_MS);
   594→   594→    if (typeof timer.unref === 'function') timer.unref();
   595→   595→  });
   596→   596→  return promise.then(
   597→   597→    (value) => {
   598→   598→      clearTimeout(timer);
   599→   599→      return value;
   600→   600→    },
   601→   601→    (err) => {
   602→   602→      clearTimeout(timer);
   603→   603→      throw err;
   604→   604→    },
   605→   605→  ).finally(() => {
   606→   606→    clearTimeout(timer);
   607→   607→    promise.catch((loser) => {
   608→   608→      // The lost race (if it also failed) must not become an unhandledRejection.
   609→   609→      logger.warn(`[search] query lost the timeout race (${describe}): ${loser?.message ?? loser}`);
   610→   610→    });
   611→   611→  });
   612→   612→}
