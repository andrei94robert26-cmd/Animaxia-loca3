     3→/**
     4→ * player-view.tsx — Stream player for library titles. Renders mp4 <video>,
     5→ * HLS (m3u8) via dynamic hls.js import, YouTube embeds and sandboxed iframes,
     6→ * with a source switcher, playback-policy line and an inline stream miner
     7→ * for titles without sources — plus the Phase-4 personal "AI companion" card
     8→ * for signed-in users (before-you-watch, predicted rating, more like this,
     9→ * next up).
    10→ *
    11→ * FAZA 17 — hardening: a real fallback chain over `sources`. mp4/HLS errors
    12→ * and a 12s load watchdog advance `sourceIndex` to the next source (with a
    13→ * toast per hop); when every source has failed, a friendly end-of-chain card
    14→ * replaces the player instead of crashing. hls.js instances are destroyed,
    15→ * the <video> decoder/buffer is released on every source change/unmount and
    16→ * all timers are cleared.
    17→ */

    48→function isNotFound(message?: string | null): boolean {
    49→  return Boolean(message && message.toLowerCase().includes('not found'));

    52→/** FAZA 21 — mm:ss clock for the resume toast. */
    53→function formatClock(sec: number): string {
    54→  const m = Math.floor(sec / 60);
    55→  const s = Math.floor(sec % 60);
    56→  return `${m}:${String(s).padStart(2, '0')}`;
    57→}

    84→export function PlayerView({ route }: { route: AppRoute }) {
    85→  const { user } = useAuth();
    86→  const id = Number(route.params[0]);
    87→  const validId = Number.isInteger(id) && id > 0;
    88→  const player = useApi(`player-${id}`, () => api.player(id), { enabled: validId });
    89→
    90→  /* FAZA 17 — source fallback chain. `sourceIndex` walks data.sources on any
    91→     load failure (mp4/HLS error, 12s load timeout). `failedState` marks the
    92→     friendly end-of-chain card and is scoped to the content id so it can
    93→     never leak across titles. `reloadNonce` forces a fresh <video> mount when
    94→     retrying the same source. */
    95→  const [sourceIndex, setSourceIndex] = useState(0);
    96→  const [failedState, setFailedState] = useState<{ contentId: number; reason: string } | null>(null);
    97→  const [reloadNonce, setReloadNonce] = useState(0);
    98→  const [mineUrl, setMineUrl] = useState('');
    99→  const [mining, setMining] = useState(false);
   100→
   101→  const videoRef = useRef<HTMLVideoElement>(null);
   102→  const sourcesRef = useRef<PlayerSource[]>([]);
   103→  const sourceIndexRef = useRef(0);
   104→  const loadTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   105→  const lastAdvanceRef = useRef<{ key: string; at: number } | null>(null);
   106→
   107→  /* Only trust data matching the requested id — useApi keeps the previous
   108→     record alive while the key changes (stale data across titles). */
   109→  const current = player.data && player.data.content.id === id ? player.data : null;
   110→  const sources = current?.sources ?? [];
   111→  const exhausted = failedState != null && failedState.contentId === id && sources.length > 0;
   112→  const activeSource: PlayerSource | null = exhausted
   113→    ? null
   114→    : sources[sourceIndex] ?? current?.primary ?? null;
   115→
   116→  /** Clears the 12s load watchdog — any progress event calls this. */
   117→  const clearLoadTimer = useCallback(() => {
   118→    if (loadTimerRef.current) {
   119→      clearTimeout(loadTimerRef.current);
   120→      loadTimerRef.current = null;
   121→    }
   122→  }, []);
   123→
   124→  /** Advance to the next source on failure; when none remain, switch to the
   125→      friendly end-of-chain card instead of crashing. Duplicate signals for the
   126→      same attempt (e.g. hls.js error + element error) are coalesced. */
   127→  const advanceSource = useCallback(
   128→    (reason: string) => {
   129→      clearLoadTimer();
   130→      const list = sourcesRef.current;
   131→      const i = sourceIndexRef.current;
   132→      const key = `${id}:${i}:${reason}`;
   133→      const now = Date.now();
   134→      if (lastAdvanceRef.current && lastAdvanceRef.current.key === key && now - lastAdvanceRef.current.at < 1500) {
   135→        return;
   136→      }
   137→      lastAdvanceRef.current = { key, at: now };
   138→      if (i + 1 < list.length) {
   139→        sourceIndexRef.current = i + 1;
   140→        setSourceIndex(i + 1);
   141→        toast.info(`Sursa ${i + 1} nu răspunde (${reason}) — trec la următoarea…`);
   142→      } else {
   143→        setFailedState({ contentId: id, reason });
   144→      }
   145→    },
   146→    [id, clearLoadTimer],
   147→  );
   148→
   149→  /** Manual source chip click — clears the chain failure; clicking the chip of
   150→      the current source forces a clean reload of that source. */
   151→  const selectSource = (i: number) => {
   152→    if (i < 0 || i >= sources.length) return;
   153→    setFailedState(null);
   154→    if (i === sourceIndex) {
   155→      setReloadNonce((n) => n + 1);
   156→    } else {
   157→      sourceIndexRef.current = i;
   158→      setSourceIndex(i);
   159→    }
   160→  };
   161→
   162→  /** "Reîncearcă de la prima sursă" — reset the chain and remount the video. */
   163→  const retryFromFirst = () => {
   164→    setFailedState(null);
   165→    sourceIndexRef.current = 0;
   166→    setSourceIndex(0);
   167→    setReloadNonce((n) => n + 1);
   168→  };
   169→
   170→  /* New title (route change) → reset the fallback chain before the new data
   171→     ever renders, so no stale index/failure can flash. */
   172→  useEffect(() => {
   173→    sourceIndexRef.current = 0;
   174→    setSourceIndex(0);
   175→    setFailedState(null);
   176→  }, [id]);
   177→
   178→  /* Fresh sources (first load or refetch after mining) → start at the primary
   179→     source and force a clean reload of the player surface. */
   180→  useEffect(() => {
   181→    const d = player.data;
   182→    if (!d || d.content.id !== id) return;
   183→    sourcesRef.current = d.sources;
   184→    const primary = d.primary;
   185→    const primaryIndex = primary
   186→      ? d.sources.findIndex((s) => s.url === primary.url && s.type === primary.type)
   187→      : -1;
   188→    const start = primaryIndex >= 0 ? primaryIndex : 0;
   189→    sourceIndexRef.current = start;
   190→    setSourceIndex(start);
   191→    setFailedState(null);
   192→    setReloadNonce((n) => n + 1);
   193→  }, [player.data, id]);
   194→
   195→  /* ------------------------- analytics tracking ------------------------- */
   196→
   197→  const trackedViewRef = useRef<number | null>(null);
   198→  const trackedPlayRef = useRef<number | null>(null);
   199→
   200→  /* Fire-and-forget 'view' event, once per content id per mount. */
   201→  useEffect(() => {
   202→    if (!player.data || player.data.content.id !== id) return;
   203→    if (trackedViewRef.current === id) return;
   204→    trackedViewRef.current = id;
   205→    platform.recordView(id, 'view').catch(() => {});
   206→  }, [player.data, id]);
   207→
   208→  /* 'play' event for embed renderers (youtube/iframe), once per content id. */
   209→  useEffect(() => {
   210→    if (!player.data || player.data.content.id !== id) return;
   211→    const src = activeSource;
   212→    if (!src || (src.type !== 'youtube' && src.type !== 'iframe')) return;
   213→    if (trackedPlayRef.current === id) return;
   214→    trackedPlayRef.current = id;
   215→    platform.recordView(id, 'play').catch(() => {});
   216→  }, [player.data, activeSource, id]);
   217→
   218→  const trackPlay = () => {
   219→    if (trackedPlayRef.current === id) return;
   220→    trackedPlayRef.current = id;
   221→    platform.recordView(id, 'play').catch(() => {});
   222→  };
   223→
   224→  const trackComplete = () => {
   225→    platform.recordView(id, 'complete').catch(() => {});
   226→  };
   227→
   228→  /* ------------------- FAZA 21: history / resume / subtitles ------------------- */
   229→
   230→  /** Resume position from ?t= (seconds) — validated once per route param. */
   231→  const tParam = route.query.get('t');
   232→  const resumeT = (() => {
   233→    const t = Number(tParam);
   234→    return Number.isFinite(t) && t > 0 && t < 86_400 ? Math.floor(t) : 0;
   235→  })();
   236→
   237→  /** recordHistory — POST /api/users/me/history {contentId, progress} via the
   238→   *  existing platform client. Fire-and-forget, silent failures: never breaks
   239→   *  playback. Only native renderers expose video.currentTime (YouTube/iframe
   240→   *  embeds are honestly skipped). */
   241→  const recordTick = useCallback(() => {
   242→    if (!user || !validId) return;
   243→    const el = videoRef.current;
   244→    if (!el) return;
   245→    const t = Math.floor(el.currentTime);
   246→    if (!Number.isFinite(t) || t <= 0) return;
   247→    platform.recordHistory(id, t).catch(() => {});
   248→  }, [user, id, validId]);
   249→
   250→  /* Progress sync: every 10s while playing + a final flush on pause handled
   251→     by the onPause prop; the cleanup below flushes on unmount AND on source
   252→     change (deps include the active source), recording the last position of
   253→     the outgoing element before the decoder-release effect resets it. */
   254→  useEffect(() => {
   255→    if (!user || !validId) return undefined;
   256→    recordTick(); /* mount-time flush (e.g. after a resume seek) */
   257→    const timer = setInterval(() => {
   258→      const el = videoRef.current;
   259→      if (el && !el.paused && !el.ended) recordTick();
   260→    }, 10_000);
   261→    return () => {
   262→      clearInterval(timer);
   263→      recordTick(); /* best-effort final flush */
   264→    };
   265→  }, [user, validId, id, recordTick, activeSource?.type, activeSource?.url, sourceIndex, reloadNonce]);
   266→
   267→  /** Resume seek — applied once per (content id + source + reload nonce) on
   268→   *  the first loadedmetadata of the current <video>. The applied key is
   269→   *  stamped on the DOM element itself (dataset), so a remount that somehow
   270→   *  preserves React state (e.g. dev Fast Refresh) still re-arms the seek,
   271→   *  while the same playing element never seeks twice. */
   272→  const maybeResume = useCallback(() => {
   273→    if (resumeT <= 0) return;
   274→    const el = videoRef.current;
   275→    if (!el) return;
   276→    const key = `${id}:${sourceIndex}:${activeSource?.url ?? ''}:${reloadNonce}`;
   277→    if (el.dataset.faza21Resume === key) return;
   278→    try {
   279→      el.currentTime = resumeT;
   280→      el.dataset.faza21Resume = key;
   281→      toast.info(`Reia de la ${formatClock(resumeT)}`);
   282→    } catch {
   283→      /* seek not possible yet — allow a later loadedmetadata to retry */
   284→    }
   285→  }, [resumeT, id, sourceIndex, activeSource?.url, reloadNonce]);
   286→
   287→  /** WebVTT subtitles — public listing, fetched once per content id (silent
   288→   *  fail: the player works fine without tracks). */
   289→  const subs = useApi(`player-subs-${id}`, () => api.subtitles.forContent(id), { enabled: validId });
   290→  const subtitleTracks = subs.data && subs.data.contentId === id ? subs.data.data : [];
   291→
   292→  /** CC toggle: off → first subtitle track showing → off. Tracks default to
   293→   *  disabled (no `default` attr) so nothing shows until the user opts in. */
   294→  const [subsOn, setSubsOn] = useState(false);
   295→  useEffect(() => {
   296→    setSubsOn(false); /* the element remounts per source — tracks reset too */
   297→  }, [id, sourceIndex, reloadNonce]);
   298→  const toggleSubs = useCallback(() => {
   299→    const el = videoRef.current;
   300→    if (!el) return;
   301→    const tracks = Array.from(el.textTracks ?? []);
   302→    const anyShowing = tracks.some((t) => t.mode === 'showing');
   303→    tracks.forEach((t) => {
   304→      t.mode = 'disabled';
   305→    });
   306→    if (anyShowing) {
   307→      setSubsOn(false);
   308→      return;
   309→    }
   310→    const first = tracks.find((t) => t.kind === 'subtitles');
   311→    if (first) {
   312→      first.mode = 'showing';
   313→      setSubsOn(true);
   314→    }
   315→  }, []);
   316→
   317→  useEffect(() => {
   318→    const url = activeSource?.url;
   319→    const el = videoRef.current;
   320→    if (activeSource?.type !== 'm3u8' || !url || !el) return;
   321→    let hls: { destroy: () => void } | null = null;
   322→    let cancelled = false;
   323→    (async () => {
   324→      try {
   325→        const mod = await import('hls.js');
   326→        const Hls = mod.default;
   327→        if (cancelled) return;
   328→        if (Hls.isSupported()) {
   329→          const instance = new Hls({ enableWorker: true });
   330→          /* FAZA 17 — fatal HLS errors destroy the instance and advance the
   331→             source chain; non-fatal ones get a light startLoad() retry. */
   332→          instance.on(Hls.Events.ERROR, (_event, data) => {
   333→            if (cancelled) return;
   334→            if (data?.fatal) {
   335→              try {
   336→                instance.destroy();
   337→              } catch {
   338→                /* already destroyed */
   339→              }
   340→              advanceSource('eroare HLS');
   341→            } else {
   342→              try {
   343→                instance.startLoad();
   344→              } catch {
   345→                /* retry not possible — the load watchdog will advance */
   346→              }
   347→            }
   348→          });
   349→          instance.loadSource(url);
   350→          instance.attachMedia(el);
   351→          hls = instance;
   352→        } else if (el.canPlayType('application/vnd.apple.mpegurl')) {
   353→          el.src = url;
   354→        }
   355→      } catch {
   356→        /* hls.js failed to init — hand over to the fallback chain. */
   357→        if (!cancelled) advanceSource('eroare HLS');
   358→      }
   359→    })();
   360→    return () => {
   361→      cancelled = true;
   362→      hls?.destroy();
   363→    };
   364→  }, [activeSource?.type, activeSource?.url, sourceIndex, reloadNonce, advanceSource]);
   365→
   366→  /* FAZA 17 — 12s load watchdog: if a video source produces no loadeddata /
   367→     canplay within the window, advance to the next source. */
   368→  useEffect(() => {
   369→    const t = activeSource?.type;
   370→    if (!activeSource?.url || exhausted) return undefined;
   371→    if (t !== 'mp4' && t !== 'm3u8') return undefined;
   372→    clearLoadTimer();
   373→    loadTimerRef.current = setTimeout(() => {
   374→      loadTimerRef.current = null;
   375→      advanceSource('timeout sursă');
   376→    }, 12_000);
   377→    return clearLoadTimer;
   378→  }, [activeSource?.type, activeSource?.url, sourceIndex, reloadNonce, exhausted, advanceSource, clearLoadTimer]);
   379→
   380→  /* FAZA 17 — release the decoder/buffer whenever the current video element
   381→     goes away (source change, retry or unmount): pause, drop the src and run
   382→     load() so the browser frees the media pipeline. */
   383→  useEffect(() => {
   384→    const el = videoRef.current;
   385→    if (!el) return undefined;
   386→    return () => {
   387→      clearLoadTimer();
   388→      try {
   389→        el.pause();
   390→        el.removeAttribute('src');
   391→        el.load();
   392→      } catch {
   393→        /* element already detached */
   394→      }
   395→    };
   396→  }, [activeSource?.type, activeSource?.url, sourceIndex, reloadNonce, clearLoadTimer]);
   397→
   398→  async function handleMine(e: React.FormEvent) {
   399→    e.preventDefault();
   400→    if (!mineUrl.trim() || !player.data || mining) return;
   401→    setMining(true);
   402→    try {
   403→      await api.miner({
   404→        url: mineUrl.trim(),
   405→        externalId: player.data.content.externalid,
   406→        type: player.data.content.type,
   407→      });
   408→      toast.success('Stream attached');
   409→      setMineUrl('');
   410→      await player.refetch();
   411→    } catch (err) {
   412→      toast.error(err instanceof Error ? err.message : 'Mining failed');
   413→    } finally {
   414→      setMining(false);
   415→    }
   416→  }
   417→
   418→  if (!validId) {
   419→    return (
   420→      <Container className="py-20">
   421→        <EmptyState
   422→          title="Title not found"
   423→          message="The player URL is missing a valid title id."
   424→          action={
   425→            <Button className="bg-rose-600 font-bold hover:bg-rose-500" onClick={() => navigate('/home')}>
   426→              Back to home
   427→            </Button>
   428→          }
   429→        />
   430→      </Container>
   431→    );
   432→  }
   433→
   434→  if (player.loading && !current) {
   435→    return (
   436→      <Container className="py-8 sm:py-12">
   437→        <div className="mx-auto w-full max-w-5xl space-y-5">
   438→          <div className="flex items-center justify-between">
   439→            <Skeleton className="h-9 w-36" />
   440→            <Skeleton className="h-7 w-28 rounded-full" />
   441→          </div>
   442→          <Skeleton className="aspect-video w-full rounded-2xl" />
   443→          <Skeleton className="h-24 w-full rounded-xl" />
   444→        </div>
   445→      </Container>
   446→    );
   447→  }
   448→
   449→  if (player.error && !current) {
   450→    if (isNotFound(player.error)) {
   451→      return (
   452→        <Container className="py-20">
   453→          <EmptyState
   454→            title="Title not found"
   455→            message={player.error}
   456→            action={
   457→              <Button className="bg-rose-600 font-bold hover:bg-rose-500" onClick={() => navigate('/home')}>
   458→                Back to home
   459→              </Button>
   460→            }
   461→          />
   462→        </Container>
   463→      );
   464→    }
   465→    return (
   466→      <Container className="py-20">
   467→        <ErrorState message={player.error} onRetry={player.refetch} />
   468→      </Container>
   469→    );
   470→  }
   471→
   472→  const data = current;
   473→  if (!data) return null;
   474→
   475→  const content = data.content;
   476→  const md = content.metadata || {};
   477→  const overview =
   478→    [md.overview, md.synopsis, md.description].find((v) => typeof v === 'string' && v.trim() !== '') ||
   479→    'No synopsis available for this title.';
   480→  const thumb = content.thumbnail || (typeof md.backdrop === 'string' ? md.backdrop : null);
   481→  const noSource = data.mode === 'none' || !data.primary;
   482→
   483→  /* FAZA 17 — "open the stream externally" suggestion: the original page
   484→     URL from metadata when available, else the last source in the chain. */
   485→  const externalHref =
   486→    (typeof md.url === 'string' && /^https?:\/\//.test(md.url) ? md.url : null) ??
   487→    (typeof md.mined?.url === 'string' && /^https?:\/\//.test(md.mined.url) ? md.mined.url : null) ??
   488→    (sources.length > 0 ? sources[sources.length - 1].url : null);
   489→
   490→  return (
   491→    <Container className="py-6 sm:py-8">
   492→      <div className="mx-auto w-full max-w-5xl space-y-5">
   493→        {/* Top bar */}
   494→        <div className="flex flex-wrap items-center justify-between gap-3">
   495→          <Button
   496→            variant="outline"
   497→            className="border-white/15 bg-white/5 font-semibold hover:bg-white/10"
   498→            onClick={() => navigate(`/content/${id}`)}
   499→          >
   500→            <ArrowLeft aria-hidden /> Back to details
   501→          </Button>
   502→          <PlayerModeBadge mode={data.mode} />
   503→        </div>
   504→
   505→        {/* Player surface */}
   506→        {noSource ? (
   507→          <div className="space-y-4">
   508→            <EmptyState
   509→              title="No stream sources available"
   510→              message={
   511→                data.note ||
   512→                'This title has no playable sources yet. Mine a stream below and it will be attached instantly.'
   513→              }
   514→              icon={<MonitorPlay className="h-6 w-6" aria-hidden />}
   515→            />
   516→            <Card className="border-white/10 bg-white/[0.03]">
   517→              <CardHeader>
   518→                <CardTitle className="flex items-center gap-2 text-base">
   519→                  <Pickaxe className="h-4 w-4 text-rose-400" aria-hidden />
   520→                  Mine a stream
   521→                </CardTitle>
   522→                <CardDescription>
   523→                  Point the miner at a page or direct media URL — it extracts mp4, m3u8, iframe and YouTube sources
   524→                  and attaches them to this title.
   525→                </CardDescription>
   526→              </CardHeader>
   527→              <CardContent>
   528→                <form className="flex flex-col gap-3 sm:flex-row" onSubmit={handleMine}>
