   442→async function handleMine(req, res) {
   443→  const url = req.method === 'GET' ? req.query.url : req.body?.url;
   444→  if (!url || typeof url !== 'string') {
   445→    return res.status(400).json({ error: 'Missing url' });
   446→  }
   447→  const options =
   448→    req.method === 'GET'
   449→      ? { type: req.query.type, externalId: req.query.externalId }
   450→      : { type: req.body?.type, externalId: req.body?.externalId };
   451→  try {
   452→    const { extraction, record } = await mineUrl(url, options);
   453→    invalidateHttpCache();
   454→    // TS1: targeted id event for the incremental searchdocs sync (upsert path).
   455→    if (record?.id) events.emit('content:changed:ids', { ids: [record.id], action: 'update' });
   456→    await auditLog({ userid: req.user?.id ?? null, action: 'miner.run', ip: clientIpOf(req), meta: { url, id: record?.id } });
   457→    return res.json({ ok: true, extraction, record });
   458→  } catch (err) {
   459→    return res.status(err.status || 502).json({ error: err.message || 'Mining failed' });
   460→  }
   461→}
