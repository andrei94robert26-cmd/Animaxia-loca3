    94→function formatUptime(sec: number): string {
    95→  const h = Math.floor(sec / 3600);
    96→  const m = Math.floor((sec % 3600) / 60);
    97→  const s = Math.floor(sec % 60);
    98→  if (h > 0) return `${h}h ${m}m`;
    99→  if (m > 0) return `${m}m ${s}s`;
   100→  return `${s}s`;
   101→}
   102→
   103→function formatBytes(b?: number): string {
   104→  if (!b || b <= 0) return '0 B';
   105→  const units = ['B', 'KB', 'MB', 'GB'];
   106→  let v = b;
   107→  let i = 0;
   108→  while (v >= 1024 && i < units.length - 1) {
   109→    v /= 1024;
   110→    i += 1;
   111→  }
   112→  return `${v >= 100 ? Math.round(v) : v.toFixed(1)} ${units[i]}`;
   113→}
   114→
   115→function formatMs(ms?: number): string {
   116→  if (ms == null) return '—';
   117→  if (ms < 1000) return `${Math.round(ms)}ms`;
   118→  return `${(ms / 1000).toFixed(1)}s`;
   119→}
   120→
   121→function formatDate(iso?: string | null): string {
   122→  if (!iso) return '—';
   123→  const d = new Date(iso);
   124→  if (Number.isNaN(d.getTime())) return '—';
   125→  return d.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
   126→}

   128→function getErrorMessage(e: unknown): string {
   129→  return e instanceof Error ? e.message : 'Request failed';
   130→}

   134→  return Boolean(message && /admin access required|forbidden|status 40[13]/i.test(message));
   135→}
