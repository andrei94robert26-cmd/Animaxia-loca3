   134→async function cloudHandler({ url, method, headers, bodyText }) {
   135→  const path = url.pathname;
   136→  cloud.calls.push({ path: `${path}${url.search}`, method, shard: headers.get(SHARD_HEADER) }); // FAZA 13: shard header seen upstream

   138→  if (cloud.failPaths.has(path)) {
   139→    return json({ error: 'forced_failure' }, 503);

   142→  if (path === '/api/all') return json({ data: CATALOG, totalPages: 1, total: CATALOG.length });
   143→  if (path === '/api/search/advanced') return json({ took: 1, total: CATALOG.length, max_score: 5, data: CATALOG.slice(0, 5) });
   144→  if (path === '/api/search/autocomplete') {
   145→    return json({ suggestions: CATALOG.slice(0, 5).map(({ id, title, type, thumbnail }) => ({ id, title, type, thumbnail })) });
   146→  }
   147→  if (path === '/api/auth/login' || path === '/api/auth/register') {
   148→    return json({ ok: true, user: { id: 42, email: 'smoke@animaxia.io', plan: 'free' }, accessToken: 'stub-access', refreshToken: 'stub-refresh-1', expiresIn: 900 });
   149→  }

   151→    const body = JSON.parse((await bodyText()) || '{}');
   152→    return json({ ok: true, accessToken: `rotated-${body.refreshToken}`, refreshToken: 'new-refresh', expiresIn: 900 });
   153→  }
   154→  if (path === '/api/auth/logout') return json({ ok: true, revoked: true });
   155→  if (path === '/api/auth/me') return json({ user: { id: 42, email: 'smoke@animaxia.io' }, stats: { watchlist: 0, history: 0, comments: 0 } });

   179→async function hlsHandler({ url }) {
   180→  if (url.pathname === '/x/manifest.m3u8') {
   181→    return new Response(MASTER_M3U8, { status: 200, headers: { 'content-type': 'application/vnd.apple.mpegurl' } });
   182→  }
   183→  if (url.pathname === '/x/360/playlist.m3u8') {
   184→    return new Response(MEDIA_M3U8, { status: 200, headers: { 'content-type': 'application/vnd.apple.mpegurl' } });
   185→  }
   186→  if (url.pathname.endsWith('.ts')) {
   187→    hlsState.segmentCalls += 1;
   188→    return new Response(SEGMENT_BYTES, { status: 200, headers: { 'content-type': 'video/mp2t' } });
   189→  }
   190→  return new Response('not found', { status: 404 });
   191→}
