   216→   216→/* TS3 (observability): W3C Trace Context propagation — when an incoming
   217→   217→ * `traceparent` header (format `00-<trace-id>-<span-id>-<flags>`, trace-id =
   218→   218→ * exactly 32 hex chars) is present its trace-id is adopted; otherwise a fresh
   219→   219→ * 32-hex id is generated (crypto.randomUUID without dashes). Every response
   220→   220→ * echoes X-Trace-Id, and requestLogger puts it on the access line (or in the
   221→   221→ * JSON log entry under LOGFORMAT=json).
   222→   222→ * OpenTelemetry-ready WITHOUT any otel dependency (documented honestly): an
   223→   223→ * otel collector / gateway in front of the backend can pick up X-Trace-Id for
   224→   224→ * trace correlation — TRACINGENABLED / OTELSENDPOINT / OTELSAMPLERATE in .env
   225→   225→ * are consumed by the edge/gateway staging configs, not by this process.
   226→   226→ * Runs BEFORE requestLogger so access lines always carry the trace id. */
   227→   227→app.use((req, res, next) => {
   228→   228→  let traceId = null;
   229→   229→  const traceparent = req.headers['traceparent'];
   230→   230→  if (typeof traceparent === 'string') {
   231→   231→    const m = /^[\da-f]{2}-([\da-f]{32})-[\da-f]{16}-[\da-f]{2}$/i.exec(traceparent.trim());
   232→   232→    if (m) traceId = m[1].toLowerCase(); // safe-parse: only well-formed ids propagate
   233→   233→  }
   234→   234→  if (!traceId) traceId = crypto.randomUUID().replace(/-/g, '');
   235→   235→  req.traceId = traceId;
   236→   236→  res.set('X-Trace-Id', traceId);
   237→   237→  next();
   238→   238→});

   487→   487→async function handleMine(req, res) {
   488→   488→  const url = req.method === 'GET' ? req.query.url : req.body?.url;
   489→   489→  if (!url || typeof url !== 'string') {
   490→   490→    return res.status(400).json({ error: 'Missing url' });
   491→   491→  }
   492→   492→  const options =
   493→   493→    req.method === 'GET'
   494→   494→      ? { type: req.query.type, externalId: req.query.externalId }
   495→   495→      : { type: req.body?.type, externalId: req.body?.externalId };
   496→   496→  try {
   497→   497→    const { extraction, record } = await mineUrl(url, options);
   498→   498→    invalidateHttpCache();
   499→   499→    // TS1: targeted id event for the incremental searchdocs sync (upsert path).
   500→   500→    if (record?.id) events.emit('content:changed:ids', { ids: [record.id], action: 'update' });
   501→   501→    await auditLog({ userid: req.user?.id ?? null, action: 'miner.run', ip: clientIpOf(req), meta: { url, id: record?.id } });
   502→   502→    return res.json({ ok: true, extraction, record });
   503→   503→  } catch (err) {
   504→   504→    return res.status(err.status || 502).json({ error: err.message || 'Mining failed' });
   505→   505→  }
   506→   506→}
