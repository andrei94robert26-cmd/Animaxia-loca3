    75→async function fetchWithTimeout(
    76→  url: string,
    77→  timeoutMs = 6000,
    78→): Promise<{ ok: boolean; status: number; text: string; error: string | null; latencyMs: number }> {
    79→  const started = Date.now();
    80→  try {
    81→    const controller = new AbortController();
    82→    const timer = setTimeout(() => controller.abort(), timeoutMs);
    83→    const res = await fetch(url, { signal: controller.signal, headers: { accept: '*/*' } });
    84→    const text = await res.text();
    85→    clearTimeout(timer);
    86→    return { ok: res.ok, status: res.status, text, error: null, latencyMs: Date.now() - started };
    87→  } catch (err) {
    88→    return {
    89→      ok: false,
    90→      status: 0,
    91→      text: '',
    92→      error: err instanceof Error ? err.message : String(err),
    93→      latencyMs: Date.now() - started,
    94→    };
    95→  }
    96→}

    98→function tryJson(text: string): Record<string, unknown> | null {
    99→  try {
   100→    const v = JSON.parse(text);
   101→    return v && typeof v === 'object' ? (v as Record<string, unknown>) : null;
   102→  } catch {
   103→    return null;
   104→  }
   105→}

   161→  private async scrapeJob(url: string, job: string): Promise<string | null> {
   162→    const res = await fetchWithTimeout(url);
   163→    if (res.ok && res.text) {
   164→      const parsed = this.store.replaceScraped(job, res.text);
   165→      if (parsed.series === 0) {
   166→        this.store.scrapeError(job);
   167→        return null;
   168→      }
   169→      return res.text;
   170→    }
   171→    this.store.scrapeError(job);
   172→    return null;
   173→  }

   175→  private async probe(
   176→    url: string,
   177→    label: string,
   178→  ): Promise<{ up: boolean; json: Record<string, unknown> | null; status: number; latencyMs: number }> {
   179→    const res = await fetchWithTimeout(url);
   180→    const up = res.status > 0 && res.status < 500; // orice răspuns HTTP = serviciul e viu
   181→    if (!up) this.store.scrapeError(label);
   182→    return { up, json: tryJson(res.text), status: res.status, latencyMs: res.latencyMs };
   183→  }

   185→  async tick(): Promise<void> {
   186→    this.state.ticks++;
   187→    this.state.lastTick = Date.now();
   188→    const now = Date.now();
   189→
   190→    // ── 1. scrape Prometheus text (FAZA 9 + sursele 23-b tolerate la 404) ─
   191→    await this.scrapeJob(SOURCES.backendMetrics, 'animaxia-backend');
   192→    await this.scrapeJob(SOURCES.backendObservability, 'animaxia-backend-faza23');
   193→    await this.scrapeJob(SOURCES.gatewayMetrics, 'animaxia-global-gateway');
   194→
   195→    // ── 2. probe backend health ───────────────────────────────────────────
   196→    const backend = await this.probe(SOURCES.backendHealth, 'animaxia-backend-health');
   197→    const bj = backend.json;
   198→    const db = (bj?.db ?? {}) as { connected?: boolean; totalItems?: number };
   199→    const totalItems =
   200→      typeof db.totalItems === 'number'
   201→        ? db.totalItems
   202→        : typeof bj?.totalItems === 'number'
   203→          ? (bj.totalItems as number)
   204→          : 0;
   205→    this.state.backend = {
   206→      up: backend.up && backend.status === 200,
   207→      neonConnected: db.connected === true,
   208→      totalItems,
   209→      uptimeSec: typeof bj?.uptimeSec === 'number' ? (bj.uptimeSec as number) : 0,
   210→      latencyMs: backend.latencyMs,
   211→    };
   212→
   213→    // ── 3. gateway health + status ────────────────────────────────────────
   214→    const gwHealth = await this.probe(SOURCES.gatewayHealth, 'global-gateway-health');
   215→    const gwStatus = await this.probe(SOURCES.gatewayStatus, 'global-gateway-status');
   216→    const gwj = gwStatus.json;
   217→    const counters = (gwj?.counters ?? null) as Record<string, unknown> | null;
   218→    const perRegion = (counters?.perRegion ?? null) as Record<string, number> | null;
   219→    const regionsRaw = Array.isArray(gwj?.regions) ? (gwj!.regions as unknown[]) : [];
   220→    const regions = regionsRaw.map((r) => {
   221→      const ro = (r ?? {}) as {
   222→        name?: string;
   223→        upstreams?: { healthy?: boolean; latencyMs?: number | null }[];
   224→      };
   225→      const healthyUps = (ro.upstreams ?? []).filter((u) => u.healthy);
   226→      const firstHealthy = healthyUps[0];
   227→      return {
   228→        name: ro.name ?? 'unknown',
   229→        healthy: healthyUps.length > 0,
   230→        latencyMs: firstHealthy?.latencyMs ?? null,
   231→      };
   232→    });
   233→    this.state.gateway = {
   234→      up: gwHealth.up && gwHealth.json?.ok === true,
   235→      redisMode: typeof gwj?.redisMode === 'string' ? (gwj.redisMode as string) : null,
   236→      regions,
   237→      failovers: counters && typeof counters.failovers === 'number' ? counters.failovers : null,
   238→      cacheHits: counters && typeof counters.cacheHits === 'number' ? counters.cacheHits : null,
   239→      cacheMisses: counters && typeof counters.cacheMisses === 'number' ? counters.cacheMisses : null,
   240→      requests: counters && typeof counters.requests === 'number' ? counters.requests : null,
   241→      requestsPerRegion: perRegion ? { ...perRegion } : {},
   242→    };
   243→
   244→    // ── 4. redis global (RESP propriu) ────────────────────────────────────
   245→    const redis = await probeRedis();
   246→    this.state.redis = { up: redis.ok, keys: redis.keys, latencyMs: redis.latencyMs };
   247→    if (!redis.ok) this.store.scrapeError('redis-global');
   248→
   249→    // ── 5. watchdog / frontend / realtime ─────────────────────────────────
   250→    const watchdog = await this.probe(SOURCES.watchdogHealth, 'animaxia-watchdog');
   251→    this.state.watchdog = { up: watchdog.up && watchdog.json?.ok === true };
   252→    const frontend = await this.probe(SOURCES.frontend, 'animaxia-frontend');
   253→    this.state.frontend = { up: frontend.status === 200 };
   254→    // realtime: root răspunde 400 — orice răspuns HTTP = UP (ECONNREFUSED = DOWN)
   255→    const realtime = await this.probe(SOURCES.realtime, 'animaxia-realtime');
   256→    this.state.realtime = { up: realtime.status > 0 };
   257→
   258→    // ── 6. search status (pipeline FAZA 18) ───────────────────────────────
   259→    const search = await this.probe(SOURCES.searchStatus, 'animaxia-search-status');
   260→    const sj = search.json;
   261→    const stages = Array.isArray(sj?.pipelineStages) ? (sj!.pipelineStages as PipelineStageStat[]) : [];
   262→    const latency = (sj?.latency ?? null) as { p95Ms?: number } | null;
   263→    const cluster = (sj?.cluster ?? null) as
   264→      | { enabled?: boolean; healthy?: boolean; nodes?: { healthy?: boolean }[] }
   265→      | null;
   266→    this.state.search = {
   267→      pipelineIntegrated: sj ? sj.pipelineIntegrated === true : null,
   268→      pipelineStages: stages,
   269→      ftsP95Ms: this.stageNumber(stages, 'neon-fts', 'p95Ms'),
   270→      ftsP50Ms: this.stageNumber(stages, 'neon-fts', 'p50Ms'),
   271→      clusterEnabled: cluster ? cluster.enabled === true : null,
   272→      clusterHealthy: cluster ? cluster.healthy === true : null,
   273→      clusterNodesHealthy: cluster?.nodes
   274→        ? cluster.nodes.filter((n) => n.healthy === true).length
   275→        : null,
   276→      docs: typeof sj?.docs === 'number' ? (sj.docs as number) : null,
   277→      latencyP95Ms: latency && typeof latency.p95Ms === 'number' ? latency.p95Ms : null,
   278→    };
   279→
   280→    // istoric cache/rate-limit/cooldown din pipeline (pentru ferestre 10m)
   281→    const cacheHits = this.stageNumber(stages, 'cache', 'hits') ?? 0;
   282→    const cacheMisses = this.stageNumber(stages, 'cache', 'misses') ?? 0;
   283→    const limited = this.stageNumber(stages, 'rate-limit', 'limited') ?? 0;
   284→    const cooldownServed = this.stageNumber(stages, 'cooldown', 'servedFromCooldown') ?? 0;
   285→    const cooldownDepri = this.stageNumber(stages, 'cooldown', 'deprioritized') ?? 0;
   286→    this.state.pipelineCacheHistory.push({ tsMs: now, hits: cacheHits, misses: cacheMisses });
   287→    this.state.rateLimitHistory.push({ tsMs: now, limited });
   288→    this.state.cooldownHistory.push({ tsMs: now, served: cooldownServed, deprioritized: cooldownDepri });
   289→    const historyCutoff = now - 11 * 60 * 1000; // puțin peste fereastra de 10m
   290→    this.state.pipelineCacheHistory = this.state.pipelineCacheHistory.filter((p) => p.tsMs >= historyCutoff);
   291→    this.state.rateLimitHistory = this.state.rateLimitHistory.filter((p) => p.tsMs >= historyCutoff);
   292→    this.state.cooldownHistory = this.state.cooldownHistory.filter((p) => p.tsMs >= historyCutoff);
   293→
   294→    // ── 7. hub stats (FAZA 20) ────────────────────────────────────────────
   295→    const hub = await this.probe(SOURCES.hubStats, 'animaxia-hub-stats');
   296→    this.state.hub = {
   297→      up: hub.up && hub.json?.ok === true,
   298→      totalItems: typeof hub.json?.totalItems === 'number' ? (hub.json.totalItems as number) : null,
   299→      raw: hub.json,
   300→    };
   301→
   302→    // ── 7b. hub feed 3-surse (personal + global + ai) ─────────────────────
   303→    const feedUrl = SOURCES.hubStats.replace('/api/hub/stats', '/api/hub/feed');
   304→    const feed = await this.probe(`${feedUrl}?limit=1`, 'animaxia-hub-feed');
   305→    const fj = feed.json;
   306→    const countItems = (v: unknown): number | null => {
   307→      if (v && typeof v === 'object' && Array.isArray((v as { items?: unknown }).items)) {
   308→        return (v as { items: unknown[] }).items.length;
   309→      }
   310→      return null;
   311→    };
   312→    this.state.hubFeed = {
   313→      up: feed.up && fj?.ok === true,
   314→      personal: countItems(fj?.personal),
   315→      global: countItems(fj?.global),
   316→      ai: countItems(fj?.ai),
   317→      combined: typeof fj?.combinedCount === 'number' ? (fj.combinedCount as number) : null,
   318→    };
   319→
   320→    // ── 8. expune familiile proprii ───────────────────────────────────────
   321→    this.exposeOwnMetrics();
   322→  }

   324→  private stageNumber(stages: PipelineStageStat[], name: string, key: string): number | null {
   325→    const st = stages.find((s) => s.name === name);
   326→    if (!st) return null;
   327→    const v = st.stats?.[key];
   328→    return typeof v === 'number' ? v : null;
   329→  }

   579→  /** rate pe fereastră (per minut) din istoric — pentru alerte. */
   580→  rateOverWindow(history: { tsMs: number; value: number }[], windowMs: number): number | null {
   581→    if (history.length < 2) return null;
   582→    const now = Date.now();
   583→    const recent = history.filter((h) => now - h.tsMs <= windowMs);
   584→    if (recent.length < 2) return null;
   585→    const first = recent[0];
   586→    const last = recent[recent.length - 1];
   587→    const delta = last.value - first.value;
   588→    if (delta < 0) return null; // counter reset
   589→    const elapsedMin = Math.max(1 / 60, (last.tsMs - first.tsMs) / 60000);
   590→    return delta / elapsedMin;
   591→  }

   593→  /** cache miss rate pe fereastră (0..1) din istoricul hits/misses. */
   594→  cacheMissRateOverWindow(windowMs: number): number | null {
   595→    const h = this.state.pipelineCacheHistory;
   596→    if (h.length < 2) return null;
   597→    const now = Date.now();
   598→    const recent = h.filter((p) => now - p.tsMs <= windowMs);
   599→    if (recent.length < 2) return null;
   600→    const dHits = recent[recent.length - 1].hits - recent[0].hits;
   601→    const dMisses = recent[recent.length - 1].misses - recent[0].misses;
   602→    const total = dHits + dMisses;
   603→    if (total <= 0) return null;
   604→    return dMisses / total;
   605→  }

   607→  /** p95 real al cererilor /api/search/* din histograma FAZA 9 (bucket-e). */
   608→  searchLatencyP95(): number | null {
   609→    const fam = this.store.scrapedFamily('animaxia_http_request_latency_ms_bucket');
   610→    if (!fam) return null;
   611→    const buckets: { le: number; count: number }[] = [];
   612→    for (const s of fam.series.values()) {
   613→      if (!s.labels.route || !s.labels.route.startsWith('/api/search')) continue;
   614→      const le = s.labels.le;
   615→      if (le === undefined) continue;
   616→      const leNum = le === '+Inf' ? Infinity : Number(le);
   617→      if (!Number.isFinite(leNum) && le !== '+Inf') continue;
   618→      buckets.push({ le: leNum, count: s.value });
   619→    }
   620→    if (buckets.length === 0) return null;
   621→    return histogramQuantile(buckets, 0.95);
   622→  }
