    59→/** Rezolvă un path JSON de forma "a.b.0.c". */
    60→function resolvePath(obj: unknown, path: string): { found: boolean; value: unknown } {
    61→  let cur: unknown = obj;
    62→  for (const part of path.split('.')) {
    63→    if (cur === null || cur === undefined) return { found: false, value: undefined };
    64→    if (Array.isArray(cur)) {
    65→      const idx = Number(part);
    66→      if (!Number.isInteger(idx) || idx < 0 || idx >= cur.length) return { found: false, value: undefined };
    67→      cur = cur[idx];
    68→      continue;
    69→    }
    70→    if (typeof cur === 'object') {
    71→      cur = (cur as Record<string, unknown>)[part] ?? undefined;
    72→      continue;
    73→    }
    74→    return { found: false, value: undefined };
    75→  }
    76→  return { found: cur !== undefined, value: cur };
    77→}
    78→
    79→function compare(op: Assertion['op'], actual: unknown, expected: unknown): boolean {
    80→  switch (op) {
    81→    case 'eq':
    82→      return actual === expected;
    83→    case 'ne':
    84→      return actual !== expected;
    85→    case 'gt':
    86→      return typeof actual === 'number' && typeof expected === 'number' && actual > expected;
    87→    case 'gte':
    88→      return typeof actual === 'number' && typeof expected === 'number' && actual >= expected;
    89→    case 'lt':
    90→      return typeof actual === 'number' && typeof expected === 'number' && actual < expected;
    91→    case 'lte':
    92→      return typeof actual === 'number' && typeof expected === 'number' && actual <= expected;
    93→    case 'truthy':
    94→      return Boolean(actual);
    95→    case 'falsy':
    96→      return !actual;
    97→    case 'exists':
    98→      return actual !== undefined && actual !== null;
    99→    case 'includes':
   100→      return (
   101→        Array.isArray(actual) &&
   102→        actual.length > 0 &&
   103→        (expected === undefined || actual.some((x) => JSON.stringify(x).includes(String(expected))))
   104→      );
   105→    default:
   106→      return false;
   107→  }
   108→}

   213→  /** rulează un singur check și înregistrează rezultatul. */
   214→  async runCheck(def: CheckDef): Promise<CheckRun> {
   215→    const started = Date.now();
   216→    let status: CheckResult = 'pass';
   217→    let error: string | null = null;
   218→    let bodyText = '';
   219→    let statusCode = 0;
   220→    let json: unknown = null;
   221→
   222→    try {
   223→      if (def.method === 'RESP') {
   224→        const { probeRedis } = await import('./resp-client.ts');
   225→        const probe = await probeRedis();
   226→        statusCode = probe.ok ? 200 : 0;
   227→        bodyText = probe.ok ? '+PONG' : probe.error ?? 'redis probe failed';
   228→        json = { __status__: statusCode, pong: probe.ok, keys: probe.keys };
   229→        if (!probe.ok) {
   230→          status = 'fail';
   231→          error = probe.error ?? 'redis probe failed';
   232→        }
   233→      } else {
   234→        const controller = new AbortController();
   235→        const timer = setTimeout(() => controller.abort(), SYNTHETIC_TIMEOUT_MS);
   236→        const res = await fetch(def.target, { signal: controller.signal });
   237→        clearTimeout(timer);
   238→        statusCode = res.status;
   239→        bodyText = await res.text();
   240→        try {
   241→          json = JSON.parse(bodyText);
   242→        } catch {
   243→          json = {};
   244→        }
   245→        // expune status-ul HTTP ca path special pentru asserții
   246→        if (json && typeof json === 'object' && !Array.isArray(json)) {
   247→          (json as Record<string, unknown>).__status__ = res.status;
   248→        } else {
   249→          json = { __status__: res.status, __value__: json };
   250→        }
   251→      }
   252→
   253→      if (status === 'pass') {
   254→        for (const a of def.assertions) {
   255→          if (a.type === 'status') {
   256→            if (statusCode !== (a.expected ?? 200)) {
   257→              status = 'fail';
   258→              error = `status ${statusCode} != ${a.expected}`;
   259→              break;
   260→            }
   261→          } else if (a.type === 'bodyIncludes') {
   262→            if (!bodyText.includes(a.text ?? '')) {
   263→              status = 'fail';
   264→              error = `body nu conține "${a.text}"`;
   265→              break;
   266→            }
   267→          } else if (a.type === 'json') {
   268→            const { found, value } = resolvePath(json, a.path ?? '');
   269→            if (!found) {
   270→              status = 'fail';
   271→              error = `path lipsă: ${a.path}`;
   272→              break;
   273→            }
   274→            if (!compare(a.op, value, a.value)) {
   275→              status = 'fail';
   276→              error = `${a.path}=${JSON.stringify(value)} nu satisface ${a.op} ${JSON.stringify(a.value)}`;
   277→              break;
   278→            }
   279→          }
   280→        }
   281→      }
   282→    } catch (err) {
   283→      status = 'fail';
   284→      error = err instanceof Error ? err.message : String(err);
   285→    }
   286→
   287→    const run: CheckRun = {
   288→      ts: new Date().toISOString(),
   289→      status,
   290→      durationMs: Date.now() - started,
   291→      error,
   292→    };
   293→    this.record(def, run);
   294→    return run;
   295→  }
   296→
   297→  private record(def: CheckDef, run: CheckRun): void {
   298→    let hist = this.history.get(def.id);
   299→    if (!hist) {
   300→      hist = [];
   301→      this.history.set(def.id, hist);
   302→    }
   303→    hist.push(run);
   304→    if (hist.length > SYNTHETIC_HISTORY_MAX) hist.splice(0, hist.length - SYNTHETIC_HISTORY_MAX);
   305→
   306→    // metrici
   307→    this.store.inc(
   308→      'animaxia_synthetic_checks_total',
   309→      { check: def.id, region: def.region, result: run.status },
   310→      1,
   311→      'Rulări synthetics per check/regiune/rezultat (FAZA 23)',
   312→    );
   313→    this.store.set(
   314→      'animaxia_synthetic_duration_ms',
   315→      { check: def.id },
   316→      run.durationMs,
   317→      'gauge',
   318→      'Durata ultimei rulări a check-ului (ms)',
   319→    );
   320→  }
   321→
   322→  /** rulează toate cele 46 (staggered, limită concurență).
   323→   *  staggerMs = pacing între porniri (default 1100ms — distribuie cele 46
   324→   *  pe fereastra de 60s ca să rămână SUB propriul rate-limit al platformei:
   325→   *  backend-ul FAZA 9/18 aplică 120 req/min/IP — good citizen, onest). */
   326→  async runAll(
   327→    concurrency: number,
   328→    staggerMs = SYNTHETIC_STAGGER_MS,
   329→  ): Promise<{
   330→    total: number;
   331→    passing: number;
   332→    failing: number;
   333→    durationMs: number;
   334→  }> {
   335→    if (this.running && this.inFlight) {
   336→      // o rulare programată e deja în desfășurare — _run așteaptă finalizarea
   337→      await this.inFlight;
   338→      return this.summaryCounts();
   339→    }
   340→    this.running = true;
   341→    const started = Date.now();
   342→    const task = (async () => {
   343→      try {
   344→        const defs = Synthetics.CHECKS;
   345→        const queue = [...defs];
   346→        const workers: Promise<void>[] = [];
   347→        // stagger ABSOLUT: check-ul N pornește la runStart + N×staggerMs
   348→        // (delay relativ la pornirea rulării, nu la momentul dequeue —
   349→        // evită compoundarea întârzierilor când execuția întârzie).
   350→        let stagger = 0;
   351→        const runStart = Date.now();
   352→        for (let i = 0; i < concurrency; i++) {
   353→          workers.push(
   354→            (async () => {
   355→              while (queue.length > 0) {
   356→                const def = queue.shift();
   357→                if (!def) break;
   358→                const startAt = runStart + stagger;
   359→                stagger += staggerMs;
   360→                const sleepFor = Math.max(0, startAt - Date.now());
   361→                await new Promise((r) => setTimeout(r, sleepFor));
   362→                await this.runCheck(def);
   363→              }
   364→            })(),
   365→          );
   366→        }
   367→        await Promise.all(workers);
   368→        this.lastRunAt = new Date().toISOString();
   369→
   370→        const counts = this.summaryCounts();
   371→        this.store.set(
   372→          'animaxia_synthetic_pass_ratio',
   373→          {},
   374→          counts.total > 0 ? counts.passing / counts.total : 0,
   375→          'gauge',
   376→          'Rata de trecere a celor 46 synthetics (ultimele rezultate)',
   377→        );
   378→      } finally {
   379→        this.running = false;
   380→        this.inFlight = null;
   381→      }
   382→    })();
   383→    this.inFlight = task;
   384→    await task;
   385→    const counts = this.summaryCounts();
   386→    return { ...counts, durationMs: Date.now() - started };
   387→  }
   388→
   389→  private summaryCounts(): { total: number; passing: number; failing: number; durationMs: 0 } {
   390→    let passing = 0;
   391→    let failing = 0;
   392→    for (const def of Synthetics.CHECKS) {
   393→      const hist = this.history.get(def.id);
   394→      const last = hist?.[hist.length - 1];
   395→      if (!last) continue;
   396→      if (last.status === 'pass') passing++;
   397→      else failing++;
   398→    }
   399→    return { total: Synthetics.CHECKS.length, passing, failing, durationMs: 0 };
   400→  }
   401→
   402→  /** GET /api/synthetics. */
   403→  view() {
   404→    const counts = this.summaryCounts();
   405→    const byRegion: Record<string, { passing: number; total: number }> = {
   406→      eu: { passing: 0, total: 0 },
   407→      us: { passing: 0, total: 0 },
   408→      asia: { passing: 0, total: 0 },
   409→      local: { passing: 0, total: 0 },
   410→    };
   411→    const checks = Synthetics.CHECKS.map((def) => {
   412→      const hist = this.history.get(def.id) ?? [];
   413→      const last = hist[hist.length - 1];
   414→      const passing = last?.status === 'pass';
   415→      if (last) {
   416→        byRegion[def.region].total++;
   417→        if (passing) byRegion[def.region].passing++;
   418→      }
   419→      const failures24h = hist.filter(
   420→        (h) => h.status === 'fail' && Date.now() - Date.parse(h.ts) < 24 * 3600 * 1000,
   421→      ).length;
   422→      return {
   423→        id: def.id,
   424→        name: def.name,
   425→        category: def.category,
   426→        region: def.region,
   427→        target: def.target,
   428→        method: def.method,
   429→        assertions: def.assertions,
   430→        intervalSec: def.intervalSec,
   431→        note: def.note,
   432→        lastStatus: last?.status ?? 'never',
   433→        lastDurationMs: last?.durationMs ?? null,
   434→        lastError: last?.error ?? null,
   435→        lastRunAt: last?.ts ?? null,
   436→        passing,
   437→        runs: hist.length,
   438→        failures24h,
   439→      };
   440→    });
   441→    return {
   442→      total: counts.total,
   443→      passing: counts.passing,
   444→      failing: counts.failing,
   445→      lastRunAt: this.lastRunAt,
   446→      byRegion,
   447→      checks,
   448→    };
   449→  }

   451→  historyFor(id: string): { id: string; history: CheckRun[] } | null {
   452→    if (!Synthetics.CHECKS.some((c) => c.id === id)) return null;
   453→    return { id, history: [...(this.history.get(id) ?? [])].reverse() };
   454→  }
   455→
   456→  recentFailures(n: number): { id: string; name: string; error: string | null; ts: string }[] {
   457→    const out: { id: string; name: string; error: string | null; ts: string }[] = [];
   458→    for (const def of Synthetics.CHECKS) {
   459→      const last = (this.history.get(def.id) ?? []).slice(-1)[0];
   460→      if (last && last.status === 'fail') {
   461→        out.push({ id: def.id, name: def.name, error: last.error, ts: last.ts });
   462→      }
   463→    }
   464→    return out.slice(0, n);
   465→  }

   467→  async runNow(): Promise<{ total: number; passing: number; failing: number; durationMs: number }> {
   468→    // pacing mai alert decât rularea programată, dar sub rate-limit-ul platformei
   469→    return this.runAll(SYNTHETIC_RUN_CONCURRENCY, SYNTHETIC_RUN_STAGGER_MS);
