    52→const RECENT_SEARCHES_KEY = 'animaxia:recent-searches';
    53→const RECENT_SEARCHES_MAX = 8;
    54→
    55→function readRecentSearches(): string[] {
    56→  try {
    57→    const raw = window.localStorage.getItem(RECENT_SEARCHES_KEY);
    58→    const list = raw ? (JSON.parse(raw) as unknown) : [];
    59→    if (!Array.isArray(list)) return [];
    60→    return list
    61→      .filter((x): x is string => typeof x === 'string' && x.trim().length > 0)
    62→      .slice(0, RECENT_SEARCHES_MAX);
    63→  } catch {
    64→    return [];
    65→  }
    66→}
    67→
    68→function writeRecentSearches(list: string[]): void {
    69→  try {
    70→    window.localStorage.setItem(RECENT_SEARCHES_KEY, JSON.stringify(list.slice(0, RECENT_SEARCHES_MAX)));
    71→  } catch {
    72→    /* private mode — istoricul nu persistă */
    73→  }
    74→}
    75→
    76→function rememberRecentSearch(current: string[], term: string): string[] {
    77→  const trimmed = term.trim();
    78→  if (!trimmed) return current;
    79→  return [trimmed, ...current.filter((r) => r.toLowerCase() !== trimmed.toLowerCase())].slice(
    80→    0,
    81→    RECENT_SEARCHES_MAX,
    82→  );
    83→}

   138→function activeFilterChips(filters: LibraryFilters): { key: keyof LibraryFilters; label: string }[] {
   139→  const chips: { key: keyof LibraryFilters; label: string }[] = [];
   140→  if (filters.year !== 'any') chips.push({ key: 'year', label: `An: ${filters.year}` });
   141→  if (filters.genre !== 'any') chips.push({ key: 'genre', label: `Gen: ${filters.genre}` });
   142→  if (filters.category !== 'any') chips.push({ key: 'category', label: `Categorie: ${filters.category}` });
   143→  if (filters.actor.trim()) chips.push({ key: 'actor', label: `Actor: ${filters.actor.trim()}` });
   144→  if (filters.studio.trim()) chips.push({ key: 'studio', label: `Studio: ${filters.studio.trim()}` });
   145→  if (filters.minRating !== 'any') chips.push({ key: 'minRating', label: `Notă ≥ ${filters.minRating}` });
   146→  return chips;
   147→}

   150→
   151→export function SearchView({ route }: { route: AppRoute }) {
   152→  const q = route.query.get('q')?.trim() || '';
   153→  const [input, setInput] = useState(q);
   154→  const [activeTab, setActiveTab] = useState<SearchTab>('library');
   155→
   156→  /* TS2b: căutări recente — încărcate după montare (localStorage e doar client). */
   157→  const [recentSearches, setRecentSearches] = useState<string[]>([]);
   158→
   159→  useEffect(() => {
   160→    const timer = setTimeout(() => setRecentSearches(readRecentSearches()), 0);
   161→    return () => clearTimeout(timer);
   162→  }, []);
   163→
   164→  /* Every executed search (URL q — submit, debounce sau navbar) is recorded. */
   165→  useEffect(() => {
   166→    if (!q) return;
   167→    const timer = setTimeout(() => {
   168→      setRecentSearches((prev) => {
   169→        const next = rememberRecentSearch(prev, q);
   170→        writeRecentSearches(next);
   171→        return next;
   172→      });
   173→    }, 0);
   174→    return () => clearTimeout(timer);
   175→  }, [q]);
   176→
   177→  function removeRecentSearch(term: string) {
   178→    setRecentSearches((prev) => {
   179→      const next = prev.filter((r) => r !== term);
   180→      writeRecentSearches(next);
   181→      return next;
   182→    });
   183→  }
   184→
   185→  function clearRecentSearches() {
   186→    setRecentSearches([]);
   187→    writeRecentSearches([]);
   188→  }
   189→
   190→  /* Keep the box in sync when the URL query changes. */
   191→  useEffect(() => {
   192→    setInput(q);
   193→  }, [q]);
   194→
   195→  /* Live search: debounce 400 ms, navigate only when the query really changed. */
   196→  useEffect(() => {
   197→    const trimmed = input.trim();
   198→    if (trimmed === q) return;
   199→    const timer = setTimeout(() => {
   200→      navigate(trimmed ? `/search?q=${encodeURIComponent(trimmed)}` : '/search');
   201→    }, 400);
   202→    return () => clearTimeout(timer);
   203→  }, [input, q]);
   204→
   205→  /* External tabs still search live providers with the URL query. */
   206→  const externalRes = useApi(
   207→    `search-${activeTab}-${q}`,
   208→    async (): Promise<{ items: NormalizedItem[]; total: number }> => {
   209→      if (activeTab === 'anime') {
   210→        const res = await api.animeSearch(q);
   211→        return { items: res.data.map(animeToItem), total: res.data.length };
   212→      }
   213→      const res = await api.moviesSearch(q);
   214→      return { items: res.data.map(mediaToItem), total: res.data.length };
   215→    },
   216→    { enabled: q.length > 0 && activeTab !== 'library' },
   217→  );
   218→
   219→  const submit = (event: React.FormEvent<HTMLFormElement>) => {
   220→    event.preventDefault();
   221→    const trimmed = input.trim();
   222→    if (trimmed) {
   223→      /* TS2b: înregistrează căutarea și la submit direct (acoperă re-căutarea
   224→         aceluiași termen, când q din URL nu se schimbă și efectul nu rulează). */
   225→      setRecentSearches((prev) => {
   226→        const next = rememberRecentSearch(prev, trimmed);
   227→        writeRecentSearches(next);
   228→        return next;
   229→      });
   230→    }
   231→    navigate(trimmed ? `/search?q=${encodeURIComponent(trimmed)}` : '/search');
   232→  };
   233→
   234→  /* ----------------------------- autocomplete ----------------------------- */
   235→
   236→  const searchBoxRef = useRef<HTMLDivElement>(null);
   237→  const [acOpen, setAcOpen] = useState(false);
   238→  const [acItems, setAcItems] = useState<Suggestion[]>([]);
   239→  const [acLoading, setAcLoading] = useState(false);
   240→  const [acHighlight, setAcHighlight] = useState(-1);
   241→
   242→  /* Debounced autocomplete fetch while the dropdown is open (Neon prefix+trigram). */
   243→  const prefetchedRef = useRef<Set<string>>(new Set()); // FAZA 29-P: hover-prefetch (o singură dată per titlu)
   244→
   245→  /* FAZA 29-P: la hover pe o sugestie, încălzim în fundal căutarea completă —
   246→   * la click rezultatul vine din cache accel în ~2ms în loc de ~600ms. */
   247→  const prefetchSuggestion = (title: string) => {
   248→    if (!title || prefetchedRef.current.has(title)) return;
   249→    prefetchedRef.current.add(title);
   250→    /* Semnătura EXACTĂ pe care o va cere search-view la click (sort=score,
   251→     * limit=24, page=1) — altfel warming-ul ar încălzi o cheie nefolosită. */
   252→    void api.searchPrefetch({ q: title, sort: 'score', limit: PAGE_LIMIT, page: 1 });
   253→  };
   254→
   255→  useEffect(() => {
   256→    const query = input.trim();
   257→    if (!acOpen || query.length < 1) {
   258→      setAcItems([]);
   259→      setAcLoading(false);
   260→      setAcHighlight(-1);
   261→      return;
   262→    }
   263→    setAcLoading(true);
   264→    let cancelled = false;
   265→    const timer = setTimeout(async () => {
   266→      try {
   267→        const res = await api.searchAutocomplete(query);
   268→        if (!cancelled) setAcItems(res.suggestions);
   269→      } catch {
   270→        if (!cancelled) setAcItems([]);
   271→      } finally {
   272→        if (!cancelled) setAcLoading(false);
   273→      }
   274→    }, 300);
   275→    return () => {
   276→      cancelled = true;
   277→      clearTimeout(timer);
   278→    };
   279→  }, [input, acOpen]);
   280→
   281→  /* Close on click-outside. */
   282→  useEffect(() => {
   283→    if (!acOpen) return undefined;
   284→    const onDown = (e: MouseEvent) => {
   285→      if (searchBoxRef.current && !searchBoxRef.current.contains(e.target as Node)) {
   286→        setAcOpen(false);
   287→      }
   288→    };
   289→    document.addEventListener('mousedown', onDown);
   290→    return () => document.removeEventListener('mousedown', onDown);
   291→  }, [acOpen]);
   292→
   293→  function pickSuggestion(s: Suggestion) {
   294→    /* TS2b: și click-ul pe un rezultat înregistrează interogarea căutată. */
   295→    const term = input.trim();
   296→    if (term) {
   297→      setRecentSearches((prev) => {
   298→        const next = rememberRecentSearch(prev, term);
   299→        writeRecentSearches(next);
   300→        return next;
   301→      });
   302→    }
   303→    setInput('');
   304→    setAcOpen(false);
   305→    setAcItems([]);
   306→    navigate(`/content/${s.id}`);
   307→  }
   308→
   309→  function handleAcKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
   310→    if (!acOpen) return;
   311→    if (e.key === 'Escape') {
   312→      e.preventDefault();
   313→      setAcOpen(false);
   314→      return;
   315→    }
   316→    if (!acItems.length) return;
   317→    if (e.key === 'ArrowDown') {
   318→      e.preventDefault();
   319→      const next = (acHighlight + 1) % acItems.length;
   320→      setAcHighlight(next);
   321→      prefetchSuggestion(acItems[next]?.title ?? ''); // FAZA 29-P: warming predictiv și la tastatură
   322→    } else if (e.key === 'ArrowUp') {
   323→      e.preventDefault();
   324→      const next = acHighlight <= 0 ? acItems.length - 1 : acHighlight - 1;
   325→      setAcHighlight(next);
   326→      prefetchSuggestion(acItems[next]?.title ?? ''); // FAZA 29-P: warming predictiv și la tastatură
   327→    } else if (e.key === 'Enter' && acHighlight >= 0 && acItems[acHighlight]) {
   328→      e.preventDefault();
   329→      pickSuggestion(acItems[acHighlight]);
   330→    }
   331→  }
   332→
   333→  const showDropdown = acOpen && input.trim().length >= 1;
   334→
   335→  return (
   336→    <Container className="space-y-8 py-8 sm:py-10">
   337→      <PageHeader
   338→        title="Căutare"
   339→        subtitle="O căsuță, toate sursele — biblioteca ta Neon (motor FTS), MyAnimeList și TMDB."
   340→      />
   341→
   342→      <form onSubmit={submit} role="search" aria-label="Caută pe Animaxia" className="flex flex-col gap-3 sm:flex-row">
   343→        <div className="relative flex-1" ref={searchBoxRef}>
   344→          <Search
   345→            className="pointer-events-none absolute left-4 top-1/2 h-5 w-5 -translate-y-1/2 text-zinc-500"
   346→            aria-hidden
   347→          />
   348→          <Input
   349→            value={input}
   350→            onChange={(event) => {
   351→              setInput(event.target.value);
   352→              setAcOpen(true);
   353→              setAcHighlight(-1);
   354→            }}
   355→            onFocus={() => setAcOpen(true)}
   356→            onKeyDown={handleAcKeyDown}
   357→            placeholder="Caută anime, filme, seriale…"
   358→            aria-label="Text de căutat"
   359→            aria-autocomplete="list"
   360→            aria-expanded={showDropdown}
   361→            aria-controls={showDropdown ? 'search-suggestions' : undefined}
   362→            className="h-12 rounded-xl border-white/10 bg-white/[0.04] pl-12 pr-11 text-base text-zinc-100 placeholder:text-zinc-500 focus-visible:border-rose-500/50 focus-visible:ring-rose-500/30"
   363→          />
   364→          {input ? (
   365→            <button
   366→              type="button"
   367→              onClick={() => setInput('')}
   368→              aria-label="Golește căutarea"
   369→              className="absolute right-3 top-1/2 -translate-y-1/2 rounded-full p-1.5 text-zinc-500 transition hover:bg-white/10 hover:text-zinc-200 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-rose-500/60"
   370→            >
   371→              <X className="h-4 w-4" aria-hidden />
   372→            </button>
   373→          ) : null}
   374→
   375→          {/* Autocomplete dropdown (Neon prefix + trigram) */}
   376→          {showDropdown && (
   377→            <div
   378→              id="search-suggestions"
   379→              role="listbox"
   380→              aria-label="Sugestii de căutare"
   381→              className="custom-scrollbar absolute inset-x-0 top-full z-20 mt-2 max-h-80 overflow-y-auto rounded-xl border border-white/10 bg-[#12121c] shadow-2xl shadow-black/60"
   382→            >
   383→              {acLoading ? (
   384→                <div className="flex items-center gap-2 px-4 py-3 text-sm text-zinc-500">
   385→                  <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> Caut…
   386→                </div>
   387→              ) : acItems.length === 0 ? (
   388→                <p className="px-4 py-3 text-sm text-zinc-500">Nicio sugestie</p>
   389→              ) : (
   390→                acItems.map((s, i) => (
   391→                  <button
   392→                    key={s.id}
   393→                    type="button"
   394→                    role="option"
   395→                    aria-selected={acHighlight === i}
   396→                    onMouseEnter={() => {
   397→                      setAcHighlight(i);
   398→                    }}
   399→                    onMouseOver={() => {
   400→                      prefetchSuggestion(s.title); // FAZA 29-P: predictive warming (mouseover = delegat fiabil)
   401→                    }}
   402→                    onFocus={() => {
   403→                      prefetchSuggestion(s.title); // FAZA 29-P: și la navigarea cu tastatura
   404→                    }}
   405→                    onClick={() => pickSuggestion(s)}
   406→                    className={cn(
   407→                      'flex w-full items-center gap-3 px-3 py-2.5 text-left transition',
   408→                      acHighlight === i ? 'bg-white/[0.06]' : 'hover:bg-white/[0.06]',
   409→                    )}
   410→                  >
   411→                    {s.thumbnail ? (
   412→                      <img
   413→                        src={s.thumbnail}
   414→                        alt=""
   415→                        loading="lazy"
   416→                        className="h-10 w-8 shrink-0 rounded object-cover"
   417→                      />
   418→                    ) : (
   419→                      <span className="grid h-10 w-8 shrink-0 place-items-center rounded bg-gradient-to-br from-white/10 to-white/[0.03]">
   420→                        <Search className="h-3.5 w-3.5 text-zinc-600" aria-hidden />
   421→                      </span>
   422→                    )}
   423→                    <span className="min-w-0 flex-1 truncate text-sm font-medium text-zinc-200">{s.title}</span>
   424→                    {s.rating != null ? <RatingBadge value={s.rating} /> : null}
   425→                    <TypeBadge type={s.type} />
   426→                  </button>
   427→                ))
   428→              )}
   429→            </div>
   430→          )}
   431→        </div>
   432→        <Button
   433→          type="submit"
   434→          className="h-12 rounded-xl bg-rose-600 px-7 text-base font-bold text-white hover:bg-rose-500"
   435→        >
   436→          Caută
   437→        </Button>
   438→      </form>
   439→
   440→      {/* TS2b: căutări recente — doar când nu există niciun termen activ. */}
   441→      {!q && input.trim() === '' && recentSearches.length > 0 && (
   442→        <section aria-label="Căutări recente" className="space-y-2.5">
   443→          <div className="flex items-center justify-between gap-3">
   444→            <h2 className="flex items-center gap-2 text-sm font-bold text-zinc-300">
   445→              <History className="h-4 w-4 text-rose-400" aria-hidden />
   446→              Căutări recente
   447→            </h2>
   448→            <button
   449→              type="button"
   450→              onClick={clearRecentSearches}
   451→              className="text-xs font-semibold text-zinc-500 transition hover:text-rose-300"
   452→            >
   453→              Golește istoricul
   454→            </button>
   455→          </div>
   456→          <div className="flex flex-wrap gap-2">
   457→            {recentSearches.map((term) => (
   458→              <span
   459→                key={term}
   460→                className="inline-flex h-9 items-center gap-1.5 rounded-full border border-white/10 bg-white/[0.04] pl-3.5 pr-1.5 text-sm text-zinc-300"
   461→              >
   462→                <button
   463→                  type="button"
   464→                  onClick={() => navigate(`/search?q=${encodeURIComponent(term)}`)}
   465→                  className="font-medium transition hover:text-rose-300"
   466→                >
   467→                  {term}
   468→                </button>
   469→                <button
   470→                  type="button"
   471→                  onClick={() => removeRecentSearch(term)}
   472→                  aria-label={`Elimină „${term}” din istoric`}
   473→                  className="grid h-6 w-6 place-items-center rounded-full text-zinc-500 transition hover:bg-rose-500/20 hover:text-rose-300 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-rose-500/60"
   474→                >
   475→                  <X className="h-3.5 w-3.5" aria-hidden />
   476→                </button>
   477→              </span>
   478→            ))}
   479→          </div>
   480→        </section>
   481→      )}
   482→
   483→      <Tabs value={activeTab} onValueChange={(value) => setActiveTab(value as SearchTab)}>
   484→        <TabsList className="h-10 w-full justify-start gap-1 rounded-xl border border-white/10 bg-white/[0.03] p-1 sm:w-auto">
   485→          {(['library', 'anime', 'media'] as const).map((tab) => (
   486→            <TabsTrigger
   487→              key={tab}
   488→              value={tab}
   489→              className="px-4 text-zinc-400 hover:text-zinc-100 data-[state=active]:bg-rose-600 data-[state=active]:text-white data-[state=active]:shadow-rose-900/40 sm:px-5"
   490→            >
   491→              {TAB_LABELS[tab]}
   492→            </TabsTrigger>
   493→          ))}
   494→        </TabsList>
   495→
   496→        <TabsContent value="library" className="pt-6">
   497→          <LibrarySearchPanel q={q} />
   498→        </TabsContent>
   499→
   500→        {(['anime', 'media'] as const).map((tab) => (
   501→          <TabsContent key={tab} value={tab} className="pt-6">
   502→            {q ? (
   503→              <ResultsBlock label={TAB_LABELS[tab]} q={q} res={externalRes} />
   504→            ) : (
   505→              <EmptyState
   506→                icon={<SearchX className="h-6 w-6" aria-hidden />}
   507→                title="Caută pe Animaxia"
   508→                message="Scrie un titlu mai sus — căutăm live pe MyAnimeList și TMDB."
   509→              />
   510→            )}
   511→          </TabsContent>
   512→        ))}
   513→      </Tabs>
   514→    </Container>
   515→  );
   516→}

   519→
   520→function ResultsBlock({

   879→function ActorInput({
   880→  actors,
   881→  value,
   882→  onChange,
   883→}: {
   884→  actors: { value: string; count: number }[];
   885→  value: string;

   899→

   912→          onChange={(e) => {
   913→            onChange(e.target.value);
   914→            setOpen(true);
   915→          }}
   916→          onFocus={() => setOpen(true)}
   917→          placeholder="ex. Dwayne Johnson"
   918→          className="border-white/10 bg-black/40"

   923→          <div className="custom-scrollbar absolute inset-x-0 top-full z-20 mt-1 max-h-48 overflow-y-auto rounded-xl border border-white/10 bg-[#12121c] shadow-2xl shadow-black/60">

   946→/* ------------------- dedup panel: verify link before adding ------------------ */
   947→
   948→function DedupCheckPanel() {
   949→  const [open, setOpen] = useState(false);
   950→  const [url, setUrl] = useState('');
   951→  const [checking, setChecking] = useState(false);
   952→  const [registering, setRegistering] = useState(false);
   953→  const [result, setResult] = useState<IngestCheckResponse | null>(null);
   954→  const [error, setError] = useState<string | null>(null);
   955→
   956→  const check = async (e: React.FormEvent) => {
   957→    e.preventDefault();
   958→    const trimmed = url.trim();
   959→    if (!trimmed) return;
   960→    setChecking(true);
   961→    setError(null);
   962→    setResult(null);
   963→    try {
   964→      const res = await api.ingestCheck(trimmed);
   965→      setResult(res);
   966→    } catch (err) {
   967→      setError(err instanceof Error ? err.message : 'Verificarea a eșuat');
   968→    } finally {
   969→      setChecking(false);
   970→    }
   971→  };
   972→
   973→  const register = async () => {
   974→    const trimmed = url.trim();
   975→    if (!trimmed) return;
   976→    setRegistering(true);
   977→    try {
   978→      const res = await api.ingestRegister({ url: trimmed, addedBy: 'manual' });
   979→      if (res.registered) {
   980→        toast.success('Link înregistrat în Neon', { description: res.source.url });
   981→      } else {
   982→        toast.warning('Linkul există deja în registru', { description: `${res.source.hits} accesări înregistrate.` });
   983→      }
   984→      setResult({ ok: true, exists: true, duplicate: true, source: res.source, content: null });
   985→    } catch (err) {
   986→      toast.error('Înregistrarea a eșuat', {
   987→        description: err instanceof Error ? err.message : 'Încearcă din nou peste un minut.',
   988→      });
   989→    } finally {
   990→      setRegistering(false);
   991→    }
   992→  };
   993→
   994→  const fmtDate = (iso: string) => {
   995→    try {
   996→      return new Date(iso).toLocaleDateString('ro-RO', { day: 'numeric', month: 'short', year: 'numeric' });
   997→    } catch {
   998→      return iso;
   999→    }
  1000→  };
  1001→
  1002→  return (
  1003→    <Collapsible open={open} onOpenChange={setOpen} className="rounded-xl border border-white/10 bg-white/[0.02]">
  1004→      <CollapsibleTrigger className="flex w-full items-center gap-2.5 px-4 py-3 text-left text-sm font-semibold text-zinc-300 transition hover:bg-white/[0.04] hover:text-white focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-rose-500/60">
