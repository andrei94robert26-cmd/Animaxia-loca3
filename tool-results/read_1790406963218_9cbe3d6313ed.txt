    61→function formatClock(sec: number): string {
    62→  const m = Math.floor(sec / 60);
    63→  const s = Math.floor(sec % 60);
    64→  return `${m}:${String(s).padStart(2, '0')}`;

    67→/** TS2a — m:ss / h:mm:ss clock for the custom control bar. */
    68→function formatTime(sec: number): string {
    69→  const s = Math.max(0, Math.floor(Number.isFinite(sec) ? sec : 0));
    70→  const h = Math.floor(s / 3600);
    71→  const m = Math.floor((s % 3600) / 60);
    72→  const ss = s % 60;
    73→  return h > 0
    74→    ? `${h}:${String(m).padStart(2, '0')}:${String(ss).padStart(2, '0')}`
    75→    : `${m}:${String(ss).padStart(2, '0')}`;
    76→}

   290→export function PlayerView({ route }: { route: AppRoute }) {
   291→  const { user } = useAuth();
   292→  const id = Number(route.params[0]);
   293→  const validId = Number.isInteger(id) && id > 0;
   294→  const player = useApi(`player-${id}`, () => api.player(id), { enabled: validId });
   295→
   296→  /* FAZA 17 — source fallback chain. `sourceIndex` walks data.sources on any
   297→     load failure (mp4/HLS error, 12s load timeout). `failedState` marks the
   298→     friendly end-of-chain card and is scoped to the content id so it can
   299→     never leak across titles. `reloadNonce` forces a fresh <video> mount when
   300→     retrying the same source. */
   301→  const [sourceIndex, setSourceIndex] = useState(0);
   302→  const [failedState, setFailedState] = useState<{ contentId: number; reason: string } | null>(null);
   303→  const [reloadNonce, setReloadNonce] = useState(0);
   304→  const [mineUrl, setMineUrl] = useState('');
   305→  const [mining, setMining] = useState(false);
   306→
   307→  const videoRef = useRef<HTMLVideoElement>(null);
   308→  const sourcesRef = useRef<PlayerSource[]>([]);
   309→  const sourceIndexRef = useRef(0);
   310→  const loadTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   311→  const lastAdvanceRef = useRef<{ key: string; at: number } | null>(null);
   312→
   313→  /* Only trust data matching the requested id — useApi keeps the previous
   314→     record alive while the key changes (stale data across titles). */
   315→  const current = player.data && player.data.content.id === id ? player.data : null;
   316→  const sources = current?.sources ?? [];
   317→  const exhausted = failedState != null && failedState.contentId === id && sources.length > 0;
   318→  const activeSource: PlayerSource | null = exhausted
   319→    ? null
   320→    : sources[sourceIndex] ?? current?.primary ?? null;
   321→
   322→  /** Clears the 12s load watchdog — any progress event calls this. */
   323→  const clearLoadTimer = useCallback(() => {
   324→    if (loadTimerRef.current) {
   325→      clearTimeout(loadTimerRef.current);
   326→      loadTimerRef.current = null;
   327→    }
   328→  }, []);
   329→
   330→  /** Advance to the next source on failure; when none remain, switch to the
   331→      friendly end-of-chain card instead of crashing. Duplicate signals for the
   332→      same attempt (e.g. hls.js error + element error) are coalesced. */
   333→  const advanceSource = useCallback(
   334→    (reason: string) => {
   335→      clearLoadTimer();
   336→      const list = sourcesRef.current;
   337→      const i = sourceIndexRef.current;
   338→      const key = `${id}:${i}:${reason}`;
   339→      const now = Date.now();
   340→      if (lastAdvanceRef.current && lastAdvanceRef.current.key === key && now - lastAdvanceRef.current.at < 1500) {
   341→        return;
   342→      }
   343→      lastAdvanceRef.current = { key, at: now };
   344→      if (i + 1 < list.length) {
   345→        sourceIndexRef.current = i + 1;
   346→        setSourceIndex(i + 1);
   347→        toast.info(`Sursa ${i + 1} nu răspunde (${reason}) — trec la următoarea…`);
   348→      } else {
   349→        setFailedState({ contentId: id, reason });
   350→      }
   351→    },
   352→    [id, clearLoadTimer],
   353→  );
   354→
   355→  /** Manual source chip click — clears the chain failure; clicking the chip of
   356→      the current source forces a clean reload of that source. */
   357→  const selectSource = (i: number) => {
   358→    if (i < 0 || i >= sources.length) return;
   359→    setFailedState(null);
   360→    if (i === sourceIndex) {
   361→      setReloadNonce((n) => n + 1);
   362→    } else {
   363→      sourceIndexRef.current = i;
   364→      setSourceIndex(i);
   365→    }
   366→  };
   367→
   368→  /** "Reîncearcă de la prima sursă" — reset the chain and remount the video. */
   369→  const retryFromFirst = () => {
   370→    setFailedState(null);
   371→    sourceIndexRef.current = 0;
   372→    setSourceIndex(0);
   373→    setReloadNonce((n) => n + 1);
   374→  };
   375→
   376→  /* New title (route change) → reset the fallback chain before the new data
   377→     ever renders, so no stale index/failure can flash. */
   378→  useEffect(() => {
   379→    sourceIndexRef.current = 0;
   380→    setSourceIndex(0);
   381→    setFailedState(null);
   382→  }, [id]);
   383→
   384→  /* Fresh sources (first load or refetch after mining) → start at the primary
   385→     source and force a clean reload of the player surface. */
   386→  useEffect(() => {
   387→    const d = player.data;
   388→    if (!d || d.content.id !== id) return;
   389→    sourcesRef.current = d.sources;
   390→    const primary = d.primary;
   391→    const primaryIndex = primary
   392→      ? d.sources.findIndex((s) => s.url === primary.url && s.type === primary.type)
   393→      : -1;
   394→    const start = primaryIndex >= 0 ? primaryIndex : 0;
   395→    sourceIndexRef.current = start;
   396→    setSourceIndex(start);
   397→    setFailedState(null);
   398→    setReloadNonce((n) => n + 1);
   399→  }, [player.data, id]);
   400→
   401→  /* ------------------------- analytics tracking ------------------------- */
   402→
   403→  const trackedViewRef = useRef<number | null>(null);
   404→  const trackedPlayRef = useRef<number | null>(null);
   405→
   406→  /* Fire-and-forget 'view' event, once per content id per mount. */
   407→  useEffect(() => {
   408→    if (!player.data || player.data.content.id !== id) return;
   409→    if (trackedViewRef.current === id) return;
   410→    trackedViewRef.current = id;
   411→    platform.recordView(id, 'view').catch(() => {});
   412→  }, [player.data, id]);
   413→
   414→  /* 'play' event for embed renderers (youtube/iframe), once per content id. */
   415→  useEffect(() => {
   416→    if (!player.data || player.data.content.id !== id) return;
   417→    const src = activeSource;
   418→    if (!src || (src.type !== 'youtube' && src.type !== 'iframe')) return;
   419→    if (trackedPlayRef.current === id) return;
   420→    trackedPlayRef.current = id;
   421→    platform.recordView(id, 'play').catch(() => {});
   422→  }, [player.data, activeSource, id]);
   423→
   424→  const trackPlay = () => {
   425→    if (trackedPlayRef.current === id) return;
   426→    trackedPlayRef.current = id;
   427→    platform.recordView(id, 'play').catch(() => {});
   428→  };
   429→
   430→  const trackComplete = () => {
   431→    platform.recordView(id, 'complete').catch(() => {});
   432→  };
   433→
   434→  /* ------------------- FAZA 21: history / resume / subtitles ------------------- */
   435→
   436→  /** Resume position from ?t= (seconds) — validated once per route param. */
   437→  const tParam = route.query.get('t');
   438→  const resumeT = (() => {
   439→    const t = Number(tParam);
   440→    return Number.isFinite(t) && t > 0 && t < 86_400 ? Math.floor(t) : 0;
   441→  })();
   442→
   443→  /** recordHistory — POST /api/users/me/history {contentId, progress} via the
   444→   *  existing platform client. Fire-and-forget, silent failures: never breaks
   445→   *  playback. Only native renderers expose video.currentTime (YouTube/iframe
   446→   *  embeds are honestly skipped). */
   447→  const recordTick = useCallback(() => {
   448→    if (!user || !validId) return;
   449→    const el = videoRef.current;
   450→    if (!el) return;
   451→    const t = Math.floor(el.currentTime);
   452→    if (!Number.isFinite(t) || t <= 0) return;
   453→    platform.recordHistory(id, t).catch(() => {});
   454→  }, [user, id, validId]);
   455→
   456→  /* Progress sync: every 10s while playing + a final flush on pause handled
   457→     by the onPause prop; the cleanup below flushes on unmount AND on source
   458→     change (deps include the active source), recording the last position of
   459→     the outgoing element before the decoder-release effect resets it. */
   460→  useEffect(() => {
   461→    if (!user || !validId) return undefined;
   462→    recordTick(); /* mount-time flush (e.g. after a resume seek) */
   463→    const timer = setInterval(() => {
   464→      const el = videoRef.current;
   465→      if (el && !el.paused && !el.ended) recordTick();
   466→    }, 10_000);
   467→    return () => {
   468→      clearInterval(timer);
   469→      recordTick(); /* best-effort final flush */
   470→    };
   471→  }, [user, validId, id, recordTick, activeSource?.type, activeSource?.url, sourceIndex, reloadNonce]);
   472→
   473→  /** Resume seek — applied once per (content id + source + reload nonce) on
   474→   *  the first loadedmetadata of the current <video>. The applied key is
   475→   *  stamped on the DOM element itself (dataset), so a remount that somehow
   476→   *  preserves React state (e.g. dev Fast Refresh) still re-arms the seek,
   477→   *  while the same playing element never seeks twice. */
   478→  const maybeResume = useCallback(() => {
   479→    if (resumeT <= 0) return;
   480→    const el = videoRef.current;
   481→    if (!el) return;
   482→    const key = `${id}:${sourceIndex}:${activeSource?.url ?? ''}:${reloadNonce}`;
   483→    if (el.dataset.faza21Resume === key) return;
   484→    try {
   485→      el.currentTime = resumeT;
   486→      el.dataset.faza21Resume = key;
   487→      toast.info(`Reia de la ${formatClock(resumeT)}`);
   488→    } catch {
   489→      /* seek not possible yet — allow a later loadedmetadata to retry */
   490→    }
   491→  }, [resumeT, id, sourceIndex, activeSource?.url, reloadNonce]);
   492→
