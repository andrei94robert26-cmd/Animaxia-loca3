   224→/* TS3 (observability): W3C Trace Context propagation — when an incoming
   225→ * `traceparent` header (format `00-<trace-id>-<span-id>-<flags>`, trace-id =
   226→ * exactly 32 hex chars) is present its trace-id is adopted; otherwise a fresh
   227→ * 32-hex id is generated (crypto.randomUUID without dashes). Every response
   228→ * echoes X-Trace-Id, and requestLogger puts it on the access line (or in the
   229→ * JSON log entry under LOGFORMAT=json).
   230→ * OpenTelemetry-ready WITHOUT any otel dependency (documented honestly): an
   231→ * otel collector / gateway in front of the backend can pick up X-Trace-Id for
   232→ * trace correlation — TRACINGENABLED / OTELSENDPOINT / OTELSAMPLERATE in .env
   233→ * are consumed by the edge/gateway staging configs, not by this process.
   234→ * Runs BEFORE requestLogger so access lines always carry the trace id. */
   235→app.use((req, res, next) => {
   236→  let traceId = null;
   237→  const traceparent = req.headers['traceparent'];
   238→  if (typeof traceparent === 'string') {
   239→    const m = /^[\da-f]{2}-([\da-f]{32})-[\da-f]{16}-[\da-f]{2}$/i.exec(traceparent.trim());
   240→    if (m) traceId = m[1].toLowerCase(); // safe-parse: only well-formed ids propagate
   241→  }
   242→  if (!traceId) traceId = crypto.randomUUID().replace(/-/g, '');
   243→  req.traceId = traceId;
   244→  res.set('X-Trace-Id', traceId);
   245→  next();
   246→});

   512→async function handleMine(req, res) {
   513→  const url = req.method === 'GET' ? req.query.url : req.body?.url;
   514→  if (!url || typeof url !== 'string') {
   515→    return res.status(400).json({ error: 'Missing url' });
   516→  }
   517→  const options =
   518→    req.method === 'GET'
   519→      ? { type: req.query.type, externalId: req.query.externalId }
   520→      : { type: req.body?.type, externalId: req.body?.externalId };
   521→  try {
   522→    const { extraction, record } = await mineUrl(url, options);
   523→    invalidateHttpCache();
   524→    // TS1: targeted id event for the incremental searchdocs sync (upsert path).
   525→    if (record?.id) events.emit('content:changed:ids', { ids: [record.id], action: 'update' });
   526→    await auditLog({ userid: req.user?.id ?? null, action: 'miner.run', ip: clientIpOf(req), meta: { url, id: record?.id } });
   527→    return res.json({ ok: true, extraction, record });
   528→  } catch (err) {
   529→    return res.status(err.status || 502).json({ error: err.message || 'Mining failed' });
   530→  }
   531→}

   536→app.post('/api/miner', requireAuth, wrap(handleMine));
