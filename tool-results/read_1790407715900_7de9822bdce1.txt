   108→/** mm:ss or h:mm:ss. */
   109→function fmtTime(sec: number): string {
   110→  const n = Number.isFinite(sec) && sec > 0 ? Math.floor(sec) : 0;
   111→  const s = n % 60;
   112→  const m = Math.floor(n / 60) % 60;
   113→  const h = Math.floor(n / 3600);
   114→  const p = (v: number) => String(v).padStart(2, '0');
   115→  return h > 0 ? `${h}:${p(m)}:${p(s)}` : `${m}:${p(s)}`;
   116→}

   225→/** Thin custom volume slider — pointer-driven, commits on release so the
   226→ *  settings store is not hammered 60×/s while dragging. */
   227→function VolumeBar({
   228→  value,
   229→  onChange,
   230→}: {
   231→  value: number;
   232→  onChange: (v: number) => void;
   233→}) {
   234→  const ref = useRef<HTMLDivElement>(null);
   235→  const [drag, setDrag] = useState<number | null>(null);
   236→  const shown = drag ?? value;
   237→
   238→  const ratioFrom = useCallback((clientX: number): number => {
   239→    const el = ref.current;
   240→    if (!el) return 0;
   241→    const rect = el.getBoundingClientRect();
   242→    if (rect.width <= 0) return 0;
   243→    return clamp01((clientX - rect.left) / rect.width);
   244→  }, []);
   245→
   246→  return (
   247→    <div
   248→      ref={ref}
   249→      role="slider"
   250→      aria-label="Volum"
   251→      aria-valuemin={0}
   252→      aria-valuemax={100}
   253→      aria-valuenow={Math.round(shown * 100)}
   254→      tabIndex={0}
   255→      className="relative h-9 w-16 cursor-pointer touch-none sm:w-20"
   256→      onPointerDown={(e) => {
   257→        e.currentTarget.setPointerCapture(e.pointerId);
   258→        setDrag(ratioFrom(e.clientX));
   259→      }}
   260→      onPointerMove={(e) => {
   261→        if (drag != null) setDrag(ratioFrom(e.clientX));
   262→      }}
   263→      onPointerUp={(e) => {
   264→        if (drag != null) onChange(drag);
   265→        setDrag(null);
   266→        try {
   267→          e.currentTarget.releasePointerCapture(e.pointerId);
   268→        } catch {
   269→          /* pointer already released */
   270→        }
   271→      }}
   272→    >
   273→      <div className="absolute inset-x-0 top-1/2 h-1 -translate-y-1/2 overflow-hidden rounded-full bg-white/25">
   274→        <div
   275→          className="h-full rounded-full bg-rose-500"
   276→          style={{ width: `${clamp01(shown) * 100}%` }}
   277→        />
   278→      </div>
   279→    </div>
   280→  );
   281→}

   329→  /* ---- HLS quality levels (original hls.js level indices are preserved —
   330→     duplicate heights are collapsed for the menu, but `index` is the raw

   346→
   347→  const hideTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   348→  const clickTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   349→  const seekedInitialRef = useRef(false);

   453→          hls.on(HlsCtor.Events.ERROR, (_event, data) => {
   454→            if (cancelled) return;
   455→            if (data?.fatal) {
   456→              try {
   457→                hls.destroy();
   458→              } catch {
   459→                /* already destroyed */
   460→              }
   461→              onError?.('eroare HLS');
   462→            } else {
   463→              try {
   464→                hls.startLoad();
   465→              } catch {
   466→                /* retry not possible — the load watchdog (player-view) advances */
   467→              }
   468→            }
   469→          });
   470→          /* FAZA 19 — quality levels for the custom quality popover. */
   471→          hls.on(HlsCtor.Events.MANIFEST_PARSED, (_event, data) => {
   472→            if (cancelled) return;
   473→            const seen = new Set<number>();
   474→            const list: { index: number; height: number }[] = [];
   475→            data.levels.forEach((level, i) => {
   476→              const height = Number(level?.height) || 0;
   477→              if (height <= 0 || seen.has(height)) return;
   478→              seen.add(height);
   479→              list.push({ index: i, height });
   480→            });
   481→            list.sort((a, b) => b.height - a.height);
   482→            setLevels(list);
   483→          });

   485→            if (cancelled) return;
   486→            setCurrentLevel(Number(data?.level));
   487→          });

   489→          hls.attachMedia(el);
   490→          instance = hls;
   491→          hlsRef.current = hls;
   492→        } else if (el.canPlayType('application/vnd.apple.mpegurl')) {
   493→          el.src = src;
   494→        } else {
   495→          onError?.('eroare HLS');
   496→        }
   497→      } catch {
   498→        if (!cancelled) onError?.('eroare HLS');
   499→      }
   500→    })();
   501→    return () => {
   502→      cancelled = true;
   503→      hlsRef.current = null;
   504→      try {
   505→        instance?.destroy();
   506→      } catch {
   507→        /* already destroyed */
   508→      }
   509→    };

   526→    if (!el || el.buffered.length === 0) {
   527→      setBufferedRanges((prev) => (prev.length === 0 ? prev : []));
   528→      return;
   529→    }
   530→    const out: { start: number; end: number }[] = [];
   531→    for (let i = 0; i < el.buffered.length; i += 1) {
   532→      out.push({ start: el.buffered.start(i), end: el.buffered.end(i) });
   533→    }
   534→    setBufferedRanges(out);
   535→  }, []);

   548→    }
   549→  }, [failed]);

   557→  }, [pokeUi]);
   558→

   561→    updatePlayerSettings({ volume: vol, muted: vol > 0 ? false : getPlayerSettings().muted });
   562→  }, []);
   563→
   564→  const nudgeVolume = useCallback((delta: number) => {
