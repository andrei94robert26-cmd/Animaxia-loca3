   135→export class ApiError extends Error {
   136→  status: number;
   137→  constructor(message: string, status: number) {
   138→    super(message);
   139→    this.name = 'ApiError';
   140→    this.status = status;
   141→  }
   142→}

   144→type ParamValue = string | number | boolean | undefined | null;
   145→
   146→export function apiUrl(path: string, params?: Record<string, ParamValue>): string {
   147→  const search = new URLSearchParams();
   148→  search.set('XTransformPort', String(BACKEND_PORT));
   149→  if (params) {
   150→    for (const [key, value] of Object.entries(params)) {
   151→      if (value !== undefined && value !== null && value !== '') search.set(key, String(value));
   152→    }
   153→  }
   154→  return `${path}?${search.toString()}`;
   155→}

   157→/** FAZA 23 (observability): same gateway pattern as apiUrl(), but pointed at

   159→export function obsUrl(path: string, params?: Record<string, ParamValue>): string {
   160→  const search = new URLSearchParams();
   161→  search.set('XTransformPort', String(OBSERVABILITY_PORT));
   162→  if (params) {
   163→    for (const [key, value] of Object.entries(params)) {
   164→      if (value !== undefined && value !== null && value !== '') search.set(key, String(value));
   165→    }
   166→  }
   167→  return `${path}?${search.toString()}`;
   168→}

   170→async function request<T>(path: string, init?: RequestInit): Promise<T> {
   171→  let res: Response;
   172→  try {
   173→    res = await fetch(path, {
   174→      ...init,
   175→      headers: { 'Content-Type': 'application/json', ...(init?.headers as Record<string, string> | undefined) },
   176→    });
   177→  } catch {
   178→    throw new ApiError('Cannot reach the Animaxia backend — it may be restarting', 0);
   179→  }
   180→  let body: unknown = null;
   181→  try {
   182→    body = await res.json();
   183→  } catch {
   184→    body = null;
   185→  }
   186→  if (!res.ok) {
   187→    const message =
   188→      (body as { error?: string } | null)?.error || `Request failed with status ${res.status}`;
   189→    throw new ApiError(message, res.status);
   190→  }
   191→  return body as T;
   192→}

   194→/** Bearer header for FAZA 16 auth endpoints (same storage key as platform.ts
   195→ *  tokenStore — read directly here to avoid a circular import). */
   196→function authHeaders(): Record<string, string> {
   197→  try {
   198→    const token = window.localStorage.getItem('animaxia.accessToken');
   199→    return token ? { Authorization: `Bearer ${token}` } : {};
   200→  } catch {
   201→    return {};
   202→  }
   203→}

   801→export function buildVirtualCursor(index: number, seed: string): string {
   802→  const json = JSON.stringify({ i: index, s: seed });
   803→  const b64 =
   804→    typeof btoa === 'function'
   805→      ? btoa(json)
   806→      : Buffer.from(json, 'utf8').toString('base64');
   807→  return b64.replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
   808→}

   813→  const md = c.metadata || {};

   827→    playable: Boolean(sources && sources.length > 0),
