    85→export function PartyRoom({ partyId }: { partyId: number }) {
    86→  const { user } = useAuth();
    87→  const validId = Number.isInteger(partyId) && partyId > 0;
    88→
    89→  const detail = useApi(`social-party-${partyId}`, () => social.partyDetail(partyId), { enabled: validId });
    90→  const current = detail.data && detail.data.party.id === partyId ? detail.data : null;
    91→
    92→  const [ended, setEnded] = useState(false);
    93→  const [members, setMembers] = useState<SocketMemberRow[]>([]);
    94→  const [messages, setMessages] = useState<ChatMessage[]>([]);
    95→  const [typingUser, setTypingUser] = useState<string | null>(null);
    96→  const [socketConnected, setSocketConnected] = useState(false);
    97→  const [lastPersisted, setLastPersisted] = useState<boolean | null>(null);
    98→  const [leaving, setLeaving] = useState(false);
    99→  const [copied, setCopied] = useState(false);
   100→  const [chatInput, setChatInput] = useState('');
   101→  const [lastState, setLastState] = useState<PartyState | null>(null);
   102→
   103→  const socketRef = useRef<Socket | null>(null);
   104→  const videoRef = useRef<HTMLVideoElement>(null);
   105→  const chatScrollRef = useRef<HTMLDivElement>(null);
   106→  const typingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   107→  const lastTypingEmitRef = useRef(0);
   108→  const lastStateRef = useRef<PartyState | null>(null);
   109→  const seededRef = useRef(false);
   110→  const isHostRef = useRef(false);
   111→
   112→  const party = current?.party ?? null;
   113→  const isHost = Boolean(user && party && party.host.id === user.id);
   114→  const partyActive = Boolean(party && party.status === 'active') && !ended;
   115→
   116→  /* Player sources — same client and rendering rules as player-view (with a
   117→     source switcher: the host's switch broadcasts a `source` control). */
   118→  const contentId = party?.content.id ?? 0;
   119→  const player = useApi(`party-player-${contentId}`, () => api.player(contentId), { enabled: contentId > 0 });
   120→  const playerData = player.data && player.data.content.id === contentId ? player.data : null;
   121→  const [sourceIndex, setSourceIndex] = useState(0);
   122→  const sources = playerData?.sources ?? [];
   123→  const source: PlayerSource | null =
   124→    sources.length > 0 ? sources[Math.min(sourceIndex, sources.length - 1)] : null;
   125→
   126→  /* Reset the picked source when the underlying player payload changes. */
   127→  useEffect(() => {
   128→    setSourceIndex(0);
   129→  }, [playerData]);
   130→
   131→  /* TS2b: mini-fallback chain — on a native <video> error (or a 12s
   132→     watchdog when the source never delivers data) advance LOCALLY to the
   133→     next source; when the last one dies, an honest recovery card takes over
   134→     below the player. Host sync/chat logic is untouched — the advance is
   135→     local-only for every member (each client retries independently). */
   136→  const [sourceFailed, setSourceFailed] = useState(false);
   137→
   138→  useEffect(() => {
   139→    setSourceFailed(false);
   140→  }, [playerData, sourceIndex]);
   141→
   142→  function advanceAfterFailure(reason: string): void {
   143→    if (sourceIndex < sources.length - 1) {
   144→      toast(`Sursa ${sourceIndex + 1} nu răspunde (${reason}) — trec la următoarea…`);
   145→      setSourceIndex((i) => Math.min(i + 1, sources.length - 1));
   146→    } else {
   147→      setSourceFailed(true);
   148→    }
   149→  }
   150→
   151→  /* 12s watchdog — fires only when a native source never reached
   152→     readyState ≥ 2 (HAVE_CURRENT_DATA): a paused-but-loaded video passes. */
   153→  useEffect(() => {
   154→    if (!source || (source.type !== 'mp4' && source.type !== 'm3u8')) return undefined;
   155→    const kind = source.type;
   156→    const timer = setTimeout(() => {
   157→      const v = videoRef.current;
   158→      if (!v || v.readyState < 2) {
   159→        advanceAfterFailure(kind === 'mp4' ? 'fără date mp4' : 'fără date m3u8');
   160→      }
   161→    }, 12_000);
   162→    return () => clearTimeout(timer);
   163→  }, [source?.url, source?.type, sourceIndex, sources.length]);
   164→
   165→  /* Sync isHost into a ref so socket handlers (registered once) stay fresh. */
   166→  useEffect(() => {
   167→    isHostRef.current = isHost;
   168→  }, [isHost]);
   169→
   170→  /* Seed local state from the REST detail (members / messages / state). */
   171→  useEffect(() => {
   172→    if (!current) return;
   173→    setMembers(current.members.map((m) => ({ userid: m.userid, username: m.username, avatar: m.avatar, online: false })));
   174→    setMessages(current.messages ?? []);
   175→    setLastState(current.state);
   176→    lastStateRef.current = current.state;
   177→    seededRef.current = false;
   178→  }, [current]);
   179→
   180→  /* Apply a party state to the <video> element (seek + pause/play + rate). */
   181→  function applyStateToVideo(state: PartyState): void {
   182→    const v = videoRef.current;
   183→    if (!v || !Number.isFinite(state.positionSec) || state.positionSec < 0) return;
   184→    try {
   185→      if (Math.abs(v.currentTime - state.positionSec) > 0.3 || v.readyState < 1) {
   186→        v.currentTime = state.positionSec;
   187→      }
   188→    } catch {
   189→      /* seeking before metadata is loaded — the loadedmetadata seed retries */
   190→      return;
   191→    }
   192→    if (Number.isFinite(state.playbackRate) && state.playbackRate > 0) v.playbackRate = state.playbackRate;
   193→    if (state.isPaused) v.pause();
   194→    else v.play().catch(() => {});
   195→  }
   196→
   197→  /* Seed the video position once the join ack (or REST state) arrives. */
   198→  function seedFromState(state: PartyState): void {
   199→    lastStateRef.current = state;
   200→    setLastState(state);
   201→    if (seededRef.current) return;
   202→    seededRef.current = true;
   203→    if (state.positionSec <= 0.5 && state.isPaused) return; // nothing to seed
   204→    applyStateToVideo(state);
   205→  }
   206→
   207→  /* Realtime socket — created for signed-in users while the party is active. */
   208→  useEffect(() => {
   209→    if (!user || !partyActive || !validId) return;
   210→
   211→    /* Idempotent REST join (also keeps membership when the socket degrades). */
   212→    social.partyJoin(partyId).catch(() => {
   213→      /* silent — the socket join below retries membership too */
   214→    });
   215→
   216→    const socket = createSocialSocket();
   217→    socketRef.current = socket;
   218→
   219→    /* The realtime service finishes its per-connection bootstrap (an async
   220→       profile fetch) before it registers the `party:join` handler — a join
   221→       emitted the instant `connect` fires can land too early and be dropped.
   222→       Join is idempotent server-side, so: emit immediately, re-emit on `hello`
   223→       (server sends it right after bootstrap) and retry until the join ack
   224→       (`party:state`) arrives. */
   225→    let joinTimer: ReturnType<typeof setTimeout> | null = null;
   226→    let joinAttempts = 0;
   227→    let stateAcked = false;
   228→    const emitJoin = (): void => {
   229→      socket.emit('party:join', { partyId });
   230→      joinAttempts += 1;
   231→      if (!stateAcked && joinAttempts < 6 && !joinTimer) {
   232→        joinTimer = setTimeout(() => {
   233→          joinTimer = null;
   234→          if (!stateAcked) emitJoin();
   235→        }, 1500);
   236→      }
   237→    };
   238→
   239→    socket.on('connect', () => {
   240→      setSocketConnected(true);
   241→      stateAcked = false;
   242→      if (joinTimer) {
   243→        clearTimeout(joinTimer);
   244→        joinTimer = null;
   245→      }
   246→      joinAttempts = 0;
   247→      emitJoin();
   248→    });
   249→    /* `hello` arrives after the server registered its handlers — a join sent
   250→       here always reaches the listener (covers both first connect and any
   251→       reconnect). */
   252→    socket.on('hello', () => {
   253→      if (!stateAcked) emitJoin();
   254→    });
   255→    socket.on('disconnect', () => setSocketConnected(false));
   256→    socket.on('connect_error', (err: Error) => {
   257→      setSocketConnected(false);
   258→      /* Auth rejected (e.g. the access token expired mid-session): rotate
   259→         once and re-arm the handshake for the next reconnection attempt. */
   260→      if (!/token/i.test(err.message)) return;
   261→      void (async () => {
   262→        if (tokenStore.refresh) await tryRefresh().catch(() => {});
   263→        socket.auth = tokenStore.access ? { token: tokenStore.access } : {};
   264→      })();
   265→    });
   266→
   267→    socket.on('party:state', (d: PartyStateEvent) => {
   268→      if (d.partyId !== partyId) return;
   269→      stateAcked = true;
   270→      if (joinTimer) {
   271→        clearTimeout(joinTimer);
   272→        joinTimer = null;
   273→      }
   274→      setMembers(d.members ?? []);
   275→      seedFromState(d.state);
   276→      if (d.degraded) toast('Serviciul realtime este în modul offline — este posibil ca mesajele să nu fie salvate');
   277→    });
   278→
   279→    socket.on('party:member-joined', (d: { partyId: number; user: SocketMemberRow }) => {
   280→      if (d.partyId !== partyId) return;
   281→      setMembers((prev) =>
   282→        prev.some((m) => m.userid === d.user.userid)
   283→          ? prev.map((m) => (m.userid === d.user.userid ? { ...m, online: true } : m))
   284→          : [...prev, { ...d.user, online: true }],
   285→      );
   286→    });
   287→
   288→    socket.on('party:member-left', (d: { partyId: number; user: { userid: number } }) => {
   289→      if (d.partyId !== partyId) return;
   290→      setMembers((prev) => prev.filter((m) => m.userid !== d.user.userid));
   291→    });
   292→
   293→    socket.on('party:control', (d: PartyControlEvent) => {
   294→      if (d.partyId !== partyId) return;
   295→      lastStateRef.current = d.state;
   296→      setLastState(d.state);
   297→      if (isHostRef.current) return; // the host drives — never loop back
   298→      if (d.action === 'source' && Number.isInteger(d.state.sourceIndex) && d.state.sourceIndex >= 0) {
   299→        setSourceIndex(d.state.sourceIndex);
   300→      }
   301→      const v = videoRef.current;
   302→      if (!v) return;
   303→      const drift = Number.isFinite(d.state.positionSec) ? Math.abs(v.currentTime - d.state.positionSec) : 0;
   304→      if (d.action === 'seek' || d.action === 'source' || drift > 1.5) {
   305→        try {
   306→          v.currentTime = d.state.positionSec;
   307→        } catch {
   308→          /* not seekable yet */
   309→        }
   310→      }
   311→      if (Number.isFinite(d.state.playbackRate) && d.state.playbackRate > 0 && d.state.playbackRate !== v.playbackRate) {
   312→        v.playbackRate = d.state.playbackRate;
   313→      }
   314→      if (d.state.isPaused && !v.paused) v.pause();
   315→      else if (!d.state.isPaused && v.paused) v.play().catch(() => {});
   316→    });
   317→
   318→    socket.on('party:chat', (d: PartyChatEvent) => {
   319→      if (d.partyId !== partyId) return;
   320→      setLastPersisted(d.persisted !== false);
   321→      setTypingUser((current) => (current === d.message.user.username ? null : current));
   322→      setMessages((prev) => {
   323→        if (d.message.id != null && prev.some((m) => m.id === d.message.id)) return prev;
   324→        const next = [...prev, d.message];
   325→        return next.length > 200 ? next.slice(next.length - 200) : next;
   326→      });
   327→    });
   328→
   329→    socket.on('party:typing', (d: { partyId: number; user: { userid: number; username: string } }) => {
   330→      if (d.partyId !== partyId || !d.user) return;
   331→      if (user && d.user.userid === user.id) return;
   332→      setTypingUser(d.user.username);
   333→      if (typingTimerRef.current) clearTimeout(typingTimerRef.current);
   334→      typingTimerRef.current = setTimeout(() => setTypingUser(null), 3000);
   335→    });
   336→
   337→    socket.on('party:error', (d: { code: string; message: string }) => {
   338→      if (d.code === 'ended') {
   339→        setEnded(true);
   340→        toast('Această vizionare în grup s-a încheiat');
   341→        return;
   342→      }
   343→      if (d.code === 'auth') return; // anon flow is handled by the sign-in prompt
   344→      toast.error(d.message || 'Eroare de vizionare în grup');
   345→    });
   346→
   347→    return () => {
   348→      if (joinTimer) clearTimeout(joinTimer);
   349→      socket.removeAllListeners();
   350→      socket.disconnect();
   351→      socketRef.current = null;
   352→      setSocketConnected(false);
   353→    };
   354→  }, [user?.id, partyId, partyActive, validId, user]);
   355→  /* NOTE: `user` is listed for identity freshness (same id, refreshed profile);
   356→     the effect only ever creates one socket at a time. */
   357→
   358→  /* Host: periodic state snapshot to the monolith (fire-and-forget, 10s). */
   359→  useEffect(() => {
   360→    if (!isHost || !partyActive) return;
   361→    const timer = setInterval(() => {
   362→      const v = videoRef.current;
   363→      if (!v) return;
   364→      social
   365→        .partySync(partyId, {
   366→          positionSec: v.currentTime,
   367→          isPaused: v.paused,
   368→          playbackRate: Number.isFinite(v.playbackRate) && v.playbackRate > 0 ? v.playbackRate : 1,
   369→          sourceIndex,
   370→          updatedAt: new Date().toISOString(),
   371→        })
   372→        .then((res) => {
   373→          lastStateRef.current = res.state;
   374→        })
   375→        .catch(() => {});
   376→    }, 10_000);
   377→    return () => clearInterval(timer);
   378→  }, [isHost, partyActive, partyId, sourceIndex]);
   379→
   380→  /* HLS (m3u8) sources — identical to player-view's dynamic hls.js attach. */
   381→  useEffect(() => {
   382→    const url = source?.url;
   383→    const el = videoRef.current;
   384→    if (source?.type !== 'm3u8' || !url || !el) return;
   385→    let hls: { destroy: () => void } | null = null;
   386→    let cancelled = false;
   387→    (async () => {
   388→      try {
   389→        const mod = await import('hls.js');
   390→        const Hls = mod.default;
   391→        if (cancelled) return;
   392→        if (Hls.isSupported()) {
   393→          const instance = new Hls({ enableWorker: true });
   394→          instance.loadSource(url);
   395→          instance.attachMedia(el);
   396→          hls = instance;
   397→        } else if (el.canPlayType('application/vnd.apple.mpegurl')) {
   398→          el.src = url;
   399→        }
   400→      } catch {
   401→        /* hls failed to init */
   402→      }
   403→    })();
   404→    return () => {
   405→      cancelled = true;
   406→      hls?.destroy();
   407→    };
   408→  }, [source?.type, source?.url]);
   409→
   410→  /* Auto-scroll chat to the newest message. */
   411→  useEffect(() => {
   412→    const el = chatScrollRef.current;
   413→    if (el) el.scrollTop = el.scrollHeight;
   414→  }, [messages.length, typingUser]);
   415→
   416→  /* Clear the typing timer on unmount. */
   417→  useEffect(() => {
   418→    return () => {
   419→      if (typingTimerRef.current) clearTimeout(typingTimerRef.current);
   420→    };
   421→  }, []);
   422→
   423→  /* ------------------------------ host controls ----------------------------- */
   424→
   425→  function emitControl(action: 'play' | 'pause' | 'seek' | 'rate' | 'source', overrideIndex?: number): void {
   426→    const socket = socketRef.current;
   427→    const v = videoRef.current;
   428→    if (!socket || !isHost) return;
   429→    socket.emit('party:control', {
   430→      partyId,
   431→      action,
   432→      positionSec: v?.currentTime ?? 0,
   433→      playbackRate: v?.playbackRate ?? 1,
   434→      sourceIndex: overrideIndex ?? sourceIndex,
   435→    });
   436→  }
   437→
   438→  /** Host picks a different stream source — members follow via `source`. */
   439→  function switchSource(nextIndex: number): void {
   440→    setSourceIndex(nextIndex);
   441→    if (isHost) emitControl('source', nextIndex);
   442→  }
   443→
   444→  /* --------------------------------- actions -------------------------------- */
   445→
   446→  function sendChat(): void {
   447→    const body = chatInput.trim();
   448→    if (!body || !user) return;
   449→    setChatInput('');
   450→    const socket = socketRef.current;
   451→    if (socket && socket.connected) {
   452→      socket.emit('party:chat', { partyId, body });
   453→      return;
   454→    }
   455→    /* REST fallback (socket down): optimistic append, replaced by the reply. */
   456→    const temp: ChatMessage = {
   457→      id: `tmp-${Date.now()}`,
   458→      body,
   459→      createdat: new Date().toISOString(),
   460→      user: { id: user.id, username: user.username, avatar: user.avatar },
   461→    };
   462→    setMessages((prev) => [...prev, temp]);
   463→    social
   464→      .partySendMessage(partyId, body)
   465→      .then((res) => {
   466→        setLastPersisted(true);
   467→        setMessages((prev) => prev.map((m) => (m.id === temp.id ? res.message : m)));
   468→      })
   469→      .catch((err) => {
   470→        setMessages((prev) => prev.filter((m) => m.id !== temp.id));
   471→        toast.error(err instanceof Error ? err.message : 'Nu am putut trimite mesajul');
   472→      });
   473→  }
   474→
   475→  function handleChatInputChange(value: string): void {
   476→    setChatInput(value);
   477→    const socket = socketRef.current;
   478→    const now = Date.now();
   479→    if (socket && socket.connected && now - lastTypingEmitRef.current > 2000) {
   480→      lastTypingEmitRef.current = now;
   481→      socket.emit('party:typing', { partyId });
   482→    }
   483→  }
   484→
   485→  async function copyInviteCode(): Promise<void> {
   486→    if (!party?.code) return;
   487→    try {
   488→      await navigator.clipboard.writeText(party.code);
   489→      setCopied(true);
   490→      toast.success('Codul de invitație a fost copiat');
   491→      setTimeout(() => setCopied(false), 1500);
   492→    } catch {
   493→      toast.error('Clipboardul nu este disponibil în acest browser');
   494→    }
   495→  }
   496→
   497→  async function leaveParty(): Promise<void> {
   498→    if (leaving) return;
   499→    setLeaving(true);
   500→    try {
   501→      socketRef.current?.emit('party:leave', { partyId });
   502→      await social.partyLeave(partyId);
   503→      if (isHost) toast('Vizionarea în grup s-a încheiat');
   504→      else toast('Ai părăsit vizionarea în grup');
   505→    } catch (err) {
   506→      toast.error(err instanceof Error ? err.message : 'Nu am putut părăsi vizionarea în grup');
   507→    } finally {
   508→      navigate('/social');
   509→    }
   510→  }
   511→
   512→  /* -------------------------------- rendering ------------------------------- */
   513→
   514→  if (!validId) {
   515→    return (
   516→      <Container className="py-20">
   517→        <EmptyState
   518→          title="Vizionare în grup negăsită"
   519→          message="URL-ului îi lipsește un id de vizionare valid."
   520→          action={
   521→            <Button className="bg-rose-600 font-bold hover:bg-rose-500" onClick={() => navigate('/social')}>
   522→              Back to Social
   523→            </Button>
   524→          }
   525→        />
   526→      </Container>
   527→    );
   528→  }
   529→
   530→  if (detail.loading && !current) {
   531→    return (
   532→      <Container className="py-8 sm:py-10">
   533→        <div className="mx-auto w-full max-w-6xl space-y-5">
   534→          <div className="flex items-center gap-3">
   535→            <Skeleton className="h-9 w-9 rounded-lg" />
   536→            <div className="flex-1 space-y-2">
   537→              <Skeleton className="h-6 w-72 max-w-full" />
