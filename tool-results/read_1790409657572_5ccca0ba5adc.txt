   140→export class ApiError extends Error {
   141→  status: number;
   142→  constructor(message: string, status: number) {
   143→    super(message);
   144→    this.name = 'ApiError';
   145→    this.status = status;
   146→  }
   147→}

   149→type ParamValue = string | number | boolean | undefined | null;
   150→
   151→export function apiUrl(path: string, params?: Record<string, ParamValue>): string {
   152→  const search = new URLSearchParams();
   153→  search.set('XTransformPort', String(BACKEND_PORT));
   154→  if (params) {
   155→    for (const [key, value] of Object.entries(params)) {
   156→      if (value !== undefined && value !== null && value !== '') search.set(key, String(value));
   157→    }
   158→  }
   159→  return `${path}?${search.toString()}`;
   160→}

   162→/** FAZA 23 (observability): same gateway pattern as apiUrl(), but pointed at

   164→export function obsUrl(path: string, params?: Record<string, ParamValue>): string {
   165→  const search = new URLSearchParams();
   166→  search.set('XTransformPort', String(OBSERVABILITY_PORT));
   167→  if (params) {
   168→    for (const [key, value] of Object.entries(params)) {
   169→      if (value !== undefined && value !== null && value !== '') search.set(key, String(value));
   170→    }
   171→  }
   172→  return `${path}?${search.toString()}`;
   173→}

   175→async function request<T>(path: string, init?: RequestInit): Promise<T> {
   176→  let res: Response;
   177→  try {
   178→    res = await fetch(path, {
   179→      ...init,
   180→      headers: { 'Content-Type': 'application/json', ...(init?.headers as Record<string, string> | undefined) },
   181→    });
   182→  } catch {
   183→    throw new ApiError('Cannot reach the Animaxia backend — it may be restarting', 0);
   184→  }
   185→  let body: unknown = null;
   186→  try {
   187→    body = await res.json();
   188→  } catch {
   189→    body = null;
   190→  }
   191→  if (!res.ok) {
   192→    const message =
   193→      (body as { error?: string } | null)?.error || `Request failed with status ${res.status}`;
   194→    throw new ApiError(message, res.status);
   195→  }
   196→  return body as T;
   197→}

   199→/** Bearer header for FAZA 16 auth endpoints (same storage key as platform.ts
   200→ *  tokenStore — read directly here to avoid a circular import). */
   201→function authHeaders(): Record<string, string> {
   202→  try {
   203→    const token = window.localStorage.getItem('animaxia.accessToken');
   204→    return token ? { Authorization: `Bearer ${token}` } : {};
   205→  } catch {
   206→    return {};
   207→  }
   208→}

   858→export function buildVirtualCursor(index: number, seed: string): string {
   859→  const json = JSON.stringify({ i: index, s: seed });
   860→  const b64 =
   861→    typeof btoa === 'function'
   862→      ? btoa(json)
   863→      : Buffer.from(json, 'utf8').toString('base64');
   864→  return b64.replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
   865→}

   869→export function dbToItem(c: DbContent): NormalizedItem {
   870→  const md = c.metadata || {};
   871→  const sources = md.player?.sources || md.sources;
   872→  return {
   873→    key: `db-${c.id}`,
   874→    kind: 'db',
   875→    dbId: c.id,
   876→    title: c.title,
   877→    image: c.thumbnail || md.backdrop || null,
   878→    backdrop: md.backdrop || null,
   879→    overview: md.overview || md.synopsis || md.description || undefined,
   880→    rating: typeof md.rating === 'number' ? md.rating : typeof md.score === 'number' ? md.score : null,
   881→    year: (md.year as string | number | undefined) ?? null,
   882→    type: c.type || 'mined',
   883→    genres: Array.isArray(md.genres) ? md.genres.map(String) : undefined,
   884→    playable: Boolean(sources && sources.length > 0),
   885→    source: typeof md.source === 'string' ? md.source : undefined,
   886→  };
   887→}

   951→/**
   952→ * FAZA 15: AI infinite-feed item → NormalizedItem.
   953→ * mediaType 'db' is a Neon library row; anything else is a live external
   954→ * provider item (movie / tv / anime) with the id as a string extId.
   955→ */
   956→export function feedToItem(i: AiFeedItem): NormalizedItem {
   957→  const isDb = i.mediaType === 'db';
   958→  const extType = isDb ? undefined : (i.mediaType as 'movie' | 'tv' | 'anime');
   959→  return {
   960→    key: isDb ? `db-${i.id}` : `${i.mediaType}-${i.id}`,
   961→    kind: isDb ? 'db' : 'external',
   962→    dbId: isDb ? i.id : undefined,
   963→    extType,
   964→    extId: isDb ? undefined : String(i.id),
   965→    title: i.title,
   966→    image: i.image || i.backdrop || null,
   967→    backdrop: i.backdrop || null,
   968→    overview: i.overview || undefined,
   969→    rating: i.rating ?? null,
   970→    year: i.year ?? null,
   971→    type: isDb ? 'mined' : i.mediaType,
   972→    genres: Array.isArray(i.genres) && i.genres.length > 0 ? i.genres : undefined,
   973→    source: i.source,
   974→  };
   975→}
