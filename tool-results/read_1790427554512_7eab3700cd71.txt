    97→/* ------------------------ file-based JSON store (zero DB) -------------------- */
    98→/* Cache in-memory + scriere atomică debounced (tmp+rename — un cititor nu vede
    99→ * niciodată jumătate de fișier). Fișierele sunt create LENEȘ (la prima mutație);
   100→ * un fișier corupt ⇒ pornim de la zero + avertisment (recoverable). */
   101→
   102→class JsonFileStore {
   103→  constructor(name) {
   104→    this.file = join(STORAGE_DIR, `${name}.json`);
   105→    this.name = name;
   106→    this.data = undefined; // undefined = neîncărcat; {} = încărcat/gol
   107→    this.writeTimer = null;
   108→  }
   109→
   110→  /** Încarcă o singură dată (sincron, la prima nevoie); corupt ⇒ {} + warn. */
   111→  load() {
   112→    if (this.data !== undefined) return this.data;
   113→    try {
   114→      const raw = readFileSync(this.file, 'utf8');
   115→      const parsed = JSON.parse(raw);
   116→      this.data = parsed && typeof parsed === 'object' && !Array.isArray(parsed) ? parsed : {};
   117→    } catch (err) {
   118→      if (err && err.code === 'ENOENT') {
   119→        this.data = {}; // nu există încă — creat leneș la prima scriere
   120→      } else {
   121→        logger.warn(`[playerx] ${this.name}.json corupt/ineligibil — pornesc de la zero (${err?.message ?? err})`);
   122→        this.data = {};
   123→      }
   124→    }
   125→    return this.data;
   126→  }
   127→
   128→  /** Programează o scriere atomică (debounce) după o mutație in-memory. */
   129→  scheduleSave() {
   130→    if (this.writeTimer) return;
   131→    this.writeTimer = setTimeout(() => {
   132→      this.writeTimer = null;
   133→      this.flush();
   134→    }, WRITE_DEBOUNCE_MS);
   135→    if (typeof this.writeTimer.unref === 'function') this.writeTimer.unref();
   136→  }
   137→
   138→  /** Scriere atomică: tmp + rename. Niciodată throw. */
   139→  flush() {
   140→    if (this.data === undefined) return; // nimic încărcat ⇒ nimic de persistat
   141→    try {
   142→      mkdirSync(STORAGE_DIR, { recursive: true });
   143→      const tmp = `${this.file}.${process.pid}.tmp`;
   144→      writeFileSync(tmp, `${JSON.stringify(this.data, null, 2)}\n`);
   145→      renameSync(tmp, this.file);
   146→    } catch (err) {
   147→      logger.warn(`[playerx] persistența ${basename(this.file)} a eșuat (non-fatal): ${err?.message ?? err}`);
   148→    }
   149→  }
   150→}

   189→/* ------------------------------ generic LRU cache --------------------------- */
   190→
   191→function makeLru(limit) {
   192→  return {
   193→    map: new Map(), // key → { value, storedAt } · Map = insertion order = LRU
   194→    get(key) {
   195→      const entry = this.map.get(key);
   196→      if (!entry) return null;
   197→      if (Date.now() - entry.storedAt > CACHE_TTL_MS) {
   198→        this.map.delete(key);
   199→        return null;
   200→      }
   201→      this.map.delete(key);
   202→      this.map.set(key, entry); // refresh recency
   203→      return entry.value;
   204→    },
   205→    set(key, value) {
   206→      if (this.map.has(key)) this.map.delete(key);
   207→      else if (this.map.size >= limit) {
   208→        const oldest = this.map.keys().next().value;
   209→        if (oldest !== undefined) this.map.delete(oldest);
   210→      }
   211→      this.map.set(key, { value, storedAt: Date.now() });
   212→    },
   213→    invalidate(key) {
   214→      this.map.delete(key);
   215→    },
   216→    size() {
   217→      return this.map.size;
   218→    },
   219→  };
   220→}

   226→const useBuckets = new Map(); // ip → { count, windowStart }
   227→
   228→function useRateAllowed(ip) {
   229→  const now = Date.now();
   230→  let bucket = useBuckets.get(ip);
   231→  if (!bucket || now - bucket.windowStart >= 60_000) {
   232→    bucket = { count: 0, windowStart: now };
   233→    useBuckets.set(ip, bucket);
   234→  }
   235→  bucket.count += 1;
   236→  if (bucket.count > RATE_PER_MIN) return false;
   237→  if (useBuckets.size > 5000) useBuckets.clear();
   238→  return true;
   239→}

   245→  }
   246→}, 60_000);

   305→/** Durata în secunde din metadata reală. Câmpuri citite (inspectate pe rândurile
   306→ * reale Neon, 2026-09-26): duration (secunde, string numeric — tv), runtime
   307→ * (minute, number — tv/movie), episodeRunTime/episode_runtime (minute, convenție
   308→ * TMDB — zero rânduri azi, citite pentru compatibilitate viitoare). */
   309→function durationSecOf(metadata) {
   310→  const meta = metadata && typeof metadata === 'object' ? metadata : {};
   311→  const dur = Number(meta.duration);
   312→  if (Number.isFinite(dur) && dur > 0) return dur; // secunde
   313→  const rt = Number(meta.runtime);
   314→  if (Number.isFinite(rt) && rt > 0) return rt * 60; // minute → secunde
   315→  const ert = Number(meta.episodeRunTime ?? meta.episode_runtime);
   316→  if (Number.isFinite(ert) && ert > 0) return ert * 60; // minute → secunde
   317→  return null;
   318→}

   322→function estimateMarkers(type, durationSec) {

   328→    }
   329→  } else if (type === 'tv') {

   331→    out.intro = { startsec: 48, endsec: 110 };

   333→      out.outro = { startsec: Math.max(170, durationSec - 90), endsec: durationSec - 6 };

   339→/** Un marker e valid doar cu startsec/endsec întregi, startsec<endsec, startsec≥0
   340→ * și o durată (endsec-startsec) între 5 și 600 de secunde. */
   341→function validMarkerSpan(startsec, endsec) {
   342→  return (
   343→    Number.isInteger(startsec) && Number.isInteger(endsec) &&
   344→    startsec >= 0 && startsec < endsec &&
   345→    endsec - startsec >= 5 && endsec - startsec <= 600
   346→  );
   347→}
   348→
   349→/** Merge estimări + override-uri curate: un kind curat ÎNLOCUIEȘTE estimatul
   350→ * aceluiași kind; kind-urile curate noi se ADAUGĂ. Sortate pe startsec.
   351→ * anyEstimated = cel puțin un marker ESTIMAT supraviețuiește în rezultatul
   352→ * final (dacă toate kind-urile estimate au override curat, nota dispare —
   353→ * contractul leagă nota de source, nu de existența estimărilor). */
   354→function computeMarkers(contentId, content) {
   355→  const estimated = estimateMarkers(content?.type, durationSecOf(content?.metadata));
   356→  const curatedAll = markersStore.load()[contentId] ?? {};
   357→  const byKind = {};
   358→  for (const [kind, marker] of Object.entries(estimated)) {
   359→    byKind[kind] = { ...marker, kind, source: 'estimated' };
   360→  }
   361→  for (const kind of MARKER_KINDS) {
   362→    const curated = curatedAll[kind];
   363→    if (!curated) continue;
   364→    const { startsec, endsec } = curated;
   365→    if (!validMarkerSpan(startsec, endsec)) {
   366→      logger.warn(`[playerx] marker curat sărit pentru content ${contentId} kind ${kind} (interval invalid: ${startsec}..${endsec})`);
   367→      continue;
   368→    }
   369→    byKind[kind] = {
   370→      startsec,
   371→      endsec,
   372→      kind,
   373→      label: typeof curated.label === 'string' && curated.label.trim()
   374→        ? curated.label.trim().slice(0, LABEL_MAX)
   375→        : KIND_LABELS[kind],
   376→      source: 'curated',
   377→    };
   378→  }
   379→  const data = Object.values(byKind)
   380→    .map((m) => ({
   381→      id: m.kind,
   382→      kind: m.kind,
   383→      startsec: m.startsec,
   384→      endsec: m.endsec,
   385→      label: m.source === 'estimated' ? KIND_LABELS[m.kind] : m.label,
   386→      source: m.source,
   387→    }))
   388→    .sort((a, b) => a.startsec - b.startsec || a.endsec - b.endsec);
   389→  return { data, anyEstimated: data.some((m) => m.source === 'estimated') };
   390→}

   618→/** Tally + votanți în ordinea în care au votat (seq incremental — determinist,
   619→ * supraviețuiește și round-trip-ului JSON, care păstrează ordinea cheilor). */
   620→function tallyOf(record) {
   621→  const tally = {};
   622→  const voters = [];
   623→  const entries = Object.entries(record?.votes ?? {}).slice().sort((a, b) => (a[1]?.seq ?? 0) - (b[1]?.seq ?? 0));
   624→  for (const [userid, vote] of entries) {
   625→    const cid = Number(vote?.contentId);
   626→    tally[cid] = (tally[cid] ?? 0) + 1;
   627→    voters.push({ userid: Number(userid), username: vote?.username, contentId: cid });
   628→  }
   629→  return { tally, voters };
   630→}

   632→/** Câștigător = contentId cu cele mai multe voturi; la egalitate, contentId-ul
   633→ * votului cel mai devreme exprimat dintre cele la egalitate (determinist). */
   634→function computeWinner(record) {
   635→  const { tally, voters } = tallyOf(record);
   636→  const counts = Object.entries(tally);
   637→  if (counts.length === 0) return null;
   638→  let max = 0;
   639→  for (const [, count] of counts) max = Math.max(max, count);
   640→  const tied = counts.filter(([, count]) => count === max).map(([cid]) => Number(cid));
   641→  if (tied.length === 1) return tied[0];
   642→  const firstVotePos = new Map(); // contentId → prima poziție în ordinea votării
   643→  for (let i = 0; i < voters.length; i += 1) {
   644→    if (!firstVotePos.has(voters[i].contentId)) firstVotePos.set(voters[i].contentId, i);
   645→  }
   646→  let winner = null;
   647→  let bestPos = Infinity;
   648→  for (const cid of tied) {
   649→    const pos = firstVotePos.get(cid);
   650→    if (pos !== undefined && pos < bestPos) {
   651→      bestPos = pos;
   652→      winner = cid;
   653→    }
   654→  }
   655→  return winner;
   656→}
