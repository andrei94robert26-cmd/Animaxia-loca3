   343→/* ---------------------------- virtual categories -------------------------- */
   344→/*
   345→ * New menu entries resolve to client-side filtered views over the Neon
   346→ * library (fetched once via /api/all?limit=100). Filters receive the raw
   347→ * DbContent row so metadata (genres, language, source) is available.
   348→ */
   349→
   350→export type ContentFilter = (row: DbContent) => boolean;
   351→
   352→export interface VirtualCategory {
   353→  slug: string;
   354→  title: string;
   355→  subtitle: string;
   356→  filter: ContentFilter;
   357→  /** Optional client-side sort (e.g. blockbusters by rating). */
   358→  sort?: (a: DbContent, b: DbContent) => number;
   359→  /** Extra hint rendered in the empty state. */
   360→  emptyHint?: string;
   361→  /** FAZA 14: category is backed by a LIVE /api/discover/:slug feed (real
   362→   *  TMDB data). The library filter above still powers the “În biblioteca
   363→   *  ta” row, while the live feed fills the main grid. */
   364→  live?: boolean;
   365→  /** Human description of the real live source (rendered as a badge). */
   366→  liveSource?: string;
   367→}
   368→
   369→function genresOf(row: DbContent): string[] {
   370→  const genres = row.metadata?.genres;
   371→  return Array.isArray(genres) ? genres.map((g) => String(g).toLowerCase()) : [];
   372→}
   373→
   374→function genreMatch(row: DbContent, re: RegExp): boolean {
   375→  return genresOf(row).some((g) => re.test(g));
   376→}
   377→
   378→function titleMatch(row: DbContent, re: RegExp): boolean {
   379→  return re.test(row.title || '');
   380→}
   381→
   382→function sourceMatch(row: DbContent, re: RegExp): boolean {
   383→  const md = row.metadata ?? {};
   384→  const hay = [md.source, md.network, md.studio, md.channel]
   385→    .flatMap((v) => (Array.isArray(v) ? v.map(String) : typeof v === 'string' ? [v] : []))
   386→    .join(' ');
   387→  return re.test(hay.toLowerCase());
   388→}
   389→
   390→function ratingOf(row: DbContent): number {
   391→  const r = row.metadata?.rating ?? row.metadata?.score;
   392→  return typeof r === 'number' ? r : 0;
   393→}

   405→function continentFilter(slug: string): ContentFilter {
   406→  const langs = CONTINENT_LANGUAGES[slug] ?? [];
   407→  return (row) => {
   408→    const lang = String(row.metadata?.language ?? '').toLowerCase();
   409→    if (lang && langs.includes(lang)) return true;
   410→    const country = row.metadata?.country ?? row.metadata?.originCountry ?? row.metadata?.origin_country;
   411→    if (country) {
   412→      const list = Array.isArray(country) ? country.map(String) : [String(country)];
   413→      if (list.some((c) => langs.includes(c.toLowerCase()))) return true;
   414→    }
   415→    return false;
   416→  };
   417→}
