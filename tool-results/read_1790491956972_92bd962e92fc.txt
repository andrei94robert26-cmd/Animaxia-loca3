   226→/* TS3 (observability): W3C Trace Context propagation — when an incoming
   227→ * `traceparent` header (format `00-<trace-id>-<span-id>-<flags>`, trace-id =
   228→ * exactly 32 hex chars) is present its trace-id is adopted; otherwise a fresh
   229→ * 32-hex id is generated (crypto.randomUUID without dashes). Every response
   230→ * echoes X-Trace-Id, and requestLogger puts it on the access line (or in the
   231→ * JSON log entry under LOGFORMAT=json).
   232→ * OpenTelemetry-ready WITHOUT any otel dependency (documented honestly): an
   233→ * otel collector / gateway in front of the backend can pick up X-Trace-Id for
   234→ * trace correlation — TRACINGENABLED / OTELSENDPOINT / OTELSAMPLERATE in .env
   235→ * are consumed by the edge/gateway staging configs, not by this process.
   236→ * Runs BEFORE requestLogger so access lines always carry the trace id. */
   237→app.use((req, res, next) => {
   238→  let traceId = null;
   239→  const traceparent = req.headers['traceparent'];
   240→  if (typeof traceparent === 'string') {
   241→    const m = /^[\da-f]{2}-([\da-f]{32})-[\da-f]{16}-[\da-f]{2}$/i.exec(traceparent.trim());
   242→    if (m) traceId = m[1].toLowerCase(); // safe-parse: only well-formed ids propagate
   243→  }
   244→  if (!traceId) traceId = crypto.randomUUID().replace(/-/g, '');
   245→  req.traceId = traceId;
   246→  res.set('X-Trace-Id', traceId);
   247→  next();
   248→});

   387→ *      activitate reală recentă (≤5 min) — compute-ul e oricum treaz din
   388→ *      cauza traficului real, deci refresh-ul costă ZERO treziri suplimentare.
   389→ *   3. Dacă DB-ul e idle (Neon suspendat, zero trafic), /api/health servește
   390→ *      valorile din cache marcate onest `stale:true` — NU trezește Neon.
   391→ *   4. `?refresh=1` forțează reinterogarea (deep-probe explicit, admin/QA).
   392→ *   5. `content:changed` marchează cache-ul dirty → reîmprospătare piggyback
   393→ *      la prima ocazie cu DB activ (write-urile reale țin compute-ul treaz).

   403→async function refreshHealthDbStats() {
   404→  if (!healthStats.inflight) {
   405→    healthStats.inflight = Promise.all([countContent(), countsByType()])
   406→      .then(([totalItems, byType]) => {
   407→        healthStats = { totalItems, byType, fetchedAt: Date.now(), dirty: false, inflight: null };
   408→        return { totalItems, byType, fetchedAt: healthStats.fetchedAt, stale: false };
   409→      })
   410→      .catch((err) => {
   411→        healthStats.inflight = null; // la eșec păstrăm valorile vechi (stale) — health nu pică din cauza statisticelor
   412→        throw err;
   413→      });
   414→  }
   415→  return healthStats.inflight;
   416→}

   422→  }
   423→  if (healthStats.totalItems === null) {
   424→    // Primul apel după boot — avem nevoie de numere reale (boot-ul tocmai a
   425→    // folosit DB-ul, compute-ul e activ — refresh piggyback, zero treziri).
   426→    return refreshHealthDbStats();

   428→  const dbActiveRecently = Date.now() - lastDbActivityAt() < HEALTH_DB_ACTIVE_WINDOW_MS;
   429→  if (dbActiveRecently) {
   430→    // DB-ul e treaz din trafic real — refresh piggyback (gratuit).
   431→    return refreshHealthDbStats();
   432→  }
   433→  // Neon idle/suspendat — servim cache-ul marcat stale, NU trezim compute-ul.
   434→  return { totalItems: healthStats.totalItems, byType: healthStats.byType, fetchedAt: healthStats.fetchedAt, stale: true };
   435→}

   604→async function handleMine(req, res) {

   608→  }

   610→    req.method === 'GET'
   611→      ? { type: req.query.type, externalId: req.query.externalId }
   612→      : { type: req.body?.type, externalId: req.body?.externalId };
   613→  try {
   614→    const { extraction, record } = await mineUrl(url, options);
   615→    invalidateHttpCache();
   616→    // TS1: targeted id event for the incremental searchdocs sync (upsert path).
   617→    if (record?.id) events.emit('content:changed:ids', { ids: [record.id], action: 'update' });

   619→    return res.json({ ok: true, extraction, record });
   620→  } catch (err) {
   621→    return res.status(err.status || 502).json({ error: err.message || 'Mining failed' });
   622→  }
