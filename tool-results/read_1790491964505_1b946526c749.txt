   101→   101→
   102→   102→class JsonFileStore {
   103→   103→  constructor(name) {
   104→   104→    this.file = join(STORAGE_DIR, `${name}.json`);
   105→   105→    this.name = name;
   106→   106→    this.data = undefined; // undefined = neîncărcat; {} = încărcat/gol
   107→   107→    this.writeTimer = null;
   108→   108→  }
   109→   109→
   110→   110→  /** Încarcă o singură dată (sincron, la prima nevoie); corupt ⇒ {} + warn. */
   111→   111→  load() {
   112→   112→    if (this.data !== undefined) return this.data;
   113→   113→    try {
   114→   114→      const raw = readFileSync(this.file, 'utf8');
   115→   115→      const parsed = JSON.parse(raw);
   116→   116→      this.data = parsed && typeof parsed === 'object' && !Array.isArray(parsed) ? parsed : {};
   117→   117→    } catch (err) {
   118→   118→      if (err && err.code === 'ENOENT') {
   119→   119→        this.data = {}; // nu există încă — creat leneș la prima scriere
   120→   120→      } else {
   121→   121→        logger.warn(`[playerx] ${this.name}.json corupt/ineligibil — pornesc de la zero (${err?.message ?? err})`);
   122→   122→        this.data = {};
   123→   123→      }
   124→   124→    }
   125→   125→    return this.data;
   126→   126→  }
   127→   127→
   128→   128→  /** Programează o scriere atomică (debounce) după o mutație in-memory. */
   129→   129→  scheduleSave() {
   130→   130→    if (this.writeTimer) return;
   131→   131→    this.writeTimer = setTimeout(() => {
   132→   132→      this.writeTimer = null;
   133→   133→      this.flush();
   134→   134→    }, WRITE_DEBOUNCE_MS);
   135→   135→    if (typeof this.writeTimer.unref === 'function') this.writeTimer.unref();
   136→   136→  }
   137→   137→
   138→   138→  /** Scriere atomică: tmp + rename. Niciodată throw. */
   139→   139→  flush() {
   140→   140→    if (this.data === undefined) return; // nimic încărcat ⇒ nimic de persistat
   141→   141→    try {
   142→   142→      mkdirSync(STORAGE_DIR, { recursive: true });
   143→   143→      const tmp = `${this.file}.${process.pid}.tmp`;
   144→   144→      writeFileSync(tmp, `${JSON.stringify(this.data, null, 2)}\n`);
   145→   145→      renameSync(tmp, this.file);
   146→   146→    } catch (err) {
   147→   147→      logger.warn(`[playerx] persistența ${basename(this.file)} a eșuat (non-fatal): ${err?.message ?? err}`);
   148→   148→    }
   149→   149→  }
   150→   150→}

   189→   189→/* ------------------------------ generic LRU cache --------------------------- */
   190→   190→
   191→   191→function makeLru(limit) {
   192→   192→  return {
   193→   193→    map: new Map(), // key → { value, storedAt } · Map = insertion order = LRU
   194→   194→    get(key) {
   195→   195→      const entry = this.map.get(key);
   196→   196→      if (!entry) return null;
   197→   197→      if (Date.now() - entry.storedAt > CACHE_TTL_MS) {
   198→   198→        this.map.delete(key);
   199→   199→        return null;
   200→   200→      }
   201→   201→      this.map.delete(key);
   202→   202→      this.map.set(key, entry); // refresh recency
   203→   203→      return entry.value;
   204→   204→    },
   205→   205→    set(key, value) {
   206→   206→      if (this.map.has(key)) this.map.delete(key);
   207→   207→      else if (this.map.size >= limit) {
   208→   208→        const oldest = this.map.keys().next().value;
   209→   209→        if (oldest !== undefined) this.map.delete(oldest);
   210→   210→      }
   211→   211→      this.map.set(key, { value, storedAt: Date.now() });
   212→   212→    },
   213→   213→    invalidate(key) {
   214→   214→      this.map.delete(key);
   215→   215→    },
   216→   216→    size() {
   217→   217→      return this.map.size;
   218→   218→    },
   219→   219→  };
   220→   220→}

   309→   309→function durationSecOf(metadata) {
   310→   310→  const meta = metadata && typeof metadata === 'object' ? metadata : {};
   311→   311→  const dur = Number(meta.duration);
   312→   312→  if (Number.isFinite(dur) && dur > 0) return dur; // secunde
   313→   313→  const rt = Number(meta.runtime);

   315→   315→  const ert = Number(meta.episodeRunTime ?? meta.episode_runtime);
   316→   316→  if (Number.isFinite(ert) && ert > 0) return ert * 60; // minute → secunde

   353→   353→ * contractul leagă nota de source, nu de existența estimărilor). */
   354→   354→function computeMarkers(contentId, content) {
   355→   355→  const estimated = estimateMarkers(content?.type, durationSecOf(content?.metadata));
   356→   356→  const curatedAll = markersStore.load()[contentId] ?? {};
   357→   357→  const byKind = {};
   358→   358→  for (const [kind, marker] of Object.entries(estimated)) {
   359→   359→    byKind[kind] = { ...marker, kind, source: 'estimated' };
   360→   360→  }
   361→   361→  for (const kind of MARKER_KINDS) {
   362→   362→    const curated = curatedAll[kind];
   363→   363→    if (!curated) continue;
   364→   364→    const { startsec, endsec } = curated;
   365→   365→    if (!validMarkerSpan(startsec, endsec)) {
   366→   366→      logger.warn(`[playerx] marker curat sărit pentru content ${contentId} kind ${kind} (interval invalid: ${startsec}..${endsec})`);
   367→   367→      continue;
   368→   368→    }
   369→   369→    byKind[kind] = {
   370→   370→      startsec,
   371→   371→      endsec,
   372→   372→      kind,
   373→   373→      label: typeof curated.label === 'string' && curated.label.trim()
   374→   374→        ? curated.label.trim().slice(0, LABEL_MAX)
   375→   375→        : KIND_LABELS[kind],
   376→   376→      source: 'curated',
   377→   377→    };
   378→   378→  }
   379→   379→  const data = Object.values(byKind)
   380→   380→    .map((m) => ({
   381→   381→      id: m.kind,
   382→   382→      kind: m.kind,
   383→   383→      startsec: m.startsec,
   384→   384→      endsec: m.endsec,
   385→   385→      label: m.source === 'estimated' ? KIND_LABELS[m.kind] : m.label,
   386→   386→      source: m.source,
   387→   387→    }))
   388→   388→    .sort((a, b) => a.startsec - b.startsec || a.endsec - b.endsec);
   389→   389→  return { data, anyEstimated: data.some((m) => m.source === 'estimated') };
   390→   390→}

   618→   618→/** Tally + votanți în ordinea în care au votat (seq incremental — determinist,
   619→   619→ * supraviețuiește și round-trip-ului JSON, care păstrează ordinea cheilor). */
   620→   620→function tallyOf(record) {
   621→   621→  const tally = {};
   622→   622→  const voters = [];
   623→   623→  const entries = Object.entries(record?.votes ?? {}).slice().sort((a, b) => (a[1]?.seq ?? 0) - (b[1]?.seq ?? 0));
   624→   624→  for (const [userid, vote] of entries) {
   625→   625→    const cid = Number(vote?.contentId);
   626→   626→    tally[cid] = (tally[cid] ?? 0) + 1;
   627→   627→    voters.push({ userid: Number(userid), username: vote?.username, contentId: cid });
   628→   628→  }
   629→   629→  return { tally, voters };
   630→   630→}

   632→   632→/** Câștigător = contentId cu cele mai multe voturi; la egalitate, contentId-ul
   633→   633→ * votului cel mai devreme exprimat dintre cele la egalitate (determinist). */
   634→   634→function computeWinner(record) {
   635→   635→  const { tally, voters } = tallyOf(record);
   636→   636→  const counts = Object.entries(tally);
   637→   637→  if (counts.length === 0) return null;
   638→   638→  let max = 0;
   639→   639→  for (const [, count] of counts) max = Math.max(max, count);
   640→   640→  const tied = counts.filter(([, count]) => count === max).map(([cid]) => Number(cid));
   641→   641→  if (tied.length === 1) return tied[0];
   642→   642→  const firstVotePos = new Map(); // contentId → prima poziție în ordinea votării
   643→   643→  for (let i = 0; i < voters.length; i += 1) {
   644→   644→    if (!firstVotePos.has(voters[i].contentId)) firstVotePos.set(voters[i].contentId, i);
   645→   645→  }
   646→   646→  let winner = null;
   647→   647→  let bestPos = Infinity;
   648→   648→  for (const cid of tied) {
   649→   649→    const pos = firstVotePos.get(cid);
   650→   650→    if (pos !== undefined && pos < bestPos) {
   651→   651→      bestPos = pos;
   652→   652→      winner = cid;
   653→   653→    }
   654→   654→  }
   655→   655→  return winner;
   656→   656→}
