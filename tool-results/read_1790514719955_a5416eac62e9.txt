   410→/* ---------------------------- virtual categories -------------------------- */
   411→/*
   412→ * New menu entries resolve to client-side filtered views over the Neon
   413→ * library (fetched once via /api/all?limit=100). Filters receive the raw
   414→ * DbContent row so metadata (genres, language, source) is available.
   415→ */
   416→
   417→export type ContentFilter = (row: DbContent) => boolean;
   418→
   419→export interface VirtualCategory {
   420→  slug: string;
   421→  title: string;
   422→  subtitle: string;
   423→  filter: ContentFilter;
   424→  /** Optional client-side sort (e.g. blockbusters by rating). */
   425→  sort?: (a: DbContent, b: DbContent) => number;
   426→  /** Extra hint rendered in the empty state. */
   427→  emptyHint?: string;
   428→  /** FAZA 14: category is backed by a LIVE /api/discover/:slug feed (real
   429→   *  TMDB data). The library filter above still powers the “În biblioteca
   430→   *  ta” row, while the live feed fills the main grid. */
   431→  live?: boolean;
   432→  /** Human description of the real live source (rendered as a badge). */
   433→  liveSource?: string;
   434→}
   435→
   436→function genresOf(row: DbContent): string[] {
   437→  const genres = row.metadata?.genres;
   438→  return Array.isArray(genres) ? genres.map((g) => String(g).toLowerCase()) : [];
   439→}
   440→
   441→function genreMatch(row: DbContent, re: RegExp): boolean {
   442→  return genresOf(row).some((g) => re.test(g));
   443→}
   444→
   445→function titleMatch(row: DbContent, re: RegExp): boolean {
   446→  return re.test(row.title || '');
   447→}
   448→
   449→function sourceMatch(row: DbContent, re: RegExp): boolean {
   450→  const md = row.metadata ?? {};
   451→  const hay = [md.source, md.network, md.studio, md.channel]
   452→    .flatMap((v) => (Array.isArray(v) ? v.map(String) : typeof v === 'string' ? [v] : []))
   453→    .join(' ');
   454→  return re.test(hay.toLowerCase());
   455→}
   456→
   457→function ratingOf(row: DbContent): number {
   458→  const r = row.metadata?.rating ?? row.metadata?.score;
   459→  return typeof r === 'number' ? r : 0;
   460→}

   462→/** Language → continent mapping for the "Lumea" group (TMDB original_language). */
   463→const CONTINENT_LANGUAGES: Record<string, string[]> = {
   464→  europa: ['fr', 'de', 'es', 'it', 'pl', 'ro', 'sv', 'nl', 'cs', 'hu', 'el', 'uk', 'da', 'fi', 'no', 'pt', 'ru', 'bg', 'hr', 'sr', 'sk', 'sl', 'lt', 'lv', 'et'],
   465→  'america-nord': ['en', 'es-MX', 'fr-CA'],
   466→  'america-sud': ['pt-BR', 'es-AR', 'es-CO', 'es-CL', 'es-PE', 'es-419', 'gn', 'qu'],
   467→  asia: ['ja', 'ko', 'te', 'hi', 'zh', 'th', 'vi', 'id', 'tl', 'fa', 'he', 'ur', 'bn', 'ta', 'ml', 'kn', 'mr', 'gu', 'pa', 'si', 'km', 'my', 'ka', 'hy', 'az', 'kk', 'uz', 'mn'],
   468→  africa: ['ar', 'sw', 'af', 'zu', 'ha', 'yo', 'ig', 'am', 'so', 'xh', 'sn', 'amh'],

   471→
   472→function continentFilter(slug: string): ContentFilter {
   473→  const langs = CONTINENT_LANGUAGES[slug] ?? [];
   474→  return (row) => {
   475→    const lang = String(row.metadata?.language ?? '').toLowerCase();
   476→    if (lang && langs.includes(lang)) return true;
   477→    const country = row.metadata?.country ?? row.metadata?.originCountry ?? row.metadata?.origin_country;
   478→    if (country) {
   479→      const list = Array.isArray(country) ? country.map(String) : [String(country)];
