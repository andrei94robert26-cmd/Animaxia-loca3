   414→   414→ * DbContent row so metadata (genres, language, source) is available.
   415→   415→ */

   436→   436→function genresOf(row: DbContent): string[] {
   437→   437→  const genres = row.metadata?.genres;
   438→   438→  return Array.isArray(genres) ? genres.map((g) => String(g).toLowerCase()) : [];
   439→   439→}
   440→   440→
   441→   441→function genreMatch(row: DbContent, re: RegExp): boolean {
   442→   442→  return genresOf(row).some((g) => re.test(g));
   443→   443→}
   444→   444→
   445→   445→function titleMatch(row: DbContent, re: RegExp): boolean {
   446→   446→  return re.test(row.title || '');
   447→   447→}
   448→   448→
   449→   449→function sourceMatch(row: DbContent, re: RegExp): boolean {
   450→   450→  const md = row.metadata ?? {};
   451→   451→  const hay = [md.source, md.network, md.studio, md.channel]
   452→   452→    .flatMap((v) => (Array.isArray(v) ? v.map(String) : typeof v === 'string' ? [v] : []))
   453→   453→    .join(' ');
   454→   454→  return re.test(hay.toLowerCase());
   455→   455→}
   456→   456→
   457→   457→function ratingOf(row: DbContent): number {
   458→   458→  const r = row.metadata?.rating ?? row.metadata?.score;
   459→   459→  return typeof r === 'number' ? r : 0;
   460→   460→}

   472→   472→function continentFilter(slug: string): ContentFilter {
   473→   473→  const langs = CONTINENT_LANGUAGES[slug] ?? [];
   474→   474→  return (row) => {
   475→   475→    const lang = String(row.metadata?.language ?? '').toLowerCase();
   476→   476→    if (lang && langs.includes(lang)) return true;
   477→   477→    const country = row.metadata?.country ?? row.metadata?.originCountry ?? row.metadata?.origin_country;
   478→   478→    if (country) {
   479→   479→      const list = Array.isArray(country) ? country.map(String) : [String(country)];
   480→   480→      if (list.some((c) => langs.includes(c.toLowerCase()))) return true;
   481→   481→    }
   482→   482→    return false;
   483→   483→  };
   484→   484→}
