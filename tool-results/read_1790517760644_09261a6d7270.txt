   388→ * New menu entries resolve to client-side filtered views over the Neon

   401→  sort?: (a: DbContent, b: DbContent) => number;
   402→  /** Extra hint rendered in the empty state. */
   403→  emptyHint?: string;
   404→  /** FAZA 14: category is backed by a LIVE /api/discover/:slug feed (real
   405→   *  TMDB data). The library filter above still powers the “În biblioteca
   406→   *  ta” row, while the live feed fills the main grid. */
   407→  live?: boolean;
   408→  /** Human description of the real live source (rendered as a badge). */
   409→  liveSource?: string;

   412→function genresOf(row: DbContent): string[] {
   413→  const genres = row.metadata?.genres;
   414→  return Array.isArray(genres) ? genres.map((g) => String(g).toLowerCase()) : [];
   415→}
   416→
   417→function genreMatch(row: DbContent, re: RegExp): boolean {
   418→  return genresOf(row).some((g) => re.test(g));
   419→}
   420→
   421→function titleMatch(row: DbContent, re: RegExp): boolean {
   422→  return re.test(row.title || '');
   423→}
   424→
   425→function sourceMatch(row: DbContent, re: RegExp): boolean {
   426→  const md = row.metadata ?? {};
   427→  const hay = [md.source, md.network, md.studio, md.channel]
   428→    .flatMap((v) => (Array.isArray(v) ? v.map(String) : typeof v === 'string' ? [v] : []))
   429→    .join(' ');
   430→  return re.test(hay.toLowerCase());
   431→}
   432→
   433→function ratingOf(row: DbContent): number {
   434→  const r = row.metadata?.rating ?? row.metadata?.score;
   435→  return typeof r === 'number' ? r : 0;
   436→}

   439→const CONTINENT_LANGUAGES: Record<string, string[]> = {

   443→  asia: ['ja', 'ko', 'te', 'hi', 'zh', 'th', 'vi', 'id', 'tl', 'fa', 'he', 'ur', 'bn', 'ta', 'ml', 'kn', 'mr', 'gu', 'pa', 'si', 'km', 'my', 'ka', 'hy', 'az', 'kk', 'uz', 'mn'],

   445→  oceania: ['mi', 'sm', 'haw', 'fj', 'to', 'ty'],
   446→};
   447→
   448→function continentFilter(slug: string): ContentFilter {
   449→  const langs = CONTINENT_LANGUAGES[slug] ?? [];
   450→  return (row) => {
   451→    const lang = String(row.metadata?.language ?? '').toLowerCase();
   452→    if (lang && langs.includes(lang)) return true;
   453→    const country = row.metadata?.country ?? row.metadata?.originCountry ?? row.metadata?.origin_country;
   454→    if (country) {
   455→      const list = Array.isArray(country) ? country.map(String) : [String(country)];
   456→      if (list.some((c) => langs.includes(c.toLowerCase()))) return true;
   457→    }
   458→    return false;
   459→  };
   460→}
