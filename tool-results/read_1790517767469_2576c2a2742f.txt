   172→export class ApiError extends Error {
   173→  status: number;
   174→  constructor(message: string, status: number) {
   175→    super(message);
   176→    this.name = 'ApiError';
   177→    this.status = status;
   178→  }
   179→}

   181→type ParamValue = string | number | boolean | undefined | null;
   182→
   183→export function apiUrl(path: string, params?: Record<string, ParamValue>): string {
   184→  const search = new URLSearchParams();
   185→  search.set('XTransformPort', String(BACKEND_PORT));
   186→  if (params) {
   187→    for (const [key, value] of Object.entries(params)) {
   188→      if (value !== undefined && value !== null && value !== '') search.set(key, String(value));
   189→    }
   190→  }
   191→  return `${path}?${search.toString()}`;
   192→}

   194→/** FAZA 23 (observability): same gateway pattern as apiUrl(), but pointed at
   195→ *  the observability stack on :4050 (Caddy forwards via XTransformPort). */
   196→export function obsUrl(path: string, params?: Record<string, ParamValue>): string {
   197→  const search = new URLSearchParams();
   198→  search.set('XTransformPort', String(OBSERVABILITY_PORT));
   199→  if (params) {
   200→    for (const [key, value] of Object.entries(params)) {
   201→      if (value !== undefined && value !== null && value !== '') search.set(key, String(value));
   202→    }
   203→  }
   204→  return `${path}?${search.toString()}`;
   205→}
   206→
   207→async function request<T>(path: string, init?: RequestInit): Promise<T> {
   208→  let res: Response;
   209→  try {
   210→    res = await fetch(path, {
   211→      ...init,
   212→      headers: { 'Content-Type': 'application/json', ...(init?.headers as Record<string, string> | undefined) },
   213→    });
   214→  } catch {
   215→    throw new ApiError('Cannot reach the Animaxia backend — it may be restarting', 0);
   216→  }
   217→  let body: unknown = null;
   218→  try {
   219→    body = await res.json();
   220→  } catch {
   221→    body = null;
   222→  }
   223→  if (!res.ok) {
   224→    const message =
   225→      (body as { error?: string } | null)?.error || `Request failed with status ${res.status}`;
   226→    throw new ApiError(message, res.status);
   227→  }
   228→  return body as T;
   229→}

   231→/** Bearer header for FAZA 16 auth endpoints (same storage key as platform.ts
   232→ *  tokenStore — read directly here to avoid a circular import). */
   233→function authHeaders(): Record<string, string> {
   234→  try {
   235→    const token = window.localStorage.getItem('animaxia.accessToken');
   236→    return token ? { Authorization: `Bearer ${token}` } : {};
   237→  } catch {
   238→    return {};
   239→  }
   240→}

  1163→export function buildVirtualCursor(index: number, seed: string): string {
  1164→  const json = JSON.stringify({ i: index, s: seed });
  1165→  const b64 =
  1166→    typeof btoa === 'function'
  1167→      ? btoa(json)
  1168→      : Buffer.from(json, 'utf8').toString('base64');
  1169→  return b64.replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
  1170→}

  1174→export function dbToItem(c: DbContent): NormalizedItem {
  1175→  const md = c.metadata || {};
  1176→  const sources = md.player?.sources || md.sources;
  1177→  return {
  1178→    key: `db-${c.id}`,
  1179→    kind: 'db',
  1180→    dbId: c.id,
  1181→    title: c.title,
  1182→    image: c.thumbnail || md.backdrop || null,
  1183→    backdrop: md.backdrop || null,
  1184→    overview: md.overview || md.synopsis || md.description || undefined,
  1185→    rating: typeof md.rating === 'number' ? md.rating : typeof md.score === 'number' ? md.score : null,
  1186→    year: (md.year as string | number | undefined) ?? null,
  1187→    type: c.type || 'mined',
  1188→    genres: Array.isArray(md.genres) ? md.genres.map(String) : undefined,
  1189→    playable: Boolean(sources && sources.length > 0),
  1190→    source: typeof md.source === 'string' ? md.source : undefined,
  1191→  };
