    67→/* Safe JSONB array extraction ('[]' when the key is missing or not an array)
    68→ * and safe numeric casts (regex-guarded — a malformed value degrades to NULL
    69→ * instead of failing the whole query). Same approach as search.js filters. */
    70→const GENRES_ARRAY = `CASE WHEN jsonb_typeof(metadata->'genres') = 'array' THEN metadata->'genres' ELSE '[]'::jsonb END`;
    71→const ACTORS_ARRAY = `CASE WHEN jsonb_typeof(metadata->'actors') = 'array' THEN metadata->'actors' ELSE '[]'::jsonb END`;

    73→const RATING_NUM = `CASE WHEN (metadata->>'rating') ~ '^[0-9]+\\.?[0-9]*$' THEN (metadata->>'rating')::numeric ELSE NULL END`;

    78→/** Shape a content row exactly like db.js normalizeRow (house contract). */
    79→function contentRow(row) {
    80→  return {
    81→    id: num(row.id),

    84→    title: row.title ?? null,

    87→    createdat: isoOrNull(row.createdat),

    91→/** Parse an int query param (null when absent/malformed). */
    92→function intOrNull(value) {
    93→  if (value === undefined || value === null || value === '') return null;
    94→  const n = parseInt(value, 10);
    95→  return Number.isInteger(n) ? n : null;
    96→}

   100→/** Build the parameterized WHERE clause for the browse filters.
   101→ *  `exclude` drops a dimension's clauses — used for facet queries, which are
   102→ *  computed on the filtered set WITHOUT their own dimension (standard
   103→ *  faceted browse: filtering by genre still shows all genre counts). */
   104→function buildBrowseWhere(filters, exclude = new Set()) {
   105→  const clauses = [];
   106→  const params = [];
   107→  const add = (value) => {
   108→    params.push(value);
   109→    return `$${params.length}`;
   110→  };
   111→  if (!exclude.has('type') && filters.types.length) {
   112→    clauses.push(`type = ANY(${add(filters.types)}::text[])`);
   113→  }
   114→  if (!exclude.has('genre') && filters.genre) {
   115→    // Exact case-insensitive match inside metadata.genres[] (ILIKE without wildcards).
   116→    clauses.push(`EXISTS (SELECT 1 FROM jsonb_array_elements_text(${GENRES_ARRAY}) g WHERE g ILIKE ${add(filters.genre)})`);
   117→  }
   118→  if (!exclude.has('year') && (filters.year != null || filters.yearFrom != null || filters.yearTo != null)) {
   119→    clauses.push(`${YEAR_INT} IS NOT NULL`);
   120→    if (filters.year != null) clauses.push(`${YEAR_INT} = ${add(filters.year)}`);
   121→    if (filters.yearFrom != null) clauses.push(`${YEAR_INT} >= ${add(filters.yearFrom)}`);
   122→    if (filters.yearTo != null) clauses.push(`${YEAR_INT} <= ${add(filters.yearTo)}`);
   123→  }
   124→  if (!exclude.has('q') && filters.q) {
   125→    clauses.push(`title ILIKE '%' || ${add(filters.q)} || '%'`);
   126→  }
   127→  if (!exclude.has('actor') && filters.actor) {
   128→    clauses.push(`EXISTS (SELECT 1 FROM jsonb_array_elements_text(${ACTORS_ARRAY}) a WHERE a ILIKE '%' || ${add(filters.actor)} || '%')`);
   129→  }
   130→  if (!exclude.has('studio') && filters.studio) {
   131→    clauses.push(`EXISTS (SELECT 1 FROM jsonb_array_elements_text(${STUDIOS_ARRAY}) st WHERE st ILIKE '%' || ${add(filters.studio)} || '%')`);
   132→  }
   133→  if (!exclude.has('minRating') && filters.minRating != null) {
   134→    clauses.push(`${RATING_NUM} >= ${add(filters.minRating)}`);
   135→  }
   136→  return { sql: clauses.length ? `WHERE ${clauses.join(' AND ')}` : '', params };
   137→}
   138→
   139→/** Build the ORDER BY clause (its params are appended after the WHERE params). */
   140→function buildBrowseOrder(sort, filters, startIdx) {
   141→  const params = [];
   142→  const add = (value) => {
   143→    params.push(value);
   144→    return `$${startIdx + params.length}`;
   145→  };
   146→  let sql;
   147→  if (sort === 'rating') {
   148→    sql = `ORDER BY ${RATING_NUM} DESC NULLS LAST, id ASC`;
   149→  } else if (sort === 'year') {
   150→    sql = `ORDER BY ${YEAR_INT} DESC NULLS LAST, id ASC`;
   151→  } else if (sort === 'title') {
   152→    sql = `ORDER BY title ASC, id ASC`;
   153→  } else if (sort === 'relevance' && filters.q) {
   154→    // Prefix hits outrank infix hits; ties break on recency (deterministic).
   155→    const q1 = add(filters.q);
   156→    const q2 = add(filters.q);
   157→    sql = `ORDER BY (CASE WHEN title ILIKE ${q1} || '%' THEN 2 WHEN title ILIKE '%' || ${q2} || '%' THEN 1 ELSE 0 END) DESC, createdat DESC, id ASC`;
   158→  } else {
   159→    sql = `ORDER BY createdat DESC, id DESC`; // 'recent' (default)
   160→  }
   161→  return { sql, params };
   162→}

   164→router.get('/browse', wrap(async (req, res) => {
   165→  assertPostgres();
   166→  const startedAt = Date.now();
   167→
   168→  // ---- filter parsing (echoed in the response) ----
   169→  const rawType = req.query.type;
   170→  const typeList = (Array.isArray(rawType) ? rawType.join(',') : String(rawType || ''))
   171→    .split(',').map((s) => s.trim()).filter(Boolean);
   172→  const filters = {
   173→    types: typeList,
   174→    genre: String(req.query.genre || '').trim().slice(0, 120) || null,
   175→    year: intOrNull(req.query.year),
   176→    yearFrom: intOrNull(req.query.yearFrom),
   177→    yearTo: intOrNull(req.query.yearTo),
   178→    q: String(req.query.q || '').trim().slice(0, 120) || null,
   179→    actor: String(req.query.actor || '').trim().slice(0, 120) || null,
   180→    studio: String(req.query.studio || '').trim().slice(0, 120) || null,
   181→    minRating: (() => {
   182→      const v = req.query.minRating;
   183→      if (v === undefined || v === null || v === '') return null;
   184→      const n = Number(v);
   185→      return Number.isFinite(n) ? n : null;
   186→    })(),
   187→  };
   188→
   189→  const page = Math.max(1, intOrNull(req.query.page) ?? 1);
   190→  const limit = Math.min(MAX_PAGE_LIMIT, Math.max(1, intOrNull(req.query.limit) ?? DEFAULT_PAGE_LIMIT));
   191→  const wantFacets = String(req.query.facet ?? 'true') !== 'false';
   192→
   193→  const requestedSort = String(req.query.sort || 'recent');
   194→  const allowedSorts = ['relevance', 'rating', 'year', 'title', 'recent'];
   195→  const sort = allowedSorts.includes(requestedSort) ? requestedSort : 'recent';
   196→  // 'relevance' without a q falls back to 'recent' (nothing to rank on).
   197→  const effectiveSort = sort === 'relevance' && !filters.q ? 'recent' : sort;
   198→
   199→  // ---- one parallel round: COUNT + data page + the 3 facet queries ----
   200→  // (facets are computed on the filtered set minus each facet's own dimension)
   201→  const where = buildBrowseWhere(filters);
   202→  const order = buildBrowseOrder(effectiveSort, filters, where.params.length);
   203→  const dataParams = [...where.params, ...order.params, limit, (page - 1) * limit];
   204→  const whereNoType = buildBrowseWhere(filters, new Set(['type']));
   205→  const whereNoGenre = buildBrowseWhere(filters, new Set(['genre']));
   206→  const whereNoYear = buildBrowseWhere(filters, new Set(['year']));
   207→  const queries = [
   208→    one(`SELECT COUNT(*)::int AS total FROM content ${where.sql}`, where.params),
   209→    rows(`SELECT * FROM content ${where.sql} ${order.sql} LIMIT $${dataParams.length - 1} OFFSET $${dataParams.length}`, dataParams),
   210→  ];
   211→  if (wantFacets) {
   212→    queries.push(
   213→      rows(`SELECT type AS value, COUNT(*)::int AS count FROM content ${whereNoType.sql} GROUP BY type ORDER BY count DESC, value ASC`, whereNoType.params),
   214→      rows(`SELECT g AS value, COUNT(DISTINCT content.id)::int AS count
   215→            FROM content, jsonb_array_elements_text(${GENRES_ARRAY}) AS g
   216→            ${whereNoGenre.sql}
   217→            GROUP BY g ORDER BY count DESC, value ASC LIMIT ${FACET_GENRES_CAP}`, whereNoGenre.params),
   218→      rows(`SELECT ${YEAR_INT} AS value, COUNT(*)::int AS count
   219→            FROM content ${whereNoYear.sql ? `${whereNoYear.sql} AND` : 'WHERE'} ${YEAR_INT} IS NOT NULL
   220→            GROUP BY 1 ORDER BY 1 DESC LIMIT ${FACET_YEARS_CAP}`, whereNoYear.params),
   221→    );
   222→  }
   223→  const [countRow, dataRows, typesFacet, genresFacet, yearsFacet] = await Promise.all(queries);
   224→  const total = num(countRow?.total);
   225→  const facets = wantFacets ? { types: typesFacet, genres: genresFacet, years: yearsFacet } : null;
   226→
   227→  const filtersEcho = {};
   228→  if (filters.types.length) filtersEcho.types = filters.types;
   229→  if (filters.genre) filtersEcho.genre = filters.genre;
   230→  if (filters.year != null) filtersEcho.year = filters.year;
   231→  if (filters.yearFrom != null) filtersEcho.yearFrom = filters.yearFrom;
   232→  if (filters.yearTo != null) filtersEcho.yearTo = filters.yearTo;
   233→  if (filters.q) filtersEcho.q = filters.q;
   234→  if (filters.actor) filtersEcho.actor = filters.actor;
   235→  if (filters.studio) filtersEcho.studio = filters.studio;
   236→  if (filters.minRating != null) filtersEcho.minRating = filters.minRating;
   237→
   238→  res.json({
   239→    ok: true,
   240→    source: 'neon',
   241→    data: dataRows.map(contentRow),
   242→    total,
   243→    page,
   244→    pages: Math.max(1, Math.ceil(total / limit)),
   245→    limit,
   246→    facets,
   247→    sort: effectiveSort,
   248→    sortRequested: sort,
   249→    filters: filtersEcho,
   250→    tookMs: Date.now() - startedAt,
   251→  });
   252→}));

   256→/** Live platform stats — shared verbatim by /stats and /scale (no HTTP reuse). */
   257→async function collectHubStats() {
   258→  const [totalRow, byType, completeRow, registryRow, registryByKind, providers, viewsRow, topViewed, usersRow] = await Promise.all([
   259→    one('SELECT COUNT(*)::int AS n FROM content'),
   260→    rows('SELECT type AS value, COUNT(*)::int AS count FROM content GROUP BY type ORDER BY count DESC, value ASC'),
   261→    // Metadata completeness — CASE/FILTER WHEN per field + all three combined (0-100).
   262→    one(`SELECT
   263→           COUNT(*)::int AS rows,
   264→           ROUND(100.0 * COUNT(*) FILTER (WHERE COALESCE(thumbnail, '') <> '') / COUNT(*))::int AS pct_thumbnail,
   265→           ROUND(100.0 * COUNT(*) FILTER (WHERE COALESCE(metadata->>'overview', '') <> '') / COUNT(*))::int AS pct_overview,
   266→           ROUND(100.0 * COUNT(*) FILTER (WHERE jsonb_array_length(${GENRES_ARRAY}) > 0) / COUNT(*))::int AS pct_genres,
   267→           ROUND(100.0 * COUNT(*) FILTER (WHERE COALESCE(thumbnail, '') <> ''
   268→             AND COALESCE(metadata->>'overview', '') <> ''
   269→             AND jsonb_array_length(${GENRES_ARRAY}) > 0) / COUNT(*))::int AS pct_complete
   270→         FROM content`),
   271→    // contentsources registry (structure per modules/ingest/dedup.js — urlhash/url/kind/addedby).
   272→    one('SELECT COUNT(*)::int AS total, COUNT(DISTINCT url)::int AS distinct_urls FROM contentsources'),
   273→    rows('SELECT kind AS value, COUNT(*)::int AS count FROM contentsources GROUP BY kind ORDER BY count DESC'),
   274→    // Ingest provider labels embedded in content metadata (honest supplement —
   275→    // the registry itself is authoritative and may legitimately be empty).
   276→    rows(`SELECT metadata->>'source' AS value, COUNT(*)::int AS count
   277→          FROM content WHERE COALESCE(metadata->>'source', '') <> ''
   278→          GROUP BY metadata->>'source' ORDER BY count DESC, value ASC`),
   279→    one(`SELECT COUNT(*)::int AS total,
   280→                COUNT(*) FILTER (WHERE createdat > NOW() - INTERVAL '24 hours')::int AS last24h
   281→         FROM viewevents`),
   282→    rows(`SELECT c.id, c.title, c.type, c.thumbnail, COUNT(*)::int AS views
   283→          FROM viewevents v JOIN content c ON c.id = v.contentid
   284→          GROUP BY c.id, c.title, c.type, c.thumbnail
   285→          ORDER BY views DESC, c.id ASC LIMIT 5`),
   286→    one('SELECT COUNT(*)::int AS n FROM users'),
   287→  ]);
   288→
   289→  const countsByType = {};
   290→  for (const r of byType) countsByType[r.value] = num(r.count);
   291→
   292→  return {
   293→    totalItems: num(totalRow?.n),
   294→    countsByType,
   295→    countsByTypeRows: byType.map((r) => ({ value: r.value, count: num(r.count) })),
   296→    completeness: {
   297→      rows: num(completeRow?.rows),
   298→      thumbnailPct: num(completeRow?.pct_thumbnail),
   299→      overviewPct: num(completeRow?.pct_overview),
   300→      genresPct: num(completeRow?.pct_genres),
   301→      completePct: num(completeRow?.pct_complete), // thumbnail+overview+genres all non-empty
   302→    },
   303→    sources: {
   304→      registry: {
   305→        total: num(registryRow?.total),
   306→        distinct: num(registryRow?.distinct_urls),
   307→        byKind: registryByKind.map((r) => ({ value: r.value, count: num(r.count) })),
   308→      },
   309→      providers, // [{value,count}] from content.metadata.source
   310→    },
   311→    viewevents: {
   312→      total: num(viewsRow?.total),
   313→      last24h: num(viewsRow?.last24h),
   314→      topViewed: topViewed.map((r) => ({ id: num(r.id), title: r.title, type: r.type, thumbnail: r.thumbnail, views: num(r.views) })),
   315→    },
   316→    users: num(usersRow?.n),
   317→  };
   318→}

   422→/** Distinct values of a metadata array (+ optional network supplement for
   423→ *  studios) with count + best-rated sample per value, sorted count DESC. */
   424→async function listDimension({ arrayExpr, extraValues, cap }) {
   425→  const extra = extraValues
   426→    ? `UNION ALL
   427→       SELECT content.id, content.metadata->>'network' AS value
   428→       FROM content WHERE COALESCE(content.metadata->>'network', '') <> ''`
   429→    : '';
   430→  const data = await rows(
   431→    `WITH exploded AS (
   432→       SELECT content.id, e AS value
   433→       FROM content, jsonb_array_elements_text(${arrayExpr}) AS e
   434→       ${extra}
   435→     ),
   436→     counts AS (
   437→       SELECT value, COUNT(DISTINCT id)::int AS count FROM exploded GROUP BY value
   438→     ),
   439→     samples AS (
   440→       SELECT DISTINCT ON (e.value)
   441→         e.value AS value, content.id AS sid, content.title AS stitle,
   442→         content.thumbnail AS sthumbnail, content.type AS stype
   443→       FROM exploded e JOIN content ON content.id = e.id
   444→       ORDER BY e.value, ${RATING_NUM} DESC NULLS LAST, content.id ASC
   445→     )

   447→            samples.sthumbnail, samples.stype
   448→     FROM counts LEFT JOIN samples ON samples.value = counts.value
   449→     ORDER BY counts.count DESC, counts.value ASC
   450→     LIMIT $1`,
   451→    [cap],
   452→  );
   453→  return data.map((r) => ({
   454→    value: r.value,
   455→    count: num(r.count),
   456→    sample: r.sid != null
   457→      ? { id: num(r.sid), title: r.stitle, thumbnail: r.sthumbnail, type: r.stype }
   458→      : null,
   459→  }));

   504→/* 8. GET /feed — 3-source feed: personal / global / ai + 1:1:1 combined. */
   505→router.get('/feed', wrap(async (req, res) => {
   506→  assertPostgres();
   507→
   508→  // Source A — personal: latest watchlist + history items (requires login).
   509→  let personal;
   510→  if (req.user) {
   511→    const items = await rows(
   512→      `SELECT c.*, 'watchlist' AS origin, w.createdat AS at
   513→       FROM watchlist w JOIN content c ON c.id = w.contentid
   514→       WHERE w.userid = $1
   515→       UNION ALL
   516→       SELECT c.*, 'history' AS origin, h.watchedat AS at
   517→       FROM watchhistory h JOIN content c ON c.id = h.contentid
   518→       WHERE h.userid = $1
   519→       ORDER BY at DESC LIMIT $2`,
   520→      [num(req.user.id), FEED_LIMIT],
   521→    );
   522→    personal = {
   523→      items: items.map((r) => ({ content: contentRow(r), origin: r.origin, at: isoOrNull(r.at) })),
   524→      source: 'neon:watchlist+history',
   525→    };
   526→  } else {
   527→    personal = {
   528→      items: [],
   529→      source: 'neon:watchlist+history',
   530→      note: 'personal feed requires a bearer token (attachUser found no user) — honest empty list',
   531→    };
   532→  }
   533→
   534→  // Source B — global: latest public activity; honest viewevents fallback when empty.
   535→  let global;
   536→  const acts = await rows(
   537→    `SELECT ae.id, ae.type, ae.createdat, ae.contentid, ae.metadata,
   538→            u.username AS actorname, c.title AS contenttitle,
   539→            c.type AS contenttype, c.thumbnail AS contentthumbnail
   540→     FROM activityevents ae
   541→     JOIN users u ON u.id = ae.actorid
   542→     LEFT JOIN content c ON c.id = ae.contentid
   543→     ORDER BY ae.createdat DESC, ae.id DESC LIMIT $1`,
   544→    [FEED_LIMIT],
   545→  );
   546→  if (acts.length > 0) {
   547→    global = {
   548→      items: acts.map((r) => ({
   549→        id: num(r.id),
   550→        type: r.type,
   551→        createdat: isoOrNull(r.createdat),
   552→        actor: r.actorname ?? null,
   553→        content: r.contentid != null
   554→          ? { id: num(r.contentid), title: r.contenttitle, type: r.contenttype, thumbnail: r.contentthumbnail }
   555→          : null,
   556→        preview: (r.metadata && (r.metadata.preview || r.metadata.body))?.slice?.(0, 140) || null,
   557→      })),
   558→      source: 'neon:activity',
   559→    };
   560→  } else {
   561→    const recent = await rows(
   562→      `SELECT c.*, MAX(v.createdat) AS at, COUNT(*)::int AS views
   563→       FROM viewevents v JOIN content c ON c.id = v.contentid
   564→       GROUP BY c.id ORDER BY at DESC LIMIT $1`,
   565→      [FEED_LIMIT],
   566→    );
   567→    global = {
   568→      items: recent.map((r) => ({ content: contentRow(r), at: isoOrNull(r.at), views: num(r.views) })),
   569→      source: 'neon:viewevents-fallback',
   570→      note: 'activityevents is empty — honest fallback to the most recently viewed titles',
   571→    };
   572→  }
   573→
   574→  // Source C — ai: the protected AI Home engine (modules/ai/home.js) does not
   575→  // export its feed builder, so the honest in-process fallback is a simple
   576→  // top-rated-title-per-top-genre query, clearly labeled 'ai-fallback'.
   577→  const picks = await rows(
   578→    `WITH top_genres AS (
   579→       SELECT g AS genre, COUNT(DISTINCT content.id)::int AS count
   580→       FROM content, jsonb_array_elements_text(${GENRES_ARRAY}) AS g
   581→       GROUP BY g ORDER BY count DESC, g ASC LIMIT $1
   582→     ),
   583→     best_per_genre AS (
   584→       SELECT DISTINCT ON (g) g AS genre, content.*
   585→       FROM content, jsonb_array_elements_text(${GENRES_ARRAY}) AS g
   586→       ORDER BY g, ${RATING_NUM} DESC NULLS LAST, content.id ASC
   587→     )
   588→     SELECT best_per_genre.*, top_genres.count AS genre_count
   589→     FROM best_per_genre JOIN top_genres ON top_genres.genre = best_per_genre.genre
   590→     ORDER BY top_genres.count DESC, best_per_genre.genre ASC
   591→     LIMIT $1`,
   592→    [FEED_LIMIT],
   593→  );
   594→  const ai = {
   595→    items: picks.map((r) => ({ ...contentRow(r), genre: r.genre, criteria: 'top-rated-per-genre' })),
   596→    source: 'neon:ai-fallback',
   597→    note: 'AI Home engine (modules/ai/home.js) is protected/not exportable — honest fallback: best-rated title per top genre',
   598→  };
   599→
   600→  // Combined — interleave 1:1:1 (personal → global → ai per round), first 24.
   601→  const combined = [];
   602→  const sources = [
   603→    ['personal', personal.items],
   604→    ['global', global.items],
   605→    ['ai', ai.items],
   606→  ];
   607→  for (let i = 0; combined.length < FEED_COMBINED_CAP; i += 1) {
   608→    let added = false;
   609→    for (const [name, list] of sources) {
   610→      if (combined.length >= FEED_COMBINED_CAP) break;
   611→      if (list[i]) {
   612→        combined.push({ origin: name, item: list[i] });
   613→        added = true;
   614→      }
   615→    }
   616→    if (!added) break;
   617→  }
   618→
   619→  res.json({
   620→    ok: true,
   621→    personal,
   622→    global,
   623→    ai,
   624→    combined,
   625→    combinedCount: combined.length,
   626→    generatedAt: new Date().toISOString(),
   627→  });
   628→}));

   630→/* 9. GET /recommendations — honest split: real Neon picks | logical 30B sample. */
   631→
   632→/** First N items of the protected virtual 30B catalog, fetched via an internal
   633→ *  localhost call to the REAL engine (/api/platform/scale/virtual — stress.js
   634→ *  is protected/read-only, so the engine is reused, never duplicated). */
   635→async function virtualCatalogSample(limit) {
   636→  const port = parseInt(process.env.PORT || '4000', 10);
   637→  const url = `http://localhost:${port}/api/platform/scale/virtual?limit=${limit}`;
   638→  try {
   639→    const res = await fetch(url, { signal: AbortSignal.timeout(VIRTUAL_CALL_TIMEOUT_MS) });
   640→    if (!res.ok) throw new Error(`HTTP ${res.status}`);
   641→    const body = await res.json();
   642→    if (!body?.ok || !Array.isArray(body.items)) throw new Error('unexpected payload');
   643→    return { items: body.items, error: null };
   644→  } catch (err) {
   645→    return { items: [], error: err?.message || String(err) };
   646→  }
   647→}

   649→router.get('/recommendations', wrap(async (req, res) => {
   650→  assertPostgres();
   651→
   652→  // Real — trending: most-viewed titles in the last 7 days.
   653→  const trending = await rows(
   654→    `SELECT c.*, COUNT(*)::int AS views
   655→     FROM viewevents v JOIN content c ON c.id = v.contentid
   656→     WHERE v.createdat > NOW() - INTERVAL '7 days'
   657→     GROUP BY c.id
   658→     ORDER BY views DESC, c.id ASC LIMIT $1`,
   659→    [RECOMMENDATIONS_TRENDING_LIMIT],
   660→  );
   661→  const trendingIds = trending.map((r) => num(r.id));
   662→
   663→  // Real — top-rated fill (high completeness first): covers the "too few
   664→  // trending" case and completes the 12-item real section.
   665→  const fillCount = Math.max(0, RECOMMENDATIONS_REAL_LIMIT - trending.length);
   666→  const fill = fillCount > 0
   667→    ? await rows(
   668→        `SELECT * FROM content
   669→         WHERE id <> ANY($1::int[])
   670→         ORDER BY (CASE WHEN COALESCE(thumbnail, '') <> ''
   671→                        AND COALESCE(metadata->>'overview', '') <> ''
   672→                        AND jsonb_array_length(${GENRES_ARRAY}) > 0
   673→                   THEN 1 ELSE 0 END) DESC,
   674→                  ${RATING_NUM} DESC NULLS LAST, id ASC
   675→         LIMIT $2`,
   676→        [trendingIds.length ? trendingIds : [0], fillCount],
   677→      )
   678→    : [];
   679→  const realItems = [
   680→    ...trending.map((r) => ({ ...contentRow(r), criteria: 'trending', views: num(r.views) })),
   681→    ...fill.map((r) => ({ ...contentRow(r), criteria: 'top-rated' })),
   682→  ].slice(0, RECOMMENDATIONS_REAL_LIMIT);
   683→
   684→  // Logical — first items of the virtual 30B catalog (labeled simulated).
   685→  const virtual = await virtualCatalogSample(RECOMMENDATIONS_LOGICAL_LIMIT);
   686→  const logicalItems = virtual.items.map((item) => ({
   687→    index: item.index,
   688→    title: item.title,
   689→    type: item.type,
   690→    year: item.year,
   691→    rating: item.rating,
   692→    genres: item.genres,
   693→    space: 'logical-30b',
   694→    simulated: true,
   695→  }));
   696→
   697→  logger.debug(
   698→    `[hub] recommendations: real=${realItems.length} (trending=${trending.length}, fill=${fill.length}) logical=${logicalItems.length}${virtual.error ? ` (virtual error: ${virtual.error})` : ''}`,
   699→  );
   700→
   701→  res.json({
   702→    ok: true,
   703→    split: 'onest',
   704→    real: { items: realItems, count: realItems.length, source: 'neon' },
   705→    logical: {
   706→      items: logicalItems,
   707→      count: logicalItems.length,
   708→      source: 'virtual-30b',
   709→      space: 'logical-30b',
   710→      simulated: true,
   711→      note: 'spațiu logic de adresare, nu conținut real — generated by the protected /api/platform/scale/virtual engine via an internal localhost call',
   712→      ...(virtual.error ? { error: `virtual catalog unavailable: ${virtual.error} (honest empty — no mock)` } : {}),
   713→    },
   714→    generatedAt: new Date().toISOString(),
   715→  });
   716→}));
